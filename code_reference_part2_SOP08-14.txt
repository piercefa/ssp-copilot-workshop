"""
===============================================================================
SOP-08 — JAR PENALTY ANALYSIS + BOOTSTRAP
Protein Shake Category Appraisal
===============================================================================

Approved question
-----------------
Which measured JAR deviations are associated with lower overall liking, and
which eligible deviations meet the workshop optimization-priority threshold?

Approved method
---------------
Within each product × measured JAR attribute, collapse the original 5-point
JAR response only for analysis/reporting:

    1–2 = Too Little
      3 = JAR
    4–5 = Too Much

For each direction:

    penalty = mean overall liking(JAR) - mean overall liking(deviation)

This script operationalizes:
    1. Dataset Capability Gate.
    2. Study-design and scale validation.
    3. G9-B cell eligibility before substantive penalty calculation.
    4. minimum_group_n = max(10, ceil(0.20 × N_valid)).
    5. HARD STOP for a product × attribute × direction when either the JAR
       reference group or deviation group fails minimum_group_n.
    6. 9,999-resample BCa bootstrap, 95% CI, RANDOM_STATE = 42.
    7. HARD STOP on bootstrap-supported interpretation for NaN intervals,
       degenerate-data warnings, or non-finite bootstrap SE.
    8. Workshop optimization-priority classification only when:
           deviation prevalence >= 20%
           penalty >= 0.50 hedonic points
           95% bootstrap CI lower bound > 0
    9. One self-contained HTML dashboard with the approved penalty bubble plot,
       uncertainty, priority summary, and collapsed cell-level diagnostics.

IMPORTANT GUARDRAILS
--------------------
- overall_liking is analyzed only as overall liking; it is not purchase intent.
- Only measured JAR variables are eligible. desc_* variables are never used as
  JAR proxies.
- There is no berry JAR or protein/off-note JAR in the approved schema.
- The 400 rows are not treated as 400 independent consumers.
- Penalties are calculated within product, using one consumer × product row.
- Penalty analysis is diagnostic/associative, not causal proof.
- A high penalty with low prevalence is not automatically a priority.
- A high prevalence with negligible mean drop is not automatically a priority.
- The 0.50-point / 20% / CI>0 rule is a project workshop decision threshold,
  not a universal industry standard.
- No mass p-value testing is added; SOP-08 uses bootstrap uncertainty and the
  approved project decision rule rather than a multiplicity-adjusted p-value
  family.
- No separate sensitivity model is invented. For SOP-08, the required
  robustness procedure is the G9-B group-eligibility gate plus BCa bootstrap.

Dependencies
------------
pip install pandas numpy scipy

Usage
-----
Default project structure:

    project/
      SOP08_jar_penalty_dashboard.py
      data/
        protein_shake_category_appraisal_mock_redesigned.csv

Press Run in your IDE.

Optional command-line data-path override:

    python SOP08_jar_penalty_dashboard.py "C:/path/to/data.csv"
"""

def remove_obvious_export_index_columns(
    df: pd.DataFrame,
) -> Tuple[
    pd.DataFrame,
    List[str],
]:
    """
    Remove only obvious CSV export index columns.

    A column is removed only when:
        - its name looks like an exported index; and
        - its values exactly equal 0..N-1 or 1..N.
    """

    work = df.copy()

    removed: List[str] = []

    for col in list(
        work.columns
    ):

        name = (
            str(col)
            .strip()
            .lower()
        )

        if not (
            name.startswith(
                "unnamed:"
            )
            or name == "index"
        ):
            continue

        values = pd.to_numeric(
            work[col],
            errors="coerce",
        )

        if not values.notna().all():
            continue

        arr = (
            values
            .to_numpy()
        )

        n = len(arr)

        zero_index = np.arange(
            n
        )

        one_index = np.arange(
            1,
            n + 1,
        )

        if (
            np.array_equal(
                arr,
                zero_index,
            )
            or np.array_equal(
                arr,
                one_index,
            )
        ):

            work = work.drop(
                columns=[col]
            )

            removed.append(
                str(col)
            )

    return (
        work,
        removed,
    )

def read_dataset(
    path: Path,
) -> Tuple[
    pd.DataFrame,
    List[str],
]:
    """
    Read the active CSV and remove only obvious
    export-index artifacts.
    """

    df = pd.read_csv(
        path
    )

    if df.empty:

        raise AnalysisStop(
            "The active dataset "
            "contains no rows."
        )

    df.columns = [
        str(c).strip()
        for c in df.columns
    ]

    return (
        remove_obvious_export_index_columns(
            df
        )
    )

def dataset_capability_gate(
    df: pd.DataFrame,
) -> Dict[str, object]:
    """
    Determine whether SOP-08 has the measured inputs
    it actually requires.

    No proxy variables are authorized.
    """

    required_base = [
        PANELIST,
        PRODUCT,
        LIKING,
    ]

    missing_base = [
        c
        for c in required_base
        if c not in df.columns
    ]

    present_jar = [
        c
        for c in APPROVED_JAR_ATTRIBUTES
        if c in df.columns
    ]

    missing_jar = [
        c
        for c in APPROVED_JAR_ATTRIBUTES
        if c not in df.columns
    ]

    if missing_base:

        classification = (
            "UNSUPPORTED"
        )

        reason = (
            "Missing required base field(s): "
            + ", ".join(
                missing_base
            )
        )

    elif not present_jar:

        classification = (
            "UNSUPPORTED"
        )

        reason = (
            "No approved measured JAR "
            "variables are present."
        )

    elif missing_jar:

        classification = (
            "PARTIALLY SUPPORTED"
        )

        reason = (
            "Base fields are present, but "
            "some approved JAR variables are "
            "absent; only present measured "
            "JAR variables will be analyzed."
        )

    else:

        classification = (
            "SUPPORTED"
        )

        reason = (
            "All required base fields and "
            "all six approved JAR variables "
            "are present."
        )

    return {
        "classification":
            classification,

        "reason":
            reason,

        "required_base":
            required_base,

        "missing_base":
            missing_base,

        "present_jar":
            present_jar,

        "missing_jar":
            missing_jar,
    }

def validate_numeric_scale(
    series: pd.Series,
    name: str,
    minimum: int,
    maximum: int,
    require_integer: bool = True,
) -> pd.Series:
    """
    Validate a numeric response scale without silently
    clipping, rounding, or substituting invalid values.
    """

    originally_nonmissing = (
        series.notna()
    )

    numeric = pd.to_numeric(
        series,
        errors="coerce",
    )

    bad_non_numeric = (
        originally_nonmissing
        & numeric.isna()
    )

    if bad_non_numeric.any():

        raise AnalysisStop(
            f"{name} contains "
            f"{int(bad_non_numeric.sum())} "
            "non-numeric nonmissing value(s)."
        )

    observed = (
        numeric.dropna()
    )

    outside = (
        (observed < minimum)
        | (observed > maximum)
    )

    if outside.any():

        examples = ", ".join(
            map(
                str,
                observed[
                    outside
                ]
                .head(5)
                .tolist(),
            )
        )

        raise AnalysisStop(
            f"{name} contains "
            f"{int(outside.sum())} value(s) "
            "outside the approved "
            f"{minimum}–{maximum} scale. "
            f"Example(s): {examples}"
        )

    if require_integer:

        non_integer = (
            ~np.isclose(
                observed.to_numpy(),
                np.round(
                    observed.to_numpy()
                ),
            )
        )

        if bool(
            non_integer.any()
        ):

            raise AnalysisStop(
                f"{name} contains "
                "non-integer values, but "
                "the approved scale uses "
                "integer responses."
            )

    return numeric

def validate_study_design(
    df: pd.DataFrame,
) -> Tuple[
    pd.DataFrame,
    List[str],
    Dict[str, object],
]:
    """
    Validate the active study rather than assuming the
    workshop snapshot is always present.
    """

    work = df.copy()

    warnings_out: List[str] = []

    # -----------------------------------------------------------------
    # Identifier integrity
    # -----------------------------------------------------------------

    if (
        work[PANELIST].isna().any()
        or work[PRODUCT].isna().any()
    ):

        raise AnalysisStop(
            "panelist_id and product_code "
            "must be present on every row; "
            "missing identifiers cannot be "
            "assigned to a valid consumer × "
            "product evaluation."
        )

    work[PRODUCT] = (
        work[PRODUCT]
        .astype(str)
        .str.strip()
    )

    if (
        work[PRODUCT]
        == ""
    ).any():

        raise AnalysisStop(
            "product_code contains "
            "blank values."
        )

    # -----------------------------------------------------------------
    # Duplicate consumer × product rows
    # -----------------------------------------------------------------

    duplicate_mask = (
        work.duplicated(
            [
                PANELIST,
                PRODUCT,
            ],
            keep=False,
        )
    )

    if duplicate_mask.any():

        dup_n = int(
            duplicate_mask.sum()
        )

        raise AnalysisStop(
            f"Found {dup_n} row(s) "
            "involved in duplicate "
            "panelist_id × product_code cells. "
            "The script will not average or "
            "silently choose among duplicate "
            "evaluations."
        )

    # -----------------------------------------------------------------
    # Liking scale
    # -----------------------------------------------------------------

    work[LIKING] = (
        validate_numeric_scale(
            work[LIKING],
            LIKING,
            LIKING_MIN,
            LIKING_MAX,
            require_integer=True,
        )
    )

    # -----------------------------------------------------------------
    # Structural counts
    # -----------------------------------------------------------------

    n_rows = len(
        work
    )

    n_panelists = int(
        work[PANELIST]
        .nunique(
            dropna=True
        )
    )

    products = sorted(
        work[PRODUCT]
        .dropna()
        .astype(str)
        .unique()
        .tolist()
    )

    n_products = len(
        products
    )

    if n_panelists < 2:

        raise AnalysisStop(
            "Fewer than two unique "
            "panelists are available."
        )

    if n_products < 1:

        raise AnalysisStop(
            "No products are available "
            "for analysis."
        )

    cells = (
        work[
            [
                PANELIST,
                PRODUCT,
            ]
        ]
        .drop_duplicates()
    )

    expected_cells = (
        n_panelists
        * n_products
    )

    observed_cells = len(
        cells
    )

    missing_cells = (
        expected_cells
        - observed_cells
    )

    evals_per_panelist = (
        work
        .groupby(
            PANELIST,
            observed=True,
        )[PRODUCT]
        .nunique()
    )

    repeat_panelists = int(
        (
            evals_per_panelist
            > 1
        ).sum()
    )

    # -----------------------------------------------------------------
    # Disclose deviations from workshop snapshot
    # -----------------------------------------------------------------

    if (
        n_rows
        != EXPECTED_ROWS
    ):

        warnings_out.append(
            f"Observed {n_rows} rows; "
            "workshop schema snapshot "
            f"expects {EXPECTED_ROWS}."
        )

    if (
        n_panelists
        != EXPECTED_PANELISTS
    ):

        warnings_out.append(
            f"Observed {n_panelists} panelists; "
            "workshop schema snapshot expects "
            f"{EXPECTED_PANELISTS}."
        )

    if (
        n_products
        != EXPECTED_PRODUCTS
    ):

        warnings_out.append(
            f"Observed {n_products} products; "
            "workshop schema snapshot expects "
            f"{EXPECTED_PRODUCTS}."
        )

    if (
        products
        != EXPECTED_PRODUCT_CODES
    ):

        warnings_out.append(
            "Observed product codes are "
            f"{products}; schema snapshot is "
            f"{EXPECTED_PRODUCT_CODES}."
        )

    if missing_cells > 0:

        warnings_out.append(
            f"Observed {observed_cells}/"
            f"{expected_cells} possible "
            "panelist × product cells; "
            f"{missing_cells} cell(s) are "
            "missing. SOP-08 proceeds "
            "cell-by-cell using explicit "
            "valid N."
        )

    if (
        repeat_panelists
        < n_panelists
    ):

        warnings_out.append(
            f"{n_panelists - repeat_panelists} "
            "panelist(s) evaluated only one "
            "observed product. Cross-product "
            "independence is not assumed; "
            "each penalty is still computed "
            "within product."
        )

    missing_liking = int(
        work[LIKING]
        .isna()
        .sum()
    )

    if missing_liking:

        warnings_out.append(
            f"overall_liking is missing on "
            f"{missing_liking} row(s); "
            "those rows are excluded only "
            "from product × JAR cells that "
            "require liking."
        )

    design = {
        "n_rows":
            n_rows,

        "n_panelists":
            n_panelists,

        "n_products":
            n_products,

        "products":
            products,

        "observed_cells":
            observed_cells,

        "expected_cells":
            expected_cells,

        "missing_cells":
            missing_cells,

        "repeat_panelists":
            repeat_panelists,

        "missing_liking":
            missing_liking,
    }

    return (
        work,
        warnings_out,
        design,
    )

def validate_jar_attribute(
    df: pd.DataFrame,
    attribute: str,
) -> Tuple[
    pd.Series,
    List[str],
]:
    """
    Validate one approved measured JAR attribute.

    A descriptive sensory variable is never substituted
    for a missing or invalid JAR variable.
    """

    raw = (
        df[attribute]
    )

    notes: List[str] = []

    originally_nonmissing = (
        raw.notna()
    )

    numeric = pd.to_numeric(
        raw,
        errors="coerce",
    )

    bad_non_numeric = (
        originally_nonmissing
        & numeric.isna()
    )

    if bad_non_numeric.any():

        raise AnalysisStop(
            f"{attribute} contains "
            f"{int(bad_non_numeric.sum())} "
            "non-numeric nonmissing value(s)."
        )

    observed = (
        numeric.dropna()
    )

    outside = (
        (observed < JAR_MIN)
        | (observed > JAR_MAX)
    )

    if outside.any():

        examples = ", ".join(
            map(
                str,
                observed[
                    outside
                ]
                .head(5)
                .tolist(),
            )
        )

        raise AnalysisStop(
            f"{attribute} contains "
            f"{int(outside.sum())} value(s) "
            "outside the approved 1–5 "
            "JAR scale. "
            f"Example(s): {examples}"
        )

    non_integer = (
        ~np.isclose(
            observed.to_numpy(),
            np.round(
                observed.to_numpy()
            ),
        )
    )

    if bool(
        non_integer.any()
    ):

        raise AnalysisStop(
            f"{attribute} contains "
            "non-integer values, but the "
            "approved JAR scale is 1–5 "
            "integer."
        )

    missing_n = int(
        numeric.isna()
        .sum()
    )

    if missing_n:

        notes.append(
            f"{attribute}: {missing_n} "
            "missing JAR response(s); "
            "valid N is recalculated within "
            "each product."
        )

    return (
        numeric,
        notes,
    )

def jar_group(
    value: float,
) -> Optional[str]:
    """
    Approved JAR collapse.

        1–2 -> Too Little
          3 -> JAR
        4–5 -> Too Much
    """

    if pd.isna(
        value
    ):
        return None

    if value <= 2:
        return "Too Little"

    if value == 3:
        return "JAR"

    if value >= 4:
        return "Too Much"

    return None

def penalty_statistic(
    jar_values: np.ndarray,
    deviation_values: np.ndarray,
) -> float:
    """
    Core approved penalty statistic.
    """

    return float(
        np.mean(
            jar_values
        )
        - np.mean(
            deviation_values
        )
    )

def bca_bootstrap_penalty(
    jar_values: np.ndarray,
    deviation_values: np.ndarray,
) -> Tuple[
    Optional[float],
    Optional[float],
    Optional[float],
    str,
    bool,
]:
    """
    Perform the required BCa bootstrap.

    Returns
    -------
    ci_low
    ci_high
    bootstrap_se
    note
    hard_stop

    G9-B hard-stops bootstrap-supported interpretation
    when:
        - interval contains NaN / non-finite values;
        - bootstrap SE is non-finite; or
        - SciPy emits a degenerate-data warning.
    """

    if (
        len(jar_values) == 0
        or len(deviation_values) == 0
    ):

        return (
            None,
            None,
            None,
            "Empty bootstrap group.",
            True,
        )

    with warnings.catch_warnings(
        record=True
    ) as caught:

        warnings.simplefilter(
            "always"
        )

        try:

            kwargs = dict(
                data=(
                    jar_values,
                    deviation_values,
                ),
                statistic=
                    penalty_statistic,
                vectorized=False,
                paired=False,
                confidence_level=
                    CONFIDENCE_LEVEL,
                n_resamples=
                    BOOTSTRAP_RESAMPLES,
                method="BCa",
            )

            # SciPy versions differ in whether the
            # random generator keyword is "rng" or
            # "random_state". Both use seed 42 here.
            try:

                result = (
                    stats.bootstrap(
                        **kwargs,
                        rng=np.random.default_rng(
                            RANDOM_STATE
                        ),
                    )
                )

            except TypeError:

                result = (
                    stats.bootstrap(
                        **kwargs,
                        random_state=
                            RANDOM_STATE,
                    )
                )

        except Exception as exc:

            return (
                None,
                None,
                None,
                "Bootstrap failed: "
                + str(exc),
                True,
            )

    warning_text = " | ".join(
        str(w.message)
        for w in caught
    )

    warning_names = {
        w.category.__name__
        for w in caught
    }

    degenerate = (
        any(
            "degenerate"
            in name.lower()
            for name in warning_names
        )
        or "degenerate"
        in warning_text.lower()
    )

    try:

        ci_low = float(
            result
            .confidence_interval
            .low
        )

        ci_high = float(
            result
            .confidence_interval
            .high
        )

        se = float(
            result.standard_error
        )

    except Exception as exc:

        return (
            None,
            None,
            None,
            "Bootstrap result could "
            "not be read: "
            + str(exc),
            True,
        )

    nonfinite = (
        not all(
            np.isfinite(v)
            for v in [
                ci_low,
                ci_high,
                se,
            ]
        )
    )

    hard_stop = bool(
        degenerate
        or nonfinite
    )

    notes: List[str] = []

    if degenerate:

        notes.append(
            "SciPy emitted a "
            "degenerate-data warning."
        )

    if nonfinite:

        notes.append(
            "Bootstrap CI or standard "
            "error was non-finite."
        )

    if (
        warning_text
        and not degenerate
    ):

        notes.append(
            "SciPy warning(s): "
            + warning_text
        )

    note = " ".join(
        notes
    )

    return (
        ci_low,
        ci_high,
        se,
        note,
        hard_stop,
    )

def minimum_group_n(
    n_valid: int,
) -> int:
    """
    Approved G9-B minimum:

        max(10, ceil(0.20 × N_valid))
    """

    return max(
        MIN_GROUP_ABSOLUTE,
        int(
            math.ceil(
                MIN_GROUP_FRACTION
                * n_valid
            )
        ),
    )

def analyze_direction(
    product_df: pd.DataFrame,
    product: str,
    attribute: str,
    label: str,
    direction: str,
) -> PenaltyCell:
    """
    Analyze one:

        product × JAR attribute × direction

    G9-B eligibility is applied BEFORE calculating a
    substantive penalty.
    """

    valid = (
        product_df[
            [
                PANELIST,
                LIKING,
                attribute,
            ]
        ]
        .dropna()
        .copy()
    )

    valid[
        "_jar_group"
    ] = (
        valid[attribute]
        .map(
            jar_group
        )
    )

    valid = (
        valid[
            valid[
                "_jar_group"
            ].notna()
        ]
        .copy()
    )

    n_valid = len(
        valid
    )

    min_n = (
        minimum_group_n(
            n_valid
        )
    )

    jar_like = (
        valid.loc[
            valid["_jar_group"]
            == "JAR",
            LIKING,
        ]
        .astype(float)
        .to_numpy()
    )

    dev_like = (
        valid.loc[
            valid["_jar_group"]
            == direction,
            LIKING,
        ]
        .astype(float)
        .to_numpy()
    )

    n_jar = len(
        jar_like
    )

    n_dev = len(
        dev_like
    )

    prevalence = (
        100.0
        * n_dev
        / n_valid
        if n_valid
        else 0.0
    )

    cell = PenaltyCell(
        product=
            product,

        attribute=
            attribute,

        attribute_label=
            label,

        direction=
            direction,

        n_valid=
            n_valid,

        minimum_group_n=
            min_n,

        n_jar=
            n_jar,

        n_deviation=
            n_dev,

        prevalence_pct=
            prevalence,
    )

    # -----------------------------------------------------------------
    # G9-B eligibility comes FIRST.
    # -----------------------------------------------------------------

    failed: List[str] = []

    if n_jar < min_n:

        failed.append(
            f"JAR reference N={n_jar} "
            f"< {min_n}"
        )

    if n_dev < min_n:

        failed.append(
            f"deviation N={n_dev} "
            f"< {min_n}"
        )

    if failed:

        cell.eligibility_status = (
            "HARD STOP — GROUP SIZE"
        )

        cell.stop_reason = (
            "; ".join(
                failed
            )
        )

        # Do not calculate or substantively interpret
        # the penalty when the comparison itself fails
        # the approved minimum group-size criterion.
        return cell

    # -----------------------------------------------------------------
    # Descriptive group means / core penalty
    # -----------------------------------------------------------------

    cell.mean_jar = float(
        np.mean(
            jar_like
        )
    )

    cell.mean_deviation = float(
        np.mean(
            dev_like
        )
    )

    cell.penalty = (
        cell.mean_jar
        - cell.mean_deviation
    )

    # -----------------------------------------------------------------
    # Required BCa bootstrap
    # -----------------------------------------------------------------

    (
        ci_low,
        ci_high,
        se,
        note,
        bootstrap_stop,
    ) = (
        bca_bootstrap_penalty(
            jar_like,
            dev_like,
        )
    )

    cell.bootstrap_note = (
        note
    )

    if bootstrap_stop:

        cell.eligibility_status = (
            "HARD STOP — BOOTSTRAP"
        )

        cell.stop_reason = (
            note
            or
            "Bootstrap robustness "
            "criterion failed."
        )

        if (
            ci_low is not None
            and np.isfinite(
                ci_low
            )
        ):

            cell.ci_low = (
                ci_low
            )

        if (
            ci_high is not None
            and np.isfinite(
                ci_high
            )
        ):

            cell.ci_high = (
                ci_high
            )

        if (
            se is not None
            and np.isfinite(
                se
            )
        ):

            cell.bootstrap_se = (
                se
            )

        cell.priority = False

        return cell

    # -----------------------------------------------------------------
    # Bootstrap passed
    # -----------------------------------------------------------------

    cell.ci_low = (
        ci_low
    )

    cell.ci_high = (
        ci_high
    )

    cell.bootstrap_se = (
        se
    )

    cell.eligibility_status = (
        "ELIGIBLE — BOOTSTRAP PASSED"
    )

    # -----------------------------------------------------------------
    # Project-specific workshop optimization priority
    #
    # ALL conditions must be true.
    # -----------------------------------------------------------------

    cell.priority = bool(
        cell.prevalence_pct
        >= PRIORITY_PREVALENCE_PCT

        and cell.penalty
        is not None

        and cell.penalty
        >= PRIORITY_PENALTY

        and cell.ci_low
        is not None

        and cell.ci_low
        > 0
    )

    return cell

def analyze_all_penalties(
    df: pd.DataFrame,
    jar_attributes: Sequence[str],
) -> Tuple[
    List[PenaltyCell],
    List[str],
    Dict[str, str],
]:
    """
    Analyze all present approved JAR attributes.

    An invalid attribute is stopped separately rather
    than causing valid, separable JAR analyses to be
    discarded.
    """

    cells: List[
        PenaltyCell
    ] = []

    notes: List[
        str
    ] = []

    attribute_stops: Dict[
        str,
        str,
    ] = {}

    products = sorted(
        df[PRODUCT]
        .dropna()
        .astype(str)
        .unique()
        .tolist()
    )

    for attribute in jar_attributes:

        label = (
            APPROVED_JAR_ATTRIBUTES[
                attribute
            ]
        )

        try:

            (
                numeric,
                attr_notes,
            ) = (
                validate_jar_attribute(
                    df,
                    attribute,
                )
            )

            notes.extend(
                attr_notes
            )

        except AnalysisStop as exc:

            attribute_stops[
                attribute
            ] = str(
                exc
            )

            continue

        work = df.copy()

        work[
            attribute
        ] = numeric

        for product in products:

            product_df = (
                work.loc[
                    work[PRODUCT]
                    == product
                ]
                .copy()
            )

            for direction in [
                "Too Little",
                "Too Much",
            ]:

                cells.append(
                    analyze_direction(
                        product_df=
                            product_df,

                        product=
                            product,

                        attribute=
                            attribute,

                        label=
                            label,

                        direction=
                            direction,
                    )
                )

    return (
        cells,
        notes,
        attribute_stops,
    )

def eligible_cells(
    cells: Sequence[
        PenaltyCell
    ],
) -> List[
    PenaltyCell
]:

    return [
        c
        for c in cells
        if c.bootstrap_passed
    ]

def priority_cells(
    cells: Sequence[
        PenaltyCell
    ],
) -> List[
    PenaltyCell
]:
    """
    Sort final priorities by penalty magnitude, then
    prevalence.
    """

    return sorted(
        [
            c
            for c in cells
            if c.priority
        ],
        key=lambda c: (
            (
                c.penalty
                if c.penalty
                is not None
                else -np.inf
            ),
            c.prevalence_pct,
        ),
        reverse=True,
    )

def strongest_eligible(
    cells: Sequence[
        PenaltyCell
    ],
) -> Optional[
    PenaltyCell
]:

    elig = [
        c
        for c in cells
        if (
            c.bootstrap_passed
            and c.penalty
            is not None
        )
    ]

    if not elig:
        return None

    return max(
        elig,
        key=lambda c:
            c.penalty,
    )

def build_plain_language_answer(
    cells: Sequence[
        PenaltyCell
    ],
) -> str:
    """
    Lead with the answer to the optimization question,
    not statistical machinery.
    """

    priorities = (
        priority_cells(
            cells
        )
    )

    if priorities:

        top = (
            priorities[0]
        )

        return (
            f"{len(priorities)} measured JAR "
            "deviation(s) meet all workshop "
            "optimization-priority criteria. "
            "The strongest eligible priority is "
            f"Product {top.product}: "
            f"{top.attribute_label} — "
            f"{top.direction}, with "
            f"{top.prevalence_pct:.1f}% affected "
            f"and a {top.penalty:.2f}-point "
            "liking penalty "
            f"(95% bootstrap CI "
            f"{top.ci_low:.2f} to "
            f"{top.ci_high:.2f})."
        )

    strongest = (
        strongest_eligible(
            cells
        )
    )

    if strongest is not None:

        return (
            "No measured JAR deviation meets "
            "all three workshop "
            "optimization-priority criteria. "
            "The largest bootstrap-eligible "
            "observed penalty is "
            f"Product {strongest.product}: "
            f"{strongest.attribute_label} — "
            f"{strongest.direction} "
            f"({strongest.penalty:.2f} "
            "hedonic points; "
            f"{strongest.prevalence_pct:.1f}% "
            "affected), but it does not "
            "satisfy the full project "
            "priority rule."
        )

    return (
        "No product × JAR direction cell "
        "produced a bootstrap-eligible "
        "penalty estimate under the approved "
        "G9-B group-size and bootstrap rules."
    )

def map_adequacy_gate(
    cumulative_variance: float,
) -> tuple[str, str]:
    """
    Apply G9-C two-dimensional PCA map interpretation thresholds.

    These are project visualization-governance thresholds, not universal
    statistical laws.
    """

    if not np.isfinite(
        cumulative_variance
    ):

        return (
            "HARD STOP",
            (
                "Two-dimensional explained "
                "variance is unavailable."
            ),
        )

    if cumulative_variance >= 0.60:

        return (
            "PASS",
            (
                "Strong 2D workshop summary "
                "(PC1 + PC2 ≥ 60%)."
            ),
        )

    if cumulative_variance >= 0.50:

        return (
            "WARNING",
            (
                "Usable with caution: "
                "PC1 + PC2 explain "
                "50% to <60%."
            ),
        )

    if cumulative_variance >= 0.40:

        return (
            "WARNING",
            (
                "Exploratory / limited "
                "2D representation: "
                "PC1 + PC2 explain "
                "40% to <50%."
            ),
        )

    return (
        "WARNING",
        (
            "PC1 + PC2 explain <40%; "
            "do not summarize the "
            "preference structure primarily "
            "from the 2D map."
        ),
    )

def unit_rows(
    array: np.ndarray,
) -> tuple[
    np.ndarray,
    np.ndarray,
]:
    """Normalize each 2D row vector when its magnitude is non-zero."""

    norms = (
        np.linalg.norm(
            array,
            axis=1,
        )
    )

    usable = (
        norms
        > ZERO_TOL
    )

    normalized = (
        np.zeros_like(
            array,
            dtype=float,
        )
    )

    normalized[
        usable
    ] = (
        array[
            usable
        ]
        / norms[
            usable,
            None,
        ]
    )

    return (
        normalized,
        usable,
    )

def direction_summary(
    vectors: np.ndarray,
) -> dict:
    """
    Summarize descriptive directional dispersion.

    This is intentionally NOT converted to a significance test or a forced
    segmentation. IPM is exploratory.
    """

    normalized, usable = (
        unit_rows(
            vectors
        )
    )

    normalized = (
        normalized[
            usable
        ]
    )

    if len(
        normalized
    ) == 0:

        return {
            "n": 0,
            "opposite": 0,
            "concentration": np.nan,
            "mean": np.zeros(
                2
            ),
        }

    mean_vector = (
        normalized
        .mean(
            axis=0
        )
    )

    concentration = float(
        np.linalg.norm(
            mean_vector
        )
    )

    if (
        concentration
        <= ZERO_TOL
    ):

        return {
            "n": len(
                normalized
            ),
            "opposite": 0,
            "concentration": concentration,
            "mean": np.zeros(
                2
            ),
        }

    mean_direction = (
        mean_vector
        / concentration
    )

    dot_products = (
        normalized
        @ mean_direction
    )

    return {
        "n": len(
            normalized
        ),
        "opposite": int(
            np.sum(
                dot_products
                < 0
            )
        ),
        "concentration": concentration,
        "mean": mean_direction,
    }

def capability_gate(
    df: pd.DataFrame,
) -> dict:
    """
    Apply the Dataset Capability Gate before analysis.

    Core IPM requirements:
        panelist_id
        product_code
        overall_liking

    EPM requires measured descriptive sensory variables.

    Missing descriptive variables may make SOP-09 only PARTIALLY SUPPORTED,
    because IPM is still a separable supported component when the liking
    inputs exist.

    No neighboring variable is substituted.
    """

    core_required = [
        PANELIST,
        PRODUCT,
        LIKING,
    ]

    core_missing = [
        column
        for column
        in core_required
        if column
        not in df.columns
    ]

    present_descriptive = [
        column
        for column
        in DESC
        if column
        in df.columns
    ]

    missing_descriptive = [
        column
        for column
        in DESC
        if column
        not in df.columns
    ]

    if core_missing:

        return {
            "status": (
                "UNSUPPORTED"
            ),
            "core_missing": (
                core_missing
            ),
            "present": (
                present_descriptive
            ),
            "missing": (
                missing_descriptive
            ),
        }

    if (
        not present_descriptive
        or missing_descriptive
    ):

        return {
            "status": (
                "PARTIALLY SUPPORTED"
            ),
            "core_missing": [],
            "present": (
                present_descriptive
            ),
            "missing": (
                missing_descriptive
            ),
        }

    return {
        "status": (
            "SUPPORTED"
        ),
        "core_missing": [],
        "present": (
            present_descriptive
        ),
        "missing": [],
    }

def validate_design(
    raw_df: pd.DataFrame,
) -> tuple[
    pd.DataFrame,
    dict,
]:
    """
    Validate the study structure required for preference mapping.

    Critical rules:
    - one row per consumer × product;
    - complete liking matrix;
    - no silent averaging of duplicates;
    - no liking imputation;
    - liking must use the approved 1–9 scale.
    """

    df = (
        raw_df
        .copy()
    )

    info = {
        "hard": False,
        "reason": "",
        "warnings": [],
        "rows": len(
            df
        ),
        "consumers": 0,
        "products": 0,
        "duplicates": 0,
        "missing_cells": 0,
    }

    if (
        df[
            PANELIST
        ]
        .isna()
        .any()
        or
        df[
            PRODUCT
        ]
        .isna()
        .any()
    ):

        info.update(
            hard=True,
            reason=(
                "panelist_id and product_code "
                "must be present on every row."
            ),
        )

        return (
            df,
            info,
        )

    # Convert identifiers to strings only for robust labels/pivots.
    # This does not change their substantive meaning.
    df[
        PANELIST
    ] = (
        df[
            PANELIST
        ]
        .astype(
            str
        )
    )

    df[
        PRODUCT
    ] = (
        df[
            PRODUCT
        ]
        .astype(
            str
        )
    )

    original_nonmissing = (
        df[
            LIKING
        ]
        .notna()
    )

    liking_numeric = (
        pd.to_numeric(
            df[
                LIKING
            ],
            errors="coerce",
        )
    )

    nonnumeric = int(
        (
            original_nonmissing
            & liking_numeric
            .isna()
        )
        .sum()
    )

    missing = int(
        liking_numeric
        .isna()
        .sum()
    )

    out_of_range = int(
        (
            liking_numeric
            .notna()
            &
            (
                (
                    liking_numeric
                    < 1
                )
                |
                (
                    liking_numeric
                    > 9
                )
            )
        )
        .sum()
    )

    info.update(
        liking_missing=missing,
        liking_invalid=(
            nonnumeric
            + out_of_range
        ),
    )

    if nonnumeric:

        info.update(
            hard=True,
            reason=(
                f"{nonnumeric} overall_liking "
                "value(s) are non-numeric; "
                "no silent coercion or deletion "
                "is allowed."
            ),
        )

        return (
            df,
            info,
        )

    if missing:

        info.update(
            hard=True,
            reason=(
                "overall_liking has "
                f"{missing} missing value(s). "
                "IPM requires a complete "
                "consumer × product matrix; "
                "no imputation is authorized."
            ),
        )

        return (
            df,
            info,
        )

    if out_of_range:

        info.update(
            hard=True,
            reason=(
                f"{out_of_range} overall_liking "
                "value(s) fall outside the "
                "approved 1–9 hedonic scale."
            ),
        )

        return (
            df,
            info,
        )

    df[
        LIKING
    ] = (
        liking_numeric
        .astype(
            float
        )
    )

    info[
        "consumers"
    ] = (
        df[
            PANELIST
        ]
        .nunique()
    )

    info[
        "products"
    ] = (
        df[
            PRODUCT
        ]
        .nunique()
    )

    info[
        "duplicates"
    ] = int(
        df
        .duplicated(
            [
                PANELIST,
                PRODUCT,
            ]
        )
        .sum()
    )

    if info[
        "duplicates"
    ]:

        info.update(
            hard=True,
            reason=(
                "Found "
                f"{info['duplicates']} "
                "duplicate consumer × product "
                "row(s). SOP-09 will not "
                "silently average duplicates."
            ),
        )

        return (
            df,
            info,
        )

    if (
        info[
            "consumers"
        ] < 2
        or
        info[
            "products"
        ] < 3
    ):

        info.update(
            hard=True,
            reason=(
                "Preference mapping requires "
                "at least 2 consumers and "
                "3 products for a "
                "non-degenerate 2D map."
            ),
        )

        return (
            df,
            info,
        )

    expected_cells = (
        info[
            "consumers"
        ]
        * info[
            "products"
        ]
    )

    observed_cells = len(
        df[
            [
                PANELIST,
                PRODUCT,
            ]
        ]
        .drop_duplicates()
    )

    info[
        "missing_cells"
    ] = (
        expected_cells
        - observed_cells
    )

    if info[
        "missing_cells"
    ]:

        info.update(
            hard=True,
            reason=(
                "Consumer × product liking "
                "matrix is incomplete: "
                f"{info['missing_cells']} "
                "cell(s) are missing. "
                "No liking imputation is "
                "authorized."
            ),
        )

        return (
            df,
            info,
        )

    # The script inspects the actual data rather than assuming the
    # current workshop snapshot.
    if (
        info[
            "rows"
        ],
        info[
            "consumers"
        ],
        info[
            "products"
        ],
    ) != (
        400,
        50,
        8,
    ):

        info[
            "warnings"
        ].append(
            (
                "Active data differ from the "
                "workshop 400-row / "
                "50-consumer / 8-product "
                "snapshot. The actual "
                "validated structure is used."
            )
        )

    return (
        df,
        info,
    )

def validate_descriptive_variables(
    df: pd.DataFrame,
    candidates: list[str],
) -> tuple[
    pd.DataFrame,
    list[str],
    list[str],
    list[str],
]:
    """
    Validate measured descriptive variables.

    EPM uses actual measured sensory intensity values only.
    Product metadata and qualitative comments are never converted into
    quantitative predictors.
    """

    working = (
        df
        .copy()
    )

    eligible = []
    excluded = []
    notes = []

    for attribute in candidates:

        raw = (
            working[
                attribute
            ]
        )

        numeric = (
            pd.to_numeric(
                raw,
                errors="coerce",
            )
        )

        nonnumeric = int(
            (
                raw
                .notna()
                &
                numeric
                .isna()
            )
            .sum()
        )

        missing = int(
            numeric
            .isna()
            .sum()
        )

        outside_scale = int(
            (
                numeric
                .notna()
                &
                (
                    (
                        numeric
                        < 0
                    )
                    |
                    (
                        numeric
                        > 10
                    )
                )
            )
            .sum()
        )

        if (
            nonnumeric
            or missing
            or outside_scale
        ):

            excluded.append(
                attribute
            )

            notes.append(
                (
                    f"{attribute}: excluded "
                    f"({nonnumeric} nonnumeric, "
                    f"{missing} missing, "
                    f"{outside_scale} outside "
                    "the approved 0–10 "
                    "descriptive scale)."
                )
            )

            continue

        working[
            attribute
        ] = (
            numeric
            .astype(
                float
            )
        )

        eligible.append(
            attribute
        )

    return (
        working,
        eligible,
        excluded,
        notes,
    )

def run_ipm(
    df: pd.DataFrame,
) -> dict:
    """
    Perform PCA-based Internal Preference Mapping.

    Analysis-plan transformation
    ----------------------------
    1. Build consumer × product overall-liking matrix.
    2. Center each consumer across products.
    3. Do NOT variance-standardize consumers.
    4. Transpose to products × consumers.
    5. PCA creates product positions and consumer preference directions.

    Why center within consumer?
    ---------------------------
    The current question concerns relative product preference direction.
    Mean-centering removes each consumer's general high/low scale-use offset
    while preserving the size of their relative preference differences.

    Why not z-standardize each consumer?
    ------------------------------------
    Standardizing would additionally force all consumers to equal variance
    and would change the analytical question. The SOP requires such choices
    to be explicit, so this script does not make that transformation silently.
    """

    result = {
        "hard": False,
        "reason": "",
    }

    preference = (
        df
        .pivot(
            index=PANELIST,
            columns=PRODUCT,
            values=LIKING,
        )
    )

    if (
        preference
        .isna()
        .any()
        .any()
    ):

        return {
            "hard": True,
            "reason": (
                "Pivoted liking matrix contains "
                "missing values. IPM does not "
                "silently impute liking."
            ),
        }

    centered = (
        preference
        .sub(
            preference
            .mean(
                axis=1
            ),
            axis=0,
        )
    )

    within_consumer_sd = (
        centered
        .std(
            axis=1,
            ddof=0,
        )
    )

    zero_variance_consumers = (
        within_consumer_sd
        .index[
            within_consumer_sd
            <= ZERO_TOL
        ]
        .astype(
            str
        )
        .tolist()
    )

    valid_consumers = (
        within_consumer_sd
        .index[
            within_consumer_sd
            > ZERO_TOL
        ]
    )

    if len(
        valid_consumers
    ) < 2:

        return {
            "hard": True,
            "reason": (
                "Fewer than two consumers show "
                "within-consumer liking "
                "variation across products, "
                "so preference directions "
                "cannot be estimated."
            ),
        }

    # Products are observations.
    # Consumers are preference variables.
    X = (
        centered
        .loc[
            valid_consumers
        ]
        .T
        .to_numpy(
            dtype=float
        )
    )

    pca = PCA(
        n_components=2,
        svd_solver="full",
    )

    scores = (
        pca
        .fit_transform(
            X
        )
    )

    explained_variance_ratio = (
        pca
        .explained_variance_ratio_
    )

    cumulative_2d = float(
        explained_variance_ratio[
            :2
        ]
        .sum()
    )

    (
        map_status,
        map_message,
    ) = map_adequacy_gate(
        cumulative_2d
    )

    # PCA variable coordinates / correlations for consumer vectors.
    loadings = (
        pca
        .components_
        .T
        * np.sqrt(
            np.maximum(
                pca
                .explained_variance_[
                    :2
                ],
                0,
            )
        )
    )

    products = (
        preference
        .columns
        .astype(
            str
        )
        .tolist()
    )

    product_scores = (
        pd.DataFrame(
            {
                PRODUCT: (
                    products
                ),
                "PC1": (
                    scores[
                        :,
                        0
                    ]
                ),
                "PC2": (
                    scores[
                        :,
                        1
                    ]
                ),
            }
        )
    )

    consumer_vectors = (
        pd.DataFrame(
            {
                PANELIST: (
                    valid_consumers
                    .astype(
                        str
                    )
                ),
                "PC1": (
                    loadings[
                        :,
                        0
                    ]
                ),
                "PC2": (
                    loadings[
                        :,
                        1
                    ]
                ),
            }
        )
    )

    result.update(
        preference=preference,
        centered=centered,
        product_scores=(
            product_scores
        ),
        consumer_vectors=(
            consumer_vectors
        ),
        evr=(
            explained_variance_ratio
        ),
        cum=(
            cumulative_2d
        ),
        map_status=(
            map_status
        ),
        map_message=(
            map_message
        ),
        zero_consumers=(
            zero_variance_consumers
        ),
        direction=(
            direction_summary(
                consumer_vectors[
                    [
                        "PC1",
                        "PC2",
                    ]
                ]
                .to_numpy(
                    dtype=float
                )
            )
        ),
    )

    return result

def identical_sensory_pairs(
    sensory: pd.DataFrame,
    mean_liking: pd.Series,
) -> list[
    tuple[
        str,
        str,
        float,
    ]
]:
    """
    Detect product pairs with identical measured sensory profiles.

    This is an important construct guardrail. If two products are identical
    on the measured sensory variables but differ in liking, the script must
    not invent an unmeasured quantitative sensory variable to explain it.
    """

    pairs = []

    products = (
        sensory
        .index
        .astype(
            str
        )
        .tolist()
    )

    for index, first in enumerate(
        products
    ):

        for second in products[
            index
            + 1:
        ]:

            same_profile = (
                np.allclose(
                    sensory
                    .loc[
                        first
                    ]
                    .to_numpy(
                        dtype=float
                    ),
                    sensory
                    .loc[
                        second
                    ]
                    .to_numpy(
                        dtype=float
                    ),
                    rtol=1e-10,
                    atol=1e-10,
                )
            )

            if same_profile:

                liking_difference = float(
                    mean_liking
                    .loc[
                        first
                    ]
                    - mean_liking
                    .loc[
                        second
                    ]
                )

                pairs.append(
                    (
                        first,
                        second,
                        liking_difference,
                    )
                )

    return pairs

def run_epm(
    df: pd.DataFrame,
    preference: pd.DataFrame,
    descriptive_attributes: list[str],
    preexcluded: list[str],
) -> dict:
    """
    Perform External Preference Mapping.

    This is intentionally a SEPARATE measured-sensory space rather than
    cosmetically overlaying sensory arrows on the IPM.

    Steps
    -----
    1. Aggregate measured descriptive attributes to product means.
    2. Detect and explicitly remove zero-variance product-level predictors.
    3. Standardize sensory variables across products.
    4. PCA creates the measured sensory product space.
    5. For each consumer, regress product liking on sensory PC1 and PC2.
    6. The two slopes define that consumer's preference direction inside the
       measured sensory space.

    This vector-model EPM preserves individual consumer directions and does
    not force a single ideal point.
    """

    result = {
        "available": False,
        "hard": False,
        "reason": "",
        "excluded": list(
            preexcluded
        ),
    }

    if len(
        descriptive_attributes
    ) < 2:

        result.update(
            hard=True,
            reason=(
                "Fewer than two valid measured "
                "descriptive sensory attributes "
                "are available for 2D EPM."
            ),
        )

        return result

    products = (
        preference
        .columns
        .astype(
            str
        )
        .tolist()
    )

    sensory = (
        df
        .groupby(
            PRODUCT,
            observed=True,
        )[
            descriptive_attributes
        ]
        .mean()
        .reindex(
            products
        )
    )

    if (
        sensory
        .isna()
        .any()
        .any()
    ):

        result.update(
            hard=True,
            reason=(
                "Product-level sensory means "
                "contain missing values. "
                "EPM does not silently "
                "impute them."
            ),
        )

        return result

    # ---------------------------------------------------------------------
    # G9-C ZERO-VARIANCE DIAGNOSTIC
    # ---------------------------------------------------------------------
    # Zero-variance sensory attributes cannot contribute to PCA.
    # They are detected, disclosed, then removed before the PCA is rerun.
    # ---------------------------------------------------------------------

    sensory_sd = (
        sensory
        .std(
            axis=0,
            ddof=0,
        )
    )

    zero_variance = (
        sensory_sd
        .index[
            sensory_sd
            <= ZERO_TOL
        ]
        .tolist()
    )

    descriptive_attributes = [
        attribute
        for attribute
        in descriptive_attributes
        if attribute
        not in zero_variance
    ]

    result[
        "excluded"
    ].extend(
        zero_variance
    )

    if len(
        descriptive_attributes
    ) < 2:

        result.update(
            hard=True,
            reason=(
                "After disclosed "
                "zero-variance removal, "
                "fewer than two measured "
                "sensory attributes remain."
            ),
        )

        return result

    sensory = (
        sensory[
            descriptive_attributes
        ]
    )

    # ---------------------------------------------------------------------
    # STANDARDIZE PRODUCT-LEVEL MEASURED SENSORY VARIABLES
    # ---------------------------------------------------------------------

    scaler = (
        StandardScaler()
    )

    X_scaled = (
        scaler
        .fit_transform(
            sensory
            .to_numpy(
                dtype=float
            )
        )
    )

    sensory_pca = PCA(
        n_components=2,
        svd_solver="full",
    )

    sensory_scores = (
        sensory_pca
        .fit_transform(
            X_scaled
        )
    )

    explained_variance_ratio = (
        sensory_pca
        .explained_variance_ratio_
    )

    cumulative_2d = float(
        explained_variance_ratio[
            :2
        ]
        .sum()
    )

    (
        map_status,
        map_message,
    ) = map_adequacy_gate(
        cumulative_2d
    )

    attribute_loadings = (
        sensory_pca
        .components_
        .T
        * np.sqrt(
            np.maximum(
                sensory_pca
                .explained_variance_[
                    :2
                ],
                0,
            )
        )
    )

    product_scores = (
        pd.DataFrame(
            {
                PRODUCT: products,
                "PC1": (
                    sensory_scores[
                        :,
                        0
                    ]
                ),
                "PC2": (
                    sensory_scores[
                        :,
                        1
                    ]
                ),
            }
        )
    )

    attribute_vectors = (
        pd.DataFrame(
            {
                "attribute": (
                    descriptive_attributes
                ),
                "PC1": (
                    attribute_loadings[
                        :,
                        0
                    ]
                ),
                "PC2": (
                    attribute_loadings[
                        :,
                        1
                    ]
                ),
            }
        )
    )

    # ---------------------------------------------------------------------
    # CONSUMER PREFERENCE VECTORS INSIDE THE MEASURED SENSORY SPACE
    # ---------------------------------------------------------------------

    design_matrix = (
        np.column_stack(
            [
                np.ones(
                    len(
                        products
                    )
                ),
                sensory_scores[
                    :,
                    0
                ],
                sensory_scores[
                    :,
                    1
                ],
            ]
        )
    )

    vector_rows = []

    for consumer in (
        preference
        .index
        .astype(
            str
        )
    ):

        y = (
            preference
            .loc[
                consumer,
                products,
            ]
            .to_numpy(
                dtype=float
            )
        )

        total_sum_squares = float(
            np.sum(
                (
                    y
                    - y.mean()
                )
                ** 2
            )
        )

        if (
            total_sum_squares
            <= ZERO_TOL
        ):

            vector_rows.append(
                {
                    PANELIST: consumer,
                    "PC1": 0.0,
                    "PC2": 0.0,
                    "r2": np.nan,
                }
            )

            continue

        beta, *_ = (
            np.linalg.lstsq(
                design_matrix,
                y,
                rcond=None,
            )
        )

        predicted = (
            design_matrix
            @ beta
        )

        residual_sum_squares = float(
            np.sum(
                (
                    y
                    - predicted
                )
                ** 2
            )
        )

        r_squared = (
            1
            - residual_sum_squares
            / total_sum_squares
        )

        vector_rows.append(
            {
                PANELIST: consumer,
                "PC1": float(
                    beta[
                        1
                    ]
                ),
                "PC2": float(
                    beta[
                        2
                    ]
                ),
                "r2": (
                    r_squared
                ),
            }
        )

    consumer_vectors = (
        pd.DataFrame(
            vector_rows
        )
    )

    preference_direction = (
        direction_summary(
            consumer_vectors[
                [
                    "PC1",
                    "PC2",
                ]
            ]
            .to_numpy(
                dtype=float
            )
        )
    )

    # ---------------------------------------------------------------------
    # ATTRIBUTE ALIGNMENT WITH SAMPLE-AVERAGE EPM DIRECTION
    # ---------------------------------------------------------------------
    # This is used only as a descriptive directional summary.
    #
    # It is NOT:
    # - a causal effect size;
    # - a significance test;
    # - a validated driver hierarchy.
    # ---------------------------------------------------------------------

    attribute_alignment = []

    if (
        np.linalg.norm(
            preference_direction[
                "mean"
            ]
        )
        > ZERO_TOL
    ):

        (
            normalized_attributes,
            usable_attributes,
        ) = unit_rows(
            attribute_vectors[
                [
                    "PC1",
                    "PC2",
                ]
            ]
            .to_numpy(
                dtype=float
            )
        )

        for index, attribute in enumerate(
            descriptive_attributes
        ):

            if not usable_attributes[
                index
            ]:
                continue

            cosine_alignment = float(
                normalized_attributes[
                    index
                ]
                @ preference_direction[
                    "mean"
                ]
            )

            attribute_alignment.append(
                (
                    attribute,
                    cosine_alignment,
                )
            )

        attribute_alignment.sort(
            key=lambda item: (
                item[
                    1
                ]
            ),
            reverse=True,
        )

    mean_liking = (
        df
        .groupby(
            PRODUCT,
            observed=True,
        )[
            LIKING
        ]
        .mean()
        .reindex(
            products
        )
    )

    identical_pairs = (
        identical_sensory_pairs(
            sensory,
            mean_liking,
        )
    )

    median_r_squared = float(
        consumer_vectors[
            "r2"
        ]
        .dropna()
        .median()
    )

    result.update(
        available=True,
        attrs=(
            descriptive_attributes
        ),
        sensory=(
            sensory
        ),
        product_scores=(
            product_scores
        ),
        attribute_vectors=(
            attribute_vectors
        ),
        consumer_vectors=(
            consumer_vectors
        ),
        median_r2=(
            median_r_squared
        ),
        evr=(
            explained_variance_ratio
        ),
        cum=(
            cumulative_2d
        ),
        map_status=(
            map_status
        ),
        map_message=(
            map_message
        ),
        direction=(
            preference_direction
        ),
        align=(
            attribute_alignment
        ),
        identical=(
            identical_pairs
        ),
    )

    return result

def fit_pls_fold(
    X_train: np.ndarray,
    y_train: np.ndarray,
    X_test: np.ndarray,
    n_components: int,
) -> tuple[
    float,
    np.ndarray,
]:
    """
    Fit one LOPO PLS fold.

    Scaling is learned on the training products only to prevent leakage.
    """

    scaler = (
        StandardScaler()
    )

    X_train_scaled = (
        scaler
        .fit_transform(
            X_train
        )
    )

    X_test_scaled = (
        scaler
        .transform(
            X_test
        )
    )

    model = (
        PLSRegression(
            n_components=(
                n_components
            ),
            scale=False,
            max_iter=1000,
            tol=1e-8,
        )
    )

    model.fit(
        X_train_scaled,
        y_train.reshape(
            -1,
            1,
        ),
    )

    prediction = float(
        np.asarray(
            model.predict(
                X_test_scaled
            )
        )
        .ravel()[
            0
        ]
    )

    coefficients = (
        np.asarray(
            model.coef_,
            dtype=float,
        )
        .reshape(
            -1
        )
    )

    return (
        prediction,
        coefficients,
    )

def pls_leave_one_product_out(
    X: np.ndarray,
    y: np.ndarray,
    n_components: int,
) -> tuple[
    float,
    float,
    float,
    np.ndarray,
]:
    """
    Leave-one-product-out cross-validation.

    Q² reference:
        prediction error from each LOPO training-fold mean.

    Therefore Q² > 0 means the PLS prediction outperformed the corresponding
    mean-only prediction reference.
    """

    predictions = []
    baseline_predictions = []
    fold_coefficients = []

    for held_out in range(
        len(
            y
        )
    ):

        keep = (
            np.ones(
                len(
                    y
                ),
                dtype=bool,
            )
        )

        keep[
            held_out
        ] = False

        (
            prediction,
            coefficients,
        ) = fit_pls_fold(
            X[
                keep
            ],
            y[
                keep
            ],
            X[
                ~keep
            ],
            n_components,
        )

        predictions.append(
            prediction
        )

        fold_coefficients.append(
            coefficients
        )

        baseline_predictions.append(
            float(
                y[
                    keep
                ]
                .mean()
            )
        )

    predictions = (
        np.asarray(
            predictions,
            dtype=float,
        )
    )

    baseline_predictions = (
        np.asarray(
            baseline_predictions,
            dtype=float,
        )
    )

    press = float(
        np.sum(
            (
                y
                - predictions
            )
            ** 2
        )
    )

    mean_only_press = float(
        np.sum(
            (
                y
                - baseline_predictions
            )
            ** 2
        )
    )

    if (
        mean_only_press
        <= ZERO_TOL
    ):

        q_squared = (
            np.nan
        )

    else:

        q_squared = (
            1
            - press
            / mean_only_press
        )

    return (
        q_squared,
        press,
        mean_only_press,
        np.vstack(
            fold_coefficients
        ),
    )

def run_optional_pls(
    df: pd.DataFrame,
    epm_result: dict,
) -> dict:
    """
    Run the optional SOP-09 PLS sensitivity/modeling layer.

    G9-G enforcement
    ----------------
    - product-level unit of analysis;
    - maximum two components;
    - leave-one-product-out validation;
    - Q² <= 0 is a HARD STOP on interpreting a driver hierarchy;
    - coefficients changing sign in >25% of folds cannot be ranked as stable
      positive/negative directions.
    """

    result = {
        "requested": (
            RUN_OPTIONAL_PLS
        ),
        "available": False,
        "hard": False,
        "reason": "",
        "q2_by_components": {},
    }

    if not RUN_OPTIONAL_PLS:
        return result

    if (
        not epm_result
        .get(
            "available"
        )
        or epm_result
        .get(
            "hard"
        )
    ):

        result.update(
            hard=True,
            reason=(
                "PLS was requested, but "
                "a valid measured "
                "product sensory matrix "
                "is unavailable."
            ),
        )

        return result

    sensory = (
        epm_result[
            "sensory"
        ]
    )

    attributes = (
        epm_result[
            "attrs"
        ]
    )

    products = (
        sensory
        .index
        .astype(
            str
        )
        .tolist()
    )

    y = (
        df
        .groupby(
            PRODUCT,
            observed=True,
        )[
            LIKING
        ]
        .mean()
        .reindex(
            products
        )
        .to_numpy(
            dtype=float
        )
    )

    X = (
        sensory
        .to_numpy(
            dtype=float
        )
    )

    result[
        "n_products"
    ] = len(
        y
    )

    if (
        len(
            y
        ) < 4
        or
        np.std(
            y
        )
        <= ZERO_TOL
    ):

        result.update(
            hard=True,
            reason=(
                "PLS lacks enough "
                "product-level observations "
                "or outcome variation for "
                "leave-one-product-out "
                "validation."
            ),
        )

        return result

    maximum_components = min(
        MAX_PLS_COMPONENTS,
        X.shape[
            1
        ],
        len(
            y
        )
        - 2,
    )

    candidates = {}

    for n_components in range(
        1,
        maximum_components
        + 1,
    ):

        try:

            candidates[
                n_components
            ] = (
                pls_leave_one_product_out(
                    X,
                    y,
                    n_components,
                )
            )

            result[
                "q2_by_components"
            ][
                n_components
            ] = (
                candidates[
                    n_components
                ][
                    0
                ]
            )

        except Exception:

            result[
                "q2_by_components"
            ][
                n_components
            ] = (
                np.nan
            )

    valid_candidates = [
        (
            components,
            q_squared,
        )
        for (
            components,
            q_squared,
        )
        in result[
            "q2_by_components"
        ]
        .items()
        if np.isfinite(
            q_squared
        )
    ]

    if not valid_candidates:

        result.update(
            hard=True,
            reason=(
                "All eligible "
                "leave-one-product-out PLS "
                "candidate models failed."
            ),
        )

        return result

    # Choose highest Q².
    # Exact ties favor the smaller component count.
    selected_components = (
        sorted(
            valid_candidates,
            key=lambda item: (
                -item[
                    1
                ],
                item[
                    0
                ],
            ),
        )[
            0
        ][
            0
        ]
    )

    (
        q_squared,
        press,
        mean_only_press,
        fold_coefficients,
    ) = (
        candidates[
            selected_components
        ]
    )

    # Fit the selected model on all products for descriptive coefficients.
    final_scaler = (
        StandardScaler()
    )

    X_scaled = (
        final_scaler
        .fit_transform(
            X
        )
    )

    final_model = (
        PLSRegression(
            n_components=(
                selected_components
            ),
            scale=False,
            max_iter=1000,
            tol=1e-8,
        )
    )

    final_model.fit(
        X_scaled,
        y.reshape(
            -1,
            1,
        ),
    )

    final_coefficients = (
        np.asarray(
            final_model.coef_,
            dtype=float,
        )
        .reshape(
            -1
        )
    )

    coefficient_rows = []

    for index, attribute in enumerate(
        attributes
    ):

        final_coefficient = float(
            final_coefficients[
                index
            ]
        )

        if (
            abs(
                final_coefficient
            )
            <= ZERO_TOL
        ):

            final_sign = 0

        else:

            final_sign = int(
                np.sign(
                    final_coefficient
                )
            )

        if final_sign == 0:

            sign_change_fraction = (
                1.0
            )

        else:

            fold_signs = (
                np.sign(
                    fold_coefficients[
                        :,
                        index
                    ]
                )
            )

            sign_change_fraction = float(
                np.mean(
                    fold_signs
                    != final_sign
                )
            )

        stable = (
            sign_change_fraction
            <= PLS_SIGN_CHANGE_MAX
        )

        coefficient_rows.append(
            {
                "attribute": (
                    attribute
                ),
                "coefficient": (
                    final_coefficient
                ),
                "sign_change": (
                    sign_change_fraction
                ),
                "stable": (
                    stable
                ),
            }
        )

    coefficients = (
        pd.DataFrame(
            coefficient_rows
        )
        .sort_values(
            "coefficient",
            key=lambda series: (
                series
                .abs()
            ),
            ascending=False,
        )
        .reset_index(
            drop=True
        )
    )

    result.update(
        available=True,
        selected=(
            selected_components
        ),
        q2=(
            float(
                q_squared
            )
        ),
        press=(
            press
        ),
        mean_press=(
            mean_only_press
        ),
        coefficients=(
            coefficients
        ),
    )

    # ---------------------------------------------------------------------
    # G9-G HARD STOP
    # ---------------------------------------------------------------------

    if (
        not np.isfinite(
            q_squared
        )
        or
        q_squared
        <= 0
    ):

        result.update(
            hard=True,
            reason=(
                "G9-G hard stop: "
                "leave-one-product-out "
                f"Q² = {fnum(q_squared, 3)} "
                "is no better than the "
                "mean-only prediction "
                "reference. No PLS driver "
                "hierarchy is interpreted."
            ),
        )

    return result

def make_answer(
    ipm_result: dict,
    epm_result: dict,
    pls_result: dict,
) -> str:
    """Construct a concise answer-first dashboard conclusion."""

    ipm_direction = (
        ipm_result[
            "direction"
        ]
    )

    if (
        ipm_direction[
            "opposite"
        ]
        > 0
    ):

        first_sentence = (
            "Preference is heterogeneous: "
            f"{ipm_direction['opposite']} of "
            f"{ipm_direction['n']} usable "
            "IPM consumer vectors point "
            "more than 90° from the "
            "sample-average 2D preference "
            "direction."
        )

    else:

        first_sentence = (
            "Consumer preference vectors "
            "vary across the internal "
            "preference map, although none "
            "points more than 90° opposite "
            "the sample-average 2D "
            "preference direction."
        )

    if (
        epm_result
        .get(
            "available"
        )
        and
        not epm_result
        .get(
            "hard"
        )
    ):

        if epm_result[
            "align"
        ]:

            top_names = (
                " and ".join(
                    friendly_attribute(
                        attribute
                    )
                    for (
                        attribute,
                        _
                    )
                    in epm_result[
                        "align"
                    ][
                        :2
                    ]
                )
            )

            second_sentence = (
                "In the measured sensory "
                "space, the sample-average "
                "preference direction aligns "
                "most closely with "
                f"{top_names}; individual "
                "consumer directions are "
                "preserved, so this is not "
                "a single causal driver story."
            )

        else:

            second_sentence = (
                "External preference mapping "
                "relates preference to the "
                "measured sensory space, but "
                "no single sensory direction "
                "is appropriate to emphasize."
            )

        if epm_result[
            "identical"
        ]:

            second_sentence += (
                " At least one product pair "
                "is indistinguishable on the "
                "measured sensory profile but "
                "differs in liking, showing "
                "that the measured sensory "
                "variables do not explain "
                "every acceptance difference."
            )

    else:

        second_sentence = (
            "External preference mapping "
            "sensory interpretation is "
            "unavailable because the "
            "measured sensory-input gate "
            "did not pass."
        )

    third_sentence = ""

    if pls_result[
        "requested"
    ]:

        if (
            pls_result
            .get(
                "available"
            )
            and
            not pls_result
            .get(
                "hard"
            )
        ):

            unstable_count = int(
                (
                    ~pls_result[
                        "coefficients"
                    ][
                        "stable"
                    ]
                )
                .sum()
            )

            third_sentence = (
                "Optional PLS achieved "
                "leave-one-product-out "
                f"Q² = {fnum(pls_result['q2'], 2)}; "
                f"{unstable_count} "
                "coefficient(s) are "
                "sign-unstable and cannot "
                "be ranked as stable "
                "positive or negative "
                "sensory directions."
            )

        elif pls_result.get(
            "hard"
        ):

            third_sentence = (
                "The optional PLS layer "
                "failed its validation stop "
                "rule, so no PLS driver "
                "hierarchy is reported."
            )

    return (
        " ".join(
            sentence
            for sentence
            in (
                first_sentence,
                second_sentence,
                third_sentence,
            )
            if sentence
        )
    )

def analysis_status(
    capability: dict,
    ipm_result: dict,
    epm_result: dict,
    pls_result: dict,
) -> str:
    """Build a compact overall status string."""

    flags = []

    if (
        capability[
            "status"
        ]
        != "SUPPORTED"
    ):

        flags.append(
            capability[
                "status"
            ]
        )

    if (
        ipm_result[
            "map_status"
        ]
        == "WARNING"
    ):

        flags.append(
            "IPM 2D CAUTION"
        )

    if epm_result.get(
        "hard"
    ):

        flags.append(
            "EPM UNAVAILABLE"
        )

    elif (
        epm_result
        .get(
            "map_status"
        )
        == "WARNING"
    ):

        flags.append(
            "EPM 2D CAUTION"
        )

    if pls_result.get(
        "hard"
    ):

        flags.append(
            "PLS DRIVER HIERARCHY HARD STOP"
        )

    elif pls_result.get(
        "available"
    ):

        unstable_count = int(
            (
                ~pls_result[
                    "coefficients"
                ][
                    "stable"
                ]
            )
            .sum()
        )

        if unstable_count:

            flags.append(
                (
                    f"{unstable_count} "
                    "PLS COEFFICIENT(S) "
                    "UNSTABLE"
                )
            )

    if not flags:

        return (
            "Exploratory diagnostics acceptable"
        )

    return (
        "; ".join(
            flags
        )
    )

def scale_vectors_to_products(
    product_xy: np.ndarray,
    vector_sets: list[
        np.ndarray | None
    ],
) -> tuple[
    list[
        np.ndarray | None
    ],
    float,
]:
    """
    Scale vectors for display only.

    The multiplication changes arrow length for readability, not direction or
    the underlying analytical values.
    """

    product_radius = max(
        float(
            np.max(
                np.linalg.norm(
                    product_xy,
                    axis=1,
                )
            )
        ),
        ZERO_TOL,
    )

    scaled_sets = []

    display_factors = [
        0.78,
        0.70,
        0.62,
    ]

    for index, array in enumerate(
        vector_sets
    ):

        if (
            array is None
            or not len(
                array
            )
        ):

            scaled_sets.append(
                array
            )

            continue

        radius = max(
            float(
                np.max(
                    np.linalg.norm(
                        array,
                        axis=1,
                    )
                )
            ),
            ZERO_TOL,
        )

        factor = (
            display_factors[
                min(
                    index,
                    len(
                        display_factors
                    )
                    - 1,
                )
            ]
        )

        scaled_sets.append(
            array
            * (
                factor
                * product_radius
                / radius
            )
        )

    maxima = [
        np.max(
            np.abs(
                product_xy
            )
        )
    ]

    maxima.extend(
        np.max(
            np.abs(
                array
            )
        )
        for array
        in scaled_sets
        if (
            array is not None
            and len(
                array
            )
        )
    )

    plot_extent = max(
        float(
            max(
                maxima
            )
        )
        * 1.18,
        1.0,
    )

    return (
        scaled_sets,
        plot_extent,
    )

def build_diagnostic_rows(

        diagnostic,
        observed,
        threshold,

                action,

            "status"

            "and appropriately defined"

            "status"

            "do not substitute neighboring "
            "constructs."
        ),
    )

        (
            "Duplicate consumer × "
            "product cells"
        ),
        design[
            "duplicates"
        ],

                "duplicates"
            ]

        ),

            "One score is required per "
            "consumer × product."
        ),
    )

            f"{design['missing_cells']} "
            "missing cell(s)"
        ),

        (
            "PASS"
            if not design[
                "missing_cells"
            ]

        ),

            "IPM requires complete "
            "consumer preference profiles."
        ),
    )

            "missing"

        (
            "1–9 hedonic scale; "
            "complete"

        (
            "PASS"
            if (
                not design[
                    "liking_invalid"
                ]
                and
                not design[
                    "liking_missing"
                ]
            )

        ),

        len(
            ipm_result[
                "zero_consumers"

            ]

            "G9-C 2D thresholds: "
            "60% / 50% / 40%"
        ),
        ipm_result[
            "map_status"
        ],
        ipm_result[
            "map_message"
        ],
    )

    for note in descriptive_notes:

        add(
            "Measured sensory input",
            note,
            (
                "Numeric, complete, "
                "0–10 measured intensity"
            ),
            "WARNING",
            (
                "Excluded rather than "
                "repaired, inferred, or proxied."
            ),
        )

    if epm_result.get(
        "hard"
    ):

        add(
            "EPM eligibility",
            epm_result[
                "reason"
            ],
            (
                "Valid measured "
                "product sensory space"
            ),
            "HARD STOP",
            (
                "Suppress EPM interpretation; "
                "valid IPM may remain."
            ),
        )

    else:

        if epm_result[
            "excluded"
        ]:

            add(

                (
                    ", ".join(
                        epm_result[
                            "excluded"
                        ]

                ),

                ),

                (
                    "Excluded before PCA "
                    "and explicitly disclosed."
                ),
            )

        add(

                f"{len(epm_result['attrs'])} "
                "attributes × "
                f"{len(epm_result['sensory'])} "
                "products"
            ),
            (
                "Standardize measured "

                "sensory PCA"
            ),
            "PASS",
            (
                "StandardScaler applied "

            ),

        add(

            add(

                (

                ),

                    "unmeasured predictor"
                ),

                (

                ),

    if pls_result[

                "q2_by_components",
                {},

        ):

            add(
                (
                    "PLS candidate Q² "
                    f"— {components} "
                    "component(s)"
                ),
                fnum(
                    q_squared,
                    3,
                ),
                (
                    "> 0 versus mean-only "
                    "LOPO reference"
                ),
                (
                    "PASS"
                    if (
                        np.isfinite(
                            q_squared
                        )
                        and
                        q_squared
                        > 0
                    )
                    else "HARD STOP"
                ),
                (
                    "Candidate model is "
                    "used as a validation "
                    "diagnostic only until "
                    "a final model passes."
                ),
            )

        if np.isfinite(
            pls_result
            .get(
                "q2",
                np.nan,
            )
        ):

            add(
                "Selected PLS LOPO Q²",
                fnum(
                    pls_result[
                        "q2"
                    ],
                    3,
                ),
                "> 0",
                (
                    "PASS"
                    if pls_result[
                        "q2"
                    ]
                    > 0
                    else "HARD STOP"
                ),
                (
                    "Q² ≤ 0 suppresses "
                    "PLS driver hierarchy."
                ),
            )

        elif pls_result.get(
            "hard"
        ):

            add(
                "PLS validation",
                pls_result[
                    "reason"
                ],
                (
                    "Valid LOPO "
                    "prediction result"
                ),
                "HARD STOP",
                (
                    "No PLS driver hierarchy."
                ),
            )

        if (
            pls_result
            .get(
                "coefficients"
            )
            is not None
        ):

            for _, row in (
                pls_result[
                    "coefficients"
                ]
                .iterrows()
            ):

                add(
                    (
                        "PLS sign stability "
                        "— "
                        f"{friendly_attribute(row['attribute'])}"
                    ),
                    fpct(
                        row[
                            "sign_change"
                        ]
                    ),
                    (
                        "≤25% LOPO folds "
                        "change coefficient sign"
                    ),
                    (
                        "PASS"
                        if row[
                            "stable"
                        ]
                        else "UNSTABLE"
                    ),
                    (
                        "An unstable coefficient "
                        "cannot be ranked as "
                        "a stable positive or "
                        "negative sensory direction."
                    ),
                )

    return rows

def main() -> None:
    """Run the complete standalone SOP-09 workflow."""

    output_path = (
        output_directory()
    )

    dashboard_path = (
        output_path
        / DASHBOARD_FILENAME
    )

    source = None

    # ---------------------------------------------------------------------
    # DATA-FILE DISCOVERY + LOAD
    # ---------------------------------------------------------------------

    try:

        source = (
            resolve_data_path()
        )

        raw_df = (
            pd.read_csv(
                source
            )
        )

    except Exception as exc:

        dashboard_html = (
            build_stop_dashboard(
                (
                    "Dataset could not "
                    "be loaded."
                ),
                str(
                    exc
                ),
                (
                    source
                    or ""
                ),
            )
        )

        write_dashboard(
            dashboard_path,
            dashboard_html,
        )

        print_console_summary(
            None,
            (
                "SOP-09 could not begin "
                "because the dataset "
                "could not be loaded."
            ),
            "HARD STOP",
            dashboard_path,
        )

        maybe_open_dashboard(
            dashboard_path
        )

        return

    # ---------------------------------------------------------------------
    # DATASET CAPABILITY GATE
    # ---------------------------------------------------------------------

    capability = (
        capability_gate(
            raw_df
        )
    )

    if (
        capability[
            "status"
        ]
        == "UNSUPPORTED"
    ):

        reason = (
            "Missing required SOP-09 "
            "core input(s): "
            + ", ".join(
                capability[
                    "core_missing"
                ]
            )
            + ". No neighboring construct "
              "will be substituted."
        )

        dashboard_html = (
            build_stop_dashboard(
                (
                    "Dataset does not "
                    "support SOP-09 IPM."
                ),
                reason,
                str(
                    source
                ),
            )
        )

    # ---------------------------------------------------------------------
    # STUDY-DESIGN VALIDATION
    # ---------------------------------------------------------------------

    (
        df,
        design,
    ) = (
        validate_design(
            raw_df
        )
    )

    # ---------------------------------------------------------------------

    (

    ) = (

        )

        maybe_open_dashboard(
            dashboard_path
        )

        return

    # ---------------------------------------------------------------------
    # STEP 2 — EXTERNAL PREFERENCE MAPPING
    # ---------------------------------------------------------------------

    epm_result = (
        run_epm(
            df,

                "preference"
            ],

    # STEP 3 — OPTIONAL PRODUCT-LEVEL PLS SENSITIVITY LAYER

    pls_result = (

def fixed_effect_covariance(result) -> pd.DataFrame:
    names = list(result.fe_params.index)
    cov = result.cov_params()
    if isinstance(cov, pd.DataFrame):
        return cov.loc[names, names]

    arr = np.asarray(cov, dtype=float)
    k = len(names)

    return pd.DataFrame(
        arr[:k, :k],
        index=names,
        columns=names,
    )

def load_data(path: Path) -> pd.DataFrame:
    if not path.exists():
        raise FileNotFoundError(
            f"Data file not found: {path}\n"
            f"Expected default location: "
            f"{script_directory() / DATA_FILENAME}"
        )

    df = pd.read_csv(path)

    # Common accidental CSV index columns are ignored only when clearly unnamed.
    unnamed = [
        c
        for c in df.columns
        if str(c).lower().startswith("unnamed:")
    ]

    if unnamed:
        df = df.drop(columns=unnamed)

    return df

def capability_gate(
    df: pd.DataFrame,
    target_attribute: str,
) -> Tuple[str, str]:

    required = [
        PANELIST,
        PRODUCT,
        OUTCOME,
        target_attribute,
    ]

    missing = [
        c
        for c in required
        if c not in df.columns
    ]

    if missing:
        raise CapabilityStop(
            "Required field(s) missing: "
            + ", ".join(missing)
            + ". SOP-10 cannot substitute another sensory measure "
            "or another business construct for a missing requested "
            "predictor/outcome."
        )

    if target_attribute not in APPROVED_DESCRIPTIVE_ATTRIBUTES:
        raise CapabilityStop(
            f"TARGET_ATTRIBUTE='{target_attribute}' is not in this "
            "project's approved measured descriptive-attribute list. "
            "Add it only if the governing schema explicitly defines "
            "it as a measured quantitative sensory intensity."
        )

    return (
        "SUPPORTED",
        APPROVED_DESCRIPTIVE_ATTRIBUTES[target_attribute],
    )

def prepare_analysis_data(
    df: pd.DataFrame,
    target_attribute: str,
) -> Tuple[pd.DataFrame, List[str], int]:

    work = df[
        [
            PANELIST,
            PRODUCT,
            OUTCOME,
            target_attribute,
        ]
    ].copy()

    warnings_out: List[str] = []

    # Duplicate consumer × product rows would make the evaluation unit
    # ambiguous. Never silently average them.
    dupes = work.duplicated(
        [PANELIST, PRODUCT],
        keep=False,
    )

    if dupes.any():
        examples = (
            work.loc[
                dupes,
                [PANELIST, PRODUCT],
            ]
            .drop_duplicates()
            .head(8)
        )

        example_text = ", ".join(
            f"{r[PANELIST]}×{r[PRODUCT]}"
            for _, r in examples.iterrows()
        )

        raise DesignStop(
            f"Found {int(dupes.sum())} rows involved in duplicate "
            f"{PANELIST} × {PRODUCT} cells "
            f"(examples: {example_text}). "
            "The script will not silently average duplicates."
        )

    original_nonmissing_y = work[OUTCOME].notna()
    original_nonmissing_x = work[target_attribute].notna()

    work[OUTCOME] = pd.to_numeric(
        work[OUTCOME],
        errors="coerce",
    )

    work[target_attribute] = pd.to_numeric(
        work[target_attribute],
        errors="coerce",
    )

    invalid_y = int(
        (
            original_nonmissing_y
            & work[OUTCOME].isna()
        ).sum()
    )

    invalid_x = int(
        (
            original_nonmissing_x
            & work[target_attribute].isna()
        ).sum()
    )

    if invalid_y:
        raise CapabilityStop(
            f"{invalid_y} non-numeric value(s) "
            f"found in {OUTCOME}."
        )

    if invalid_x:
        raise CapabilityStop(
            f"{invalid_x} non-numeric value(s) "
            f"found in {target_attribute}."
        )

    y_bad = work[OUTCOME].dropna()

    y_bad = y_bad[
        (y_bad < LIKING_MIN)
        | (y_bad > LIKING_MAX)
    ]

    if not y_bad.empty:
        raise CapabilityStop(
            f"{OUTCOME} contains {len(y_bad)} value(s) "
            "outside the approved 1–9 hedonic scale."
        )

    x_bad = work[target_attribute].dropna()

    x_bad = x_bad[
        (x_bad < DESCRIPTIVE_MIN)
        | (x_bad > DESCRIPTIVE_MAX)
    ]

    if not x_bad.empty:
        raise CapabilityStop(
            f"{target_attribute} contains {len(x_bad)} value(s) "
            "outside the approved 0–10 descriptive-intensity scale."
        )

    before = len(work)

    work = work.dropna(
        subset=[
            PANELIST,
            PRODUCT,
            OUTCOME,
            target_attribute,
        ]
    ).copy()

    missing_rows = (
        before - len(work)
    )

    if missing_rows:
        warnings_out.append(
            f"Excluded {missing_rows} row(s) with missing "
            f"required values; {len(work)} complete evaluation "
            "rows remained."
        )

    if work.empty:
        raise DesignStop(
            "No complete rows remain for SOP-10."
        )

    if work[PANELIST].nunique() < 2:
        raise DesignStop(
            "Fewer than two consumers remain."
        )

    if work[PRODUCT].nunique() < 2:
        warnings_out.append(
            "Only one product remains in the modeled data. "
            "Product-level confounding cannot be assessed "
            "descriptively."
        )

    # Repeated-measures evidence: at least some consumers must
    # contribute >1 row.
    evals_per_panelist = (
        work.groupby(
            PANELIST,
            observed=True,
        )
        .size()
    )

    if int(
        (
            evals_per_panelist >= 2
        ).sum()
    ) < 2:
        raise DesignStop(
            "The retained data do not contain a usable "
            "repeated-measures structure: fewer than two "
            "consumers contribute at least two evaluations."
        )

    if work[OUTCOME].nunique() < 2:
        raise DesignStop(
            f"{OUTCOME} has no usable variation."
        )

    if work[target_attribute].nunique() < 2:
        raise DesignStop(
            f"{target_attribute} has no usable variation."
        )

    # Preserve IDs as strings for stable categorical handling.
    work[PANELIST] = (
        work[PANELIST]
        .astype(str)
    )

    work[PRODUCT] = (
        work[PRODUCT]
        .astype(str)
    )

    return (
        work,
        warnings_out,
        missing_rows,
    )

def apply_g9d_eligibility(
    work: pd.DataFrame,
    target_attribute: str,
) -> None:

    n = len(work)

    n_unique = int(
        work[target_attribute]
        .nunique()
    )

    failures = []

    if n < MIN_ANALYSIS_N:
        failures.append(
            f"N={n} < {MIN_ANALYSIS_N}"
        )

    if n_unique < MIN_DISTINCT_PREDICTOR_VALUES:
        failures.append(
            f"{n_unique} distinct predictor values "
            f"< {MIN_DISTINCT_PREDICTOR_VALUES}"
        )

    if failures:
        raise DesignStop(
            "G9-D quadratic eligibility HARD STOP: "
            + "; ".join(failures)
            + ". A quadratic model is not fit when "
            "observed predictor support is insufficient."
        )

def fit_mixed_model(
    formula: str,
    work: pd.DataFrame,
) -> FitBundle:

    attempts: List[
        Dict[str, object]
    ] = []

    last_error = ""

    for optimizer in OPTIMIZER_SEQUENCE:

        caught: List[str] = []

        try:
            with warnings.catch_warnings(
                record=True
            ) as ws:

                warnings.simplefilter(
                    "always"
                )

                model = smf.mixedlm(
                    formula,
                    data=work,
                    groups=work[PANELIST],
                    re_formula="1",
                )

                result = model.fit(
                    reml=False,
                    method=optimizer,
                    maxiter=2000,
                    disp=False,
                    full_output=True,
                )

                caught = [
                    str(w.message)
                    for w in ws
                ]

            converged = bool(
                getattr(
                    result,
                    "converged",
                    False,
                )
            )

            attempts.append(
                {
                    "optimizer": optimizer,
                    "converged": converged,
                    "warnings": caught,
                    "error": "",
                }
            )

            if converged:
                return FitBundle(
                    result=result,
                    optimizer=optimizer,
                    warnings=caught,
                    attempts=attempts,
                )

            last_error = (
                "Optimizer returned "
                "without convergence."
            )

        except Exception as exc:

            last_error = str(exc)

            attempts.append(
                {
                    "optimizer": optimizer,
                    "converged": False,
                    "warnings": caught,
                    "error": last_error,
                }
            )

    raise InferenceStop(
        "MixedLM failed to converge using the approved "
        f"optimizer sequence {OPTIMIZER_SEQUENCE}. "
        f"Last error: {last_error}"
    )

def run_g9a_diagnostics(
    work: pd.DataFrame,
    fit: FitBundle,
) -> Tuple[
    pd.DataFrame,
    bool,
    bool,
    List[str],
]:

    result = fit.result

    warning_blob = (
        " | ".join(
            fit.warnings
        ).lower()
    )

    rows: List[
        Dict[str, object]
    ] = []

    hard_stop = False
    sensitivity_required = False
    triggers: List[str] = []

    # =========================================================================
    # A1 — OPTIMIZER CONVERGENCE
    # =========================================================================

    converged = bool(
        getattr(
            result,
            "converged",
            False,
        )
    )

    if converged:

        rows.append(
            diag_row(
                "A1 Optimizer convergence",
                True,
                "result.converged == True",
                "PASS",
                "The requested mixed model reached "
                "an optimizer solution.",
                "Proceed.",
            )
        )

    else:

        hard_stop = True

        rows.append(
            diag_row(
                "A1 Optimizer convergence",
                False,
                "result.converged == True",
                "HARD STOP",
                "The mixed model did not converge.",
                "Suppress inferential interpretation.",
            )
        )

    # =========================================================================
    # EXTRACT RANDOM-EFFECT AND RESIDUAL VARIANCE EARLY
    #
    # This is intentionally done BEFORE A2.
    #
    # A random-effect variance near zero can cause statsmodels to describe
    # the random-effects covariance as singular. Under G9-A, that is a
    # boundary condition handled by A3/A4 + sensitivity analysis, not
    # automatically a fatal covariance failure.
    # =========================================================================

    try:

        cov_re = np.asarray(
            result.cov_re,
            dtype=float,
        )

    except Exception:

        cov_re = np.array(
            [[np.nan]],
            dtype=float,
        )

    cov_re_finite = bool(
        np.isfinite(
            cov_re
        ).all()
    )

    random_var = (
        float(
            cov_re[
                0,
                0,
            ]
        )
        if (
            cov_re.size
            and
            np.isfinite(
                cov_re[
                    0,
                    0,
                ]
            )
        )
        else float(
            "nan"
        )
    )

    residual_var = (
        float(
            result.scale
        )
        if np.isfinite(
            result.scale
        )
        else float(
            "nan"
        )
    )

    denom = (
        random_var
        + residual_var
    )

    icc = (
        random_var
        / denom
        if (
            np.isfinite(
                denom
            )
            and
            denom > 0
        )
        else float(
            "nan"
        )
    )

    boundary_random_effect = bool(
        np.isfinite(
            random_var
        )
        and
        random_var
        < BOUNDARY_RANDOM_VARIANCE
    )

    near_zero_icc = bool(
        np.isfinite(
            icc
        )
        and
        icc
        < NEAR_ZERO_ICC
    )

    # =========================================================================
    # A2 — COVARIANCE / HESSIAN VALIDITY
    #
    # IMPORTANT:
    # A singular random-effects covariance by itself is NOT a hard stop
    # when the estimated random-intercept variance is essentially zero.
    #
    # That situation is handled below by A3/A4 and requires the approved
    # panelist fixed-effect OLS + HC3 sensitivity analysis.
    #
    # A2 remains a hard stop for:
    #   • unusable/non-finite random covariance not explained by boundary
    #   • unusable fixed-effect covariance matrix
    #   • clearly invalid Hessian/information matrix not attributable only
    #     to a boundary random effect
    # =========================================================================

    min_cov_re_eig = float(
        "nan"
    )

    if (
        cov_re_finite
        and
        cov_re.size
    ):

        try:

            min_cov_re_eig = float(
                np.min(
                    np.linalg.eigvalsh(
                        (
                            cov_re
                            + cov_re.T
                        )
                        / 2
                    )
                )
            )

        except Exception:

            min_cov_re_eig = float(
                "nan"
            )

    singular_warning = (
        "random effects covariance is singular"
        in warning_blob
        or
        "covariance matrix is singular"
        in warning_blob
    )

    hessian_warning = (
        "hessian"
        in warning_blob
        and
        "not positive definite"
        in warning_blob
    )

    # -------------------------------------------------------------------------
    # Fixed-effect covariance validity
    # -------------------------------------------------------------------------

    cov_params_valid = True
    cov_params_error = ""

    try:

        cov_fe = (
            fixed_effect_covariance(
                result
            )
            .to_numpy(
                dtype=float
            )
        )

        cov_params_valid = bool(
            np.isfinite(
                cov_fe
            ).all()
            and
            np.all(
                np.diag(
                    cov_fe
                )
                >= 0
            )
        )

    except Exception as exc:

        cov_params_valid = False

        cov_params_error = str(
            exc
        )

    # -------------------------------------------------------------------------
    # Hessian / information matrix
    # -------------------------------------------------------------------------

    hessian_singular = False
    hessian_valid = False

    min_neg_hessian_eig = float(
        "nan"
    )

    hessian_error = ""

    try:

        (
            hess,
            hessian_singular,
        ) = result.model.hessian(
            result.params_object
        )

        hess = np.asarray(
            hess,
            dtype=float,
        )

        neg_hess = -(
            (
                hess
                + hess.T
            )
            / 2
        )

        if np.isfinite(
            neg_hess
        ).all():

            eigvals = np.linalg.eigvalsh(
                neg_hess
            )

            min_neg_hessian_eig = float(
                np.min(
                    eigvals
                )
            )

            hessian_valid = bool(
                min_neg_hessian_eig
                > NUMERIC_TOL
            )

    except Exception as exc:

        hessian_error = str(
            exc
        )

    # -------------------------------------------------------------------------
    # Determine whether the apparent singularity is simply the known
    # random-effect boundary condition.
    # -------------------------------------------------------------------------

    boundary_explains_singularity = bool(
        boundary_random_effect
        and
        singular_warning
        and
        cov_params_valid
    )

    boundary_explains_hessian_warning = bool(
        boundary_random_effect
        and
        hessian_warning
        and
        cov_params_valid
        and
        converged
    )

    boundary_explains_hessian_singular = bool(
        boundary_random_effect
        and
        bool(
            hessian_singular
        )
        and
        cov_params_valid
        and
        converged
    )

    boundary_explains_nonpositive_hessian = bool(
        boundary_random_effect
        and
        not hessian_valid
        and
        cov_params_valid
        and
        converged
    )

    # -------------------------------------------------------------------------
    # Genuine A2 failures
    # -------------------------------------------------------------------------

    random_covariance_unusable = bool(
        not cov_re_finite
    )

    random_covariance_negative = bool(
        np.isfinite(
            min_cov_re_eig
        )
        and
        min_cov_re_eig
        < -NUMERIC_TOL
    )

    fatal_singular_warning = bool(
        singular_warning
        and
        not boundary_explains_singularity
    )

    fatal_hessian_warning = bool(
        hessian_warning
        and
        not boundary_explains_hessian_warning
    )

    fatal_hessian_singular = bool(
        hessian_singular
        and
        not boundary_explains_hessian_singular
    )

    fatal_hessian_invalid = bool(
        not hessian_valid
        and
        not boundary_explains_nonpositive_hessian
    )

    a2_fail = bool(
        random_covariance_unusable
        or
        random_covariance_negative
        or
        fatal_singular_warning
        or
        fatal_hessian_warning
        or
        fatal_hessian_singular
        or
        fatal_hessian_invalid
        or
        not cov_params_valid
    )

    if a2_fail:

        hard_stop = True

        detail = (
            "min random-cov eigenvalue="
            f"{fmt_num(min_cov_re_eig, 4)}; "
            "min -Hessian eigenvalue="
            f"{fmt_num(min_neg_hessian_eig, 4)}"
        )

        if hessian_error:

            detail += (
                "; Hessian error="
                f"{hessian_error}"
            )

        if cov_params_error:

            detail += (
                "; covariance error="
                f"{cov_params_error}"
            )

        rows.append(
            diag_row(
                "A2 Covariance / Hessian validity",
                detail,
                "Fixed-effect covariance must be usable; "
                "non-boundary Hessian/covariance failures "
                "are hard stops.",
                "HARD STOP",
                "The model has a numerical covariance or "
                "information-matrix problem that cannot be "
                "explained solely by a near-zero consumer "
                "random effect.",
                "Suppress mixed-model inferential interpretation.",
            )
        )

    elif (
        boundary_random_effect
        and
        (
            singular_warning
            or
            hessian_warning
            or
            bool(
                hessian_singular
            )
            or
            not hessian_valid
        )
    ):

        # Boundary covariance/Hessian behavior is explicitly routed
        # to A3/A4 rather than terminating the analysis here.

        rows.append(
            diag_row(
                "A2 Covariance / Hessian validity",
                (
                    "Boundary random-effect condition detected; "
                    "random variance="
                    f"{fmt_num(random_var, 4)}; "
                    "min -Hessian eigenvalue="
                    f"{fmt_num(min_neg_hessian_eig, 4)}"
                ),
                "Boundary singularity with otherwise usable "
                "fixed effects routes to A3/A4 sensitivity.",
                "WARNING",
                "The random consumer intercept is essentially "
                "on the boundary. Statsmodels may therefore "
                "report a singular covariance or non-positive "
                "Hessian even though the fixed-effect estimates "
                "remain usable enough for the prescribed "
                "sensitivity pathway.",
                "Do not hard stop here. Run the A3/A4-approved "
                "panelist fixed-effect OLS + HC3 sensitivity check.",
            )
        )

    else:

        rows.append(
            diag_row(
                "A2 Covariance / Hessian validity",
                (
                    "min random-cov eig="
                    f"{fmt_num(min_cov_re_eig, 4)}; "
                    "min -Hessian eig="
                    f"{fmt_num(min_neg_hessian_eig, 4)}"
                ),
                "Usable covariance and information matrix",
                "PASS",
                "Model uncertainty calculations are "
                "numerically usable.",
                "Proceed.",
            )
        )

    # =========================================================================
    # A3 — BOUNDARY RANDOM-INTERCEPT VARIANCE
    # =========================================================================

    if boundary_random_effect:

        sensitivity_required = True

        triggers.append(
            "A3 random-intercept variance "
            f"{random_var:.4g} < "
            f"{BOUNDARY_RANDOM_VARIANCE:.2f}"
        )

        rows.append(
            diag_row(
                "A3 Boundary random-intercept variance",
                random_var,
                f"variance < {BOUNDARY_RANDOM_VARIANCE:.2f}",
                "SENSITIVITY REQUIRED",
                "Very little between-consumer intercept "
                "variance was estimated. The mixed model's "
                "random intercept is effectively on the boundary.",
                "Run panelist fixed-effect OLS + HC3 "
                "sensitivity analysis.",
            )
        )

    else:

        rows.append(
            diag_row(
                "A3 Boundary random-intercept variance",
                random_var,
                f"variance < {BOUNDARY_RANDOM_VARIANCE:.2f}",
                "PASS",
                "Random consumer-to-consumer intercept "
                "variation is not at the boundary threshold.",
                "Proceed.",
            )
        )

    # =========================================================================
    # A4 — NEAR-ZERO ICC
    # =========================================================================

    if near_zero_icc:

        sensitivity_required = True

        triggers.append(
            "A4 ICC "
            f"{icc:.4g} < "
            f"{NEAR_ZERO_ICC:.2f}"
        )

        rows.append(
            diag_row(
                "A4 Near-zero ICC",
                icc,
                f"ICC < {NEAR_ZERO_ICC:.2f}",
                "SENSITIVITY REQUIRED",
                "The estimated correlation among repeated "
                "evaluations from the same consumer is nearly zero.",
                "Check robustness with panelist fixed "
                "effects + HC3.",
            )
        )

    else:

        rows.append(
            diag_row(
                "A4 ICC",
                icc,
                f"ICC < {NEAR_ZERO_ICC:.2f} "
                "triggers sensitivity",
                "PASS",
                "The estimated within-consumer dependence "
                "is above the near-zero threshold.",
                "Proceed.",
            )
        )

    # =========================================================================
    # If A1/A2 truly failed, residual diagnostics may themselves be unreliable.
    #
    # We still return the rows already generated so the dashboard explains
    # exactly why inference stopped instead of crashing while trying to derive
    # residual diagnostics from an unusable model.
    # =========================================================================

    if hard_stop:

        return (
            pd.DataFrame(
                rows
            ),
            hard_stop,
            sensitivity_required,
            triggers,
        )

    # =========================================================================
    # A5 — RESIDUAL DISTRIBUTION
    # =========================================================================

    resid = np.asarray(
        result.resid,
        dtype=float,
    )

    if (
        np.isfinite(
            residual_var
        )
        and
        residual_var > 0
    ):

        pearson = (
            resid
            / math.sqrt(
                residual_var
            )
        )

    else:

        pearson = resid

    pearson = pearson[
        np.isfinite(
            pearson
        )
    ]

    if len(
        pearson
    ) >= 3:

        try:

            if len(
                pearson
            ) <= 5000:

                (
                    normality_stat,
                    normality_p,
                ) = stats.shapiro(
                    pearson
                )

                normality_name = (
                    "Shapiro-Wilk"
                )

            else:

                (
                    normality_stat,
                    normality_p,
                ) = stats.normaltest(
                    pearson
                )

                normality_name = (
                    "D'Agostino K²"
                )

        except Exception:

            normality_stat = float(
                "nan"
            )

            normality_p = float(
                "nan"
            )

            normality_name = (
                "formal normality test"
            )

        skewness = (
            float(
                stats.skew(
                    pearson,
                    bias=False,
                )
            )
            if len(
                pearson
            ) > 2
            else float(
                "nan"
            )
        )

        kurtosis = (
            float(
                stats.kurtosis(
                    pearson,
                    fisher=False,
                    bias=False,
                )
            )
            if len(
                pearson
            ) > 3
            else float(
                "nan"
            )
        )

    else:

        normality_stat = float(
            "nan"
        )

        normality_p = float(
            "nan"
        )

        skewness = float(
            "nan"
        )

        kurtosis = float(
            "nan"
        )

        normality_name = (
            "formal normality test"
        )

    normality_status = (
        "WARNING"
        if (
            np.isfinite(
                normality_p
            )
            and
            normality_p
            < NORMALITY_WARNING_P
        )
        else "PASS"
    )

    rows.append(
        diag_row(
            f"A5 Residual normality ({normality_name})",
            normality_p,
            "p < 0.01 = warning",
            normality_status,
            "A small p-value flags departure from a normal "
            "residual shape; by itself it does not invalidate "
            "the model.",
            "Inspect skewness/kurtosis sensitivity thresholds.",
        )
    )

    skew_trigger = (
        np.isfinite(
            skewness
        )
        and
        abs(
            skewness
        )
        > RESIDUAL_SKEW_SENSITIVITY
    )

    kurt_trigger = (
        np.isfinite(
            kurtosis
        )
        and
        kurtosis
        > PEARSON_KURTOSIS_SENSITIVITY
    )

    if (
        skew_trigger
        or
        kurt_trigger
    ):

        sensitivity_required = True

        if skew_trigger:

            triggers.append(
                "A5 |residual skewness|="
                f"{abs(skewness):.3f} > "
                f"{RESIDUAL_SKEW_SENSITIVITY:.1f}"
            )

        if kurt_trigger:

            triggers.append(
                "A5 Pearson kurtosis="
                f"{kurtosis:.3f} > "
                f"{PEARSON_KURTOSIS_SENSITIVITY:.1f}"
            )

    rows.append(
        diag_row(
            "A5 Residual skewness",
            skewness,
            f"|skewness| > "
            f"{RESIDUAL_SKEW_SENSITIVITY:.1f}",
            (
                "SENSITIVITY REQUIRED"
                if skew_trigger
                else "PASS"
            ),
            "Large residual asymmetry can make "
            "model-based uncertainty less robust.",
            (
                "Run panelist fixed-effect OLS + HC3 "
                "sensitivity analysis."
                if skew_trigger
                else "Proceed."
            ),
        )
    )

    rows.append(
        diag_row(
            "A5 Pearson residual kurtosis",
            kurtosis,
            f"kurtosis > "
            f"{PEARSON_KURTOSIS_SENSITIVITY:.1f}",
            (
                "SENSITIVITY REQUIRED"
                if kurt_trigger
                else "PASS"
            ),
            "Very heavy residual tails can make "
            "model-based uncertainty less robust.",
            (
                "Run panelist fixed-effect OLS + HC3 "
                "sensitivity analysis."
                if kurt_trigger
                else "Proceed."
            ),
        )
    )

    # =========================================================================
    # A6 — HETEROSCEDASTICITY BY PRODUCT
    # =========================================================================

    temp = work[
        [
            PRODUCT,
        ]
    ].copy()

    temp[
        "_resid"
    ] = resid

    product_vars = (
        temp.groupby(
            PRODUCT,
            observed=True,
        )[
            "_resid"
        ]
        .var(
            ddof=1
        )
        .dropna()
    )

    positive_vars = product_vars[
        product_vars
        > NUMERIC_TOL
    ]

    if len(
        positive_vars
    ) >= 2:

        variance_ratio = float(
            positive_vars.max()
            / positive_vars.min()
        )

    elif (
        len(
            product_vars
        ) >= 2
        and
        (
            product_vars
            <= NUMERIC_TOL
        ).any()
    ):

        variance_ratio = float(
            "inf"
        )

    else:

        variance_ratio = float(
            "nan"
        )

    groups = [
        g[
            "_resid"
        ]
        .dropna()
        .to_numpy()
        for _, g
        in temp.groupby(
            PRODUCT,
            observed=True,
        )
    ]

    groups = [
        g
        for g
        in groups
        if len(
            g
        ) >= 2
    ]

    if len(
        groups
    ) >= 2:

        try:

            _,
            bf_p = stats.levene(
                *groups,
                center="median",
            )

            bf_p = float(
                bf_p
            )

        except Exception:

            bf_p = float(
                "nan"
            )

    else:

        bf_p = float(
            "nan"
        )

    hetero_warning = bool(
        (
            np.isfinite(
                bf_p
            )
            and
            bf_p
            < HETERO_WARNING_P
        )
        or
        (
            np.isfinite(
                variance_ratio
            )
            and
            variance_ratio
            > HETERO_WARNING_RATIO
        )
        or
        math.isinf(
            variance_ratio
        )
    )

    hetero_sensitivity = bool(
        (
            (
                np.isfinite(
                    bf_p
                )
                and
                bf_p
                < HETERO_WARNING_P
            )
            and
            (
                (
                    np.isfinite(
                        variance_ratio
                    )
                    and
                    variance_ratio
                    > HETERO_WARNING_RATIO
                )
                or
                math.isinf(
                    variance_ratio
                )
            )
        )
        or
        (
            (
                np.isfinite(
                    variance_ratio
                )
                and
                variance_ratio
                > HETERO_SENSITIVITY_RATIO
            )
            or
            math.isinf(
                variance_ratio
            )
        )
    )

    if hetero_sensitivity:

        sensitivity_required = True

        triggers.append(
            "A6 Brown-Forsythe p="
            f"{fmt_p(bf_p)}, "
            "variance ratio="
            f"{fmt_num(variance_ratio, 2)}"
        )

        hetero_status = (
            "SENSITIVITY REQUIRED"
        )

        hetero_action = (
            "Run panelist fixed-effect OLS + HC3 "
            "sensitivity analysis."
        )

    elif hetero_warning:

        hetero_status = (
            "WARNING"
        )

        hetero_action = (
            "Report the unequal-variance warning; "
            "sensitivity is not required by threshold."
        )

    else:

        hetero_status = (
            "PASS"
        )

        hetero_action = (
            "Proceed."
        )

    rows.append(
        diag_row(
            "A6 Brown-Forsythe residual variance test",
            bf_p,
            "p < 0.01 warns; sensitivity when "
            "p < 0.01 AND variance ratio > 2",
            hetero_status,
            "This checks whether residual spread "
            "differs across products.",
            hetero_action,
        )
    )

    rows.append(
        diag_row(
            "A6 Residual variance ratio",
            variance_ratio,
            "> 2 warns; > 4 requires sensitivity",
            hetero_status,
            "This compares the largest product residual "
            "variance with the smallest.",
            hetero_action,
        )
    )

    # =========================================================================
    # A7 — EXTREME RESIDUALS
    # =========================================================================

    abs_r = np.abs(
        pearson
    )

    n_gt3 = int(
        (
            abs_r
            > EXTREME_RESIDUAL_FLAG
        ).sum()
    )

    pct_gt3 = (
        100.0
        * n_gt3
        / len(
            abs_r
        )
        if len(
            abs_r
        )
        else float(
            "nan"
        )
    )

    max_abs = (
        float(
            abs_r.max()
        )
        if len(
            abs_r
        )
        else float(
            "nan"
        )
    )

    extreme_sensitivity = bool(
        (
            np.isfinite(
                pct_gt3
            )
            and
            pct_gt3
            > EXTREME_RESIDUAL_PCT_SENSITIVITY
        )
        or
        (
            np.isfinite(
                max_abs
            )
            and
            max_abs
            > EXTREME_RESIDUAL_HARDER
        )
    )

    if extreme_sensitivity:

        sensitivity_required = True

        triggers.append(
            f"A7 {pct_gt3:.2f}% with "
            "|Pearson residual|>3; "
            f"max={max_abs:.2f}"
        )

        extreme_status = (
            "SENSITIVITY REQUIRED"
        )

        extreme_action = (
            "Run sensitivity analysis; do not "
            "automatically delete observations."
        )

    elif n_gt3 > 0:

        extreme_status = (
            "WARNING"
        )

        extreme_action = (
            "Flag observations; do not "
            "automatically delete them."
        )

    else:

        extreme_status = (
            "PASS"
        )

        extreme_action = (
            "Proceed."
        )

    rows.append(
        diag_row(
            "A7 Extreme Pearson residuals",
            (
                f"{n_gt3} rows "
                f"({fmt_num(pct_gt3, 2)}%); "
                "max |r|="
                f"{fmt_num(max_abs, 2)}"
            ),
            ">1% above 3 OR any above 4 "
            "requires sensitivity",
            extreme_status,
            "A small number of extreme observations "
            "can disproportionately affect a fitted curve.",
            extreme_action,
        )
    )

    return (
        pd.DataFrame(
            rows
        ),
        hard_stop,
        sensitivity_required,
        triggers,
    )

def extract_quadratic_evidence(
    result,
    x_center: float,
    work: pd.DataFrame,
    target_attribute: str,
) -> Dict[str, object]:

    b1 = float(
        result.fe_params.get(
            "x_c",
            np.nan,
        )
    )

    b2 = float(
        result.fe_params.get(
            "x_c_sq",
            np.nan,
        )
    )

    se2 = float(
        result.bse_fe.get(
            "x_c_sq",
            np.nan,
        )
    )

    p2 = float(
        result.pvalues.get(
            "x_c_sq",
            np.nan,
        )
    )

    ci_low = (
        b2
        - Z_CRITICAL
        * se2
        if (
            np.isfinite(
                b2
            )
            and
            np.isfinite(
                se2
            )
        )
        else float(
            "nan"
        )
    )

    ci_high = (
        b2
        + Z_CRITICAL
        * se2
        if (
            np.isfinite(
                b2
            )
            and
            np.isfinite(
                se2
            )
        )
        else float(
            "nan"
        )
    )

    turning = float(
        "nan"
    )

    if (
        np.isfinite(
            b1
        )
        and
        np.isfinite(
            b2
        )
        and
        abs(
            b2
        )
        > NUMERIC_TOL
    ):

        turning = (
            x_center
            - b1
            / (
                2.0
                * b2
            )
        )

    x = (
        work[
            target_attribute
        ]
        .to_numpy(
            dtype=float
        )
    )

    obs_min = float(
        np.min(
            x
        )
    )

    obs_max = float(
        np.max(
            x
        )
    )

    q10 = float(
        np.quantile(
            x,
            OPTIMUM_LOW_QUANTILE,
        )
    )

    q90 = float(
        np.quantile(
            x,
            OPTIMUM_HIGH_QUANTILE,
        )
    )

    n_below = (
        int(
            (
                x
                < turning
            ).sum()
        )
        if np.isfinite(
            turning
        )
        else 0
    )

    n_above = (
        int(
            (
                x
                > turning
            ).sum()
        )
        if np.isfinite(
            turning
        )
        else 0
    )

    checks = {
        "within_observed_range": bool(
            np.isfinite(
                turning
            )
            and
            obs_min
            <= turning
            <= obs_max
        ),
        "within_10th_90th_percentiles": bool(
            np.isfinite(
                turning
            )
            and
            q10
            <= turning
            <= q90
        ),
        "at_least_10_below": (
            n_below
            >= MIN_OBS_EACH_SIDE
        ),
        "at_least_10_above": (
            n_above
            >= MIN_OBS_EACH_SIDE
        ),
    }

    if np.isfinite(
        b2
    ):

        if b2 < 0:
            shape = (
                "inverted-U"
            )
            kind = (
                "maximum"
            )

        elif b2 > 0:
            shape = (
                "U-shaped"
            )
            kind = (
                "minimum"
            )

        else:
            shape = (
                "flat quadratic term"
            )
            kind = ""

    else:

        shape = (
            "unknown"
        )
        kind = ""

    return {
        "b1": b1,
        "b2": b2,
        "se2": se2,
        "p2": p2,
        "ci_low": ci_low,
        "ci_high": ci_high,
        "turning": turning,
        "shape": shape,
        "kind": kind,
        "checks": checks,
        "n_below": n_below,
        "n_above": n_above,
        "obs_min": obs_min,
        "obs_max": obs_max,
        "q10": q10,
        "q90": q90,
    }

def build_curve(
    result,
    work: pd.DataFrame,
    target_attribute: str,
    x_center: float,
) -> pd.DataFrame:

    x_grid = np.linspace(
        work[target_attribute].min(),
        work[target_attribute].max(),
        GRID_POINTS,
    )

    x_c = (
        x_grid
        - x_center
    )

    design = pd.DataFrame(
        {
            "Intercept": np.ones_like(
                x_grid
            ),
            "x_c": x_c,
            "x_c_sq": x_c ** 2,
        }
    )

    fe_names = list(
        result.fe_params.index
    )

    X = (
        design.reindex(
            columns=fe_names,
            fill_value=0.0,
        )
        .to_numpy(
            dtype=float
        )
    )

    beta = (
        result.fe_params
        .reindex(
            fe_names
        )
        .to_numpy(
            dtype=float
        )
    )

    cov = (
        fixed_effect_covariance(
            result
        )
        .reindex(
            index=fe_names,
            columns=fe_names,
        )
        .to_numpy(
            dtype=float
        )
    )

    pred = (
        X @ beta
    )

    var = np.einsum(
        "ij,jk,ik->i",
        X,
        cov,
        X,
    )

    se = np.sqrt(
        np.clip(
            var,
            0.0,
            None,
        )
    )

    return pd.DataFrame(
        {
            "x": x_grid,
            "fit": pred,
            "low": (
                pred
                - Z_CRITICAL
                * se
            ),
            "high": (
                pred
                + Z_CRITICAL
                * se
            ),
        }
    )

def run_sensitivity(
    work: pd.DataFrame,
) -> Dict[str, object]:

    # Approved repeated-measures sensitivity:
    # panelist fixed-effect OLS + HC3.
    #
    # Same complete rows are used for linear and quadratic
    # fits so AIC is comparable.

    linear = smf.ols(
        f"{OUTCOME} ~ x_c + C({PANELIST})",
        data=work,
    ).fit(
        cov_type="HC3"
    )

    quadratic = smf.ols(
        f"{OUTCOME} ~ x_c + x_c_sq + C({PANELIST})",
        data=work,
    ).fit(
        cov_type="HC3"
    )

    b1 = float(
        quadratic.params.get(
            "x_c",
            np.nan,
        )
    )

    b2 = float(
        quadratic.params.get(
            "x_c_sq",
            np.nan,
        )
    )

    p2 = float(
        quadratic.pvalues.get(
            "x_c_sq",
            np.nan,
        )
    )

    delta_aic = float(
        linear.aic
        - quadratic.aic
    )

    curvature = bool(
        np.isfinite(
            p2
        )
        and
        p2
        < CURVATURE_P_THRESHOLD
        and
        delta_aic
        >= MIN_DELTA_AIC
    )

    turning = float(
        "nan"
    )

    if (
        np.isfinite(
            b1
        )
        and
        np.isfinite(
            b2
        )
        and
        abs(
            b2
        )
        > NUMERIC_TOL
    ):

        turning = float(
            work[
                "_x_center"
            ].iloc[
                0
            ]
            - b1
            / (
                2
                * b2
            )
        )

    x = (
        work[
            TARGET_ATTRIBUTE
        ]
        .to_numpy(
            dtype=float
        )
    )

    q10 = float(
        np.quantile(
            x,
            OPTIMUM_LOW_QUANTILE,
        )
    )

    q90 = float(
        np.quantile(
            x,
            OPTIMUM_HIGH_QUANTILE,
        )
    )

    checks = {
        "within_observed_range": bool(
            np.isfinite(
                turning
            )
            and
            x.min()
            <= turning
            <= x.max()
        ),
        "within_10th_90th_percentiles": bool(
            np.isfinite(
                turning
            )
            and
            q10
            <= turning
            <= q90
        ),
        "at_least_10_below": (
            int(
                (
                    x
                    < turning
                ).sum()
            )
            >= MIN_OBS_EACH_SIDE
            if np.isfinite(
                turning
            )
            else False
        ),
        "at_least_10_above": (
            int(
                (
                    x
                    > turning
                ).sum()
            )
            >= MIN_OBS_EACH_SIDE
            if np.isfinite(
                turning
            )
            else False
        ),
    }

    optimum = bool(
        curvature
        and
        b2 < 0
        and
        all(
            checks.values()
        )
    )

    return {
        "linear_aic": float(
            linear.aic
        ),
        "quadratic_aic": float(
            quadratic.aic
        ),
        "delta_aic": delta_aic,
        "quadratic_coef": b2,
        "quadratic_p": p2,
        "curvature_supported": curvature,
        "turning_point": turning,
        "turning_point_supported": optimum,
        "turning_point_checks": checks,
    }

def compare_sensitivity(
    primary: AnalysisResult,
    sensitivity: Dict[str, object],
) -> Tuple[
    bool,
    List[str],
]:

    reasons: List[str] = []

    sens_curv = bool(
        sensitivity.get(
            "curvature_supported",
            False,
        )
    )

    if (
        sens_curv
        != primary.curvature_supported
    ):

        reasons.append(
            "Primary MixedLM and panelist-fixed-effect "
            "HC3 sensitivity disagree on whether G9-D "
            "curvature is supported."
        )

    p_b2 = (
        primary.quadratic_coef
    )

    s_b2 = float(
        sensitivity.get(
            "quadratic_coef",
            np.nan,
        )
    )

    if (
        primary.curvature_supported
        and
        sens_curv
        and
        np.isfinite(
            p_b2
        )
        and
        np.isfinite(
            s_b2
        )
    ):

        if (
            np.sign(
                p_b2
            )
            !=
            np.sign(
                s_b2
            )
        ):

            reasons.append(
                "Primary and sensitivity models disagree "
                "on curvature direction."
            )

    sens_opt = bool(
        sensitivity.get(
            "turning_point_supported",
            False,
        )
    )

    if (
        primary.turning_point_supported
        != sens_opt
    ):

        reasons.append(
            "Primary and sensitivity models disagree on "
            "whether a supported interior liking optimum "
            "can be claimed."
        )

    return (
        bool(
            reasons
        ),
        reasons,
    )

def run_analysis(
    df: pd.DataFrame,
) -> Tuple[
    AnalysisResult,
    pd.DataFrame,
]:

    out = AnalysisResult(
        target_attribute=TARGET_ATTRIBUTE
    )

    try:

        (
            out.capability_status,
            out.target_label,
        ) = capability_gate(
            df,
            TARGET_ATTRIBUTE,
        )

        (
            work,
            design_warnings,
            missing_rows,
        ) = prepare_analysis_data(
            df,
            TARGET_ATTRIBUTE,
        )

        out.design_warnings = (
            design_warnings
        )

        out.missing_rows = (
            missing_rows
        )

        out.n_rows = len(
            work
        )

        out.n_panelists = int(
            work[
                PANELIST
            ].nunique()
        )

        out.n_products = int(
            work[
                PRODUCT
            ].nunique()
        )

        out.n_unique_x = int(
            work[
                TARGET_ATTRIBUTE
            ].nunique()
        )

        apply_g9d_eligibility(
            work,
            TARGET_ATTRIBUTE,
        )

        out.x_center = float(
            work[
                TARGET_ATTRIBUTE
            ].mean()
        )

        work[
            "_x_center"
        ] = out.x_center

        work[
            "x_c"
        ] = (
            work[
                TARGET_ATTRIBUTE
            ]
            - out.x_center
        )

        work[
            "x_c_sq"
        ] = (
            work[
                "x_c"
            ]
            ** 2
        )

        out.observed_min = float(
            work[
                TARGET_ATTRIBUTE
            ].min()
        )

        out.observed_max = float(
            work[
                TARGET_ATTRIBUTE
            ].max()
        )

        out.q10 = float(
            work[
                TARGET_ATTRIBUTE
            ].quantile(
                OPTIMUM_LOW_QUANTILE
            )
        )

        out.q90 = float(
            work[
                TARGET_ATTRIBUTE
            ].quantile(
                OPTIMUM_HIGH_QUANTILE
            )
        )

        linear_formula = (
            f"{OUTCOME} ~ x_c"
        )

        quadratic_formula = (
            f"{OUTCOME} ~ x_c + x_c_sq"
        )

        out.linear_fit = (
            fit_mixed_model(
                linear_formula,
                work,
            )
        )

        out.quadratic_fit = (
            fit_mixed_model(
                quadratic_formula,
                work,
            )
        )

        out.linear_aic = float(
            out.linear_fit
            .result
            .aic
        )

        out.quadratic_aic = float(
            out.quadratic_fit
            .result
            .aic
        )

        out.delta_aic = (
            out.linear_aic
            - out.quadratic_aic
        )

        out.linear_coef = float(
            out.linear_fit
            .result
            .fe_params
            .get(
                "x_c",
                np.nan,
            )
        )

        out.linear_p = float(
            out.linear_fit
            .result
            .pvalues
            .get(
                "x_c",
                np.nan,
            )
        )

        evidence = (
            extract_quadratic_evidence(
                out.quadratic_fit.result,
                out.x_center,
                work,
                TARGET_ATTRIBUTE,
            )
        )

        out.quadratic_linear_coef = (
            evidence[
                "b1"
            ]
        )

        out.quadratic_coef = (
            evidence[
                "b2"
            ]
        )

        out.quadratic_se = (
            evidence[
                "se2"
            ]
        )

        out.quadratic_p = (
            evidence[
                "p2"
            ]
        )

        out.quadratic_ci_low = (
            evidence[
                "ci_low"
            ]
        )

        out.quadratic_ci_high = (
            evidence[
                "ci_high"
            ]
        )

        out.turning_point = (
            evidence[
                "turning"
            ]
        )

        out.curve_shape = (
            evidence[
                "shape"
            ]
        )

        out.turning_point_kind = (
            evidence[
                "kind"
            ]
        )

        out.turning_point_checks = (
            evidence[
                "checks"
            ]
        )

        out.n_below_turning = (
            evidence[
                "n_below"
            ]
        )

        out.n_above_turning = (
            evidence[
                "n_above"
            ]
        )

        out.curvature_supported = bool(
            np.isfinite(
                out.quadratic_p
            )
            and
            out.quadratic_p
            < CURVATURE_P_THRESHOLD
            and
            np.isfinite(
                out.delta_aic
            )
            and
            out.delta_aic
            >= MIN_DELTA_AIC
        )

        # An "optimum" is specifically a supported interior MAXIMUM,
        # not merely any mathematical vertex.
        out.turning_point_supported = bool(
            out.curvature_supported
            and
            np.isfinite(
                out.quadratic_coef
            )
            and
            out.quadratic_coef
            < 0
            and
            all(
                out.turning_point_checks
                .values()
            )
        )

        (
            out.diagnostics,
            diag_hard_stop,
            out.sensitivity_required,
            out.sensitivity_triggers,
        ) = run_g9a_diagnostics(
            work,
            out.quadratic_fit,
        )

        if diag_hard_stop:

            raise InferenceStop(
                "G9-A diagnostic HARD STOP: the quadratic "
                "MixedLM covariance / convergence diagnostics "
                "do not support inferential interpretation."
            )

        if out.sensitivity_required:

            sens = run_sensitivity(
                work
            )

            out.sensitivity_summary = (
                sens
            )

            (
                out.model_sensitive,
                out.model_sensitive_reasons,
            ) = compare_sensitivity(
                out,
                sens,
            )

        # If model-sensitive, suppress strong curvature / optimum
        # conclusion even if the primary model passed.
        if out.model_sensitive:

            out.model_status = (
                "MODEL-SENSITIVE — primary and sensitivity "
                "conclusions disagree."
            )

            out.answer = (
                "The relationship between "
                f"{out.target_label.lower()} and overall liking "
                "is model-sensitive. The primary and required "
                "sensitivity analyses do not support the same "
                "substantive conclusion, so no stable curvature "
                "or optimum claim is made."
            )

        elif out.curvature_supported:

            if out.turning_point_supported:

                out.model_status = (
                    "Diagnostics acceptable; required sensitivity "
                    "check passed / stable."
                    if out.sensitivity_required
                    else
                    "Diagnostics acceptable."
                )

                out.answer = (
                    "A curved, inverted-U relationship is supported "
                    f"for {out.target_label.lower()} and overall liking. "
                    "Within the observed data, estimated liking peaks "
                    f"near {out.turning_point:.2f} intensity units. "
                    "This is an associative within-study optimum, "
                    "not a causal formulation target."
                )

            else:

                out.model_status = (
                    "Diagnostics acceptable; required sensitivity "
                    "check passed / stable."
                    if out.sensitivity_required
                    else
                    "Diagnostics acceptable."
                )

                if out.quadratic_coef > 0:

                    vertex_note = (
                        "The supported curve is U-shaped, so its "
                        "turning point is a minimum rather than a "
                        "liking optimum."
                    )

                else:

                    failed = [
                        k.replace(
                            "_",
                            " ",
                        )
                        for k, v
                        in out.turning_point_checks.items()
                        if not v
                    ]

                    vertex_note = (
                        "The mathematical peak does not satisfy "
                        "all G9-D interior-support rules"
                        + (
                            f" ({', '.join(failed)})."
                            if failed
                            else "."
                        )
                    )

                out.answer = (
                    "Curvature is supported between "
                    f"{out.target_label.lower()} and overall liking, "
                    "but an optimum is not supported. "
                    + vertex_note
                )

        else:

            out.model_status = (
                "Diagnostics acceptable; required sensitivity "
                "check passed / stable."
                if out.sensitivity_required
                else
                "Diagnostics acceptable."
            )

            out.answer = (
                "The current data do not support a nonlinear "
                "relationship between "
                f"{out.target_label.lower()} and overall liking "
                "under the SOP-10 curvature rule "
                f"(quadratic p={fmt_p(out.quadratic_p)}, "
                f"ΔAIC={fmt_num(out.delta_aic, 2)}). "
                "This is not proof that the true relationship "
                "is perfectly linear or absent."
            )

        out.curve = build_curve(
            out.quadratic_fit.result,
            work,
            TARGET_ATTRIBUTE,
            out.x_center,
        )

        out.product_summary = (
            work.groupby(
                PRODUCT,
                observed=True,
            )
            .agg(
                n=(
                    OUTCOME,
                    "size",
                ),
                mean_attribute=(
                    TARGET_ATTRIBUTE,
                    "mean",
                ),
                mean_liking=(
                    OUTCOME,
                    "mean",
                ),
            )
            .reset_index()
            .sort_values(
                "mean_attribute"
            )
        )

        return (
            out,
            work,
        )

    except (
        CapabilityStop,
        DesignStop,
        InferenceStop,
    ) as exc:

        out.hard_stop = True

        out.hard_stop_reason = str(
            exc
        )

        out.model_status = (
            "HARD STOP"
        )

        out.answer = (
            "SOP-10 inference stopped: "
            + str(
                exc
            )
        )

        return (
            out,
            pd.DataFrame(),
        )

def scale_x(
    x: float,
    x_min: float,
    x_max: float,
    left: float,
    right: float,
) -> float:

    if x_max <= x_min:
        return (
            left
            + right
        ) / 2

    return (
        left
        + (
            x
            - x_min
        )
        / (
            x_max
            - x_min
        )
        * (
            right
            - left
        )
    )

def scale_y(
    y: float,
    y_min: float,
    y_max: float,
    top: float,
    bottom: float,
) -> float:

    if y_max <= y_min:
        return (
            top
            + bottom
        ) / 2

    return (
        bottom
        - (
            y
            - y_min
        )
        / (
            y_max
            - y_min
        )
        * (
            bottom
            - top
        )
    )

def moderator_display_name() -> str:
    return MODERATOR_LABELS.get(
        MODERATOR,
        MODERATOR.replace("_", " ").title(),
    )

def remove_export_index_columns(
    df: pd.DataFrame,
) -> Tuple[pd.DataFrame, List[str]]:
    """
    Remove harmless CSV-export index fields only when they behave like
    row indices. Do not remove substantive fields merely because their
    name resembles an index.
    """
    df = df.copy()
    removed: List[str] = []

    candidates = [
        c for c in df.columns
        if str(c).lower() in {"index", "unnamed: 0"}
    ]

    for col in candidates:
        s = pd.to_numeric(df[col], errors="coerce")

        if s.notna().all():
            values = s.to_numpy()

            zero_based = np.array_equal(
                values,
                np.arange(len(df)),
            )
            one_based = np.array_equal(
                values,
                np.arange(1, len(df) + 1),
            )

            if zero_based or one_based:
                df = df.drop(columns=[col])
                removed.append(col)

    return df, removed

def determine_moderator_type(
    df: pd.DataFrame,
) -> str:
    requested = MODERATOR_TYPE.strip().lower()

    if requested not in {
        "auto",
        "categorical",
        "continuous",
    }:
        raise ValueError(
            "MODERATOR_TYPE must be "
            "'auto', 'categorical', or 'continuous'."
        )

    if requested != "auto":
        return requested

    if MODERATOR in KNOWN_CATEGORICAL_MODERATORS:
        return "categorical"

    if MODERATOR not in df.columns:
        return "unknown"

    s = df[MODERATOR]

    if (
        pd.api.types.is_numeric_dtype(s)
        and s.nunique(dropna=True) > 10
    ):
        return "continuous"

    return "categorical"

def dataset_capability_gate(
    raw_df: pd.DataFrame,
) -> Tuple[pd.DataFrame, AnalysisResult]:
    result = AnalysisResult()
    result.moderator_label = moderator_display_name()

    df, removed = remove_export_index_columns(raw_df)

    required = [
        PANELIST,
        PRODUCT,
        OUTCOME,
        MODERATOR,
    ]

    missing = [
        col for col in required
        if col not in df.columns
    ]

    if missing:
        result.capability_status = "UNSUPPORTED"
        result.hard_stop = True
        result.model_status = "HARD STOP"

        result.hard_stop_reasons.append(
            "Required variable(s) absent: "
            + ", ".join(missing)
            + ". SOP-11 cannot substitute another construct."
        )

        return df, result

    result.moderator_type = determine_moderator_type(df)

    if result.moderator_type == "unknown":
        result.capability_status = "UNSUPPORTED"
        result.hard_stop = True
        result.model_status = "HARD STOP"
        result.hard_stop_reasons.append(
            f"Could not determine moderator type for {MODERATOR}."
        )
        return df, result

    # Outcome must be numeric.
    df[OUTCOME] = pd.to_numeric(
        df[OUTCOME],
        errors="coerce",
    )

    # IDs/product should remain categorical/string-like.
    df[PRODUCT] = df[PRODUCT].astype("string")
    df[PANELIST] = df[PANELIST].astype("string")

    if result.moderator_type == "continuous":
        df[MODERATOR] = pd.to_numeric(
            df[MODERATOR],
            errors="coerce",
        )
    else:
        df[MODERATOR] = df[MODERATOR].astype("string")

    result.capability_status = "SUPPORTED"

    if removed:
        result.warnings.append(
            "Removed harmless saved CSV index column(s): "
            + ", ".join(removed)
            + "."
        )

    return df, result

def unique_panelist_moderator(
    df: pd.DataFrame,
    result: AnalysisResult,
) -> pd.DataFrame:
    """
    Verify that the consumer-level moderator is actually constant within
    consumer. A consumer characteristic that changes across product rows
    is not coherent for this SOP as coded.
    """
    temp = df[
        [PANELIST, MODERATOR]
    ].dropna().copy()

    counts = (
        temp.groupby(PANELIST, observed=True)[MODERATOR]
        .nunique(dropna=True)
    )

    inconsistent = counts[counts > 1]

    if not inconsistent.empty:
        result.hard_stop = True
        result.model_status = "HARD STOP"
        result.hard_stop_reasons.append(
            f"{MODERATOR} is not constant within "
            f"{len(inconsistent)} consumer(s). "
            "SOP-11 requires a coherent consumer-level moderator."
        )

    return (
        temp.drop_duplicates(
            subset=[PANELIST],
            keep="first",
        )
    )

def subgroup_eligibility(
    df: pd.DataFrame,
    result: AnalysisResult,
) -> Tuple[pd.DataFrame, AnalysisResult]:
    """
    Apply SOP-11 moderator eligibility to the SELECTED moderator only.

    The goal is to answer questions such as:

        Does gender change product response?
        Does age group change product response?
        Does income bracket change product response?
        Does chocolate preference change product response?
        Does usage frequency change product response?

    Categorical moderator rules are evaluated using UNIQUE CONSUMERS:

        N >= 10
            Include in the interaction normally.

        N = 5–9
            Include in the interaction, but classify the resulting
            subgroup analysis as EXPLORATORY.

        N < 5
            Do not estimate that moderator level. Exclude those
            consumers from THIS interaction analysis only and
            disclose the exclusion.

    Sparse levels do NOT cause the whole SOP-11 analysis to stop as
    long as at least two analyzable moderator levels remain.

    Sparse categories are NEVER automatically merged or recoded.
    """

    if result.moderator_type != "categorical":
        return df, result

    # ------------------------------------------------------------------
    # Confirm that the selected moderator is a consumer-level variable
    # ------------------------------------------------------------------
    people = unique_panelist_moderator(
        df[[PANELIST, MODERATOR]].copy(),
        result,
    )

    if result.hard_stop:
        return df, result

    # ------------------------------------------------------------------
    # Count UNIQUE consumers by selected moderator level
    # ------------------------------------------------------------------
    counts = (
        people[MODERATOR]
        .value_counts(dropna=False)
        .rename_axis(MODERATOR)
        .reset_index(name="N_consumers")
    )

    counts[MODERATOR] = counts[MODERATOR].astype("string")

    def classify(n: int) -> str:
        if n >= MIN_STANDARD_SUBGROUP_N:
            return "STANDARD"

        if n >= MIN_ANALYZABLE_SUBGROUP_N:
            return "EXPLORATORY"

        return "NOT ESTIMATED"

    counts["G9E_status"] = (
        counts["N_consumers"]
        .astype(int)
        .map(classify)
    )

    result.subgroup_counts = counts.copy()

    # ------------------------------------------------------------------
    # Identify usable and sparse levels
    # ------------------------------------------------------------------
    analyzable = counts[
        counts["N_consumers"]
        >= MIN_ANALYZABLE_SUBGROUP_N
    ].copy()

    sparse = counts[
        counts["N_consumers"]
        < MIN_ANALYZABLE_SUBGROUP_N
    ].copy()

    exploratory = analyzable[
        analyzable["N_consumers"]
        < MIN_STANDARD_SUBGROUP_N
    ].copy()

    # ------------------------------------------------------------------
    # HARD STOP ONLY IF <2 LEVELS CAN ACTUALLY BE COMPARED
    # ------------------------------------------------------------------
    if len(analyzable) < 2:
        result.hard_stop = True
        result.model_status = "HARD STOP"

        usable_description = ", ".join(
            f"{row[MODERATOR]} (N={int(row['N_consumers'])})"
            for _, row in analyzable.iterrows()
        )

        sparse_description = ", ".join(
            f"{row[MODERATOR]} (N={int(row['N_consumers'])})"
            for _, row in sparse.iterrows()
        )

        result.hard_stop_reasons.append(
            f"The selected moderator '{MODERATOR}' does not contain "
            "at least two adequately supported levels for comparison. "
            f"Analyzable levels: {usable_description or 'none'}. "
            f"Sparse levels: {sparse_description or 'none'}. "
            "At least two moderator levels with five or more unique "
            "consumers are required for a product × moderator interaction."
        )

        return df, result

    # ------------------------------------------------------------------
    # DISCLOSE SPARSE LEVELS, BUT DO NOT KILL THE ANALYSIS
    # ------------------------------------------------------------------
    if not sparse.empty:
        sparse_levels = [
            str(v)
            for v in sparse[MODERATOR].tolist()
        ]

        sparse_description = "; ".join(
            f"{row[MODERATOR]}: N={int(row['N_consumers'])}"
            for _, row in sparse.iterrows()
        )

        result.warnings.append(
            f"The selected moderator '{MODERATOR}' contains "
            f"insufficiently supported level(s): {sparse_description}. "
            "These level(s) are not estimated in the interaction model. "
            "They are not merged or recoded. The remaining adequately "
            "supported moderator levels are analyzed."
        )

        # --------------------------------------------------------------
        # Filter ONLY this SOP-11 analysis.
        #
        # The original dataset is not modified.
        # These consumers are simply outside the estimable scope of
        # this particular moderator interaction.
        # --------------------------------------------------------------
        df = df[
            ~df[MODERATOR].astype(str).isin(sparse_levels)
        ].copy()

    # ------------------------------------------------------------------
    # EXPLORATORY DOWNGRADE WHEN ANY INCLUDED LEVEL HAS N=5–9
    # ------------------------------------------------------------------
    if not exploratory.empty:
        result.analysis_level = "EXPLORATORY"

        exploratory_description = "; ".join(
            f"{row[MODERATOR]}: N={int(row['N_consumers'])}"
            for _, row in exploratory.iterrows()
        )

        result.warnings.append(
            f"The selected moderator '{MODERATOR}' contains included "
            f"level(s) with only 5–9 unique consumers: "
            f"{exploratory_description}. "
            "The interaction may be estimated, but subgroup conclusions "
            "are exploratory."
        )

    # ------------------------------------------------------------------
    # Verify that two levels remain after filtering
    # ------------------------------------------------------------------
    remaining_levels = (
        df[MODERATOR]
        .dropna()
        .astype(str)
        .nunique()
    )

    if remaining_levels < 2:
        result.hard_stop = True
        result.model_status = "HARD STOP"

        result.hard_stop_reasons.append(
            f"After removing unsupported '{MODERATOR}' levels, fewer "
            "than two moderator levels remain. The interaction cannot "
            "be estimated."
        )

    return df, result

def validate_study_design(
    df: pd.DataFrame,
    result: AnalysisResult,
) -> pd.DataFrame:
    """
    Validate repeated-measures structure and analysis rows.
    """
    analysis = df[
        [PANELIST, PRODUCT, OUTCOME, MODERATOR]
    ].copy()

    result.n_rows = len(analysis)
    result.n_consumers = analysis[PANELIST].nunique(dropna=True)
    result.n_products = analysis[PRODUCT].nunique(dropna=True)

    duplicates = analysis.duplicated(
        subset=[PANELIST, PRODUCT],
        keep=False,
    )

    if duplicates.any():
        n_dup_rows = int(duplicates.sum())
        result.hard_stop = True
        result.model_status = "HARD STOP"
        result.hard_stop_reasons.append(
            f"Found {n_dup_rows} rows involved in duplicate "
            f"{PANELIST} × {PRODUCT} cells. "
            "Duplicate evaluations are not silently averaged."
        )

    invalid_liking = (
        analysis[OUTCOME].notna()
        & ~analysis[OUTCOME].between(
            LIKING_MIN,
            LIKING_MAX,
            inclusive="both",
        )
    )

    if invalid_liking.any():
        result.hard_stop = True
        result.model_status = "HARD STOP"
        result.hard_stop_reasons.append(
            f"Found {int(invalid_liking.sum())} "
            f"{OUTCOME} value(s) outside the approved "
            f"{LIKING_MIN}–{LIKING_MAX} scale."
        )

    if analysis[OUTCOME].nunique(dropna=True) < 2:
        result.hard_stop = True
        result.model_status = "HARD STOP"
        result.hard_stop_reasons.append(
            f"{OUTCOME} has insufficient variation."
        )

    if result.n_products < 2:
        result.hard_stop = True
        result.model_status = "HARD STOP"
        result.hard_stop_reasons.append(
            "At least two products are required for "
            "a product × consumer interaction."
        )

    # Confirm actual repeated measures.
    evaluations_per_consumer = (
        analysis.groupby(PANELIST, observed=True)[PRODUCT]
        .nunique()
    )

    if len(evaluations_per_consumer) > 0:
        max_products_per_consumer = int(
            evaluations_per_consumer.max()
        )
    else:
        max_products_per_consumer = 0

    if max_products_per_consumer < 2:
        result.hard_stop = True
        result.model_status = "HARD STOP"
        result.hard_stop_reasons.append(
            "The data do not demonstrate repeated product "
            "evaluations within consumers."
        )

    missing_before = analysis.isna().any(axis=1)
    n_missing_rows = int(missing_before.sum())

    if n_missing_rows > 0:
        result.warnings.append(
            f"{n_missing_rows} evaluation row(s) contain a missing "
            "required SOP-11 value and are excluded from the model. "
            "No values are imputed."
        )

    analysis = analysis.dropna(
        subset=[
            PANELIST,
            PRODUCT,
            OUTCOME,
            MODERATOR,
        ]
    ).copy()

    # Recompute after complete-case filtering.
    result.n_rows = len(analysis)
    result.n_consumers = analysis[PANELIST].nunique()
    result.n_products = analysis[PRODUCT].nunique()

    if result.moderator_type == "categorical":
        analysis, result = subgroup_eligibility(
            analysis,
            result,
        )
    else:
        unique_panelist_moderator(analysis, result)

        moderator_unique = analysis[MODERATOR].nunique()

        if moderator_unique < 3:
            result.hard_stop = True
            result.model_status = "HARD STOP"
            result.hard_stop_reasons.append(
                f"Continuous moderator {MODERATOR} has only "
                f"{moderator_unique} distinct observed value(s). "
                "This is insufficient for a meaningful continuous "
                "interaction analysis."
            )

    # Check consumer × product coverage.
    coverage = (
        analysis.groupby(PANELIST, observed=True)[PRODUCT]
        .nunique()
    )

    if not coverage.empty:
        incomplete = int(
            (coverage < result.n_products).sum()
        )

        if incomplete > 0:
            result.warnings.append(
                f"{incomplete} consumer(s) do not have complete "
                "data for all modeled products after missing-value "
                "filtering. MixedLM can use incomplete repeated "
                "measures, but this is disclosed."
            )

    return analysis

def build_primary_formula(
    moderator_type: str,
) -> str:
    if moderator_type == "categorical":
        return (
            f"{OUTCOME} ~ "
            f"C({PRODUCT}) * C({MODERATOR})"
        )

    return (
        f"{OUTCOME} ~ "
        f"C({PRODUCT}) * {MODERATOR}"
    )

def build_sensitivity_formula(
    moderator_type: str,
) -> str:
    """
    Panelist fixed-effect sensitivity model.

    For a categorical moderator, its standalone main effect is
    redundant with consumer fixed effects because the moderator is
    consumer-level and constant within person. Patsy/statsmodels may
    therefore represent a rank-deficient main-effect block, but the
    product × moderator interaction remains the quantity of interest.

    To avoid unnecessary redundant moderator main effects, construct:
        outcome ~ C(product) + C(panelist)
                  + C(product):C(moderator)

    For continuous moderator:
        outcome ~ C(product) + C(panelist)
                  + C(product):moderator
    """
    if moderator_type == "categorical":
        return (
            f"{OUTCOME} ~ "
            f"C({PRODUCT}) + "
            f"C({PANELIST}) + "
            f"C({PRODUCT}):C({MODERATOR})"
        )

    return (
        f"{OUTCOME} ~ "
        f"C({PRODUCT}) + "
        f"C({PANELIST}) + "
        f"C({PRODUCT}):{MODERATOR}"
    )

def fit_mixed_model(
    df: pd.DataFrame,
    result: AnalysisResult,
):
    formula = build_primary_formula(
        result.moderator_type
    )
    result.formula = formula

    last_error: Optional[Exception] = None
    warning_messages: List[str] = []

    for optimizer in MIXEDLM_OPTIMIZERS:
        try:
            with warnings.catch_warnings(record=True) as caught:
                warnings.simplefilter("always")

                model = smf.mixedlm(
                    formula,
                    data=df,
                    groups=df[PANELIST],
                )

                fit = model.fit(
                    method=optimizer,
                    reml=False,
                    disp=False,
                )

            current_warnings = [
                str(w.message)
                for w in caught
            ]

            warning_messages.extend(current_warnings)

            if bool(getattr(fit, "converged", False)):
                result.optimizer_used = optimizer.upper()

                # A1 passed.
                result.diagnostics.append(
                    DiagnosticRow(
                        diagnostic="A1 — Optimizer convergence",
                        observed=f"Converged with {optimizer.upper()}",
                        threshold="result.converged == True",
                        status="PASS",
                        action="Proceed to model diagnostics.",
                        tooltip=(
                            "Optimizer convergence checks whether the "
                            "model-fitting algorithm reached a stable "
                            "solution. A non-converged model cannot "
                            "support standard inference."
                        ),
                    )
                )

                # Check A2 warnings before returning.
                lower_warnings = " ".join(
                    warning_messages
                ).lower()

                a2_phrases = [
                    "random effects covariance is singular",
                    "covariance matrix is singular",
                    "hessian matrix at the estimated parameter values "
                    "is not positive definite",
                    "hessian is not positive definite",
                ]

                a2_hit = any(
                    phrase in lower_warnings
                    for phrase in a2_phrases
                )

                # Also ensure fixed-effect covariance is finite.
                try:
                    cov = fit.cov_params()
                    fe_names = list(fit.fe_params.index)
                    cov_fe = cov.loc[
                        fe_names,
                        fe_names,
                    ].to_numpy(dtype=float)

                    covariance_valid = np.all(
                        np.isfinite(cov_fe)
                    )
                except Exception:
                    covariance_valid = False

                if a2_hit or not covariance_valid:
                    result.hard_stop = True
                    result.model_status = "HARD STOP"

                    reason = (
                        "A2 hard stop: the fitted mixed model showed "
                        "a singular random-effects covariance, invalid "
                        "Hessian/covariance warning, or unusable "
                        "standard-error covariance."
                    )
                    result.hard_stop_reasons.append(reason)

                    result.diagnostics.append(
                        DiagnosticRow(
                            diagnostic=(
                                "A2 — Random-effects covariance / Hessian"
                            ),
                            observed=(
                                "Problem detected in model covariance "
                                "or Hessian."
                            ),
                            threshold=(
                                "No singular covariance; finite inference "
                                "covariance; positive-definite Hessian"
                            ),
                            status="HARD STOP",
                            action=(
                                "Suppress mixed-model inferential "
                                "interpretation."
                            ),
                            tooltip=(
                                "This diagnostic checks whether the model "
                                "estimated its uncertainty reliably. A "
                                "singular covariance or invalid Hessian "
                                "means ordinary Wald-style inference is "
                                "not defensible."
                            ),
                        )
                    )

                else:
                    result.diagnostics.append(
                        DiagnosticRow(
                            diagnostic=(
                                "A2 — Random-effects covariance / Hessian"
                            ),
                            observed="No hard-stop condition detected",
                            threshold=(
                                "No singular covariance or invalid "
                                "inference covariance"
                            ),
                            status="PASS",
                            action="Proceed.",
                            tooltip=(
                                "Checks whether the random-effect and "
                                "parameter uncertainty estimates are "
                                "mathematically usable."
                            ),
                        )
                    )

                return fit

        except Exception as exc:
            last_error = exc
            continue

    result.hard_stop = True
    result.model_status = "HARD STOP"

    message = (
        "A1 hard stop: the mixed-effects model did not converge "
        "successfully after the approved optimizer sequence "
        "BFGS → L-BFGS → CG."
    )

    if last_error is not None:
        message += f" Final fitting error: {last_error}"

    result.hard_stop_reasons.append(message)

    result.diagnostics.append(
        DiagnosticRow(
            diagnostic="A1 — Optimizer convergence",
            observed="Failed approved optimizer sequence",
            threshold="result.converged == True",
            status="HARD STOP",
            action="Suppress mixed-model inference.",
            tooltip=(
                "The model did not reach a stable fitted solution "
                "using the approved optimizers. The SOP does not "
                "authorize trying arbitrary additional optimizers "
                "until one produces favorable results."
            ),
        )
    )

    return None

def interaction_parameter_names(
    param_names: Sequence[str],
    moderator_type: str,
) -> List[str]:
    """
    Return coefficient names corresponding to product × moderator
    interaction terms.
    """
    names: List[str] = []

    for name in param_names:
        if ":" not in name:
            continue

        if f"C({PRODUCT})" not in name:
            continue

        if moderator_type == "categorical":
            if f"C({MODERATOR})" not in name:
                continue
        else:
            if MODERATOR not in name:
                continue

        names.append(name)

    return names

def joint_wald_test(
    fit,
    moderator_type: str,
) -> Tuple[Optional[float], Optional[int], Optional[float]]:
    """
    Joint Wald test that every product × moderator coefficient equals 0.
    """
    param_names = list(fit.params.index)

    interaction_names = interaction_parameter_names(
        param_names,
        moderator_type,
    )

    if not interaction_names:
        return None, None, None

    r = np.zeros(
        (
            len(interaction_names),
            len(param_names),
        ),
        dtype=float,
    )

    name_to_idx = {
        name: i
        for i, name in enumerate(param_names)
    }

    for row_idx, name in enumerate(interaction_names):
        r[row_idx, name_to_idx[name]] = 1.0

    try:
        test = fit.wald_test(
            r,
            scalar=True,
        )

        statistic = float(
            np.asarray(test.statistic).squeeze()
        )
        pvalue = float(
            np.asarray(test.pvalue).squeeze()
        )

        return (
            statistic,
            len(interaction_names),
            pvalue,
        )

    except Exception:
        return None, len(interaction_names), None

def calculate_mixed_diagnostics(
    df: pd.DataFrame,
    fit,
    result: AnalysisResult,
) -> None:
    if fit is None or result.hard_stop:
        return

    # ---------------------------------------------------------------------
    # A3 / A4 — random variance and ICC
    # ---------------------------------------------------------------------
    try:
        cov_re = np.asarray(
            fit.cov_re,
            dtype=float,
        )

        random_variance = float(
            cov_re[0, 0]
        )
    except Exception:
        random_variance = np.nan

    try:
        residual_variance = float(
            fit.scale
        )
    except Exception:
        residual_variance = np.nan

    result.random_variance = random_variance
    result.residual_variance = residual_variance

    denominator = (
        random_variance + residual_variance
        if (
            np.isfinite(random_variance)
            and np.isfinite(residual_variance)
        )
        else np.nan
    )

    if (
        np.isfinite(denominator)
        and denominator > 0
    ):
        icc = random_variance / denominator
    else:
        icc = np.nan

    result.icc = icc

    if (
        np.isfinite(random_variance)
        and random_variance < BOUNDARY_RANDOM_VARIANCE
    ):
        result.sensitivity_required = True

        result.diagnostics.append(
            DiagnosticRow(
                diagnostic="A3 — Random-intercept variance",
                observed=fmt_num(random_variance, 4),
                threshold="< 0.01 → sensitivity required",
                status="SENSITIVITY REQUIRED",
                action=(
                    "Run panelist fixed-effect OLS + HC3 "
                    "sensitivity model."
                ),
                tooltip=(
                    "Random-intercept variance estimates how much "
                    "consumers differ in baseline liking. A value "
                    "below 0.01 places the random effect near the "
                    "model boundary, so the SOP requires a repeated-"
                    "measures sensitivity analysis."
                ),
            )
        )
    else:
        result.diagnostics.append(
            DiagnosticRow(
                diagnostic="A3 — Random-intercept variance",
                observed=fmt_num(random_variance, 4),
                threshold="< 0.01 → sensitivity required",
                status="PASS",
                action="None.",
                tooltip=(
                    "Estimates between-consumer differences in "
                    "baseline scale use."
                ),
            )
        )

    if np.isfinite(icc) and icc < NEAR_ZERO_ICC:
        result.sensitivity_required = True

        result.diagnostics.append(
            DiagnosticRow(
                diagnostic="A4 — ICC",
                observed=fmt_num(icc, 4),
                threshold="< 0.01 → sensitivity required",
                status="SENSITIVITY REQUIRED",
                action=(
                    "Run panelist fixed-effect OLS + HC3 "
                    "sensitivity model."
                ),
                tooltip=(
                    "ICC estimates the share of modeled variation "
                    "associated with between-consumer baseline "
                    "differences. ICC below 0.01 means the random "
                    "consumer effect contributes very little "
                    "variance, triggering the approved sensitivity "
                    "check."
                ),
            )
        )
    else:
        result.diagnostics.append(
            DiagnosticRow(
                diagnostic="A4 — ICC",
                observed=fmt_num(icc, 4),
                threshold="< 0.01 → sensitivity required",
                status="PASS",
                action="None.",
                tooltip=(
                    "ICC summarizes the amount of variation tied "
                    "to differences between consumers."
                ),
            )
        )

    # ---------------------------------------------------------------------
    # Residual diagnostics
    # ---------------------------------------------------------------------
    residuals = np.asarray(
        fit.resid,
        dtype=float,
    )

    residuals = residuals[
        np.isfinite(residuals)
    ]

    # Standardized residuals for A7.
    if (
        np.isfinite(residual_variance)
        and residual_variance > 0
    ):
        standardized = (
            residuals
            / math.sqrt(residual_variance)
        )
    else:
        standardized = residuals.copy()

    # ---------------------------------------------------------------------
    # A5 — distribution
    # ---------------------------------------------------------------------
    if len(residuals) >= 8:
        try:
            normality_stat, normality_p = stats.normaltest(
                residuals
            )
            normality_p = float(normality_p)
        except Exception:
            normality_p = np.nan
    else:
        normality_p = np.nan

    try:
        skewness = float(
            stats.skew(
                residuals,
                bias=False,
                nan_policy="omit",
            )
        )
    except Exception:
        skewness = np.nan

    try:
        kurtosis = float(
            stats.kurtosis(
                residuals,
                fisher=False,
                bias=False,
                nan_policy="omit",
            )
        )
    except Exception:
        kurtosis = np.nan

    result.normality_p = normality_p
    result.residual_skewness = skewness
    result.residual_kurtosis = kurtosis

    formal_warning = (
        np.isfinite(normality_p)
        and normality_p < NORMALITY_WARNING_P
    )

    distribution_sensitivity = (
        (
            np.isfinite(skewness)
            and abs(skewness) > SKEWNESS_SENSITIVITY
        )
        or (
            np.isfinite(kurtosis)
            and kurtosis > KURTOSIS_SENSITIVITY
        )
    )

    if distribution_sensitivity:
        result.sensitivity_required = True
        a5_status = "SENSITIVITY REQUIRED"
        a5_action = (
            "Run panelist fixed-effect OLS + HC3 "
            "sensitivity model."
        )
    elif formal_warning:
        a5_status = "WARNING"
        a5_action = (
            "Disclose formal normality warning; sensitivity "
            "not required from normality p-value alone."
        )
    else:
        a5_status = "PASS"
        a5_action = "None."

    result.diagnostics.append(
        DiagnosticRow(
            diagnostic="A5 — Residual distribution",
            observed=(
                f"normality p={fmt_p(normality_p)}; "
                f"skew={fmt_num(skewness, 2)}; "
                f"Pearson kurtosis={fmt_num(kurtosis, 2)}"
            ),
            threshold=(
                "p<0.01 warning; |skew|>1 or "
                "kurtosis>5 → sensitivity"
            ),
            status=a5_status,
            action=a5_action,
            tooltip=(
                "Residual-distribution checks assess whether model "
                "errors are strongly asymmetric or unusually "
                "heavy-tailed. The formal normality test alone is "
                "only a warning at p<0.01; sensitivity is required "
                "when |skewness| exceeds 1 or Pearson kurtosis "
                "exceeds 5."
            ),
        )
    )

    # ---------------------------------------------------------------------
    # A6 — residual heteroscedasticity
    # ---------------------------------------------------------------------
    diag_frame = df.copy()

    # statsmodels fit rows should match the complete modeling frame.
    if len(diag_frame) == len(fit.resid):
        diag_frame["_resid"] = np.asarray(
            fit.resid,
            dtype=float,
        )
    else:
        diag_frame = diag_frame.iloc[
            :len(fit.resid)
        ].copy()
        diag_frame["_resid"] = np.asarray(
            fit.resid,
            dtype=float,
        )

    variances = (
        diag_frame.groupby(
            PRODUCT,
            observed=True,
        )["_resid"]
        .var(ddof=1)
        .dropna()
    )

    positive_variances = variances[
        variances > 0
    ]

    if len(positive_variances) >= 2:
        variance_ratio = float(
            positive_variances.max()
            / positive_variances.min()
        )
    elif len(positive_variances) == 1:
        variance_ratio = 1.0
    else:
        variance_ratio = np.nan

    groups = [
        group["_resid"].dropna().to_numpy()
        for _, group in diag_frame.groupby(
            PRODUCT,
            observed=True,
        )
        if group["_resid"].notna().sum() >= 2
    ]

    if len(groups) >= 2:
        try:
            _, brown_p = stats.levene(
                *groups,
                center="median",
            )
            brown_p = float(brown_p)
        except Exception:
            brown_p = np.nan
    else:
        brown_p = np.nan

    result.variance_ratio = variance_ratio
    result.brown_forsythe_p = brown_p

    a6_warning = (
        (
            np.isfinite(brown_p)
            and brown_p < BROWN_FORSYTHE_WARNING_P
        )
        or (
            np.isfinite(variance_ratio)
            and variance_ratio > VARIANCE_RATIO_WARNING
        )
    )

    a6_sensitivity = (
        (
            np.isfinite(brown_p)
            and brown_p < BROWN_FORSYTHE_WARNING_P
            and np.isfinite(variance_ratio)
            and variance_ratio > VARIANCE_RATIO_WARNING
        )
        or (
            np.isfinite(variance_ratio)
            and variance_ratio > VARIANCE_RATIO_SENSITIVITY
        )
    )

    if a6_sensitivity:
        result.sensitivity_required = True
        a6_status = "SENSITIVITY REQUIRED"
        a6_action = (
            "Run panelist fixed-effect OLS + HC3 "
            "sensitivity model."
        )
    elif a6_warning:
        a6_status = "WARNING"
        a6_action = (
            "Disclose unequal-residual-variance warning."
        )
    else:
        a6_status = "PASS"
        a6_action = "None."

    result.diagnostics.append(
        DiagnosticRow(
            diagnostic="A6 — Residual heteroscedasticity",
            observed=(
                f"variance ratio={fmt_num(variance_ratio, 2)}; "
                f"Brown-Forsythe p={fmt_p(brown_p)}"
            ),
            threshold=(
                "warning: p<0.01 or ratio>2; "
                "sensitivity: p<0.01 & ratio>2, or ratio>4"
            ),
            status=a6_status,
            action=a6_action,
            tooltip=(
                "This checks whether unexplained variability "
                "differs substantially across products. The "
                "variance ratio compares the largest and smallest "
                "product residual variance, while Brown-Forsythe "
                "tests equality of residual variability using a "
                "median-centered robust approach."
            ),
        )
    )

    # ---------------------------------------------------------------------
    # A7 — extreme residuals
    # ---------------------------------------------------------------------
    abs_resid = np.abs(standardized)

    n_gt3 = int(
        np.sum(abs_resid > EXTREME_RESIDUAL_FLAG)
    )
    pct_gt3 = (
        n_gt3 / len(abs_resid)
        if len(abs_resid) > 0
        else 0.0
    )
    n_gt4 = int(
        np.sum(abs_resid > EXTREME_RESIDUAL_SENSITIVITY)
    )

    result.residual_gt3_n = n_gt3
    result.residual_gt3_pct = pct_gt3
    result.residual_gt4_n = n_gt4

    a7_sensitivity = (
        pct_gt3 > EXTREME_RESIDUAL_PERCENT_SENSITIVITY
        or n_gt4 > 0
    )

    if a7_sensitivity:
        result.sensitivity_required = True
        a7_status = "SENSITIVITY REQUIRED"
        a7_action = (
            "Keep all observations; run approved "
            "sensitivity model."
        )
    elif n_gt3 > 0:
        a7_status = "WARNING"
        a7_action = (
            "Flag observations for robustness review; "
            "do not delete automatically."
        )
    else:
        a7_status = "PASS"
        a7_action = "None."

    result.diagnostics.append(
        DiagnosticRow(
            diagnostic="A7 — Extreme residuals",
            observed=(
                f"|r|>3: {n_gt3} "
                f"({100 * pct_gt3:.1f}%); "
                f"|r|>4: {n_gt4}"
            ),
            threshold=(
                ">1% with |r|>3 or any |r|>4 "
                "→ sensitivity"
            ),
            status=a7_status,
            action=a7_action,
            tooltip=(
                "Extreme residuals are observations substantially "
                "farther from the model prediction than expected. "
                "They are robustness flags, not automatic exclusion "
                "criteria. More than 1% beyond |3| or any beyond "
                "|4| triggers sensitivity analysis."
            ),
        )
    )

def ordered_levels(
    values: Sequence[str],
    variable: str,
) -> List[str]:
    observed = [
        str(v)
        for v in pd.Series(values)
        .dropna()
        .unique()
    ]

    preferred = CATEGORY_ORDER.get(
        variable,
        [],
    )

    first = [
        v for v in preferred
        if v in observed
    ]

    remaining = sorted(
        v for v in observed
        if v not in first
    )

    return first + remaining

def build_design_row(
    fit,
    row_data: Dict[str, object],
) -> np.ndarray:
    """
    Rebuild fixed-effect design vector using the exact Patsy design
    specification used for model fitting.
    """
    design_info = fit.model.data.design_info

    matrix = patsy.build_design_matrices(
        [design_info],
        pd.DataFrame([row_data]),
        return_type="dataframe",
    )[0]

    return matrix.to_numpy(dtype=float)[0]

def fixed_effect_covariance(
    fit,
) -> Tuple[np.ndarray, List[str]]:
    fe_names = list(fit.fe_params.index)

    cov = fit.cov_params()

    cov_fe = cov.loc[
        fe_names,
        fe_names,
    ].to_numpy(dtype=float)

    return cov_fe, fe_names

def estimate_categorical_cells(
    df: pd.DataFrame,
    fit,
) -> pd.DataFrame:
    products = ordered_levels(
        df[PRODUCT],
        PRODUCT,
    )

    groups = ordered_levels(
        df[MODERATOR],
        MODERATOR,
    )

    beta = fit.fe_params.to_numpy(dtype=float)
    cov_fe, fe_names = fixed_effect_covariance(fit)

    # MixedLM fixed-effect model matrix excludes groups automatically.
    rows = []

    for product in products:
        for group in groups:
            row_data = {
                PRODUCT: product,
                MODERATOR: group,
            }

            x = build_design_row(
                fit,
                row_data,
            )

            # Defensive alignment.
            if len(x) != len(beta):
                raise RuntimeError(
                    "Prediction design vector does not match "
                    "mixed-model fixed-effect parameter length."
                )

            mean = float(x @ beta)
            var = float(x @ cov_fe @ x)
            se = math.sqrt(max(var, 0.0))

            z = stats.norm.ppf(
                1 - (1 - CI_LEVEL) / 2
            )

            rows.append(
                {
                    PRODUCT: str(product),
                    MODERATOR: str(group),
                    "estimated_mean": mean,
                    "se": se,
                    "ci95_low": mean - z * se,
                    "ci95_high": mean + z * se,
                }
            )

    return pd.DataFrame(rows)

def continuous_reference_values(
    df: pd.DataFrame,
) -> List[Tuple[str, float]]:
    """
    For a continuous moderator, visualize at the 25th, 50th, and
    75th percentiles. These are observed-range summaries, not
    consumer subgroups and not separate inferential categories.
    """
    s = pd.to_numeric(
        df[MODERATOR],
        errors="coerce",
    ).dropna()

    values = [
        ("25th percentile", float(s.quantile(0.25))),
        ("Median", float(s.quantile(0.50))),
        ("75th percentile", float(s.quantile(0.75))),
    ]

    return values

def estimate_continuous_cells(
    df: pd.DataFrame,
    fit,
) -> pd.DataFrame:
    products = ordered_levels(
        df[PRODUCT],
        PRODUCT,
    )

    moderator_values = continuous_reference_values(df)

    beta = fit.fe_params.to_numpy(dtype=float)
    cov_fe, fe_names = fixed_effect_covariance(fit)

    rows = []

    for product in products:
        for label, value in moderator_values:
            row_data = {
                PRODUCT: product,
                MODERATOR: value,
            }

            x = build_design_row(
                fit,
                row_data,
            )

            mean = float(x @ beta)
            var = float(x @ cov_fe @ x)
            se = math.sqrt(max(var, 0.0))

            z = stats.norm.ppf(
                1 - (1 - CI_LEVEL) / 2
            )

            rows.append(
                {
                    PRODUCT: str(product),
                    "moderator_display": label,
                    "moderator_value": value,
                    "estimated_mean": mean,
                    "se": se,
                    "ci95_low": mean - z * se,
                    "ci95_high": mean + z * se,
                }
            )

    return pd.DataFrame(rows)

def categorical_simple_contrasts(
    df: pd.DataFrame,
    fit,
) -> pd.DataFrame:
    """
    Compare moderator levels WITHIN each product.

    These are secondary simple contrasts and are only interpreted in
    the context of the omnibus product × moderator interaction.

    The entire requested family is Holm-adjusted together.
    """
    products = ordered_levels(
        df[PRODUCT],
        PRODUCT,
    )
    groups = ordered_levels(
        df[MODERATOR],
        MODERATOR,
    )

    beta = fit.fe_params.to_numpy(dtype=float)
    cov_fe, _ = fixed_effect_covariance(fit)

    rows = []

    for product in products:
        design = {}

        for group in groups:
            design[group] = build_design_row(
                fit,
                {
                    PRODUCT: product,
                    MODERATOR: group,
                },
            )

        for i in range(len(groups)):
            for j in range(i + 1, len(groups)):
                group_1 = groups[i]
                group_2 = groups[j]

                contrast = (
                    design[group_1]
                    - design[group_2]
                )

                estimate = float(
                    contrast @ beta
                )

                variance = float(
                    contrast
                    @ cov_fe
                    @ contrast
                )

                se = math.sqrt(
                    max(variance, 0.0)
                )

                if se > 0:
                    z_stat = estimate / se
                    p_raw = float(
                        2
                        * stats.norm.sf(
                            abs(z_stat)
                        )
                    )
                else:
                    z_stat = np.nan
                    p_raw = np.nan

                z_crit = stats.norm.ppf(
                    1 - (1 - CI_LEVEL) / 2
                )

                rows.append(
                    {
                        PRODUCT: product,
                        "group_1": group_1,
                        "group_2": group_2,
                        "difference": estimate,
                        "se": se,
                        "ci95_low": estimate - z_crit * se,
                        "ci95_high": estimate + z_crit * se,
                        "p_raw": p_raw,
                    }
                )

    contrasts = pd.DataFrame(rows)

    if contrasts.empty:
        return contrasts

    valid = contrasts["p_raw"].notna()

    contrasts["p_holm"] = np.nan
    contrasts["holm_supported"] = False

    if valid.any():
        _, adjusted, _, _ = multipletests(
            contrasts.loc[
                valid,
                "p_raw",
            ].to_numpy(),
            alpha=ALPHA,
            method="holm",
        )

        contrasts.loc[
            valid,
            "p_holm",
        ] = adjusted

        contrasts.loc[
            valid,
            "holm_supported",
        ] = adjusted < ALPHA

    return contrasts

def fit_sensitivity_model(
    df: pd.DataFrame,
    result: AnalysisResult,
):
    formula = build_sensitivity_formula(
        result.moderator_type
    )

    try:
        fit = smf.ols(
            formula,
            data=df,
        ).fit(
            cov_type="HC3"
        )

        return fit

    except Exception as exc:
        result.warnings.append(
            "Required sensitivity model could not be fitted: "
            f"{exc}"
        )

        result.model_sensitive = True
        result.sensitivity_stable = False

        return None

def compare_sensitivity(
    result: AnalysisResult,
) -> None:
    """
    Determine whether the primary interaction conclusion is stable.

    At minimum compare omnibus interaction support. If the primary
    analysis includes Holm-supported simple contrasts, preserve their
    direction where the corresponding sensitivity contrast can be
    evaluated indirectly through predicted coefficients.

    For this workshop dashboard the principal stability criterion is
    whether the product × moderator interaction remains supported at
    the same alpha level.
    """
    if not result.sensitivity_required:
        result.sensitivity_stable = None
        return

    if result.sensitivity_result is None:
        result.sensitivity_stable = False
        result.model_sensitive = True
        return

    primary_support = (
        result.interaction_p is not None
        and np.isfinite(result.interaction_p)
        and result.interaction_p < ALPHA
    )

    sensitivity_support = (
        result.sensitivity_interaction_p is not None
        and np.isfinite(
            result.sensitivity_interaction_p
        )
        and result.sensitivity_interaction_p < ALPHA
    )

    result.interaction_supported = primary_support
    result.sensitivity_interaction_supported = (
        sensitivity_support
    )

    stable = (
        primary_support == sensitivity_support
    )

    result.sensitivity_stable = stable
    result.model_sensitive = not stable

    if stable:
        result.diagnostics.append(
            DiagnosticRow(
                diagnostic="Sensitivity analysis",
                observed=(
                    "Primary and panelist fixed-effect HC3 "
                    "interaction conclusions agree."
                ),
                threshold=(
                    "Direction/substantive interaction conclusion "
                    "must remain consistent"
                ),
                status="PASS",
                action="Sensitivity check passed / stable.",
                tooltip=(
                    "The model diagnostics triggered an approved "
                    "alternative repeated-measures analysis. The "
                    "main conclusion about whether product response "
                    "varies with the moderator did not change."
                ),
            )
        )
    else:
        result.diagnostics.append(
            DiagnosticRow(
                diagnostic="Sensitivity analysis",
                observed=(
                    "Primary and sensitivity interaction "
                    "conclusions differ."
                ),
                threshold=(
                    "Direction/substantive interaction conclusion "
                    "must remain consistent"
                ),
                status="MODEL-SENSITIVE",
                action=(
                    "Do not present one definitive interaction "
                    "conclusion."
                ),
                tooltip=(
                    "The result changed materially under the "
                    "approved sensitivity model. This means the "
                    "conclusion depends on modeling assumptions, "
                    "so the dashboard suppresses overconfident "
                    "subgroup claims."
                ),
            )
        )

def build_answer(
    result: AnalysisResult,
) -> str:
    label = result.moderator_label

    if result.hard_stop:
        if result.subgroup_counts is not None:
            low = result.subgroup_counts[
                result.subgroup_counts["N_consumers"]
                < MIN_EXPLORATORY_SUBGROUP_N
            ]

            if not low.empty:
                smallest = low.sort_values(
                    "N_consumers"
                ).iloc[0]

                return (
                    f"The {label.lower()} interaction cannot be "
                    f"interpreted as coded because "
                    f"'{smallest[MODERATOR]}' contains only "
                    f"{int(smallest['N_consumers'])} consumer(s). "
                    "SOP-11 requires at least 5 unique consumers "
                    "in every categorical moderator level and does "
                    "not permit silently dropping or combining "
                    "sparse categories."
                )

        return (
            f"The SOP-11 {label.lower()} interaction could not "
            "be estimated defensibly because a required "
            "eligibility or model diagnostic hard stop occurred."
        )

    if result.model_sensitive:
        return (
            f"The evidence that {label.lower()} changes product "
            "response is model-sensitive. The primary and approved "
            "sensitivity models did not lead to the same substantive "
            "interaction conclusion, so no definitive subgroup "
            "claim is presented."
        )

    if result.interaction_p is None:
        return (
            f"The product × {label.lower()} interaction could "
            "not be tested reliably."
        )

    exploratory_prefix = (
        "Exploratorily, "
        if result.analysis_level == "EXPLORATORY"
        else ""
    )

    if result.interaction_p < ALPHA:
        return (
            f"{exploratory_prefix}product liking patterns differed "
            f"across {label.lower()} levels "
            f"(interaction p {fmt_p(result.interaction_p)}). "
            "This indicates that the relative response to products "
            "was not the same across the measured consumer groups; "
            "it does not establish a demographic cause or a "
            "population-wide truth."
        )

    return (
        f"{exploratory_prefix}the current data did not provide "
        f"statistically detectable evidence that {label.lower()} "
        f"changed the product-liking pattern "
        f"(interaction p {fmt_p(result.interaction_p)}). "
        "This is not evidence that all consumer groups respond "
        "identically."
    )

def run_analysis(
    raw_df: pd.DataFrame,
) -> Tuple[pd.DataFrame, AnalysisResult]:
    # ---------------------------------------------------------------------
    # 1. Dataset capability
    # ---------------------------------------------------------------------
    df, result = dataset_capability_gate(
        raw_df
    )

    if result.hard_stop:
        result.answer = build_answer(result)
        return df, result

    # ---------------------------------------------------------------------
    # 2. Study-design and G9-E validation
    # ---------------------------------------------------------------------
    analysis_df = validate_study_design(
        df,
        result,
    )

    if result.hard_stop:
        result.answer = build_answer(result)
        return analysis_df, result

    # ---------------------------------------------------------------------
    # 3. Primary mixed model
    # ---------------------------------------------------------------------
    fit = fit_mixed_model(
        analysis_df,
        result,
    )

    result.model_result = fit

    if fit is None or result.hard_stop:
        result.answer = build_answer(result)
        return analysis_df, result

    # ---------------------------------------------------------------------
    # 4. Interaction test
    # ---------------------------------------------------------------------
    (
        result.interaction_stat,
        result.interaction_df,
        result.interaction_p,
    ) = joint_wald_test(
        fit,
        result.moderator_type,
    )

    if (
        result.interaction_p is None
        or not np.isfinite(
            result.interaction_p
        )
    ):
        result.hard_stop = True
        result.model_status = "HARD STOP"
        result.hard_stop_reasons.append(
            "The joint product × moderator interaction test "
            "could not be estimated reliably."
        )

        result.answer = build_answer(result)
        return analysis_df, result

    result.interaction_supported = (
        result.interaction_p < ALPHA
    )

    # ---------------------------------------------------------------------
    # 5. G9-A diagnostics
    # ---------------------------------------------------------------------
    calculate_mixed_diagnostics(
        analysis_df,
        fit,
        result,
    )

    if result.hard_stop:
        result.answer = build_answer(result)
        return analysis_df, result

    # ---------------------------------------------------------------------
    # 6. Estimated means / simple contrasts
    # ---------------------------------------------------------------------
    try:
        if result.moderator_type == "categorical":
            result.estimated_means = (
                estimate_categorical_cells(
                    analysis_df,
                    fit,
                )
            )

            result.simple_contrasts = (
                categorical_simple_contrasts(
                    analysis_df,
                    fit,
                )
            )

        else:
            result.estimated_means = (
                estimate_continuous_cells(
                    analysis_df,
                    fit,
                )
            )

    except Exception as exc:
        result.warnings.append(
            "Model inference succeeded, but one or more "
            "estimated-mean visual calculations could not be "
            f"completed: {exc}"
        )

    # ---------------------------------------------------------------------
    # 7. Required sensitivity analysis
    # ---------------------------------------------------------------------
    if result.sensitivity_required:
        sensitivity = fit_sensitivity_model(
            analysis_df,
            result,
        )

        result.sensitivity_result = sensitivity

        if sensitivity is not None:
            (
                result.sensitivity_interaction_stat,
                result.sensitivity_interaction_df,
                result.sensitivity_interaction_p,
            ) = joint_wald_test(
                sensitivity,
                result.moderator_type,
            )

        compare_sensitivity(result)

    # ---------------------------------------------------------------------
    # 8. Final model status
    # ---------------------------------------------------------------------
    if result.model_sensitive:
        result.model_status = "MODEL-SENSITIVE"

    elif result.sensitivity_required:
        result.model_status = (
            "COMPLETED — SENSITIVITY STABLE"
            if result.sensitivity_stable
            else "COMPLETED — WARNING"
        )

    elif result.analysis_level == "EXPLORATORY":
        result.model_status = "COMPLETED — EXPLORATORY"

    else:
        result.model_status = "COMPLETED"

    result.answer = build_answer(result)

    return analysis_df, result

def main() -> None:
    data_path = resolve_data_path()
    dashboard_path = get_output_path()

    try:
        raw_df = pd.read_csv(
            data_path
        )
    except Exception as exc:
        raise RuntimeError(
            f"Could not read dataset: {data_path}\n{exc}"
        ) from exc

    analysis_df, result = run_analysis(
        raw_df
    )

    dashboard_html = build_dashboard(
        result,
        data_path,
    )

    dashboard_path.write_text(
        dashboard_html,
        encoding="utf-8",
    )

    save_technical_exports(
        result,
        dashboard_path,
    )

    print_console_summary(
        result,
        dashboard_path,
    )

    if AUTO_OPEN_DASHBOARD:
        try:
            webbrowser.open(
                dashboard_path.resolve().as_uri()
            )
        except Exception:
            # Dashboard creation is the required output.
            # Browser opening failure should not invalidate analysis.
            pass

if __name__ == "__main__":
    main()

def remove_export_index_columns(
    df: pd.DataFrame,
) -> Tuple[
    pd.DataFrame,
    List[str],
]:
    """
    Remove obvious CSV-export index columns.

    This does NOT remove arbitrary columns based only on position.
    """

    out = df.copy()

    removed: List[str] = []

    for col in list(out.columns):

        name = (
            str(col)
            .strip()
            .lower()
        )

        possible_export_index = (
            name.startswith("unnamed:")
            or name
            in {
                "index",
                "row.names",
                "row_names",
            }
        )

        if not possible_export_index:
            continue

        series = out[col]

        numeric = pd.to_numeric(
            series,
            errors="coerce",
        )

        if not numeric.notna().all():
            continue

        vals = numeric.to_numpy()

        n = len(vals)

        zero_based = np.array_equal(
            vals,
            np.arange(n),
        )

        one_based = np.array_equal(
            vals,
            np.arange(
                1,
                n + 1,
            ),
        )

        if (
            zero_based
            or one_based
            or name.startswith(
                "unnamed:"
            )
        ):

            removed.append(
                str(col)
            )

            out = out.drop(
                columns=[
                    col,
                ]
            )

    return (
        out,
        removed,
    )

def prepare_analysis_data(
    raw_df: pd.DataFrame,
) -> Tuple[
    pd.DataFrame,
    List[str],
    List[str],
]:
    """
    Apply the SOP-12 Dataset Capability Gate.

    Required substantive variables:
        panelist_id
        product_code
        overall_liking

    The method will not substitute:
        purchase intent;
        descriptive sensory ratings;
        JAR ratings;
        demographics;
        flavor impact;
        comments;
        product metadata.
    """

    (
        df,
        removed,
    ) = remove_export_index_columns(
        raw_df
    )

    design_notes: List[str] = []

    missing_required = [
        col
        for col
        in REQUIRED_COLUMNS
        if col
        not in df.columns
    ]

    if missing_required:

        raise AnalysisHalt(
            (
                "SOP-12 CAPABILITY GATE: "
                "UNSUPPORTED — required variables are missing."
            ),
            [
                (
                    "Missing required variable(s): "
                    + ", ".join(
                        missing_required
                    )
                    + "."
                ),
                (
                    "SOP-12 requires panelist_id, "
                    "product_code, and overall_liking."
                ),
                (
                    "The script will not invent, infer, "
                    "or substitute a missing construct."
                ),
            ],
        )

    work = df[
        REQUIRED_COLUMNS
    ].copy()


    # -------------------------------------------------------------------------
    # Required identifiers
    # -------------------------------------------------------------------------

    missing_identifier_rows = (
        work[
            [
                "panelist_id",
                "product_code",
            ]
        ]
        .isna()
        .any(
            axis=1
        )
    )

    if missing_identifier_rows.any():

        n_bad = int(
            missing_identifier_rows.sum()
        )

        raise AnalysisHalt(
            (
                "SOP-12 HARD STOP — panelist or product "
                "identifiers are missing."
            ),
            [
                (
                    "Rows with a missing panelist_id "
                    f"or product_code: {n_bad}."
                ),
            ],
        )


    work[
        "product_code"
    ] = (
        work[
            "product_code"
        ]
        .astype(
            "string"
        )
        .str.strip()
    )

    blank_product = (
        work[
            "product_code"
        ]
        == ""
    )

    if blank_product.any():

        raise AnalysisHalt(
            (
                "SOP-12 HARD STOP — blank product "
                "identifiers are present."
            ),
            [
                (
                    "Resolve blank product_code values "
                    "before clustering."
                ),
            ],
        )


    # -------------------------------------------------------------------------
    # Overall-liking scale validity
    # -------------------------------------------------------------------------

    original_nonmissing = (
        work[
            "overall_liking"
        ]
        .notna()
    )

    work[
        "overall_liking"
    ] = pd.to_numeric(
        work[
            "overall_liking"
        ],
        errors="coerce",
    )

    converted_to_na = (
        original_nonmissing
        & work[
            "overall_liking"
        ].isna()
    )

    if converted_to_na.any():

        raise AnalysisHalt(
            (
                "SOP-12 HARD STOP — non-numeric "
                "overall_liking values were detected."
            ),
            [
                (
                    "Non-numeric overall_liking values: "
                    f"{int(converted_to_na.sum())}."
                ),
            ],
        )


    invalid_range = (
        work[
            "overall_liking"
        ]
        .dropna()
    )

    invalid_range = invalid_range[
        (
            invalid_range
            < LIKING_MIN
        )
        |
        (
            invalid_range
            > LIKING_MAX
        )
    ]

    if len(
        invalid_range
    ) > 0:

        raise AnalysisHalt(
            (
                "SOP-12 HARD STOP — overall_liking "
                "contains values outside the approved "
                "1–9 hedonic scale."
            ),
            [
                (
                    "Out-of-range values detected: "
                    f"{len(invalid_range)}."
                ),
            ],
        )


    # -------------------------------------------------------------------------
    # One consumer × product rating per cell
    # -------------------------------------------------------------------------

    duplicate_mask = (
        work.duplicated(
            [
                "panelist_id",
                "product_code",
            ],
            keep=False,
        )
    )

    if duplicate_mask.any():

        dup_count = int(
            duplicate_mask.sum()
        )

        raise AnalysisHalt(
            (
                "SOP-12 HARD STOP — duplicate "
                "consumer × product liking cells "
                "were detected."
            ),
            [
                (
                    "Rows involved in duplicated "
                    "panelist_id × product_code cells: "
                    f"{dup_count}."
                ),
                (
                    "SOP-12 requires one liking score "
                    "per consumer × product cell."
                ),
                (
                    "Duplicate observations are not "
                    "silently averaged."
                ),
            ],
        )


    # -------------------------------------------------------------------------
    # Basic scope
    # -------------------------------------------------------------------------

    n_consumers = int(
        work[
            "panelist_id"
        ].nunique()
    )

    n_products = int(
        work[
            "product_code"
        ].nunique()
    )

    if n_consumers < 2:

        raise AnalysisHalt(
            (
                "SOP-12 HARD STOP — fewer than "
                "two consumers are available."
            ),
            [
                (
                    "Unique consumers: "
                    f"{n_consumers}."
                ),
            ],
        )

    if n_products < 2:

        raise AnalysisHalt(
            (
                "SOP-12 HARD STOP — fewer than "
                "two products are available."
            ),
            [
                (
                    "Unique products: "
                    f"{n_products}."
                ),
            ],
        )


    if removed:

        design_notes.append(
            (
                "Removed exported index column(s): "
                + ", ".join(
                    removed
                )
                + "."
            )
        )


    return (
        work,
        removed,
        design_notes,
    )

def build_preference_matrix(
    df: pd.DataFrame,
) -> Tuple[
    pd.DataFrame,
    List[str],
]:
    """
    Construct the SOP-12 input matrix.

    Rows:
        consumers

    Columns:
        products

    Values:
        overall liking

    The matrix must be complete.

    Missing values are not silently imputed.
    """

    notes: List[str] = []

    pref = df.pivot(
        index="panelist_id",
        columns="product_code",
        values="overall_liking",
    )

    pref = (
        pref
        .sort_index(
            axis=0
        )
        .sort_index(
            axis=1
        )
    )


    # -------------------------------------------------------------------------
    # Missing consumer × product cells
    # -------------------------------------------------------------------------

    missing_cells = int(
        pref
        .isna()
        .sum()
        .sum()
    )

    if missing_cells > 0:

        consumers_incomplete = int(
            pref
            .isna()
            .any(
                axis=1
            )
            .sum()
        )

        products_incomplete = int(
            pref
            .isna()
            .any(
                axis=0
            )
            .sum()
        )

        raise AnalysisHalt(
            (
                "SOP-12 HARD STOP — the consumer × product "
                "liking matrix is incomplete."
            ),
            [
                (
                    "Missing consumer × product liking cells: "
                    f"{missing_cells}."
                ),
                (
                    "Consumers with at least one missing "
                    "product liking: "
                    f"{consumers_incomplete}."
                ),
                (
                    "Products with at least one missing "
                    "consumer liking: "
                    f"{products_incomplete}."
                ),
                (
                    "SOP-12 does not silently impute "
                    "missing liking values."
                ),
            ],
        )


    # -------------------------------------------------------------------------
    # Same product set for each consumer
    # -------------------------------------------------------------------------

    row_counts = (
        df.groupby(
            "panelist_id"
        )[
            "product_code"
        ]
        .nunique()
    )

    if (
        row_counts.nunique()
        != 1
    ):

        raise AnalysisHalt(
            (
                "SOP-12 HARD STOP — consumers do not "
                "share a common complete product set."
            ),
            [
                (
                    "The validated segmentation requires "
                    "a coherent consumer × product matrix."
                ),
            ],
        )


    expected_per_consumer = int(
        pref.shape[1]
    )

    if int(
        row_counts.iloc[0]
    ) != expected_per_consumer:

        raise AnalysisHalt(
            (
                "SOP-12 HARD STOP — the consumer × product "
                "study structure is inconsistent."
            ),
            [
                (
                    "Expected product ratings per consumer "
                    f"from the matrix: {expected_per_consumer}."
                ),
                (
                    "Observed unique product count per "
                    f"consumer: {int(row_counts.iloc[0])}."
                ),
            ],
        )


    # -------------------------------------------------------------------------
    # Global outcome variation
    # -------------------------------------------------------------------------

    overall_variance = float(
        np.nanvar(
            pref.to_numpy(
                dtype=float
            )
        )
    )

    if overall_variance == 0.0:

        raise AnalysisHalt(
            (
                "SOP-12 HARD STOP — overall liking has "
                "no variation in the preference matrix."
            ),
            [
                (
                    "KMeans cannot identify meaningful "
                    "preference structure when all liking "
                    "values are identical."
                ),
            ],
        )


    # -------------------------------------------------------------------------
    # Consumers with flat profiles
    # -------------------------------------------------------------------------

    zero_variance_consumers = (
        pref.std(
            axis=1,
            ddof=0,
        )
        == 0
    )

    if zero_variance_consumers.any():

        notes.append(
            (
                f"{int(zero_variance_consumers.sum())} "
                "consumer(s) gave the same liking score "
                "to every product. "
                "They remain eligible for raw-profile "
                "clustering. Consumer z-scoring would "
                "be undefined for those consumers."
            )
        )


    return (
        pref,
        notes,
    )

def transform_preference_matrix(
    pref: pd.DataFrame,
) -> Tuple[
    pd.DataFrame,
    str,
]:
    """
    Apply the explicitly selected preference-profile transformation.

    This transformation is documented because changing from raw liking to
    centered/z-scored profiles changes the segmentation question.
    """

    mode = (
        PROFILE_TRANSFORM
        .strip()
        .lower()
    )


    if mode not in ALLOWED_PROFILE_TRANSFORMS:

        raise AnalysisHalt(
            (
                "SOP-12 CONFIGURATION STOP — "
                "PROFILE_TRANSFORM is not recognized."
            ),
            [
                (
                    "Allowed values are: "
                    "raw, center_consumer, "
                    "zscore_consumer."
                ),
            ],
        )


    # -------------------------------------------------------------------------
    # Raw 1–9 liking profiles
    # -------------------------------------------------------------------------

    if mode == "raw":

        return (
            pref.astype(
                float
            ).copy(),
            (
                "Raw 1–9 overall-liking profiles"
            ),
        )


    # -------------------------------------------------------------------------
    # Consumer-centered profiles
    # -------------------------------------------------------------------------

    centered = pref.sub(
        pref.mean(
            axis=1
        ),
        axis=0,
    )


    if mode == "center_consumer":

        return (
            centered.astype(
                float
            ),
            (
                "Consumer-centered liking profiles"
            ),
        )


    # -------------------------------------------------------------------------
    # Within-consumer z-scored profiles
    # -------------------------------------------------------------------------

    if mode == "zscore_consumer":

        sd = pref.std(
            axis=1,
            ddof=0,
        )

        zero_sd = (
            sd
            <= 0
        )

        if zero_sd.any():

            ids = ", ".join(
                map(
                    str,
                    pref.index[
                        zero_sd
                    ]
                    .tolist()[
                        :10
                    ],
                )
            )

            raise AnalysisHalt(
                (
                    "SOP-12 HARD STOP — consumer "
                    "z-scoring was requested but at "
                    "least one consumer has zero "
                    "within-profile variance."
                ),
                [
                    (
                        "Consumers with zero profile SD: "
                        f"{int(zero_sd.sum())}."
                    ),
                    (
                        "Example panelist_id values: "
                        f"{ids}."
                    ),
                    (
                        "The script will not divide by "
                        "zero or silently drop consumers."
                    ),
                ],
            )


        z = centered.div(
            sd,
            axis=0,
        )

        return (
            z.astype(
                float
            ),
            (
                "Within-consumer z-scored liking profiles"
            ),
        )


    raise AnalysisHalt(
        (
            "SOP-12 CONFIGURATION STOP — "
            "profile transformation could not "
            "be resolved."
        )
    )

def fit_kmeans(
    X: np.ndarray,
    k: int,
    random_state: int,
) -> KMeans:
    """
    Fit one KMeans model using the approved reproducibility settings.
    """

    model = KMeans(
        n_clusters=k,
        n_init=N_INIT,
        random_state=random_state,
        algorithm="lloyd",
    )

    model.fit(
        X
    )

    return model

def stability_against_full_solution(
    X: np.ndarray,
    full_labels: np.ndarray,
    k: int,
) -> Tuple[
    float,
    float,
    int,
]:
    """
    Repeated 80% subsampling stability.

    Procedure
    ---------
    1. Randomly sample 80% of consumers without replacement.
    2. Refit the same k-cluster KMeans model.
    3. Compare the subsample labels with the corresponding labels from the
       full-data fit for the same consumers.
    4. Use Adjusted Rand Index.
    5. Repeat STABILITY_RESAMPLES times.

    ARI is permutation-invariant; arbitrary KMeans numeric cluster labels do
    not need to be manually aligned.
    """

    n = int(
        X.shape[0]
    )

    sample_n = int(
        math.floor(
            STABILITY_SUBSAMPLE_FRACTION
            * n
        )
    )

    sample_n = max(
        sample_n,
        k + 1,
    )

    sample_n = min(
        sample_n,
        n,
    )


    if (
        sample_n <= k
        or n <= k
    ):

        return (
            float("nan"),
            float("nan"),
            0,
        )


    rng = np.random.default_rng(
        RANDOM_STATE
        + 1000 * k
    )

    ari_values: List[float] = []


    for b in range(
        STABILITY_RESAMPLES
    ):

        idx = np.sort(
            rng.choice(
                n,
                size=sample_n,
                replace=False,
            )
        )

        X_sub = X[
            idx
        ]

        try:

            sub_model = fit_kmeans(
                X_sub,
                k,
                (
                    RANDOM_STATE
                    + 10_000
                    + b
                    + 1000 * k
                ),
            )

            ari = adjusted_rand_score(
                full_labels[
                    idx
                ],
                sub_model.labels_,
            )

            if np.isfinite(
                ari
            ):

                ari_values.append(
                    float(
                        ari
                    )
                )

        except Exception:

            # A failed resampling fit is NOT converted to a favorable value.
            #
            # Successful runs remain available for the transparent stability
            # summary, and the number completed is disclosed.

            continue


    if not ari_values:

        return (
            float("nan"),
            float("nan"),
            0,
        )


    arr = np.asarray(
        ari_values,
        dtype=float,
    )

    median_ari = float(
        np.median(
            arr
        )
    )

    p10_ari = float(
        np.percentile(
            arr,
            10,
        )
    )

    return (
        median_ari,
        p10_ari,
        int(
            len(arr)
        ),
    )

def evaluate_candidate(
    pref_model: pd.DataFrame,
    k: int,
) -> CandidateResult:
    """
    Apply all G9-F validation rules to one candidate cluster count.
    """

    X = pref_model.to_numpy(
        dtype=float
    )

    n_consumers = int(
        X.shape[0]
    )


    # -------------------------------------------------------------------------
    # Mathematical minimum-size eligibility
    # -------------------------------------------------------------------------

    # If N < k * MIN_CLUSTER_N, there is no possible KMeans partition that can
    # satisfy the approved minimum cluster-size rule.

    minimum_total_needed = (
        MIN_CLUSTER_N
        * k
    )

    if (
        n_consumers
        < minimum_total_needed
    ):

        return CandidateResult(
            k=k,
            fitted=False,
            silhouette=float(
                "nan"
            ),
            cluster_sizes=[],
            min_cluster_size=0,
            median_ari=float(
                "nan"
            ),
            p10_ari=float(
                "nan"
            ),
            n_stability_runs=0,
            passes_silhouette=False,
            passes_size=False,
            passes_stability=False,
            retainable=False,
            note=(
                f"N={n_consumers} cannot support "
                f"{k} clusters with at least "
                f"{MIN_CLUSTER_N} consumers each."
            ),
        )


    # -------------------------------------------------------------------------
    # Candidate fit
    # -------------------------------------------------------------------------

    try:

        model = fit_kmeans(
            X,
            k,
            RANDOM_STATE,
        )

        labels = (
            model.labels_
            .astype(
                int
            )
        )


        # ---------------------------------------------------------------------
        # Silhouette
        # ---------------------------------------------------------------------

        unique_labels = np.unique(
            labels
        )

        if (
            len(
                unique_labels
            )
            < 2
            or len(
                unique_labels
            )
            >= n_consumers
        ):

            silhouette = float(
                "nan"
            )

        else:

            silhouette = float(
                silhouette_score(
                    X,
                    labels,
                    metric="euclidean",
                )
            )


        # ---------------------------------------------------------------------
        # Cluster sizes
        # ---------------------------------------------------------------------

        sizes = (
            np.bincount(
                labels,
                minlength=k,
            )
            .astype(
                int
            )
            .tolist()
        )

        min_size = (
            int(
                min(
                    sizes
                )
            )
            if sizes
            else 0
        )


        # ---------------------------------------------------------------------
        # Resampling stability
        # ---------------------------------------------------------------------

        (
            median_ari,
            p10_ari,
            n_runs,
        ) = stability_against_full_solution(
            X,
            labels,
            k,
        )


        # ---------------------------------------------------------------------
        # G9-F decisions
        # ---------------------------------------------------------------------

        pass_silhouette = bool(
            np.isfinite(
                silhouette
            )
            and silhouette
            >= MIN_SILHOUETTE
        )

        pass_size = bool(
            min_size
            >= MIN_CLUSTER_N
        )

        pass_stability = bool(
            np.isfinite(
                median_ari
            )
            and np.isfinite(
                p10_ari
            )
            and median_ari
            >= MIN_MEDIAN_ARI
            and p10_ari
            >= MIN_P10_ARI
        )


        retainable = bool(
            pass_silhouette
            and pass_size
            and pass_stability
        )


        return CandidateResult(
            k=k,
            fitted=True,
            silhouette=silhouette,
            cluster_sizes=sizes,
            min_cluster_size=min_size,
            median_ari=median_ari,
            p10_ari=p10_ari,
            n_stability_runs=n_runs,
            passes_silhouette=pass_silhouette,
            passes_size=pass_size,
            passes_stability=pass_stability,
            retainable=retainable,
            labels=labels,
            centers_model_space=(
                model
                .cluster_centers_
                .copy()
            ),
        )


    except Exception as exc:

        return CandidateResult(
            k=k,
            fitted=False,
            silhouette=float(
                "nan"
            ),
            cluster_sizes=[],
            min_cluster_size=0,
            median_ari=float(
                "nan"
            ),
            p10_ari=float(
                "nan"
            ),
            n_stability_runs=0,
            passes_silhouette=False,
            passes_size=False,
            passes_stability=False,
            retainable=False,
            note=(
                "Candidate fit failed: "
                f"{type(exc).__name__}: "
                f"{exc}"
            ),
        )

def choose_retained_candidate(
    candidates: Sequence[
        CandidateResult
    ],
) -> Optional[
    CandidateResult
]:
    """
    Select among candidates ONLY after G9-F eligibility.

    Failed solutions never compete with passing solutions.

    When more than one candidate passes:
        1. higher silhouette is preferred;
        2. higher median ARI is the secondary tie-breaker;
        3. smaller k is preferred if otherwise tied.

    This keeps the selection validation-led and reasonably parsimonious.
    """

    passing = [
        candidate
        for candidate
        in candidates
        if candidate.retainable
    ]


    if not passing:

        return None


    passing = sorted(
        passing,
        key=lambda candidate: (
            -candidate.silhouette,
            -candidate.median_ari,
            candidate.k,
        ),
    )


    return passing[
        0
    ]

def normalize_segment_labels(
    pref_raw: pd.DataFrame,
    candidate: CandidateResult,
) -> Tuple[
    pd.Series,
    pd.DataFrame,
    pd.Series,
]:
    """
    KMeans numeric labels have no inherent ordering.

    For stable dashboard display only, relabel clusters 1..k according to:
        1. descending highest product mean;
        2. descending overall segment mean;
        3. original KMeans label.

    This does NOT alter:
        - KMeans fitting;
        - silhouette;
        - ARI;
        - candidate validation.
    """

    if candidate.labels is None:

        raise ValueError(
            (
                "Cannot normalize segment labels "
                "without retained KMeans labels."
            )
        )


    raw_labels = pd.Series(
        candidate.labels,
        index=pref_raw.index,
        name="raw_cluster",
    )


    tmp = pref_raw.copy()

    tmp[
        "__cluster__"
    ] = raw_labels


    raw_profiles = (
        tmp.groupby(
            "__cluster__"
        )
        .mean(
            numeric_only=True
        )
    )


    order_rows = []


    for (
        cluster_id,
        row,
    ) in raw_profiles.iterrows():

        vals = row.to_numpy(
            dtype=float
        )

        order_rows.append(
            (
                int(
                    cluster_id
                ),
                float(
                    np.max(
                        vals
                    )
                ),
                float(
                    np.mean(
                        vals
                    )
                ),
            )
        )


    ordered = sorted(
        order_rows,
        key=lambda item: (
            -item[1],
            -item[2],
            item[0],
        ),
    )


    mapping = {
        old_label: new_label
        for (
            new_label,
            (
                old_label,
                _,
                __,
            ),
        )
        in enumerate(
            ordered,
            start=1,
        )
    }


    assignments = (
        raw_labels
        .map(
            mapping
        )
        .astype(
            int
        )
    )

    assignments.name = "segment"


    profile_work = pref_raw.copy()

    profile_work[
        "segment"
    ] = assignments


    profiles = (
        profile_work
        .groupby(
            "segment"
        )
        .mean(
            numeric_only=True
        )
        .sort_index()
    )


    sizes = (
        assignments
        .value_counts()
        .sort_index()
    )


    return (
        assignments,
        profiles,
        sizes,
    )

def segment_story(
    profiles: pd.DataFrame,
    sizes: pd.Series,
) -> List[str]:
    """
    Describe liking patterns only.

    Do not attach demographics, motives, sensory causes, purchase behavior,
    or psychographic identities to clusters.
    """

    stories: List[str] = []


    for (
        segment_id,
        row,
    ) in profiles.iterrows():

        ranked = row.sort_values(
            ascending=False
        )

        top_product = str(
            ranked.index[
                0
            ]
        )

        bottom_product = str(
            ranked.index[
                -1
            ]
        )

        top_value = float(
            ranked.iloc[
                0
            ]
        )

        bottom_value = float(
            ranked.iloc[
                -1
            ]
        )

        n = int(
            sizes.loc[
                segment_id
            ]
        )


        stories.append(
            (
                f"Segment {segment_id} (N={n}) "
                f"rated Product {top_product} highest "
                f"on average ({top_value:.1f}/9) and "
                f"Product {bottom_product} lowest "
                f"({bottom_value:.1f}/9)."
            )
        )


    return stories

def hex_to_rgb(
    hex_color: str,
) -> Tuple[
    int,
    int,
    int,
]:

    value = (
        hex_color
        .lstrip(
            "#"
        )
    )

    return (
        int(
            value[
                0:2
            ],
            16,
        ),
        int(
            value[
                2:4
            ],
            16,
        ),
        int(
            value[
                4:6
            ],
            16,
        ),
    )

def rgb_to_hex(
    rgb: Tuple[
        int,
        int,
        int,
    ],
) -> str:

    return (
        "#"
        + "".join(
            f"{max(0, min(255, int(v))):02x}"
            for v
            in rgb
        )
    )

def mix_colors(
    color_1: str,
    color_2: str,
    t: float,
) -> str:

    t = max(
        0.0,
        min(
            1.0,
            float(
                t
            ),
        ),
    )

    first = hex_to_rgb(
        color_1
    )

    second = hex_to_rgb(
        color_2
    )

    mixed = tuple(
        round(
            first[i]
            + (
                second[i]
                - first[i]
            )
            * t
        )
        for i
        in range(
            3
        )
    )

    return rgb_to_hex(
        mixed
    )

def liking_fill(
    value: float,
) -> str:
    """
    Restrained sequential palette:
        pale neutral blue-gray -> deep teal.
    """

    t = (
        (
            float(
                value
            )
            - LIKING_MIN
        )
        /
        (
            LIKING_MAX
            - LIKING_MIN
        )
    )

    return mix_colors(
        "#eef3f5",
        "#1b4b5a",
        t,
    )

def liking_text_fill(
    value: float,
) -> str:

    if float(
        value
    ) >= 6.0:

        return "#ffffff"

    return "#172b4d"

def run_analysis(
    data_path: Path,
) -> AnalysisBundle:
    """
    Complete SOP-12 executable workflow.
    """

    # -------------------------------------------------------------------------
    # Data-file discovery
    # -------------------------------------------------------------------------

    if not data_path.exists():

        raise AnalysisHalt(
            (
                "SOP-12 DATA STOP — "
                "the dataset file was not found."
            ),
            [
                (
                    "Expected or requested path: "
                    f"{data_path}"
                ),
                (
                    "Default expected path: "
                    "data/"
                    "protein_shake_category_appraisal_mock_redesigned.csv"
                ),
            ],
        )


    # -------------------------------------------------------------------------
    # Read CSV
    # -------------------------------------------------------------------------

    try:

        raw_df = pd.read_csv(
            data_path
        )

    except Exception as exc:

        raise AnalysisHalt(
            (
                "SOP-12 DATA STOP — "
                "the dataset could not be read as CSV."
            ),
            [
                (
                    f"{type(exc).__name__}: "
                    f"{exc}"
                ),
            ],
        ) from exc


    # -------------------------------------------------------------------------
    # Dataset Capability Gate
    # -------------------------------------------------------------------------

    (
        analysis_df,
        _removed,
        design_notes,
    ) = prepare_analysis_data(
        raw_df
    )


    # -------------------------------------------------------------------------
    # Study-design validation
    # -------------------------------------------------------------------------

    (
        pref_raw,
        matrix_notes,
    ) = build_preference_matrix(
        analysis_df
    )


    design_notes.extend(
        matrix_notes
    )


    # -------------------------------------------------------------------------
    # Required preprocessing
    # -------------------------------------------------------------------------

    (
        pref_model,
        transform_label,
    ) = transform_preference_matrix(
        pref_raw
    )


    # -------------------------------------------------------------------------
    # Candidate statistical analysis
    # -------------------------------------------------------------------------

    candidates = [
        evaluate_candidate(
            pref_model,
            k,
        )
        for k
        in CANDIDATE_K
    ]


    # -------------------------------------------------------------------------
    # Hard post-fit validation logic
    # -------------------------------------------------------------------------

    retained = choose_retained_candidate(
        candidates
    )


    # -------------------------------------------------------------------------
    # Dataset summary
    # -------------------------------------------------------------------------

    n_rows = int(
        len(
            analysis_df
        )
    )

    n_consumers = int(
        pref_raw.shape[
            0
        ]
    )

    n_products = int(
        pref_raw.shape[
            1
        ]
    )

    products = list(
        map(
            str,
            pref_raw.columns,
        )
    )


    # -------------------------------------------------------------------------
    # No passing candidate
    # -------------------------------------------------------------------------

    if retained is None:

        return AnalysisBundle(
            capability="SUPPORTED",
            answer=(
                "The current liking data do not support "
                "a stable consumer segmentation."
            ),
            model_status=(
                "HARD POST-FIT VALIDATION HALT"
            ),
            n_rows=n_rows,
            n_consumers=n_consumers,
            n_products=n_products,
            products=products,
            transform_label=transform_label,
            candidate_results=candidates,
            retained=None,
            pref_raw=pref_raw,
            pref_model=pref_model,
            assignments=None,
            profiles=None,
            segment_sizes=None,
            design_notes=design_notes,
        )


    # -------------------------------------------------------------------------
    # Retained candidate passed all thresholds
    # -------------------------------------------------------------------------

    (
        assignments,
        profiles,
        segment_sizes,
    ) = normalize_segment_labels(
        pref_raw,
        retained,
    )


    answer = (
        "The liking data support a validated "
        f"{retained.k}-segment exploratory "
        "preference solution. "
        "The retained solution passed separation, "
        "minimum-size, and resampling-stability "
        "requirements."
    )


    return AnalysisBundle(
        capability="SUPPORTED",
        answer=answer,
        model_status=(
            "VALIDATED EXPLORATORY SEGMENTATION"
        ),
        n_rows=n_rows,
        n_consumers=n_consumers,
        n_products=n_products,
        products=products,
        transform_label=transform_label,
        candidate_results=candidates,
        retained=retained,
        pref_raw=pref_raw,
        pref_model=pref_model,
        assignments=assignments,
        profiles=profiles,
        segment_sizes=segment_sizes,
        design_notes=design_notes,
    )

def main() -> None:

    data_path = resolve_data_path()

    out_dir = ensure_output_dir()

    dashboard_path = (
        out_dir
        / DASHBOARD_FILENAME
    )


    try:

        bundle = run_analysis(
            data_path
        )


        # ---------------------------------------------------------------------
        # Validated solution
        # ---------------------------------------------------------------------

        if (
            bundle.retained
            is not None
        ):

            dashboard = build_success_dashboard(
                bundle
            )


        # ---------------------------------------------------------------------
        # Post-fit hard validation halt
        # ---------------------------------------------------------------------

        else:

            dashboard = build_validation_halt_dashboard(
                bundle
            )


        write_dashboard(
            dashboard_path,
            dashboard,
        )


        save_optional_exports(
            bundle,
            out_dir,
        )


        print_console_summary(
            bundle,
            dashboard_path,
        )


        open_dashboard(
            dashboard_path
        )


    # -------------------------------------------------------------------------
    # Controlled SOP / capability / design hard stop
    # -------------------------------------------------------------------------

    except AnalysisHalt as exc:

        dashboard = build_preanalysis_halt_dashboard(
            exc.message,
            exc.details,
            data_path,
        )


        write_dashboard(
            dashboard_path,
            dashboard,
        )


        print_console_summary(
            None,
            dashboard_path,
            halt_message=(
                exc.message
            ),
        )


        open_dashboard(
            dashboard_path
        )


    # -------------------------------------------------------------------------
    # Unexpected runtime/programming failure
    # -------------------------------------------------------------------------

    except Exception as exc:

        message = (
            "SOP-12 RUNTIME FAILURE — "
            "analysis could not be completed."
        )


        details = [
            (
                f"{type(exc).__name__}: "
                f"{exc}"
            )
        ]


        dashboard = build_preanalysis_halt_dashboard(
            message,
            details,
            data_path,
        )


        write_dashboard(
            dashboard_path,
            dashboard,
        )


        print_console_summary(
            None,
            dashboard_path,
            halt_message=message,
        )


        open_dashboard(
            dashboard_path
        )


        # Re-raise unexpected programming failures so the IDE still provides
        # the traceback needed for debugging.
        raise

if __name__ == "__main__":

    main()

    """

    df = pd.read_csv(

    unnamed = [
        column

        if str(

    if unnamed:
        df = df.drop(
            columns=unnamed
        )

    return df

def dataset_capability_gate(
    df: pd.DataFrame,
) -> CapabilityResult:
    """

    missing = [

            ),
        ],
    )

def validate_study_design(
    df: pd.DataFrame,
) -> DesignResult:
    """
    Validate the consumer × product repeated-measures structure.

    SOP-13 requires:
    - one value per panelist × product;
    - complete liking matrix;
    - sufficient consumers;
    - sufficient products;
    - numeric valid liking scores.
    """

    result = DesignResult()

    result.n_rows = len(
        df
    )

    work = df.copy()

    work[PANELIST] = (
        work[PANELIST]
        .astype(
            "string"
        )
    )

    work[PRODUCT] = (
        work[PRODUCT]
        .astype(
            "string"
        )
    )

    work[OUTCOME] = pd.to_numeric(
        work[OUTCOME],
        errors="coerce",
    )

    result.n_consumers = int(
        work[
            PANELIST
        ].nunique(
            dropna=True
        )
    )

    result.n_products = int(
        work[
            PRODUCT
        ].nunique(
            dropna=True
        )
    )

    result.missing_outcome = int(
        work[
            OUTCOME
        ].isna().sum()
    )

    valid_keys = work.dropna(
        subset=[
            PANELIST,
            PRODUCT,
        ]
    )

    duplicate_mask = (
        valid_keys
        .duplicated(
            [
                PANELIST,
                PRODUCT,
            ],
            keep=False,
        )
    )

    result.duplicates = int(
        duplicate_mask.sum()
    )

    result.observed_unique_cells = int(
        valid_keys[
            [
                PANELIST,
                PRODUCT,
            ]
        ]
        .drop_duplicates()
        .shape[0]
    )

    result.expected_cells = (
        result.n_consumers
        * result.n_products
    )

    # ---------------------------------------------------------------------
    # STRUCTURAL ELIGIBILITY
    # ---------------------------------------------------------------------

    if (
        result.n_consumers
        < MIN_CONSUMERS
    ):
        result.hard_stop_reasons.append(
            (
                f"Only {result.n_consumers} "
                "consumers are available; at "
                f"least {MIN_CONSUMERS} are "
                "required for this 2D "
                "preference-map workflow."
            )
        )

    if (
        result.n_products
        < MIN_PRODUCTS
    ):
        result.hard_stop_reasons.append(
            (
                f"Only {result.n_products} "
                "products are available; at "
                f"least {MIN_PRODUCTS} are "
                "required for a 2D preference "
                "map."
            )
        )

    if (
        result.duplicates
        > 0
    ):
        result.hard_stop_reasons.append(
            (
                f"Detected {result.duplicates} "
                "rows involved in duplicate "
                "panelist × product cells. "
                "SOP-13 requires one liking "
                "score per consumer × product "
                "and does not silently average "
                "duplicates."
            )
        )

    if (
        work[
            PANELIST
        ].isna().any()
        or work[
            PRODUCT
        ].isna().any()
    ):
        result.hard_stop_reasons.append(
            (
                "Missing panelist_id or "
                "product_code values prevent "
                "a defensible consumer × "
                "product matrix."
            )
        )

    # ---------------------------------------------------------------------
    # SCALE VALIDITY
    # ---------------------------------------------------------------------

    numeric = pd.to_numeric(
        work[
            OUTCOME
        ],
        errors="coerce",
    )

    finite = numeric[
        np.isfinite(
            numeric
        )
    ]

    if len(
        finite
    ) == 0:
        result.hard_stop_reasons.append(
            (
                "overall_liking contains no "
                "finite numeric values."
            )
        )

    else:
        invalid_range = (
            (finite < 1)
            | (finite > 9)
        )

        if invalid_range.any():
            bad_n = int(
                invalid_range.sum()
            )

            result.hard_stop_reasons.append(
                (
                    f"Found {bad_n} "
                    "overall_liking value(s) "
                    "outside the approved "
                    "1–9 hedonic range."
                )
            )

    # ---------------------------------------------------------------------
    # MATRIX COMPLETENESS
    # ---------------------------------------------------------------------

    if not result.hard_stop_reasons:
        pivot = (
            work.pivot(
                index=PANELIST,
                columns=PRODUCT,
                values=OUTCOME,
            )
        )

        result.complete_consumers = int(
            pivot
            .notna()
            .all(
                axis=1
            )
            .sum()
        )

        missing_cells = int(
            pivot
            .isna()
            .sum()
            .sum()
        )

        if (
            missing_cells
            > 0
        ):
            result.hard_stop_reasons.append(
                (
                    "The consumer × product "
                    "liking matrix contains "
                    f"{missing_cells} missing "
                    "cell(s). SOP-13 requires "
                    "a complete matrix and does "
                    "not silently impute liking "
                    "values."
                )
            )

        counts = (
            pivot
            .notna()
            .sum(
                axis=1
            )
        )

        too_sparse = counts[
            counts
            < MIN_VALID_LIKING_VALUES_PER_CONSUMER
        ]

        if len(
            too_sparse
        ) > 0:
            result.hard_stop_reasons.append(
                (
                    f"{len(too_sparse)} "
                    "consumer(s) have fewer "
                    f"than "
                    f"{MIN_VALID_LIKING_VALUES_PER_CONSUMER} "
                    "valid liking scores."
                )
            )

    if (
        result.missing_outcome
        > 0
    ):
        result.warnings.append(
            (
                "overall_liking contains "
                f"{result.missing_outcome} "
                "missing row value(s)."
            )
        )

    result.hard_stop = bool(
        result.hard_stop_reasons
    )

    return result

def build_preference_matrix(
    df: pd.DataFrame,
) -> pd.DataFrame:
    """
    Construct the approved consumer × product liking matrix.

    Rows:
        panelist_id

    Columns:
        product_code

    Values:
        overall_liking
    """

    work = df[
        [
            PANELIST,
            PRODUCT,
            OUTCOME,
        ]
    ].copy()

    work[PANELIST] = (
        work[PANELIST]
        .astype(
            str
        )
    )

    work[PRODUCT] = (
        work[PRODUCT]
        .astype(
            str
        )
    )

    work[OUTCOME] = pd.to_numeric(
        work[OUTCOME],
        errors="coerce",
    )

    pref = work.pivot(
        index=PANELIST,
        columns=PRODUCT,
        values=OUTCOME,
    )

    pref = (
        pref
        .sort_index(
            axis=0
        )
        .sort_index(
            axis=1
        )
    )

    return pref

def preprocess_preference_matrix(
    pref: pd.DataFrame,
) -> Tuple[
    pd.DataFrame,
    List[str],
]:
    """
    Apply the explicitly documented IPM transformation.

    Returns:
        transformed consumer × product matrix
        consumers with zero post-transformation variance
    """

    if (
        pref
        .isna()
        .any()
        .any()
    ):
        raise ValueError(
            (
                "Preference matrix contains "
                "missing values; no imputation "
                "is authorized."
            )
        )

    transformed = (
        pref
        .astype(
            float
        )
        .copy()
    )

    # ---------------------------------------------------------------------
    # MEAN-CENTER WITHIN CONSUMER
    # ---------------------------------------------------------------------

    if CENTER_WITHIN_CONSUMER:
        transformed = (
            transformed
            .sub(
                transformed.mean(
                    axis=1
                ),
                axis=0,
            )
        )

    # ---------------------------------------------------------------------
    # OPTIONAL WITHIN-CONSUMER STANDARDIZATION
    # ---------------------------------------------------------------------
    #
    # OFF by default because it changes the question.
    # ---------------------------------------------------------------------

    if STANDARDIZE_WITHIN_CONSUMER:
        consumer_sd = transformed.std(
            axis=1,
            ddof=0,
        )

        zero_before_scale = consumer_sd[
            np.isclose(
                consumer_sd,
                0.0,
                atol=1e-12,
            )
        ]

        if len(
            zero_before_scale
        ) > 0:
            return (
                transformed,
                zero_before_scale
                .index
                .astype(
                    str
                )
                .tolist(),
            )

        transformed = (
            transformed
            .div(
                consumer_sd,
                axis=0,
            )
        )

    # ---------------------------------------------------------------------
    # ZERO-VARIANCE CONSUMER CHECK
    # ---------------------------------------------------------------------

    consumer_sd = transformed.std(
        axis=1,
        ddof=0,
    )

    zero_variance = consumer_sd[
        np.isclose(
            consumer_sd,
            0.0,
            atol=1e-12,
        )
    ]

    zero_variance_consumers = (
        zero_variance
        .index
        .astype(
            str
        )
        .tolist()
    )

    return (
        transformed,
        zero_variance_consumers,
    )

def fit_ipm(
    pref: pd.DataFrame,
) -> IPMResult:
    """
    Fit PCA-based internal preference mapping.

    The input is consumer × product liking.

    After within-consumer centering, the matrix is transposed to:

        rows    = products
        columns = consumers

    Product PCA scores form the internal preference map.
    Consumer PCA loadings form individual preference vectors.
    """

    result = IPMResult(
        pref_matrix=pref
    )

    # ---------------------------------------------------------------------
    # PREPROCESS
    # ---------------------------------------------------------------------

    try:
        (
            centered,
            zero_var_consumers,
        ) = preprocess_preference_matrix(
            pref
        )

    except Exception as exc:
        result.hard_stop = True

        result.hard_stop_reasons.append(
            str(
                exc
            )
        )

        return result

    result.centered_matrix = centered

    result.zero_variance_consumers = (
        zero_var_consumers
    )

    # ---------------------------------------------------------------------
    # G9-C C1 — ZERO VARIANCE
    # ---------------------------------------------------------------------

    if zero_var_consumers:
        result.hard_stop = True

        display_ids = ", ".join(
            zero_var_consumers[
                :12
            ]
        )

        if (
            len(
                zero_var_consumers
            )
            > 12
        ):
            display_ids += "…"

        result.hard_stop_reasons.append(
            (
                "G9-C zero-variance stop: "
                "consumer(s) "
                f"{display_ids} "
                "have no variation in product "
                "liking after the documented "
                "centering step. They cannot "
                "define a preference direction, "
                "and SOP-13 does not authorize "
                "silently dropping them."
            )
        )

        result.diagnostics.append(
            {
                "Diagnostic": (
                    "Zero-variance "
                    "preference variables"
                ),
                "Observed Value": str(
                    len(
                        zero_var_consumers
                    )
                ),
                "Threshold": (
                    "0 zero-variance "
                    "consumer vectors"
                ),
                "Status": "HARD STOP",
                "Required Action": (
                    "Do not silently drop; "
                    "review data/design "
                    "before rerun."
                ),
            }
        )

        return result

    # ---------------------------------------------------------------------
    # PCA ORIENTATION
    # ---------------------------------------------------------------------
    #
    # centered:
    #
    #     consumers × products
    #
    # PCA input:
    #
    #     products × consumers
    #
    # ---------------------------------------------------------------------

    X_products_by_consumers = (
        centered
        .T
        .to_numpy(
            dtype=float
        )
    )

    if not np.isfinite(
        X_products_by_consumers
    ).all():
        result.hard_stop = True

        result.hard_stop_reasons.append(
            (
                "Non-finite values remain "
                "after preference preprocessing."
            )
        )

        return result

    max_components = min(
        X_products_by_consumers.shape[
            0
        ],
        X_products_by_consumers.shape[
            1
        ],
    )

    if (
        max_components
        < 2
    ):
        result.hard_stop = True

        result.hard_stop_reasons.append(
            (
                "Fewer than two PCA "
                "components are estimable."
            )
        )

        return result

    # ---------------------------------------------------------------------
    # FIT PCA
    # ---------------------------------------------------------------------

    pca = PCA(
        n_components=max_components,
        svd_solver="full",
    )

    product_scores_all = (
        pca.fit_transform(
            X_products_by_consumers
        )
    )

    explained = (
        pca.explained_variance_ratio_
    )

    if (
        len(
            explained
        )
        < 2
        or not np.isfinite(
            explained[
                :2
            ]
        ).all()
    ):
        result.hard_stop = True

        result.hard_stop_reasons.append(
            (
                "PC1/PC2 explained variance "
                "could not be estimated "
                "reliably."
            )
        )

        return result

    products = (
        centered
        .columns
        .astype(
            str
        )
        .tolist()
    )

    consumers = (
        centered
        .index
        .astype(
            str
        )
        .tolist()
    )

    result.explained_variance = (
        explained
    )

    result.cumulative_2d = float(
        explained[0]
        + explained[1]
    )

    # ---------------------------------------------------------------------
    # PRODUCT SCORES
    # ---------------------------------------------------------------------

    result.product_scores = pd.DataFrame(
        {
            PRODUCT: products,
            "PC1": (
                product_scores_all[
                    :,
                    0
                ]
            ),
            "PC2": (
                product_scores_all[
                    :,
                    1
                ]
            ),
        }
    )

    # ---------------------------------------------------------------------
    # CONSUMER LOADINGS / PREFERENCE DIRECTIONS
    # ---------------------------------------------------------------------
    #
    # sklearn:
    #
    # pca.components_
    #
    # rows    = principal axes
    # columns = consumer variables
    #
    # Multiplying component coefficients by sqrt(eigenvalue) gives
    # loading coordinates that are useful for directional visualization.
    # ---------------------------------------------------------------------

    eigen = (
        pca.explained_variance_
    )

    loading_pc1 = (
        pca.components_[
            0,
            :
        ]
        * math.sqrt(
            max(
                float(
                    eigen[0]
                ),
                0.0,
            )
        )
    )

    loading_pc2 = (
        pca.components_[
            1,
            :
        ]
        * math.sqrt(
            max(
                float(
                    eigen[1]
                ),
                0.0,
            )
        )
    )

    result.consumer_loadings = (
        pd.DataFrame(
            {
                PANELIST: consumers,
                "PC1": loading_pc1,
                "PC2": loading_pc2,
            }
        )
    )

    # ---------------------------------------------------------------------
    # G9-C C2 — TWO-DIMENSIONAL MAP ADEQUACY
    # ---------------------------------------------------------------------

    if (
        result.cumulative_2d
        >= STRONG_2D_THRESHOLD
    ):
        result.map_adequacy = (
            "Strong 2D workshop summary"
        )

        result.map_adequacy_status = (
            "PASS"
        )

        required_action = (
            "Interpret PC1/PC2 as the "
            "primary workshop preference map."
        )

    elif (
        result.cumulative_2d
        >= CAUTION_2D_THRESHOLD
    ):
        result.map_adequacy = (
            "Usable with caution"
        )

        result.map_adequacy_status = (
            "WARNING"
        )

        required_action = (
            "Use PC1/PC2 with explicit "
            "caution; later dimensions "
            "contain material preference "
            "structure."
        )

    elif (
        result.cumulative_2d
        >= LIMITED_2D_THRESHOLD
    ):
        result.map_adequacy = (
            "Exploratory / limited "
            "2D representation"
        )

        result.map_adequacy_status = (
            "WARNING"
        )

        required_action = (
            "Treat the 2D map as limited "
            "and avoid reducing all "
            "heterogeneity to PC1/PC2."
        )

    else:
        result.map_adequacy = (
            "PC1/PC2 inadequate as "
            "the primary summary"
        )

        result.map_adequacy_status = (
            "WARNING"
        )

        required_action = (
            "PCA remains exploratory, "
            "but do not summarize "
            "preference heterogeneity "
            "primarily from PC1/PC2."
        )

    # ---------------------------------------------------------------------
    # DIAGNOSTICS
    # ---------------------------------------------------------------------

    result.diagnostics.extend(
        [
            {
                "Diagnostic": (
                    "Complete consumer × "
                    "product matrix"
                ),
                "Observed Value": (
                    f"{pref.notna().sum().sum()} "
                    f"/ {pref.size} cells complete"
                ),
                "Threshold": (
                    "100% complete; "
                    "no silent imputation"
                ),
                "Status": (
                    "PASS"
                    if pref
                    .notna()
                    .all()
                    .all()
                    else "HARD STOP"
                ),
                "Required Action": (
                    "Proceed"
                    if pref
                    .notna()
                    .all()
                    .all()
                    else "Stop IPM"
                ),
            },
            {
                "Diagnostic": (
                    "Duplicate consumer × "
                    "product cells"
                ),
                "Observed Value": (
                    "0 after validated pivot"
                ),
                "Threshold": "0",
                "Status": "PASS",
                "Required Action": "Proceed",
            },
            {
                "Diagnostic": (
                    "Zero-variance "
                    "preference variables"
                ),
                "Observed Value": "0",
                "Threshold": "0",
                "Status": "PASS",
                "Required Action": "Proceed",
            },
            {
                "Diagnostic": (
                    "PC1 + PC2 "
                    "variance explained"
                ),
                "Observed Value": (
                    f"{100 * result.cumulative_2d:.1f}%"
                ),
                "Threshold": (
                    "≥60% strong; "
                    "50–<60% caution; "
                    "40–<50% limited; "
                    "<40% not primary "
                    "2D summary"
                ),
                "Status": (
                    result.map_adequacy_status
                ),
                "Required Action": (
                    required_action
                ),
            },
        ]
    )

    # ---------------------------------------------------------------------
    # DESCRIPTIVE DIRECTIONAL SUMMARY
    # ---------------------------------------------------------------------
    #
    # This does NOT create segments.
    #
    # Each consumer vector is matched to whichever product direction has
    # the greatest cosine similarity in PC1/PC2.
    #
    # The result is a descriptive map-reading aid only.
    # ---------------------------------------------------------------------

    product_vectors = (
        result.product_scores[
            [
                "PC1",
                "PC2",
            ]
        ]
        .to_numpy(
            dtype=float
        )
    )

    product_norms = np.linalg.norm(
        product_vectors,
        axis=1,
    )

    consumer_vectors = (
        result.consumer_loadings[
            [
                "PC1",
                "PC2",
            ]
        ]
        .to_numpy(
            dtype=float
        )
    )

    consumer_norms = np.linalg.norm(
        consumer_vectors,
        axis=1,
    )

    counts = {
        product: 0
        for product in products
    }

    valid_angles: List[
        float
    ] = []

    for (
        vector,
        vector_norm,
    ) in zip(
        consumer_vectors,
        consumer_norms,
    ):
        if (
            vector_norm
            <= 1e-12
        ):
            continue

        valid_angles.append(
            math.atan2(
                float(
                    vector[1]
                ),
                float(
                    vector[0]
                ),
            )
        )

        similarities: List[
            float
        ] = []

        for (
            product_vector,
            product_norm,
        ) in zip(
            product_vectors,
            product_norms,
        ):
            if (
                product_norm
                <= 1e-12
            ):
                similarities.append(
                    -np.inf
                )

            else:
                similarity = float(
                    np.dot(
                        vector,
                        product_vector,
                    )
                    / (
                        vector_norm
                        * product_norm
                    )
                )

                similarities.append(
                    similarity
                )

        best_index = int(
            np.argmax(
                similarities
            )
        )

        counts[
            products[
                best_index
            ]
        ] += 1

    result.dominant_product_counts = (
        counts
    )

    # ---------------------------------------------------------------------
    # CIRCULAR DISPERSION
    # ---------------------------------------------------------------------
    #
    # concentration R:
    #   1 = strongly aligned
    #   0 = highly dispersed
    #
    # We retain 1 - R as a descriptive orientation-dispersion quantity.
    #
    # No arbitrary significance or small/medium/large threshold is used.
    # ---------------------------------------------------------------------

    if valid_angles:
        mean_cos = float(
            np.mean(
                np.cos(
                    valid_angles
                )
            )
        )

        mean_sin = float(
            np.mean(
                np.sin(
                    valid_angles
                )
            )
        )

        concentration = math.sqrt(
            mean_cos ** 2
            + mean_sin ** 2
        )

        result.orientation_dispersion = (
            1.0
            - concentration
        )

    # ---------------------------------------------------------------------
    # PLAIN-LANGUAGE CONCLUSION
    # ---------------------------------------------------------------------

    nonzero_product_directions = sum(
        1
        for count in counts.values()
        if count > 0
    )

    if (
        result.cumulative_2d
        < LIMITED_2D_THRESHOLD
    ):
        result.answer = (
            "Consumer preferences contain "
            "multivariate structure, but PC1 "
            "and PC2 alone do not capture "
            "enough of it for a strong "
            "two-dimensional summary."
        )

        result.conclusion_detail = (
            f"PC1 + PC2 explain "
            f"{100 * result.cumulative_2d:.1f}% "
            "of preference variation. "
            "The map may still be inspected "
            "exploratorily, but the category "
            "should not be reduced to the "
            "displayed plane."
        )

    elif (
        nonzero_product_directions
        >= 2
    ):
        result.answer = (
            "Consumers show more than one "
            "preference direction across "
            "the products."
        )

        result.conclusion_detail = (
            f"PC1 + PC2 explain "
            f"{100 * result.cumulative_2d:.1f}% "
            "of preference variation, and "
            "consumer vectors orient toward "
            "multiple product directions. "
            "This suggests that average liking "
            "can conceal meaningful "
            "heterogeneity, but IPM alone "
            "does not prove discrete segments "
            "or sensory drivers."
        )

    else:
        result.answer = (
            "Consumer preference directions "
            "appear relatively aligned in "
            "the first two preference "
            "dimensions."
        )

        result.conclusion_detail = (
            f"PC1 + PC2 explain "
            f"{100 * result.cumulative_2d:.1f}% "
            "of preference variation. "
            "The displayed preference vectors "
            "largely share one direction, "
            "although IPM remains exploratory "
            "and does not establish statistical "
            "homogeneity."
        )

    return result

def nice_limit(
    values: np.ndarray,
    minimum: float = 1.0,
) -> float:
    """
    Return a padded symmetric plotting limit.
    """

    vals = np.asarray(
        values,
        dtype=float,
    )

    vals = vals[
        np.isfinite(
            vals
        )
    ]

    if len(
        vals
    ) == 0:
        return minimum

    maximum = max(
        float(
            np.max(
                np.abs(
                    vals
                )
            )
        ),
        minimum,
    )

    return (
        maximum
        * 1.18
    )

def diagnostics_table(
    rows: Sequence[
        Dict[
            str,
            str,
        ]
    ],
) -> str:
    """
    Render the SOP diagnostic table.
    """

    if not rows:
        return (
            "<p class='muted'>"
            "No diagnostic results available."
            "</p>"
        )

    headers = [
        "Diagnostic",
        "Observed Value",
        "Threshold",
        "Status",
        "Required Action",
    ]

    body_rows: List[
        str
    ] = []

    for row in rows:
        cells: List[
            str
        ] = []

        for header in headers:
            value = row.get(
                header,
                "",
            )

            if (
                header
                == "Status"
            ):
                rendered = status_badge(
                    value
                )

            else:
                rendered = html.escape(
                    str(
                        value
                    )
                )

            cells.append(
                (
                    "<td>"
                    f"{rendered}"
                    "</td>"
                )
            )

        body_rows.append(
            (
                "<tr>"
                + "".join(
                    cells
                )
                + "</tr>"
            )
        )

    return (
        "<div class='table-wrap'>"
        "<table>"
        "<thead>"
        "<tr>"
        + "".join(
            (
                "<th>"
                f"{html.escape(header)}"
                "</th>"
            )
            for header in headers
        )
        + "</tr>"
        "</thead>"
        "<tbody>"
        + "".join(
            body_rows
        )
        + "</tbody>"
        "</table>"
        "</div>"
    )

def adequacy_explanation(
    result: IPMResult,
) -> str:
    """
    Translate the G9-C PC1 + PC2 threshold to plain English.
    """

    percentage = (
        100
        * result.cumulative_2d
        if np.isfinite(
            result.cumulative_2d
        )
        else float(
            "nan"
        )
    )

    if (
        percentage
        >= 60
    ):
        return (
            "PC1 and PC2 retain at least "
            "60% of preference variation, "
            "so the two-dimensional map is "
            "a strong workshop summary."
        )

    if (
        percentage
        >= 50
    ):
        return (
            "The two-dimensional map is "
            "usable, but a meaningful share "
            "of preference variation remains "
            "in later dimensions."
        )

    if (
        percentage
        >= 40
    ):
        return (
            "The map is useful only as a "
            "limited exploratory view; later "
            "components contain substantial "
            "preference structure."
        )

    return (
        "PC1 and PC2 retain less than 40% "
        "of preference variation, so the "
        "displayed plane should not be treated "
        "as the primary summary of all "
        "preference heterogeneity."
    )

def main() -> None:
    """
    SOP-13 execution workflow.

    1. Resolve data
    2. Load CSV
    3. Dataset Capability Gate
    4. Study-design validation
    5. Build consumer × product matrix
    6. Preprocess preference profiles
    7. Fit PCA-based IPM
    8. Apply G9-C adequacy rule
    9. Generate one self-contained HTML dashboard
    10. Optionally save technical exports
    11. Print concise console result
    12. Open dashboard
    """

    data_path = (
        resolve_data_path()
    )

    dashboard_path = (
        output_path()
    )

    capability: Optional[
        CapabilityResult
    ] = None

    design: Optional[
        DesignResult
    ] = None

    result: Optional[
        IPMResult
    ] = None

    # ---------------------------------------------------------------------
    # LOAD DATA
    # ---------------------------------------------------------------------

    try:
        df = load_data(
            data_path
        )

    except Exception as exc:
        page = (
            build_failure_dashboard(
                data_path=data_path,
                capability=None,
                design=None,
                error=str(
                    exc
                ),
            )
        )

        dashboard_path.write_text(
            page,
            encoding="utf-8",
        )

        print_console_summary(
            data_path=data_path,
            design=None,
            result=None,
            dashboard=dashboard_path,
        )

        try_open_dashboard(
            dashboard_path
        )

        return

    # ---------------------------------------------------------------------
    # DATASET CAPABILITY GATE
    # ---------------------------------------------------------------------

    capability = (
        dataset_capability_gate(
            df
        )
    )

    if (
        capability.status
        == "UNSUPPORTED"
    ):
        page = (
            build_failure_dashboard(
                data_path=data_path,
                capability=capability,
                design=None,
            )
        )

        dashboard_path.write_text(
            page,
            encoding="utf-8",
        )

        print_console_summary(
            data_path=data_path,
            design=None,
            result=None,
            dashboard=dashboard_path,
        )

        try_open_dashboard(
            dashboard_path
        )

        return

    # ---------------------------------------------------------------------
    # STUDY-DESIGN VALIDATION
    # ---------------------------------------------------------------------

    design = validate_study_design(
        df
    )

    if design.hard_stop:
        page = (
            build_failure_dashboard(
                data_path=data_path,
                capability=capability,
                design=design,
            )
        )

        dashboard_path.write_text(
            page,
            encoding="utf-8",
        )

        print_console_summary(
            data_path=data_path,
            design=design,
            result=None,
            dashboard=dashboard_path,
        )

        try_open_dashboard(
            dashboard_path
        )

        return

    # ---------------------------------------------------------------------
    # BUILD PREFERENCE MATRIX
    # ---------------------------------------------------------------------

    preference_matrix = (
        build_preference_matrix(
            df
        )
    )

    # ---------------------------------------------------------------------
    # FIT INTERNAL PREFERENCE MAP
    # ---------------------------------------------------------------------

    result = fit_ipm(
        preference_matrix
    )

    # ---------------------------------------------------------------------
    # BUILD DASHBOARD
    # ---------------------------------------------------------------------

    page = build_dashboard(
        data_path=data_path,
        capability=capability,
        design=design,
        result=result,
    )

    dashboard_path.write_text(
        page,
        encoding="utf-8",
    )

    # ---------------------------------------------------------------------
    # OPTIONAL TECHNICAL EXPORTS
    # ---------------------------------------------------------------------

    save_technical_exports(
        result=result,
        out_dir=dashboard_path.parent,
    )

    # ---------------------------------------------------------------------
    # CONSOLE SUMMARY
    # ---------------------------------------------------------------------

    print_console_summary(
        data_path=data_path,
        design=design,
        result=result,
        dashboard=dashboard_path,
    )

    # ---------------------------------------------------------------------
    # AUTO-OPEN DASHBOARD
    # ---------------------------------------------------------------------

    try_open_dashboard(
        dashboard_path
    )

if __name__ == "__main__":
    main()

def pretty_name(column: str) -> str:
    """Convert a technical descriptive-variable name to readable text."""
    name = str(column)

    if name.startswith("desc_"):
        name = name[5:]

    replacements = {
        "cocoa_authenticity": "Cocoa authenticity",
        "chocolate_intensity": "Chocolate intensity",
        "artificial_candy_note": "Artificial / candy note",
        "berry_note": "Berry note",
    }

    if name in replacements:
        return replacements[name]

    return name.replace("_", " ").strip().title()

def locate_data_file() -> Path:
    """
    Resolve the dataset.

    Priority:
    1. First command-line argument.
    2. DATA_FILENAME relative to current working directory.
    3. DATA_FILENAME relative to the script location.
    """
    candidates: list[Path] = []

    if len(sys.argv) > 1:
        candidates.append(Path(sys.argv[1]).expanduser())

    candidates.append(Path.cwd() / DATA_FILENAME)

    try:
        script_dir = Path(__file__).resolve().parent
        candidates.append(script_dir / DATA_FILENAME)
    except NameError:
        pass

    seen: set[str] = set()

    for candidate in candidates:
        try:
            candidate = candidate.resolve()
        except Exception:
            candidate = candidate.absolute()

        key = str(candidate).lower()

        if key in seen:
            continue

        seen.add(key)

        if candidate.exists() and candidate.is_file():
            return candidate

    searched = "\n".join(f"  - {x}" for x in candidates)

    raise FileNotFoundError(
        "Could not locate the SOP-14 dataset.\n\n"
        "Paths checked:\n"
        f"{searched}\n\n"
        "Expected default path:\n"
        f"  {DATA_FILENAME}\n\n"
        "You may also pass a CSV path as the first command-line argument."
    )

def dataset_capability_gate(df: pd.DataFrame) -> dict[str, Any]:
    """
    Determine whether the active dataset can support approved SOP-14 EPM.

    Required constructs:
      - panelist identifier
      - product identifier
      - overall liking
      - all approved measured descriptive sensory variables

    Missing required constructs are not substituted.
    """
    missing = [col for col in REQUIRED_COLUMNS if col not in df.columns]

    result = {
        "status": "SUPPORTED",
        "missing_columns": missing,
        "messages": [],
    }

    if missing:
        result["status"] = "UNSUPPORTED"
        result["messages"].append(
            "SOP-14 requires consumer liking plus the approved measured "
            "descriptive sensory variables. Required fields are missing: "
            + ", ".join(missing)
            + "."
        )
        result["messages"].append(
            "No substitute or proxy variable was used."
        )

    return result

def preprocess_required_columns(
    df: pd.DataFrame,
) -> tuple[pd.DataFrame, list[str]]:
    """
    Copy the data and coerce only variables that are supposed to be numeric.

    Product and panelist identifiers retain their categorical/string meaning.
    """
    work = df.copy()
    warnings: list[str] = []

    for col in [LIKING, *DESCRIPTIVE_VARS]:
        original_nonmissing = int(work[col].notna().sum())

        work[col] = pd.to_numeric(
            work[col],
            errors="coerce",
        )

        converted_nonmissing = int(work[col].notna().sum())

        n_lost = original_nonmissing - converted_nonmissing

        if n_lost > 0:
            warnings.append(
                f"{n_lost} value(s) in {col} could not be interpreted "
                "as numeric and were treated as missing."
            )

    return work, warnings

def validate_study_design(df: pd.DataFrame) -> dict[str, Any]:
    """
    Validate structural assumptions needed for EPM.

    Hard stops are reserved for conditions that make the approved analysis
    structurally impossible or would require unapproved imputation.
    """
    hard_stops: list[str] = []
    warnings: list[str] = []
    notes: list[str] = []

    n_rows = len(df)
    n_consumers = int(df[PANELIST].nunique(dropna=True))
    n_products = int(df[PRODUCT].nunique(dropna=True))

    missing_panelist = int(df[PANELIST].isna().sum())
    missing_product = int(df[PRODUCT].isna().sum())

    if missing_panelist > 0:
        hard_stops.append(
            f"{missing_panelist} row(s) have missing {PANELIST}. "
            "Consumer identity is required to preserve individual "
            "preference information."
        )

    if missing_product > 0:
        hard_stops.append(
            f"{missing_product} row(s) have missing {PRODUCT}."
        )

    duplicate_mask = df.duplicated(
        subset=[PANELIST, PRODUCT],
        keep=False,
    )
    n_duplicate_rows = int(duplicate_mask.sum())

    if n_duplicate_rows > 0:
        duplicate_cells = (
            df.loc[duplicate_mask, [PANELIST, PRODUCT]]
            .drop_duplicates()
            .shape[0]
        )

        hard_stops.append(
            f"Found {duplicate_cells} duplicated consumer × product "
            "cell(s). SOP-14 requires one liking observation per "
            "consumer × product combination; duplicates cannot be "
            "silently averaged or discarded."
        )

    if n_products < 3:
        hard_stops.append(
            f"Only {n_products} distinct product(s) are available. "
            "A two-dimensional external preference map requires at least "
            "three distinct products."
        )

    if n_consumers < 1:
        hard_stops.append(
            "No usable consumers were identified."
        )

    missing_liking = int(df[LIKING].isna().sum())

    if missing_liking > 0:
        warnings.append(
            f"{missing_liking} evaluation(s) have missing {LIKING}. "
            "The complete consumer × product preference matrix will be "
            "checked before EPM is allowed."
        )

    liking_unique = int(df[LIKING].nunique(dropna=True))

    if liking_unique < 2:
        hard_stops.append(
            f"{LIKING} does not contain sufficient variation to construct "
            "consumer preference directions."
        )

    # -------------------------------------------------------------------------
    # Consumer × product preference matrix
    # -------------------------------------------------------------------------

    try:
        pref = df.pivot(
            index=PANELIST,
            columns=PRODUCT,
            values=LIKING,
        )
    except Exception as exc:
        hard_stops.append(
            "The consumer × product liking matrix could not be constructed: "
            f"{exc}"
        )
        pref = pd.DataFrame()

    expected_cells = n_consumers * n_products
    observed_cells = 0
    missing_pref_cells = 0

    if not pref.empty:
        observed_cells = int(pref.notna().sum().sum())
        missing_pref_cells = int(pref.isna().sum().sum())

        if missing_pref_cells > 0:
            hard_stops.append(
                f"The consumer × product liking matrix contains "
                f"{missing_pref_cells} missing cell(s) out of "
                f"{expected_cells}. The approved preference mapping "
                "workflow does not authorize silent liking imputation."
            )

    counts_per_consumer = (
        df.groupby(PANELIST, dropna=False)[PRODUCT]
        .nunique()
    )

    incomplete_consumers = counts_per_consumer[
        counts_per_consumer < n_products
    ]

    if len(incomplete_consumers) > 0:
        warnings.append(
            f"{len(incomplete_consumers)} consumer(s) do not have all "
            f"{n_products} products represented."
        )

    if n_products == 8:
        notes.append(
            "The active dataset contains eight products, matching the "
            "category structure described in the approved SOP."
        )
    else:
        warnings.append(
            f"The active dataset contains {n_products} products rather "
            "than the approximately eight described for this study. "
            "The script uses the actual product count."
        )

    return {
        "n_rows": n_rows,
        "n_consumers": n_consumers,
        "n_products": n_products,
        "expected_cells": expected_cells,
        "observed_cells": observed_cells,
        "missing_pref_cells": missing_pref_cells,
        "n_duplicate_rows": n_duplicate_rows,
        "hard_stops": hard_stops,
        "warnings": warnings,
        "notes": notes,
        "preference_matrix": pref,
    }

def build_sensory_profiles(
    df: pd.DataFrame,
    products: list[Any],
) -> dict[str, Any]:
    """
    Aggregate measured descriptive sensory variables to product means.

    No metadata or intended product character is used.
    """
    hard_stops: list[str] = []
    warnings: list[str] = []
    notes: list[str] = []

    product_profiles = (
        df.groupby(PRODUCT, observed=True)[DESCRIPTIVE_VARS]
        .mean()
        .reindex(products)
    )

    missing_profile_cells = int(product_profiles.isna().sum().sum())

    if missing_profile_cells > 0:
        locations = []

        for product in product_profiles.index:
            for attr in product_profiles.columns:
                if pd.isna(product_profiles.loc[product, attr]):
                    locations.append(
                        f"{product} × {pretty_name(attr)}"
                    )

        display = locations[:8]
        suffix = "" if len(locations) <= 8 else " …"

        hard_stops.append(
            f"The product descriptive matrix contains "
            f"{missing_profile_cells} missing product-profile value(s): "
            + "; ".join(display)
            + suffix
            + ". SOP-14 does not authorize silent sensory-profile "
              "imputation."
        )

    sds = product_profiles.std(axis=0, ddof=0)

    zero_variance = [
        col
        for col in DESCRIPTIVE_VARS
        if pd.notna(sds.get(col))
        and abs(float(sds[col])) <= NUMERIC_ZERO_TOL
    ]

    usable_vars = [
        col for col in DESCRIPTIVE_VARS
        if col not in zero_variance
    ]

    if zero_variance:
        warnings.append(
            "The following measured descriptive attribute(s) have zero "
            "between-product variance and therefore cannot define a "
            "sensory direction in the map: "
            + ", ".join(pretty_name(x) for x in zero_variance)
            + ". They are retained as measured fields but excluded from "
              "the PCA calculation."
        )

    if len(usable_vars) < 2:
        hard_stops.append(
            "Fewer than two descriptive sensory attributes contain "
            "between-product variation. A two-dimensional measured sensory "
            "space cannot be constructed."
        )

    notes.append(
        "Sensory profiles were calculated from measured descriptive "
        "responses aggregated to product means."
    )

    return {
        "profiles": product_profiles,
        "usable_vars": usable_vars,
        "zero_variance_vars": zero_variance,
        "hard_stops": hard_stops,
        "warnings": warnings,
        "notes": notes,
    }

def run_ipm_review(
    preference_matrix: pd.DataFrame,
) -> dict[str, Any]:
    """
    Perform the SOP-13-style preliminary IPM review required by SOP-14.

    Orientation:
        rows    = products
        columns = individual consumers

    Each consumer's liking values are centered across products before PCA.
    This emphasizes relative preference pattern rather than absolute
    scale-use differences.

    This is used as a preference-heterogeneity review, not as a
    segmentation procedure.
    """
    product_by_consumer = preference_matrix.T.astype(float)

    centered = (
        product_by_consumer
        - product_by_consumer.mean(axis=0)
    )

    matrix = centered.to_numpy(dtype=float)

    max_components = min(matrix.shape)

    if max_components < 2:
        raise ValueError(
            "The preference matrix cannot support a two-dimensional IPM."
        )

    pca = PCA(
        n_components=max_components,
        random_state=RANDOM_STATE,
    )

    product_scores_all = pca.fit_transform(matrix)

    product_scores = pd.DataFrame(
        product_scores_all[:, :2],
        index=product_by_consumer.index,
        columns=["IPM1", "IPM2"],
    )

    # Consumer loading directions in preference space.
    consumer_loadings = pd.DataFrame(
        pca.components_[:2, :].T,
        index=product_by_consumer.columns,
        columns=["IPM1", "IPM2"],
    )

    evr = pca.explained_variance_ratio_

    pc1 = float(evr[0]) if len(evr) >= 1 else float("nan")
    pc2 = float(evr[1]) if len(evr) >= 2 else float("nan")

    # Descriptive review only:
    # determine whether consumer loading directions occupy multiple quadrants.
    directions = consumer_loadings.to_numpy(dtype=float)
    norms = np.linalg.norm(directions, axis=1)

    valid = norms > NUMERIC_ZERO_TOL

    quadrant_counts = {1: 0, 2: 0, 3: 0, 4: 0}

    for x, y in directions[valid]:
        if x >= 0 and y >= 0:
            quadrant_counts[1] += 1
        elif x < 0 <= y:
            quadrant_counts[2] += 1
        elif x < 0 and y < 0:
            quadrant_counts[3] += 1
        else:
            quadrant_counts[4] += 1

    occupied_quadrants = sum(
        int(v > 0) for v in quadrant_counts.values()
    )

    return {
        "pca": pca,
        "product_scores": product_scores,
        "consumer_loadings": consumer_loadings,
        "pc1_variance": pc1,
        "pc2_variance": pc2,
        "cumulative_2d": pc1 + pc2,
        "quadrant_counts": quadrant_counts,
        "occupied_quadrants": occupied_quadrants,
    }

def construct_sensory_space(
    product_profiles: pd.DataFrame,
    usable_vars: list[str],
) -> dict[str, Any]:
    """
    Standardize product descriptive means and construct the product sensory
    space using PCA.

    Scaling is explicit because the descriptive attributes share a conceptual
    intensity framework but can have different empirical spreads.
    """
    matrix = product_profiles[usable_vars].astype(float)

    scaler = StandardScaler(
        with_mean=True,
        with_std=True,
    )

    scaled = scaler.fit_transform(matrix)

    max_components = min(
        scaled.shape[0],
        scaled.shape[1],
    )

    if max_components < 2:
        raise ValueError(
            "The measured sensory matrix cannot support two PCA dimensions."
        )

    pca = PCA(
        n_components=max_components,
        random_state=RANDOM_STATE,
    )

    score_array = pca.fit_transform(scaled)

    scores = pd.DataFrame(
        score_array[:, :2],
        index=matrix.index,
        columns=["PC1", "PC2"],
    )

    evr = pca.explained_variance_ratio_

    pc1 = float(evr[0])
    pc2 = float(evr[1])

    # Correlation-style PCA coordinates for standardized variables.
    # These provide attribute directions for the biplot.
    loading_matrix = (
        pca.components_[:2, :].T
        * np.sqrt(pca.explained_variance_[:2])
    )

    loadings = pd.DataFrame(
        loading_matrix,
        index=usable_vars,
        columns=["PC1", "PC2"],
    )

    return {
        "profiles": matrix,
        "scaled": scaled,
        "scaler": scaler,
        "pca": pca,
        "scores": scores,
        "loadings": loadings,
        "pc1_variance": pc1,
        "pc2_variance": pc2,
        "cumulative_2d": pc1 + pc2,
    }

def fit_consumer_preference_vectors(
    preference_matrix: pd.DataFrame,
    sensory_scores: pd.DataFrame,
) -> pd.DataFrame:
    """
    Relate each consumer's individual liking pattern to the first two
    dimensions of the measured product sensory space.

    Model for each consumer:

        liking = intercept + beta1 * sensory_PC1
                           + beta2 * sensory_PC2

    beta1/beta2 define that consumer's preference direction in the sensory
    space.

    R² describes how much of that consumer's observed product-to-product
    liking pattern is represented by this two-dimensional linear preference
    vector.

    No p-values are used. EPM is exploratory.
    """
    common_products = [
        p for p in sensory_scores.index
        if p in preference_matrix.columns
    ]

    X_scores = (
        sensory_scores
        .loc[common_products, ["PC1", "PC2"]]
        .to_numpy(dtype=float)
    )

    X = np.column_stack(
        [
            np.ones(len(common_products)),
            X_scores,
        ]
    )

    records: list[dict[str, Any]] = []

    for consumer in preference_matrix.index:
        y = (
            preference_matrix
            .loc[consumer, common_products]
            .to_numpy(dtype=float)
        )

        if not np.all(np.isfinite(y)):
            continue

        if np.ptp(y) <= NUMERIC_ZERO_TOL:
            records.append(
                {
                    PANELIST: consumer,
                    "intercept": float(np.mean(y)),
                    "beta_pc1": 0.0,
                    "beta_pc2": 0.0,
                    "vector_norm": 0.0,
                    "r2": np.nan,
                    "directional": False,
                }
            )
            continue

        coef, _, _, _ = np.linalg.lstsq(
            X,
            y,
            rcond=None,
        )

        fitted = X @ coef

        sse = float(np.sum((y - fitted) ** 2))
        sst = float(np.sum((y - np.mean(y)) ** 2))

        if sst <= NUMERIC_ZERO_TOL:
            r2 = np.nan
        else:
            r2 = 1.0 - (sse / sst)

        beta1 = float(coef[1])
        beta2 = float(coef[2])

        vector_norm = float(
            math.sqrt(beta1 ** 2 + beta2 ** 2)
        )

        records.append(
            {
                PANELIST: consumer,
                "intercept": float(coef[0]),
                "beta_pc1": beta1,
                "beta_pc2": beta2,
                "vector_norm": vector_norm,
                "r2": r2,
                "directional": vector_norm > NUMERIC_ZERO_TOL,
            }
        )

    return pd.DataFrame(records)

def classify_vector_quadrant(x: float, y: float) -> int:
    if x >= 0 and y >= 0:
        return 1
    if x < 0 <= y:
        return 2
    if x < 0 and y < 0:
        return 3
    return 4

def summarize_preference_directions(
    vectors: pd.DataFrame,
    loadings: pd.DataFrame,
) -> dict[str, Any]:
    """
    Create transparent descriptive summaries of the individual preference
    vectors.

    The "closest sensory direction" is based on angular similarity (cosine)
    between a consumer preference vector and each measured sensory loading
    direction.

    It is NOT:
      - a significance result;
      - an effect size;
      - a causal-driver ranking.
    """
    directional = vectors[
        vectors["directional"] == True
    ].copy()

    quadrant_counts = {1: 0, 2: 0, 3: 0, 4: 0}

    alignment_counts = {
        attr: 0 for attr in loadings.index
    }

    alignment_rows: list[dict[str, Any]] = []

    unit_vectors: list[np.ndarray] = []

    for _, row in directional.iterrows():
        v = np.array(
            [
                float(row["beta_pc1"]),
                float(row["beta_pc2"]),
            ],
            dtype=float,
        )

        norm = float(np.linalg.norm(v))

        if norm <= NUMERIC_ZERO_TOL:
            continue

        u = v / norm
        unit_vectors.append(u)

        quadrant = classify_vector_quadrant(
            float(u[0]),
            float(u[1]),
        )
        quadrant_counts[quadrant] += 1

        best_attr = None
        best_cosine = -np.inf

        for attr in loadings.index:
            a = loadings.loc[
                attr, ["PC1", "PC2"]
            ].to_numpy(dtype=float)

            a_norm = float(np.linalg.norm(a))

            if a_norm <= NUMERIC_ZERO_TOL:
                continue

            cosine = float(
                np.dot(u, a / a_norm)
            )

            if cosine > best_cosine:
                best_cosine = cosine
                best_attr = attr

        if best_attr is not None:
            alignment_counts[best_attr] += 1

            alignment_rows.append(
                {
                    PANELIST: row[PANELIST],
                    "closest_attribute": best_attr,
                    "cosine_alignment": best_cosine,
                    "r2": row["r2"],
                    "quadrant": quadrant,
                }
            )

    occupied_quadrants = sum(
        int(v > 0) for v in quadrant_counts.values()
    )

    ranked_alignments = sorted(
        alignment_counts.items(),
        key=lambda item: (-item[1], pretty_name(item[0])),
    )

    positive_alignments = [
        (attr, count)
        for attr, count in ranked_alignments
        if count > 0
    ]

    if unit_vectors:
        mean_direction = np.mean(
            np.vstack(unit_vectors),
            axis=0,
        )

        consensus_length = float(
            np.linalg.norm(mean_direction)
        )

        if consensus_length > NUMERIC_ZERO_TOL:
            mean_direction_unit = (
                mean_direction / consensus_length
            )
        else:
            mean_direction_unit = np.array(
                [0.0, 0.0]
            )
    else:
        mean_direction = np.array([0.0, 0.0])
        mean_direction_unit = np.array([0.0, 0.0])
        consensus_length = 0.0

    return {
        "directional_vectors": directional,
        "quadrant_counts": quadrant_counts,
        "occupied_quadrants": occupied_quadrants,
        "alignment_counts": alignment_counts,
        "ranked_alignments": positive_alignments,
        "alignment_rows": pd.DataFrame(alignment_rows),
        "mean_direction": mean_direction,
        "mean_direction_unit": mean_direction_unit,
        "directional_consensus": consensus_length,
    }

def run_diagnostics(
    vectors: pd.DataFrame,
    ipm: dict[str, Any],
    sensory: dict[str, Any],
    zero_variance_vars: list[str],
) -> dict[str, Any]:
    """
    Diagnostics required by the SOP.

    No arbitrary inferential cutoffs are added. Quantities such as EPM R²
    are reported descriptively rather than converted into unsupported
    pass/fail thresholds.
    """
    hard_stops: list[str] = []
    warnings: list[str] = []
    passes: list[str] = []

    if sensory["scores"].shape[0] < 3:
        hard_stops.append(
            "The sensory space contains fewer than three products."
        )
    else:
        passes.append(
            "The product sensory space contains enough distinct products "
            "for a two-dimensional map."
        )

    if sensory["loadings"].shape[0] < 2:
        hard_stops.append(
            "Fewer than two usable descriptive attributes define the "
            "sensory space."
        )
    else:
        passes.append(
            f"{sensory['loadings'].shape[0]} measured descriptive "
            "attribute(s) contribute to the sensory space."
        )

    if zero_variance_vars:
        warnings.append(
            f"{len(zero_variance_vars)} descriptive attribute(s) were "
            "non-informative because their product means had zero variance."
        )

    directional = vectors[
        vectors["directional"] == True
    ]

    n_non_directional = int(
        len(vectors) - len(directional)
    )

    if len(directional) == 0:
        hard_stops.append(
            "No consumer has a measurable preference direction in the "
            "two-dimensional sensory space."
        )
    else:
        passes.append(
            f"Preference vectors were estimable for "
            f"{len(directional)} consumer(s)."
        )

    if n_non_directional > 0:
        warnings.append(
            f"{n_non_directional} consumer(s) had no directional liking "
            "variation that could be represented as an EPM vector."
        )

    finite_r2 = vectors["r2"].replace(
        [np.inf, -np.inf],
        np.nan,
    ).dropna()

    if len(finite_r2) > 0:
        median_r2 = float(finite_r2.median())
        mean_r2 = float(finite_r2.mean())
        q25_r2 = float(finite_r2.quantile(0.25))
        q75_r2 = float(finite_r2.quantile(0.75))
    else:
        median_r2 = np.nan
        mean_r2 = np.nan
        q25_r2 = np.nan
        q75_r2 = np.nan

    passes.append(
        "Individual consumer information was retained; liking was not "
        "collapsed to a single product-level mean response for the EPM."
    )

    passes.append(
        "Sensory directions are based only on measured descriptive "
        "variables, not product metadata or open-ended comments."
    )

    passes.append(
        "The sensory variables were standardized before PCA, and the "
        "variance represented by the displayed dimensions is reported."
    )

    return {
        "hard_stops": hard_stops,
        "warnings": warnings,
        "passes": passes,
        "median_r2": median_r2,
        "mean_r2": mean_r2,
        "q25_r2": q25_r2,
        "q75_r2": q75_r2,
        "ipm_2d_variance": ipm["cumulative_2d"],
        "sensory_2d_variance": sensory["cumulative_2d"],
    }

def build_interpretation(
    design: dict[str, Any],
    sensory: dict[str, Any],
    directions: dict[str, Any],
    diagnostics: dict[str, Any],
) -> dict[str, str]:
    """
    Generate appropriately cautious plain-language interpretation.

    No significance claim is generated because SOP-14 is exploratory.
    """
    n_consumers = design["n_consumers"]
    n_products = design["n_products"]

    ranked = directions["ranked_alignments"]

    if ranked:
        top_attr, top_count = ranked[0]
        top_label = pretty_name(top_attr)
    else:
        top_attr = ""
        top_count = 0
        top_label = "No single sensory direction"

    occupied = directions["occupied_quadrants"]

    if occupied <= 1:
        heterogeneity_text = (
            "Consumer preference vectors were concentrated in one broad "
            "direction of the displayed sensory space."
        )
    elif occupied == 2:
        heterogeneity_text = (
            "Consumer preference vectors occupied two quadrants of the "
            "displayed sensory space, indicating more than one preference "
            "orientation."
        )
    else:
        heterogeneity_text = (
            f"Consumer preference vectors occupied {occupied} of four "
            "quadrants of the displayed sensory space, indicating "
            "substantial directional heterogeneity."
        )

    if ranked:
        top_two = [
            pretty_name(attr)
            for attr, count in ranked[:2]
            if count > 0
        ]

        association_text = (
            f"Across individual preference vectors, "
            f"{join_words(top_two)} most often provided the closest "
            "measured sensory directions."
        )
    else:
        association_text = (
            "No measured sensory direction consistently aligned with the "
            "consumer preference vectors."
        )

    answer = (
        f"The external preference map links individual liking patterns "
        f"from {n_consumers} consumers with the measured sensory positions "
        f"of {n_products} products. {heterogeneity_text} "
        f"{association_text}"
    )

    caution = (
        "These patterns are exploratory sensory–preference associations. "
        "They are not significance tests, do not establish causality, and "
        "should not be interpreted as validated driver rankings. "
        f"Only {n_products} products define the sensory space."
    )

    if ranked:
        kpi_direction = (
            f"{top_label} ({top_count}/{n_consumers})"
        )
    else:
        kpi_direction = "No dominant alignment"

    return {
        "answer": answer,
        "caution": caution,
        "top_direction": kpi_direction,
    }
