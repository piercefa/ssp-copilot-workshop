def remove_export_index(df):
    """
    Removes an accidental CSV export index column such as:
        index
        Unnamed: 0

    Only removes it if it is clearly 0,1,2,3...
    """

    df = df.copy()
    removed = []

    for column in list(df.columns):

        normalized = column.strip().lower()

        if normalized in ["index", "unnamed: 0"]:

            numeric = pd.to_numeric(df[column], errors="coerce")

            if numeric.notna().all():

                expected = np.arange(len(df))

                if np.array_equal(numeric.to_numpy(), expected):

                    df = df.drop(columns=[column])
                    removed.append(column)

    return df, removed

def confidence_interval(series, alpha=0.05):
    """
    Calculate a two-sided t-based confidence interval for a mean.
    """

    x = pd.to_numeric(series, errors="coerce").dropna()

    n = len(x)

    if n == 0:
        return np.nan, np.nan

    if n == 1:
        return x.iloc[0], x.iloc[0]

    mean = x.mean()
    sem = stats.sem(x)

    critical_value = stats.t.ppf(
        1 - alpha / 2,
        df=n - 1
    )

    margin = critical_value * sem

    return mean - margin, mean + margin

def check_scale(df, column, minimum, maximum, integers_only=False):
    """
    Check whether values fall inside the approved scale.
    """

    numeric = pd.to_numeric(
        df[column],
        errors="coerce"
    )

    invalid_numeric = (
        df[column].notna() &
        numeric.isna()
    ).sum()

    outside_range = (
        (numeric < minimum) |
        (numeric > maximum)
    ).sum()

    non_integer = 0

    if integers_only:

        valid = numeric.dropna()

        non_integer = (
            ~np.isclose(
                valid,
                np.round(valid)
            )
        ).sum()

    total_invalid = (
        invalid_numeric +
        outside_range +
        non_integer
    )

    return int(total_invalid)

def product_summary(df, variable, calculate_boxes=False):
    """
    Produce descriptive product summaries.
    """

    rows = []

    for product, subset in df.groupby("product_code"):

        values = pd.to_numeric(
            subset[variable],
            errors="coerce"
        ).dropna()

        lower_ci, upper_ci = confidence_interval(values)

        result = {
            "product_code": product,
            "n": len(values),
            "mean": values.mean(),
            "sd": values.std(ddof=1),
            "median": values.median(),
            "ci95_low": lower_ci,
            "ci95_high": upper_ci,
        }

        if calculate_boxes:

            # Overall liking is 1–9.
            result["top_box_pct_7_to_9"] = (
                values.between(7, 9).mean() * 100
            )

            result["bottom_box_pct_1_to_3"] = (
                values.between(1, 3).mean() * 100
            )

        rows.append(result)

    output = pd.DataFrame(rows)

    return output.sort_values(
        "mean",
        ascending=False
    ).reset_index(drop=True)

def dataframe_to_json_records(df, round_digits=4):
    """Serialize a DataFrame to JSON-safe records for the dashboard."""

    display_df = df.copy()

    for column in display_df.select_dtypes(include="number").columns:
        display_df[column] = display_df[column].round(round_digits)

    records = json.loads(
        display_df.to_json(
            orient="records",
            date_format="iso",
        )
    )

    return records

def find_data_file() -> Path:

    # Allow a file path to be supplied from the command line.
    if len(sys.argv) > 1:

        path = (
            Path(sys.argv[1])
            .expanduser()
            .resolve()
        )

        if path.exists():
            return path

        raise FileNotFoundError(
            f"Dataset not found: {path}"
        )


    # Otherwise look relative to:
    # 1. the IDE working directory
    # 2. the Python script location

    possible_paths = [

        Path.cwd() / DATA_FILENAME,

        Path(__file__).resolve().parent
        / DATA_FILENAME,

    ]


    for path in possible_paths:

        if path.exists():
            return path.resolve()


    raise FileNotFoundError(

        f"Could not find {DATA_FILENAME}. "

    )

def remove_export_index(
    df: pd.DataFrame
) -> pd.DataFrame:

    output = df.copy()


    for column in list(
        output.columns
    ):

        normalized = (
            str(column)
            .strip()
            .lower()
        )


        if normalized in {
            "index",
            "unnamed: 0",
        }:

            numeric = pd.to_numeric(
                output[column],
                errors="coerce"
            )


            if numeric.notna().all():

                expected = np.arange(
                    len(output),
                    dtype=float
                )


                if np.array_equal(
                    numeric.to_numpy(
                        dtype=float
                    ),
                    expected
                ):

                    output = output.drop(
                        columns=column
                    )


    return output

def prepare_data(
    raw: pd.DataFrame
):

    df = remove_export_index(
        raw
    )


    # -------------------------------------------------------------------------
    # Confirm required variables exist.
    # -------------------------------------------------------------------------

    missing = [

        column

        for column
        in REQUIRED

        if column
        not in df.columns

    ]


    if missing:

        raise ValueError(

            "SOP-02 CAPABILITY GATE FAILED.\n\n"

            f"Missing required field(s): "
            f"{missing}\n\n"

            "The script will not invent, "
            "infer, or substitute constructs."

        )


    # Only variables needed by SOP-02 are used.
    df = df[
        REQUIRED
    ].copy()


    # -------------------------------------------------------------------------
    # Overall liking must be numeric.
    # -------------------------------------------------------------------------

    originally_present = (
        df[
            "overall_liking"
        ]
        .notna()
    )


    df[
        "overall_liking"
    ] = pd.to_numeric(

        df[
            "overall_liking"
        ],

        errors="coerce"

    )


    newly_invalid = (

        originally_present

        & df[
            "overall_liking"
        ].isna()

    )


    if newly_invalid.any():

        raise ValueError(

            "Non-numeric overall_liking "
            "values were detected."

        )


    # -------------------------------------------------------------------------
    # IDs cannot be missing.
    # -------------------------------------------------------------------------

    if df[
        [
            "panelist_id",
            "product_code",
        ]
    ].isna().any(axis=None):

        raise ValueError(

            "Missing panelist_id or "
            "product_code values detected."

        )


    df[
        "product_code"
    ] = (

        df[
            "product_code"
        ]

        .astype(str)

        .str.strip()

    )


    # -------------------------------------------------------------------------
    # Validate 1–9 hedonic scale.
    # -------------------------------------------------------------------------

    invalid_scale = (

        df[
            "overall_liking"
        ].notna()

        & ~df[
            "overall_liking"
        ].between(
            1,
            9
        )

    )


    if invalid_scale.any():

        raise ValueError(

            "overall_liking contains "
            "values outside the approved "
            "1–9 hedonic scale."

        )


    # -------------------------------------------------------------------------
    # Duplicate panelist × product evaluations.
    # -------------------------------------------------------------------------

    duplicates = df.duplicated(

        [
            "panelist_id",
            "product_code",
        ],

        keep=False

    )


    if duplicates.any():

        raise ValueError(

            "Duplicate panelist × product "
            "rows detected.\n\n"

            f"Rows involved: "
            f"{int(duplicates.sum())}"

        )


    notes = []


    # -------------------------------------------------------------------------
    # Remove incomplete rows.
    # -------------------------------------------------------------------------

    before = len(
        df
    )


    df = (

        df.dropna(
            subset=REQUIRED
        )

        .copy()

    )


    if len(df) < before:

        notes.append(

            f"{before - len(df)} incomplete "
            "model row(s) were excluded."

        )


    # -------------------------------------------------------------------------
    # At least two products required.
    # -------------------------------------------------------------------------

    if (
        df[
            "product_code"
        ].nunique()
        < 2
    ):

        raise ValueError(

            "Fewer than two products "
            "remain for analysis."

        )


    # -------------------------------------------------------------------------
    # Confirm repeated-measures structure.
    # -------------------------------------------------------------------------

    products_per_panelist = (

        df.groupby(
            "panelist_id"
        )[
            "product_code"
        ]

        .nunique()

    )


    repeated_panelists = (

        products_per_panelist
        >= 2

    ).sum()


    if repeated_panelists < 2:

        raise ValueError(

            "Repeated-measures structure "
            "is not supported by the "
            "available rows."

        )


    # -------------------------------------------------------------------------
    # Product ordering.
    # -------------------------------------------------------------------------

    products = sorted(

        df[
            "product_code"
        ].unique()

    )


    df[
        "product_code"
    ] = pd.Categorical(

        df[
            "product_code"
        ],

        categories=products,

        ordered=True

    )


    # -------------------------------------------------------------------------
    # Check complete repeated-measures design.
    # -------------------------------------------------------------------------

    expected_cells = (

        df[
            "panelist_id"
        ].nunique()

        * len(
            products
        )

    )


    if len(df) != expected_cells:

        notes.append(

            "The repeated-measures design "
            "is incomplete. The mixed model "
            "will use the available complete rows."

        )


    return (
        df,
        notes
    )

def fit_mixed_model(
    df
):

    model = smf.mixedlm(

        "overall_liking ~ C(product_code)",

        data=df,

        groups=df[
            "panelist_id"
        ],

        re_formula="1"

    )


    captured_warnings = []

    best_result = None

    optimizer_used = None


    for method in OPTIMIZERS:

        try:

            with warnings.catch_warnings(
                record=True
            ) as warning_list:

                warnings.simplefilter(
                    "always"
                )


                result = model.fit(

                    reml=False,

                    method=method,

                    maxiter=2000,

                    disp=False

                )


                captured_warnings.extend(

                    str(
                        warning.message
                    )

                    for warning
                    in warning_list

                )


            best_result = result

            optimizer_used = method


            if result.converged:
                break


        except Exception as error:

            captured_warnings.append(

                f"{method}: {error}"

            )


    if best_result is None:

        raise RuntimeError(

            "Mixed model could not be fit "
            "using the approved optimizer "
            "sequence."

        )


    return (

        best_result,

        optimizer_used,

        captured_warnings

    )

def fixed_effect_info(
    result
):

    names = list(
        result.fe_params.index
    )


    beta = (
        result.fe_params
        .to_numpy(
            dtype=float
        )
    )


    covariance = (

        result.cov_params()

        .loc[
            names,
            names
        ]

        .to_numpy(
            dtype=float
        )

    )


    return (
        names,
        beta,
        covariance
    )

def product_vectors(
    products,
    fixed_effect_names
):

    baseline_product = (
        products[0]
    )


    vectors = {}


    for product in products:

        vector = np.zeros(
            len(
                fixed_effect_names
            )
        )


        vector[
            fixed_effect_names.index(
                "Intercept"
            )
        ] = 1.0


        if product != baseline_product:

            term = (

                f"C(product_code)"
                f"[T.{product}]"

            )


            if term not in fixed_effect_names:

                raise RuntimeError(

                    "Could not locate "
                    f"fixed-effect term "
                    f"for Product {product}."

                )


            vector[
                fixed_effect_names.index(
                    term
                )
            ] = 1.0


        vectors[
            product
        ] = vector


    return vectors

def estimated_means(
    result,
    products
):

    (
        names,
        beta,
        covariance

    ) = fixed_effect_info(
        result
    )


    vectors = product_vectors(

        products,
        names

    )


    rows = []


    for product in products:

        vector = vectors[
            product
        ]


        mean = float(

            vector
            @ beta

        )


        variance = float(

            vector
            @ covariance
            @ vector

        )


        standard_error = float(

            np.sqrt(

                max(
                    variance,
                    0
                )

            )

        )


        rows.append({

            "product_code":
                product,

            "estimated_mean":
                mean,

            "ci95_low":
                mean
                - 1.96
                * standard_error,

            "ci95_high":
                mean
                + 1.96
                * standard_error,

        })


    return (

        pd.DataFrame(
            rows
        )

        .sort_values(
            "estimated_mean",
            ascending=False
        )

        .reset_index(
            drop=True
        )

    )

def omnibus_test(
    result
):

    (
        names,
        beta,
        covariance

    ) = fixed_effect_info(
        result
    )


    product_indices = [

        index

        for index, name
        in enumerate(
            names
        )

        if name != "Intercept"

    ]


    product_beta = (

        beta[
            product_indices
        ]

    )


    product_covariance = (

        covariance[

            np.ix_(

                product_indices,
                product_indices

            )

        ]

    )


    statistic = float(

        product_beta.T

        @ np.linalg.pinv(
            product_covariance
        )

        @ product_beta

    )


    degrees_freedom = len(
        product_indices
    )


    p_value = float(

        stats.chi2.sf(

            statistic,

            degrees_freedom

        )

    )


    return (

        statistic,

        degrees_freedom,

        p_value

    )

def pairwise_tests(
    result,
    products
):

    (
        names,
        beta,
        covariance

    ) = fixed_effect_info(
        result
    )


    vectors = product_vectors(

        products,
        names

    )


    rows = []


    for (
        product_1,
        product_2

    ) in itertools.combinations(
        products,
        2
    ):

        contrast = (

            vectors[
                product_1
            ]

            - vectors[
                product_2
            ]

        )


        difference = float(

            contrast
            @ beta

        )


        variance = float(

            contrast
            @ covariance
            @ contrast

        )


        standard_error = float(

            np.sqrt(

                max(
                    variance,
                    0
                )

            )

        )


        if standard_error > 0:

            z_value = (

                difference
                / standard_error

            )


            p_value = float(

                2
                * stats.norm.sf(

                    abs(
                        z_value
                    )

                )

            )


        else:

            p_value = np.nan


        rows.append({

            "product_1":
                product_1,

            "product_2":
                product_2,

            "difference":
                difference,

            "ci95_low":
                difference
                - 1.96
                * standard_error,

            "ci95_high":
                difference
                + 1.96
                * standard_error,

            "p_raw":
                p_value,

        })


    output = pd.DataFrame(
        rows
    )


    (
        reject,
        adjusted_p,
        _,
        _

    ) = multipletests(

        output[
            "p_raw"
        ].to_numpy(),

        alpha=ALPHA,

        method="holm"

    )


    output[
        "p_holm"
    ] = adjusted_p


    output[
        "significant_holm"
    ] = reject


    return output

def diagnostics(

    df,

    result,

    optimizer,

    warning_log

):

    rows = []

    hard_stop = False

    sensitivity_required = False

    sensitivity_triggers = []


    def add(
        name,
        observed,
        status
    ):

        rows.append({

            "diagnostic":
                name,

            "observed":
                observed,

            "status":
                status,

        })


    # -------------------------------------------------------------------------
    # Convergence
    # -------------------------------------------------------------------------

    if result.converged:

        add(

            "Model convergence",

            f"PASS ({optimizer})",

            "PASS"

        )


    else:

        add(

            "Model convergence",

            f"FAILED ({optimizer})",

            "HARD STOP"

        )


        hard_stop = True


    # -------------------------------------------------------------------------
    # Random intercept variance + ICC
    # -------------------------------------------------------------------------

    random_variance = float(

        np.asarray(
            result.cov_re
        )[0, 0]

    )


    residual_variance = float(
        result.scale
    )


    denominator = (

        random_variance
        + residual_variance

    )


    if denominator > 0:

        icc = (

            random_variance
            / denominator

        )


    else:

        icc = np.nan


    if (
        random_variance
        < RANDOM_VAR_BOUNDARY
    ):

        add(

            "Random-intercept variance",

            f"{random_variance:.4f}",

            "SENSITIVITY"

        )


        sensitivity_required = True


        sensitivity_triggers.append(

            "random-effect variance "
            f"< {RANDOM_VAR_BOUNDARY}"

        )


    else:

        add(

            "Random-intercept variance",

            f"{random_variance:.4f}",

            "PASS"

        )


    if (

        np.isfinite(
            icc
        )

        and icc
        < ICC_BOUNDARY

    ):

        add(

            "ICC",

            f"{icc:.4f}",

            "SENSITIVITY"

        )


        sensitivity_required = True


        sensitivity_triggers.append(

            f"ICC < {ICC_BOUNDARY}"

        )


    else:

        add(

            "ICC",

            (
                f"{icc:.4f}"

                if np.isfinite(
                    icc
                )

                else "NA"
            ),

            "PASS"

        )


    # -------------------------------------------------------------------------
    # Fixed-effect covariance / Hessian
    # -------------------------------------------------------------------------

    try:

        (
            _,
            _,
            covariance

        ) = fixed_effect_info(
            result
        )


        eigenvalues = (

            np.linalg.eigvalsh(

                (
                    covariance
                    + covariance.T
                )
                / 2

            )

        )


        covariance_ok = (

            np.isfinite(
                covariance
            ).all()

            and eigenvalues.min()
            > -1e-8

        )


    except Exception:

        covariance_ok = False


    if covariance_ok:

        add(

            "Fixed-effect covariance / Hessian",

            "valid",

            "PASS"

        )


    else:

        add(

            "Fixed-effect covariance / Hessian",

            "invalid",

            "HARD STOP"

        )


        hard_stop = True


    # -------------------------------------------------------------------------
    # Residuals
    # -------------------------------------------------------------------------

    residuals = np.asarray(

        result.resid,

        dtype=float

    )


    residuals = residuals[

        np.isfinite(
            residuals
        )

    ]


    if len(
        residuals
    ) < 3:

        add(

            "Residual diagnostics",

            "insufficient residuals",

            "HARD STOP"

        )


        return (

            pd.DataFrame(
                rows
            ),

            True,

            sensitivity_required,

            sensitivity_triggers

        )


    # -------------------------------------------------------------------------
    # Shapiro-Wilk
    # -------------------------------------------------------------------------

    shapiro_p = float(

        stats.shapiro(
            residuals
        ).pvalue

    )


    if (
        shapiro_p
        < SHAPIRO_WARNING_P
    ):

        shapiro_status = (
            "WARNING"
        )


    else:

        shapiro_status = (
            "PASS"
        )


    add(

        "Shapiro-Wilk residual normality",

        f"p={shapiro_p:.4f}",

        shapiro_status

    )


    # -------------------------------------------------------------------------
    # Residual skewness
    # -------------------------------------------------------------------------

    skewness = float(

        stats.skew(

            residuals,

            bias=False

        )

    )


    if (
        abs(
            skewness
        )
        > SKEW_SENSITIVITY
    ):

        add(

            "Residual skewness",

            f"{skewness:.3f}",

            "SENSITIVITY"

        )


        sensitivity_required = True


        sensitivity_triggers.append(

            "|residual skewness| "
            f"> {SKEW_SENSITIVITY}"

        )


    else:

        add(

            "Residual skewness",

            f"{skewness:.3f}",

            "PASS"

        )


    # -------------------------------------------------------------------------
    # Pearson kurtosis
    # -------------------------------------------------------------------------

    kurtosis = float(

        stats.kurtosis(

            residuals,

            fisher=False,

            bias=False

        )

    )


    if (
        kurtosis
        > KURTOSIS_SENSITIVITY
    ):

        add(

            "Residual Pearson kurtosis",

            f"{kurtosis:.3f}",

            "SENSITIVITY"

        )


        sensitivity_required = True


        sensitivity_triggers.append(

            "Pearson kurtosis "
            f"> {KURTOSIS_SENSITIVITY}"

        )


    else:

        add(

            "Residual Pearson kurtosis",

            f"{kurtosis:.3f}",

            "PASS"

        )


    # -------------------------------------------------------------------------
    # Heteroscedasticity by product
    # -------------------------------------------------------------------------

    temp = df.copy()


    temp[
        "_residual"
    ] = np.asarray(

        result.resid,

        dtype=float

    )


    residual_groups = [

        group[
            "_residual"
        ]

        .dropna()

        .to_numpy()

        for _,
        group

        in temp.groupby(

            "product_code",

            observed=True

        )

        if len(
            group
        ) >= 2

    ]


    brown_forsythe_p = float(

        stats.levene(

            *residual_groups,

            center="median"

        ).pvalue

    )


    variances = np.array([

        np.var(
            group,
            ddof=1
        )

        for group
        in residual_groups

    ])


    positive_variances = (

        variances[
            variances > 0
        ]

    )


    if len(
        positive_variances
    ):

        variance_ratio = float(

            positive_variances.max()

            / positive_variances.min()

        )


    else:

        variance_ratio = np.inf


    if (
        variance_ratio
        >= VAR_RATIO_SENSITIVITY
    ):

        hetero_status = (
            "SENSITIVITY"
        )


        sensitivity_required = True


        sensitivity_triggers.append(

            "residual variance ratio "
            f"≥ {VAR_RATIO_SENSITIVITY}"

        )


    elif (

        brown_forsythe_p
        < BF_WARNING_P

        or variance_ratio
        >= VAR_RATIO_WARNING

    ):

        hetero_status = (
            "WARNING"
        )


    else:

        hetero_status = (
            "PASS"
        )


    add(

        "Brown-Forsythe / residual variance ratio",

        (
            f"p={brown_forsythe_p:.4f}; "
            f"ratio={variance_ratio:.2f}"
        ),

        hetero_status

    )


    # -------------------------------------------------------------------------
    # Extreme residuals
    # -------------------------------------------------------------------------

    residual_sd = float(

        np.std(
            residuals,
            ddof=1
        )

    )


    if residual_sd > 0:

        standardized = (

            residuals
            / residual_sd

        )


    else:

        standardized = np.zeros_like(
            residuals
        )


    pct_beyond_3 = float(

        np.mean(

            np.abs(
                standardized
            )

            > EXTREME_RESIDUAL

        )

        * 100

    )


    maximum_absolute = float(

        np.max(

            np.abs(
                standardized
            )

        )

    )


    if (

        maximum_absolute
        > EXTREME_RESIDUAL_HARD

        and pct_beyond_3
        > EXTREME_PCT_SENSITIVITY

    ):

        extreme_status = (
            "SENSITIVITY"
        )


        sensitivity_required = True


        sensitivity_triggers.append(

            f">{EXTREME_PCT_SENSITIVITY}% "
            "residuals beyond ±3 with "
            f"max beyond ±"
            f"{EXTREME_RESIDUAL_HARD}"

        )


    elif (
        maximum_absolute
        > EXTREME_RESIDUAL
    ):

        extreme_status = (
            "WARNING"
        )


    else:

        extreme_status = (
            "PASS"
        )


    add(

        "Extreme standardized residuals",

        (
            f"{pct_beyond_3:.2f}% > |3|; "
            f"max |r|={maximum_absolute:.2f}"
        ),

        extreme_status

    )


    # -------------------------------------------------------------------------
    # Statsmodels warning log
    # -------------------------------------------------------------------------

    warning_text = (

        " | ".join(
            warning_log
        )

        .lower()

    )


    warning_phrases = [

        "random effects covariance is singular",

        "hessian matrix at the estimated "
        "parameter values is not positive definite",

        "singular matrix",

    ]


    if any(

        phrase
        in warning_text

        for phrase
        in warning_phrases

    ):

        add(

            "Model warning log",

            "singularity/Hessian warning",

            "SENSITIVITY"

        )


        sensitivity_required = True


        sensitivity_triggers.append(

            "statsmodels "
            "singularity/Hessian warning"

        )


    return (

        pd.DataFrame(
            rows
        ),

        hard_stop,

        sensitivity_required,

        sorted(
            set(
                sensitivity_triggers
            )
        )

    )

def sensitivity_analysis(
    df,
    products
):

    model = smf.ols(

        "overall_liking "
        "~ C(product_code) "
        "+ C(panelist_id)",

        data=df

    ).fit(

        cov_type="HC3"

    )


    design_info = (
        model.model.data.design_info
    )


    # IMPORTANT:
    # Preserve the original datatype of panelist_id.
    # This fixes the Patsy categorical error that occurs
    # when numeric IDs such as 1 are converted to strings such as "1".

    reference_panelist = (

        df[
            "panelist_id"
        ]

        .dropna()

        .iloc[0]

    )


    vectors = {}


    for product in products:

        new_data = pd.DataFrame({

            "product_code":
                [product],

            "panelist_id":
                [reference_panelist],

        })


        matrix = (

            patsy.build_design_matrices(

                [
                    design_info
                ],

                new_data,

                return_type="dataframe"

            )[0]

        )


        vectors[
            product
        ] = (

            matrix
            .to_numpy(
                dtype=float
            )

            .ravel()

        )


    beta = np.asarray(

        model.params,

        dtype=float

    )


    covariance = np.asarray(

        model.cov_params(),

        dtype=float

    )


    rows = []


    for (
        product_1,
        product_2

    ) in itertools.combinations(
        products,
        2
    ):

        contrast = (

            vectors[
                product_1
            ]

            - vectors[
                product_2
            ]

        )


        difference = float(

            contrast
            @ beta

        )


        variance = float(

            contrast
            @ covariance
            @ contrast

        )


        standard_error = float(

            np.sqrt(

                max(
                    variance,
                    0
                )

            )

        )


        if standard_error > 0:

            z_value = (

                difference
                / standard_error

            )


            p_value = float(

                2
                * stats.norm.sf(

                    abs(
                        z_value
                    )

                )

            )


        else:

            p_value = np.nan


        rows.append({

            "product_1":
                product_1,

            "product_2":
                product_2,

            "difference":
                difference,

            "p_raw":
                p_value,

        })


    pairwise = pd.DataFrame(
        rows
    )


    (
        reject,
        adjusted_p,
        _,
        _

    ) = multipletests(

        pairwise[
            "p_raw"
        ].to_numpy(),

        alpha=ALPHA,

        method="holm"

    )


    pairwise[
        "p_holm"
    ] = adjusted_p


    pairwise[
        "significant_holm"
    ] = reject


    # -------------------------------------------------------------------------
    # Overall product test for sensitivity model.
    # -------------------------------------------------------------------------

    parameter_names = list(
        model.params.index
    )


    product_indices = [

        index

        for index, name
        in enumerate(
            parameter_names
        )

        if name.startswith(

            "C(product_code)[T."

        )

    ]


    restriction_matrix = np.zeros(

        (
            len(
                product_indices
            ),

            len(
                parameter_names
            )
        )

    )


    for row_index, parameter_index in enumerate(
        product_indices
    ):

        restriction_matrix[
            row_index,
            parameter_index
        ] = 1


    wald_test = model.wald_test(

        restriction_matrix,

        scalar=True

    )


    overall_p = float(
        wald_test.pvalue
    )


    return (

        overall_p,

        pairwise

    )

def sensitivity_changed(

    primary_p,

    primary_pairs,

    sensitivity_p,

    sensitivity_pairs

):

    reasons = []


    if (

        (
            primary_p
            < ALPHA
        )

        !=

        (
            sensitivity_p
            < ALPHA
        )

    ):

        reasons.append(

            "overall product-effect "
            "conclusion changes"

        )


    primary_decisions = (

        primary_pairs

        .set_index(
            [
                "product_1",
                "product_2"
            ]
        )[
            "significant_holm"
        ]

    )


    sensitivity_decisions = (

        sensitivity_pairs

        .set_index(
            [
                "product_1",
                "product_2"
            ]
        )[
            "significant_holm"
        ]

    )


    changed_pairs = int(

        (
            primary_decisions
            != sensitivity_decisions
        ).sum()

    )


    if changed_pairs:

        reasons.append(

            f"{changed_pairs} "
            "Holm pairwise decision(s) change"

        )


    return (

        bool(
            reasons
        ),

        reasons

    )

def compact_letters(
    means,
    pairs
):

    """
    Highest estimated mean gets the first letter ("a").

    Significant pairs share no letters.

    Non-significant pairs share at least one letter.
    """


    ordered = (

        means

        .sort_values(
            "estimated_mean",
            ascending=False
        )[
            "product_code"
        ]

        .tolist()

    )


    significant_pairs = {

        frozenset([
            row.product_1,
            row.product_2
        ])

        for row
        in pairs.itertuples()

        if bool(
            row.significant_holm
        )

    }


    letter_columns = []


    for product in ordered:

        placed = False


        for column in letter_columns:

            compatible = all(

                frozenset([
                    product,
                    other
                ])

                not in
                significant_pairs

                for other
                in column

            )


            if compatible:

                column.add(
                    product
                )

                placed = True


        if not placed:

            letter_columns.append(
                {
                    product
                }
            )


        # Ensure each non-significant pair shares
        # at least one letter.

        for other in ordered:

            if (
                other
                == product
            ):

                continue


            pair = frozenset([
                product,
                other
            ])


            if pair in significant_pairs:
                continue


            already_shared = any(

                product in column

                and other in column

                for column
                in letter_columns

            )


            if not already_shared:

                new_column = {
                    product,
                    other
                }


                for candidate in ordered:

                    if candidate in new_column:
                        continue


                    compatible = all(

                        frozenset([
                            candidate,
                            member
                        ])

                        not in
                        significant_pairs

                        for member
                        in new_column

                    )


                    if compatible:

                        new_column.add(
                            candidate
                        )


                letter_columns.append(
                    new_column
                )


    # -------------------------------------------------------------------------
    # Remove redundant subset columns.
    # -------------------------------------------------------------------------

    reduced_columns = []


    for index, column in enumerate(
        letter_columns
    ):

        redundant = any(

            column
            < other

            for other_index,
            other

            in enumerate(
                letter_columns
            )

            if other_index
            != index

        )


        if not redundant:

            reduced_columns.append(
                column
            )


    top_product = ordered[
        0
    ]


    reduced_columns.sort(

        key=lambda column: (

            top_product
            not in column,

            -len(
                column
            ),

            sorted(
                column
            )

        )

    )


    labels = {

        product: ""

        for product
        in ordered

    }


    for index, column in enumerate(
        reduced_columns
    ):

        letter = chr(

            ord("a")
            + index

        )


        for product in ordered:

            if product in column:

                labels[
                    product
                ] += letter


    return labels

def is_export_index(series: pd.Series) -> bool:
    """Return True only for an obvious 0..n-1 CSV export index."""
    numeric = pd.to_numeric(series, errors="coerce")
    if numeric.isna().any():
        return False
    expected = np.arange(len(series), dtype=float)
    return np.array_equal(numeric.to_numpy(dtype=float), expected)

def remove_export_index(df: pd.DataFrame) -> Tuple[pd.DataFrame, List[str]]:
    """Remove only obvious export-index artifacts, never substantive columns."""
    out = df.copy()
    removed: List[str] = []

    for col in list(out.columns):
        normalized = str(col).strip().lower()
        if normalized in {"index", "unnamed: 0"} and is_export_index(out[col]):
            out = out.drop(columns=[col])
            removed.append(str(col))

    return out, removed

        classification = "SUPPORTED"
    else:
        # SOP-03 has no separable inferential sub-question. All inputs are required.
        classification = "UNSUPPORTED"

    return {
        "classification": classification,
        "required": REQUIRED_COLUMNS.copy(),
        "missing": missing,
    }

def prepare_analysis_data(
    raw_df: pd.DataFrame,
) -> Tuple[pd.DataFrame, Dict[str, object], List[str], List[str]]:
    """Apply capability, scale, and repeated-measures design checks."""
    cleaned, removed_columns = remove_export_index(raw_df)
    capability = capability_gate(cleaned)

    if capability["classification"] != "SUPPORTED":
        raise CapabilityStop(
            f"{SOP_ID} DATASET CAPABILITY GATE: UNSUPPORTED. "
            f"Missing required variable(s): {capability['missing']}. "
            "No proxy or substitute construct is authorized."
        )

    df = cleaned[REQUIRED_COLUMNS].copy()

    # Product labels are required identifiers.
    df[PRODUCT] = df[PRODUCT].astype("string").str.strip()
    blank_product = df[PRODUCT].fillna("").eq("")
    if blank_product.any():
        df.loc[blank_product, PRODUCT] = pd.NA

    missing_id_rows = int(df[[PRODUCT, PANELIST]].isna().any(axis=1).sum())
    if missing_id_rows > 0:
        raise DesignStop(
            f"{SOP_ID} HARD STOP: {missing_id_rows} row(s) have missing "
            "product_code or panelist_id. These identifiers are required to "
            "preserve the repeated-measures design."
        )

    # Outcome must be numeric when present.
    originally_nonmissing = df[OUTCOME].notna()
    df[OUTCOME] = pd.to_numeric(df[OUTCOME], errors="coerce")

    newly_invalid = int((originally_nonmissing & df[OUTCOME].isna()).sum())
    if newly_invalid > 0:
        raise CapabilityStop(
            f"{SOP_ID} HARD STOP: {newly_invalid} non-numeric {OUTCOME} value(s) "
            "were found. The approved outcome is a numeric 1–9 intensity rating."
        )

    # Validate the approved 1–9 scale.
    out_of_range = df[OUTCOME].dropna()
    out_of_range = out_of_range[
        (out_of_range < SCALE_MIN) | (out_of_range > SCALE_MAX)
    ]

    if not out_of_range.empty:
        raise CapabilityStop(
            f"{SOP_ID} HARD STOP: {OUTCOME} contains {len(out_of_range)} "
            f"value(s) outside the approved {int(SCALE_MIN)}–{int(SCALE_MAX)} scale."
        )

    # One panelist × product row is expected. Do not silently average duplicates.
    duplicates = df.duplicated(
        subset=[PANELIST, PRODUCT],
        keep=False,
    )

    if duplicates.any():
        n_rows = int(duplicates.sum())
        n_cells = int(
            df.loc[duplicates, [PANELIST, PRODUCT]]
            .drop_duplicates()
            .shape[0]
        )
        raise DesignStop(
            f"{SOP_ID} HARD STOP: duplicate panelist × product evaluation cells "
            f"detected ({n_rows} rows across {n_cells} duplicated cells). "
            "Resolve duplicates before modeling."
        )

    design_warnings: List[str] = []

    # MixedLM uses complete model rows.
    before = len(df)
    df = df.dropna(subset=[OUTCOME, PRODUCT, PANELIST]).copy()
    dropped_missing = before - len(df)

    if dropped_missing > 0:
        design_warnings.append(
            f"Excluded {dropped_missing} row(s) with missing required model values; "
            f"{len(df)} complete evaluation rows remained."
        )

    if len(df) == 0:
        raise CapabilityStop(
            f"{SOP_ID} HARD STOP: no complete rows remain for analysis."
        )

    if df[PRODUCT].nunique() < 2:
        raise DesignStop(
            f"{SOP_ID} HARD STOP: fewer than two products remain for comparison."
        )

    if df[PANELIST].nunique() < 2:
        raise DesignStop(
            f"{SOP_ID} HARD STOP: fewer than two panelists remain."
        )

    # Confirm actual repeated observations within panelists.
    product_counts_by_panelist = (
        df.groupby(PANELIST, observed=True)[PRODUCT].nunique()
    )
    repeated_panelists = int((product_counts_by_panelist >= 2).sum())

    if repeated_panelists < 2:
        raise DesignStop(
            f"{SOP_ID} HARD STOP: insufficient repeated product evaluations "
            "within panelist for the approved mixed-effects model."
        )

    if df[OUTCOME].nunique(dropna=True) < 2:
        raise DesignStop(
            f"{SOP_ID} HARD STOP: {OUTCOME} has no usable variation "
            "in the analyzed rows."
        )

    # Stable categorical coding for Patsy / statsmodels.
    products = sorted(df[PRODUCT].astype(str).unique().tolist())
    df[PRODUCT] = pd.Categorical(
        df[PRODUCT],
        categories=products,
        ordered=True,
    )

    # Expected workshop counts are disclosures, not assumptions.
    if products != EXPECTED_PRODUCTS:
        design_warnings.append(
            f"Observed products: {products}; "
            f"workshop expectation: {EXPECTED_PRODUCTS}."
        )

    n_panelists = int(df[PANELIST].nunique())

    if n_panelists != EXPECTED_PANELISTS:
        design_warnings.append(
            f"Observed {n_panelists} unique panelists; "
            f"workshop expectation: {EXPECTED_PANELISTS}."
        )

    if len(df) != EXPECTED_ROWS:
        design_warnings.append(
            f"Analyzed {len(df)} complete evaluation rows; "
            f"workshop expectation: {EXPECTED_ROWS}."
        )

    evals_per_panelist = df.groupby(PANELIST, observed=True).size()

    if evals_per_panelist.nunique() != 1:
        design_warnings.append(
            "The analyzed dataset is unbalanced: panelists do not all contribute "
            "the same number of product evaluations. MixedLM can accommodate "
            "imbalance, but it is disclosed."
        )

    # At least two observations per product are required for residual variance checks.
    cells_per_product = df.groupby(PRODUCT, observed=True).size()

    if (cells_per_product < 2).any():
        low_products = cells_per_product[cells_per_product < 2].to_dict()
        raise DesignStop(
            f"{SOP_ID} HARD STOP: at least one product has fewer than two "
            f"analyzed observations ({low_products}), so required product-level "
            "residual variance diagnostics cannot be completed."
        )

    return df, capability, removed_columns, design_warnings

def fit_primary_mixed_model(df: pd.DataFrame):
    """Fit only the reviewed BFGS → L-BFGS → CG optimizer sequence."""

    model = smf.mixedlm(
        f"{OUTCOME} ~ C({PRODUCT})",
        data=df,
        groups=df[PANELIST],
        re_formula="1",
    )

    attempts: List[Dict[str, object]] = []
    last_result = None
    last_warning_text: List[str] = []
    last_exception: Optional[Exception] = None

    for method in OPTIMIZER_SEQUENCE:
        captured: List[str] = []

        try:
            with warnings.catch_warnings(record=True) as caught:
                warnings.simplefilter("always")

                result = model.fit(
                    reml=True,
                    method=method,
                    maxiter=2000,
                    disp=False,
                )

                captured = [str(w.message) for w in caught]

            converged = bool(getattr(result, "converged", False))

            attempts.append(
                {
                    "optimizer": method,
                    "completed": True,
                    "converged": converged,
                    "warnings": " | ".join(captured) if captured else "None",
                }
            )

            last_result = result
            last_warning_text = captured

            if converged:
                return result, captured, attempts

        except Exception as exc:
            last_exception = exc

            attempts.append(
                {
                    "optimizer": method,
                    "completed": False,
                    "converged": False,
                    "warnings": f"Exception: {exc}",
                }
            )

    if last_result is not None:
        # Return final candidate so A1 can record the hard stop.
        return last_result, last_warning_text, attempts

    raise InferenceStop(
        f"{SOP_ID} A1 HARD STOP: all approved MixedLM optimizer attempts failed. "
        f"Last error: {last_exception}"
    )

def fixed_effect_covariance(result) -> pd.DataFrame:
    """Extract only the fixed-effect covariance block."""
    fe_names = list(result.fe_params.index)
    covariance = result.cov_params()

    if isinstance(covariance, pd.DataFrame):
        try:
            block = covariance.loc[fe_names, fe_names]
        except Exception:
            block = covariance.iloc[: len(fe_names), : len(fe_names)]

        return pd.DataFrame(
            block,
            index=fe_names,
            columns=fe_names,
        )

    array = np.asarray(covariance, dtype=float)
    array = array[: len(fe_names), : len(fe_names)]

    return pd.DataFrame(
        array,
        index=fe_names,
        columns=fe_names,
    )

def mixedlm_design_vector(result, product: str) -> np.ndarray:
    """Build the fixed-effect design row for one product mean."""
    design_info = result.model.data.design_info
    frame = result.model.data.frame
    categories = list(frame[PRODUCT].cat.categories)

    new_data = pd.DataFrame(
        {
            PRODUCT: pd.Categorical(
                [product],
                categories=categories,
                ordered=True,
            )
        }
    )

    matrix = patsy.build_design_matrices(
        [design_info],
        new_data,
        return_type="dataframe",
    )[0]

    return matrix.to_numpy(dtype=float).ravel()

def estimate_linear_function(
    beta: np.ndarray,
    covariance: np.ndarray,
    lvector: np.ndarray,
) -> Tuple[float, float, float, float, float, float]:
    """Estimate, SE, z, raw p, and 95% CI for a linear coefficient function."""

    estimate = float(lvector @ beta)
    variance = float(lvector @ covariance @ lvector.T)

    if not np.isfinite(variance) or variance < -NUMERIC_TOL:
        raise InferenceStop(
            "Invalid variance encountered for a required model contrast."
        )

    variance = max(variance, 0.0)
    se = math.sqrt(variance)

    if not np.isfinite(se) or se <= NUMERIC_TOL:
        raise InferenceStop(
            "A required model contrast has an invalid or zero standard error."
        )

    z_value = estimate / se
    p_value = float(2 * stats.norm.sf(abs(z_value)))

    ci_low = estimate - Z_CRITICAL * se
    ci_high = estimate + Z_CRITICAL * se

    return (
        estimate,
        se,
        z_value,
        p_value,
        ci_low,
        ci_high,
    )

def mixedlm_estimated_means(
    result,
    products: Sequence[str],
) -> pd.DataFrame:

    beta = result.fe_params.to_numpy(dtype=float)
    covariance = fixed_effect_covariance(result).to_numpy(dtype=float)

    rows: List[Dict[str, object]] = []

    for product in products:
        lvec = mixedlm_design_vector(result, product)

        estimate, se, _, _, ci_low, ci_high = estimate_linear_function(
            beta,
            covariance,
            lvec,
        )

        rows.append(
            {
                PRODUCT: product,
                "estimated_mean": estimate,
                "se": se,
                "ci95_low": ci_low,
                "ci95_high": ci_high,
            }
        )

    return (
        pd.DataFrame(rows)
        .sort_values("estimated_mean", ascending=False)
        .reset_index(drop=True)
    )

def apply_holm(pairwise: pd.DataFrame) -> pd.DataFrame:
    """Apply Holm multiplicity correction to the complete pairwise family."""

    if pairwise.empty:
        raise InferenceStop(
            "No pairwise comparisons were available for Holm correction."
        )

    p_values = pairwise["p_raw"].to_numpy(dtype=float)

    if not np.isfinite(p_values).all():
        raise InferenceStop(
            "Non-finite pairwise p-values prevent valid Holm correction."
        )

    reject, p_holm, _, _ = multipletests(
        p_values,
        alpha=ALPHA,
        method="holm",
    )

    out = pairwise.copy()
    out["p_holm"] = p_holm
    out["significant_holm"] = reject.astype(bool)

    return out

def mixedlm_pairwise_contrasts(
    result,
    means: pd.DataFrame,
) -> pd.DataFrame:
    """Complete family of model-based product contrasts + Holm correction."""

    ordered_products = means[PRODUCT].astype(str).tolist()

    beta = result.fe_params.to_numpy(dtype=float)
    covariance = fixed_effect_covariance(result).to_numpy(dtype=float)

    rows: List[Dict[str, object]] = []

    for i, product_1 in enumerate(ordered_products):
        for product_2 in ordered_products[i + 1 :]:

            lvec = (
                mixedlm_design_vector(result, product_1)
                - mixedlm_design_vector(result, product_2)
            )

            (
                estimate,
                se,
                z_value,
                p_raw,
                ci_low,
                ci_high,
            ) = estimate_linear_function(
                beta,
                covariance,
                lvec,
            )

            rows.append(
                {
                    "product_1": product_1,
                    "product_2": product_2,
                    "contrast": f"{product_1} - {product_2}",
                    "difference": estimate,
                    "se": se,
                    "z": z_value,
                    "ci95_low": ci_low,
                    "ci95_high": ci_high,
                    "p_raw": p_raw,
                    "magnitude_points": abs(estimate),
                }
            )

    out = apply_holm(pd.DataFrame(rows))

    out["magnitude_note"] = out["magnitude_points"].map(
        lambda value: (
            f"{value:.2f} point(s) on the {SCALE_LABEL}"
        )
    )

    return (
        out.sort_values(
            ["significant_holm", "magnitude_points"],
            ascending=[False, False],
        )
        .reset_index(drop=True)
    )

def mixedlm_omnibus_product_test(
    result,
) -> Tuple[float, int, float]:
    """Joint Wald chi-square test for the fixed product effect."""

    names = list(result.fe_params.index)

    product_indices = [
        i
        for i, name in enumerate(names)
        if f"C({PRODUCT})" in str(name)
    ]

    if not product_indices:
        raise InferenceStop(
            "Could not identify product fixed-effect coefficients "
            "for the omnibus test."
        )

    beta = result.fe_params.to_numpy(dtype=float)
    covariance = fixed_effect_covariance(result).to_numpy(dtype=float)

    b = beta[product_indices]
    v = covariance[np.ix_(product_indices, product_indices)]

    if not np.isfinite(v).all():
        raise InferenceStop(
            "Invalid covariance matrix for the omnibus product test."
        )

    try:
        solved = np.linalg.solve(v, b)
    except np.linalg.LinAlgError as exc:
        raise InferenceStop(
            f"{SOP_ID} A2 HARD STOP: covariance inversion failed "
            "for the omnibus product test."
        ) from exc

    chi2 = float(b.T @ solved)
    df_test = len(product_indices)
    p_value = float(stats.chi2.sf(chi2, df_test))

    return chi2, df_test, p_value

def run_g9a_diagnostics(
    df: pd.DataFrame,
    result,
    warning_text: Sequence[str],
) -> Tuple[pd.DataFrame, bool, bool, List[str]]:
    """Apply every reviewed G9-A threshold required for SOP-03."""

    diagnostics: List[Dict[str, object]] = []
    hard_stop = False
    sensitivity_required = False
    triggers: List[str] = []

    warning_blob = " | ".join(warning_text).lower()

    # -------------------------------------------------------------------------
    # A1 — optimizer convergence
    # -------------------------------------------------------------------------

    converged = bool(getattr(result, "converged", False))

    final_convergence_warning = any(
        phrase in warning_blob
        for phrase in [
            "failed to converge",
            "gradient optimization failed",
            "optimization failed",
            "maximum likelihood optimization failed",
        ]
    )

    a1_pass = converged and not final_convergence_warning

    if not a1_pass:
        hard_stop = True

    diagnostics.append(
        diag_row(
            "A1 Optimizer convergence",
            (
                f"converged={converged}; "
                f"final convergence warning={final_convergence_warning}"
            ),
            (
                "result.converged=True and no final warning "
                "that optimum was not reached"
            ),
            "PASS" if a1_pass else "HARD STOP",
            (
                "Proceed."
                if a1_pass
                else "Suppress substantive mixed-model inference."
            ),
        )
    )

    # -------------------------------------------------------------------------
    # A2 — singular covariance / invalid Hessian / invalid covariance
    # -------------------------------------------------------------------------

    cov_re_finite = False
    min_cov_re_eig = float("nan")

    try:
        cov_re = np.asarray(result.cov_re, dtype=float)
        cov_re_finite = bool(np.isfinite(cov_re).all())

        if cov_re_finite:
            sym_cov_re = (cov_re + cov_re.T) / 2
            min_cov_re_eig = float(
                np.min(np.linalg.eigvalsh(sym_cov_re))
            )

    except Exception:
        cov_re_finite = False

    singular_warning = any(
        phrase in warning_blob
        for phrase in [
            "random effects covariance is singular",
            "covariance matrix is singular",
            "singular matrix",
            "pseudo-inverse",
            "pseudoinverse",
        ]
    )

    hessian_singular = False
    hessian_valid = False
    min_neg_hessian_eig = float("nan")
    hessian_error = ""

    try:
        hess, hessian_singular = result.model.hessian(
            result.params_object
        )

        hess = np.asarray(hess, dtype=float)
        neg_hess = -((hess + hess.T) / 2)

        if np.isfinite(neg_hess).all():
            eigvals = np.linalg.eigvalsh(neg_hess)
            min_neg_hessian_eig = float(np.min(eigvals))
            hessian_valid = (
                min_neg_hessian_eig > NUMERIC_TOL
            )

    except Exception as exc:
        hessian_error = str(exc)
        hessian_valid = False

    hessian_warning = (
        "hessian" in warning_blob
        and "not positive definite" in warning_blob
    )

    covariance_valid = True
    covariance_error = ""

    try:
        cov_fe = fixed_effect_covariance(
            result
        ).to_numpy(dtype=float)

        covariance_valid = bool(
            np.isfinite(cov_fe).all()
            and np.all(np.diag(cov_fe) >= 0)
        )

    except Exception as exc:
        covariance_valid = False
        covariance_error = str(exc)

    a2_fail = (
        singular_warning
        or not cov_re_finite
        or not np.isfinite(min_cov_re_eig)
        or min_cov_re_eig <= NUMERIC_TOL
        or bool(hessian_singular)
        or hessian_warning
        or not hessian_valid
        or not covariance_valid
    )

    if a2_fail:
        hard_stop = True

    a2_observed = (
        f"min random-cov eigen={min_cov_re_eig:.6g}; "
        f"min(-Hessian) eigen={min_neg_hessian_eig:.6g}; "
        f"singular warning={singular_warning}; "
        f"hessian singular={bool(hessian_singular)}; "
        f"covariance valid={covariance_valid}"
    )

    if hessian_error:
        a2_observed += f"; Hessian error={hessian_error}"

    if covariance_error:
        a2_observed += f"; covariance error={covariance_error}"

    diagnostics.append(
        diag_row(
            "A2 Random-effects covariance / Hessian / SE covariance",
            a2_observed,
            (
                "No singular covariance, pseudo-inverse handling, "
                "non-PD Hessian, or invalid SE covariance"
            ),
            "HARD STOP" if a2_fail else "PASS",
            (
                "Suppress substantive inference."
                if a2_fail
                else "Proceed."
            ),
        )
    )

    # Calculate random-intercept variance and ICC.
    random_variance = float("nan")

    try:
        random_variance = float(
            np.asarray(result.cov_re, dtype=float).ravel()[0]
        )
    except Exception:
        pass

    residual_variance = safe_float(
        getattr(result, "scale", float("nan"))
    )

    if (
        np.isfinite(random_variance)
        and np.isfinite(residual_variance)
        and random_variance >= 0
        and residual_variance >= 0
        and (random_variance + residual_variance) > 0
    ):
        icc = (
            random_variance
            / (random_variance + residual_variance)
        )
    else:
        icc = float("nan")

    # -------------------------------------------------------------------------
    # A3 — boundary random-effect variance
    # -------------------------------------------------------------------------

    a3_trigger = (
        np.isfinite(random_variance)
        and random_variance < BOUNDARY_RANDOM_VARIANCE
    )

    if a3_trigger and not a2_fail:
        sensitivity_required = True
        triggers.append(
            f"A3 random-intercept variance "
            f"{random_variance:.6f} < "
            f"{BOUNDARY_RANDOM_VARIANCE:.2f}"
        )

    diagnostics.append(
        diag_row(
            "A3 Random-intercept variance",
            random_variance,
            (
                "WARNING / SENSITIVITY if < "
                f"{BOUNDARY_RANDOM_VARIANCE}"
            ),
            (
                "WARNING / SENSITIVITY REQUIRED"
                if a3_trigger
                else "PASS"
            ),
            (
                "Run repeated-measures sensitivity analysis."
                if a3_trigger
                else "Proceed."
            ),
        )
    )

    # -------------------------------------------------------------------------
    # A4 — near-zero ICC
    # -------------------------------------------------------------------------

    a4_trigger = (
        np.isfinite(icc)
        and icc < NEAR_ZERO_ICC
    )

    if a4_trigger and not a2_fail:
        sensitivity_required = True
        triggers.append(
            f"A4 ICC {icc:.6f} < {NEAR_ZERO_ICC:.2f}"
        )

    diagnostics.append(
        diag_row(
            "A4 Intraclass correlation (ICC)",
            icc,
            f"WARNING / SENSITIVITY if < {NEAR_ZERO_ICC}",
            (
                "WARNING / SENSITIVITY REQUIRED"
                if a4_trigger
                else (
                    "PASS"
                    if np.isfinite(icc)
                    else "HARD STOP"
                )
            ),
            (
                "Run repeated-measures sensitivity analysis."
                if a4_trigger
                else (
                    "Proceed."
                    if np.isfinite(icc)
                    else (
                        "ICC could not be calculated; "
                        "suppress inference."
                    )
                )
            ),
        )
    )

    if not np.isfinite(icc):
        hard_stop = True

    # Required residual diagnostics.
    try:
        residuals = pd.Series(
            np.asarray(result.resid, dtype=float),
            index=df.index,
            name="residual",
        )

    except Exception as exc:
        hard_stop = True

        diagnostics.append(
            diag_row(
                "A5–A7 Residual diagnostics availability",
                f"Residual extraction failed: {exc}",
                "Residual diagnostics are mandatory",
                "HARD STOP",
                (
                    "Suppress inference because required "
                    "diagnostics cannot be completed."
                ),
            )
        )

        return (
            pd.DataFrame(diagnostics),
            hard_stop,
            sensitivity_required,
            triggers,
        )

    if (
        residuals.isna().any()
        or not np.isfinite(residuals.to_numpy()).all()
    ):
        hard_stop = True

        diagnostics.append(
            diag_row(
                "A5–A7 Residual diagnostics availability",
                "Residual vector contains non-finite values",
                (
                    "Residual diagnostics require "
                    "finite residuals"
                ),
                "HARD STOP",
                "Suppress inference.",
            )
        )

        return (
            pd.DataFrame(diagnostics),
            hard_stop,
            sensitivity_required,
            triggers,
        )

    # -------------------------------------------------------------------------
    # A5 — residual distribution
    # -------------------------------------------------------------------------

    if len(residuals) >= 3:
        try:
            if len(residuals) <= 5000:
                (
                    normality_stat,
                    normality_p,
                ) = stats.shapiro(
                    residuals.to_numpy()
                )

                normality_test_name = "Shapiro-Wilk"

            else:
                (
                    normality_stat,
                    normality_p,
                ) = stats.normaltest(
                    residuals.to_numpy()
                )

                normality_test_name = "D'Agostino K²"

            normality_p = float(normality_p)

        except Exception:
            normality_stat = float("nan")
            normality_p = float("nan")
            normality_test_name = (
                "formal normality test unavailable"
            )

    else:
        normality_stat = float("nan")
        normality_p = float("nan")
        normality_test_name = "insufficient residuals"

    skewness = (
        float(
            stats.skew(
                residuals.to_numpy(),
                bias=False,
            )
        )
        if len(residuals) >= 3
        else float("nan")
    )

    pearson_kurtosis = (
        float(
            stats.kurtosis(
                residuals.to_numpy(),
                fisher=False,
                bias=False,
            )
        )
        if len(residuals) >= 4
        else float("nan")
    )

    normality_warning = (
        np.isfinite(normality_p)
        and normality_p < NORMALITY_WARNING_P
    )

    skew_trigger = (
        np.isfinite(skewness)
        and abs(skewness) > RESIDUAL_SKEW_SENSITIVITY
    )

    kurtosis_trigger = (
        np.isfinite(pearson_kurtosis)
        and pearson_kurtosis
        > PEARSON_KURTOSIS_SENSITIVITY
    )

    a5_sensitivity = (
        skew_trigger
        or kurtosis_trigger
    )

    if a5_sensitivity and not hard_stop:
        sensitivity_required = True
        triggers.append(
            f"A5 residual shape: skew={skewness:.3f}, "
            f"Pearson kurtosis={pearson_kurtosis:.3f}"
        )

    diagnostics.append(
        diag_row(
            (
                "A5 Formal residual normality "
                f"({normality_test_name})"
            ),
            f"p={fmt_p(normality_p)}",
            (
                "WARNING if p < "
                f"{NORMALITY_WARNING_P}"
            ),
            (
                "WARNING"
                if normality_warning
                else "PASS"
            ),
            (
                "Disclose; formal p-value alone "
                "is not a hard stop."
                if normality_warning
                else "Proceed."
            ),
        )
    )

    diagnostics.append(
        diag_row(
            "A5 Residual skewness",
            skewness,
            (
                "SENSITIVITY if |skew| > "
                f"{RESIDUAL_SKEW_SENSITIVITY}"
            ),
            (
                "WARNING / SENSITIVITY REQUIRED"
                if skew_trigger
                else "PASS"
            ),
            (
                "Run repeated-measures sensitivity analysis."
                if skew_trigger
                else "Proceed."
            ),
        )
    )

    diagnostics.append(
        diag_row(
            "A5 Pearson residual kurtosis",
            pearson_kurtosis,
            (
                "SENSITIVITY if > "
                f"{PEARSON_KURTOSIS_SENSITIVITY}"
            ),
            (
                "WARNING / SENSITIVITY REQUIRED"
                if kurtosis_trigger
                else "PASS"
            ),
            (
                "Run repeated-measures sensitivity analysis."
                if kurtosis_trigger
                else "Proceed."
            ),
        )
    )

    # -------------------------------------------------------------------------
    # A6 — residual heteroscedasticity
    # -------------------------------------------------------------------------

    residual_frame = pd.DataFrame(
        {
            PRODUCT: df[PRODUCT].astype(str).to_numpy(),
            "residual": residuals.to_numpy(),
        },
        index=df.index,
    )

    variance_by_product = (
        residual_frame
        .groupby(PRODUCT, observed=True)["residual"]
        .var(ddof=1)
    )

    if (
        variance_by_product.isna().any()
        or (variance_by_product < 0).any()
    ):
        hard_stop = True
        variance_ratio = float("nan")
        levene_p = float("nan")

        diagnostics.append(
            diag_row(
                "A6 Residual variance diagnostics",
                (
                    "Could not calculate valid residual "
                    "variance for every product"
                ),
                (
                    "Required product-level variance + "
                    "Brown-Forsythe diagnostics"
                ),
                "HARD STOP",
                (
                    "Suppress inference because required "
                    "diagnostics are unavailable."
                ),
            )
        )

    else:
        min_variance = float(
            variance_by_product.min()
        )

        max_variance = float(
            variance_by_product.max()
        )

        variance_ratio = (
            float("inf")
            if min_variance <= NUMERIC_TOL
            else max_variance / min_variance
        )

        groups = [
            group["residual"].to_numpy(dtype=float)
            for _, group
            in residual_frame.groupby(
                PRODUCT,
                observed=True,
            )
        ]

        try:
            _, levene_p = stats.levene(
                *groups,
                center="median",
            )
            levene_p = float(levene_p)

        except Exception:
            levene_p = float("nan")

        if not np.isfinite(levene_p):
            hard_stop = True

            diagnostics.append(
                diag_row(
                    (
                        "A6 Brown-Forsythe "
                        "(median-centered Levene)"
                    ),
                    "p=NA",
                    (
                        "Required diagnostic must "
                        "be calculable"
                    ),
                    "HARD STOP",
                    (
                        "Suppress inference because the required "
                        "heteroscedasticity test failed."
                    ),
                )
            )

        else:
            hetero_warning = (
                levene_p < HETERO_WARNING_P
                or variance_ratio > HETERO_WARNING_RATIO
            )

            hetero_sensitivity = (
                (
                    levene_p < HETERO_WARNING_P
                    and variance_ratio > HETERO_WARNING_RATIO
                )
                or variance_ratio > HETERO_SENSITIVITY_RATIO
            )

            if hetero_sensitivity and not hard_stop:
                sensitivity_required = True
                triggers.append(
                    "A6 heteroscedasticity: "
                    f"Brown-Forsythe p={levene_p:.4g}, "
                    f"variance ratio={variance_ratio:.3f}"
                )

            diagnostics.append(
                diag_row(
                    (
                        "A6 Brown-Forsythe "
                        "(median-centered Levene)"
                    ),
                    f"p={fmt_p(levene_p)}",
                    (
                        "WARNING if p < "
                        f"{HETERO_WARNING_P}"
                    ),
                    (
                        "WARNING / SENSITIVITY REQUIRED"
                        if hetero_sensitivity
                        else (
                            "WARNING"
                            if hetero_warning
                            else "PASS"
                        )
                    ),
                    (
                        "Run repeated-measures sensitivity analysis."
                        if hetero_sensitivity
                        else (
                            "Disclose."
                            if hetero_warning
                            else "Proceed."
                        )
                    ),
                )
            )

            diagnostics.append(
                diag_row(
                    (
                        "A6 Residual variance ratio "
                        "(max/min across products)"
                    ),
                    variance_ratio,
                    (
                        f"WARNING if > {HETERO_WARNING_RATIO}; "
                        f"SENSITIVITY if > "
                        f"{HETERO_SENSITIVITY_RATIO}"
                    ),
                    (
                        "WARNING / SENSITIVITY REQUIRED"
                        if hetero_sensitivity
                        else (
                            "WARNING"
                            if hetero_warning
                            else "PASS"
                        )
                    ),
                    (
                        "Run repeated-measures sensitivity analysis."
                        if hetero_sensitivity
                        else (
                            "Disclose."
                            if hetero_warning
                            else "Proceed."
                        )
                    ),
                )
            )

    # -------------------------------------------------------------------------
    # A7 — extreme raw residuals
    # -------------------------------------------------------------------------

    abs_resid = residuals.abs()

    n_gt3 = int(
        (abs_resid > EXTREME_RESIDUAL_FLAG).sum()
    )

    pct_gt3 = (
        100.0 * n_gt3 / len(abs_resid)
    )

    max_abs = float(abs_resid.max())

    extreme_sensitivity = (
        pct_gt3 > EXTREME_RESIDUAL_PCT_SENSITIVITY
        or max_abs > EXTREME_RESIDUAL_HARDER
    )

    if extreme_sensitivity and not hard_stop:
        sensitivity_required = True

        triggers.append(
            f"A7 extreme residuals: {pct_gt3:.2f}% "
            f"exceed |3|; max |residual|={max_abs:.3f}"
        )

    diagnostics.append(
        diag_row(
            "A7 Rows with |residual| > 3.0",
            (
                f"{n_gt3}/{len(abs_resid)} "
                f"({pct_gt3:.2f}%)"
            ),
            (
                "Flag each; SENSITIVITY if > "
                f"{EXTREME_RESIDUAL_PCT_SENSITIVITY}% "
                "of rows"
            ),
            (
                "WARNING / SENSITIVITY REQUIRED"
                if extreme_sensitivity
                else (
                    "WARNING"
                    if n_gt3 > 0
                    else "PASS"
                )
            ),
            (
                "Do not delete automatically; "
                "run sensitivity analysis."
                if extreme_sensitivity
                else (
                    "Flag only; do not delete."
                    if n_gt3 > 0
                    else "Proceed."
                )
            ),
        )
    )

    diagnostics.append(
        diag_row(
            "A7 Maximum absolute residual",
            max_abs,
            (
                "SENSITIVITY if > "
                f"{EXTREME_RESIDUAL_HARDER}"
            ),
            (
                "WARNING / SENSITIVITY REQUIRED"
                if max_abs > EXTREME_RESIDUAL_HARDER
                else "PASS"
            ),
            (
                "Do not delete automatically; "
                "run sensitivity analysis."
                if max_abs > EXTREME_RESIDUAL_HARDER
                else "Proceed."
            ),
        )
    )

    return (
        pd.DataFrame(diagnostics),
        hard_stop,
        sensitivity_required,
        triggers,
    )

def fit_sensitivity_model(df: pd.DataFrame):
    """
    Reviewed sensitivity approach:
    panelist fixed-effect OLS with HC3 covariance.
    """

    ordinary = smf.ols(
        f"{OUTCOME} ~ C({PRODUCT}) + C({PANELIST})",
        data=df,
    ).fit()

    return ordinary.get_robustcov_results(
        cov_type="HC3"
    )

def ols_design_vector(
    result,
    product: str,
    reference_panelist,
) -> np.ndarray:

    design_info = result.model.data.design_info
    frame = result.model.data.frame
    categories = list(frame[PRODUCT].cat.categories)

    new_data = pd.DataFrame(
        {
            PRODUCT: pd.Categorical(
                [product],
                categories=categories,
                ordered=True,
            ),
            PANELIST: [reference_panelist],
        }
    )

    matrix = patsy.build_design_matrices(
        [design_info],
        new_data,
        return_type="dataframe",
    )[0]

    return matrix.to_numpy(dtype=float).ravel()

def ols_product_omnibus(
    result,
) -> Tuple[float, int, float]:

    names = list(result.model.exog_names)

    product_indices = [
        i
        for i, name in enumerate(names)
        if f"C({PRODUCT})" in str(name)
    ]

    if not product_indices:
        raise InferenceStop(
            "Could not identify product terms "
            "in the sensitivity model."
        )

    beta = np.asarray(result.params, dtype=float)
    covariance = np.asarray(
        result.cov_params(),
        dtype=float,
    )

    b = beta[product_indices]
    v = covariance[
        np.ix_(
            product_indices,
            product_indices,
        )
    ]

    if not np.isfinite(v).all():
        raise InferenceStop(
            "Sensitivity model covariance is invalid."
        )

    try:
        solved = np.linalg.solve(v, b)

    except np.linalg.LinAlgError as exc:
        raise InferenceStop(
            "Sensitivity model omnibus covariance "
            "is singular."
        ) from exc

    chi2 = float(b.T @ solved)
    df_test = len(product_indices)
    p_value = float(stats.chi2.sf(chi2, df_test))

    return chi2, df_test, p_value

def sensitivity_pairwise_contrasts(
    result,
    primary_means: pd.DataFrame,
    reference_panelist,
) -> pd.DataFrame:

    ordered_products = (
        primary_means[PRODUCT]
        .astype(str)
        .tolist()
    )

    beta = np.asarray(
        result.params,
        dtype=float,
    )

    covariance = np.asarray(
        result.cov_params(),
        dtype=float,
    )

    rows: List[Dict[str, object]] = []

    for i, product_1 in enumerate(ordered_products):
        for product_2 in ordered_products[i + 1 :]:

            lvec = (
                ols_design_vector(
                    result,
                    product_1,
                    reference_panelist,
                )
                - ols_design_vector(
                    result,
                    product_2,
                    reference_panelist,
                )
            )

            (
                estimate,
                se,
                z_value,
                p_raw,
                ci_low,
                ci_high,
            ) = estimate_linear_function(
                beta,
                covariance,
                lvec,
            )

            rows.append(
                {
                    "product_1": product_1,
                    "product_2": product_2,
                    "contrast": (
                        f"{product_1} - {product_2}"
                    ),
                    "difference": estimate,
                    "se": se,
                    "z": z_value,
                    "ci95_low": ci_low,
                    "ci95_high": ci_high,
                    "p_raw": p_raw,
                    "magnitude_points": abs(estimate),
                }
            )

    return apply_holm(
        pd.DataFrame(rows)
    )

def pair_key(
    a: str,
    b: str,
) -> Tuple[str, str]:
    return tuple(
        sorted(
            [str(a), str(b)]
        )
    )

def compare_primary_and_sensitivity(
    primary_overall: Tuple[float, int, float],
    sensitivity_overall: Tuple[float, int, float],
    primary_pairs: pd.DataFrame,
    sensitivity_pairs: pd.DataFrame,
) -> Tuple[bool, List[str], pd.DataFrame]:
    """
    Flag material changes in:
    - overall product support,
    - pairwise Holm support,
    - contrast direction.
    """

    reasons: List[str] = []

    primary_overall_sig = (
        primary_overall[2] < ALPHA
    )

    sensitivity_overall_sig = (
        sensitivity_overall[2] < ALPHA
    )

    if primary_overall_sig != sensitivity_overall_sig:
        reasons.append(
            "Overall product-effect significance changed "
            "between the mixed model and the approved "
            "sensitivity model."
        )

    p_map = {
        pair_key(
            row.product_1,
            row.product_2,
        ): row
        for row
        in primary_pairs.itertuples(index=False)
    }

    s_map = {
        pair_key(
            row.product_1,
            row.product_2,
        ): row
        for row
        in sensitivity_pairs.itertuples(index=False)
    }

    comparison_rows: List[Dict[str, object]] = []

    support_changes = 0
    direction_changes = 0

    for key in sorted(p_map.keys()):

        p_row = p_map[key]
        s_row = s_map.get(key)

        if s_row is None:
            raise InferenceStop(
                f"Sensitivity comparison is missing pair {key}."
            )

        p_diff = float(p_row.difference)
        s_diff = float(s_row.difference)

        p_sig = bool(
            p_row.significant_holm
        )

        s_sig = bool(
            s_row.significant_holm
        )

        if p_sig != s_sig:
            support_changes += 1

        direction_changed = (
            abs(p_diff) > NUMERIC_TOL
            and abs(s_diff) > NUMERIC_TOL
            and np.sign(p_diff) != np.sign(s_diff)
        )

        if direction_changed:
            direction_changes += 1

        comparison_rows.append(
            {
                "pair": (
                    f"{key[0]} vs {key[1]}"
                ),
                "primary_difference": p_diff,
                "sensitivity_difference": s_diff,
                "primary_p_holm": float(
                    p_row.p_holm
                ),
                "sensitivity_p_holm": float(
                    s_row.p_holm
                ),
                "primary_significant": p_sig,
                "sensitivity_significant": s_sig,
                "direction_changed": direction_changed,
            }
        )

    if support_changes > 0:
        reasons.append(
            "Holm-adjusted pairwise support changed for "
            f"{support_changes} product comparison(s)."
        )

    if direction_changes > 0:
        reasons.append(
            "Estimated contrast direction changed for "
            f"{direction_changes} product comparison(s)."
        )

    comparison = pd.DataFrame(
        comparison_rows
    )

    return (
        bool(reasons),
        reasons,
        comparison,
    )

def maximal_nonsignificant_cliques(
    products: Sequence[str],
    pairwise: pd.DataFrame,
) -> List[Set[str]]:
    """
    Find maximal cliques where an edge means:
    'not statistically separated after Holm correction.'
    """

    products = [
        str(x)
        for x in products
    ]

    adjacency: Dict[str, Set[str]] = {
        p: set()
        for p in products
    }

    sig_map = {
        pair_key(
            row.product_1,
            row.product_2,
        ): bool(row.significant_holm)
        for row
        in pairwise.itertuples(index=False)
    }

    for i, a in enumerate(products):
        for b in products[i + 1 :]:
            if not sig_map.get(
                pair_key(a, b),
                False,
            ):
                adjacency[a].add(b)
                adjacency[b].add(a)

    cliques: List[Set[str]] = []

    def bron_kerbosch(
        r: Set[str],
        p: Set[str],
        x: Set[str],
    ) -> None:

        if not p and not x:
            cliques.append(set(r))
            return

        for vertex in sorted(list(p)):

            bron_kerbosch(
                r | {vertex},
                p & adjacency[vertex],
                x & adjacency[vertex],
            )

            p.remove(vertex)
            x.add(vertex)

    bron_kerbosch(
        set(),
        set(products),
        set(),
    )

    return cliques

def compact_letter_display(
    means: pd.DataFrame,
    pairwise: pd.DataFrame,
) -> Dict[str, str]:
    """
    Derive letters from the Holm-adjusted comparison matrix.

    Significantly different products never share a letter.
    The highest estimated mean receives "a".
    """

    ranked = (
        means
        .sort_values(
            "estimated_mean",
            ascending=False,
        )[PRODUCT]
        .astype(str)
        .tolist()
    )

    rank_index = {
        product: i
        for i, product
        in enumerate(ranked)
    }

    cliques = maximal_nonsignificant_cliques(
        ranked,
        pairwise,
    )

    non_sig_edges: Set[
        Tuple[str, str]
    ] = set()

    for row in pairwise.itertuples(index=False):

        if not bool(row.significant_holm):
            non_sig_edges.add(
                pair_key(
                    row.product_1,
                    row.product_2,
                )
            )

    uncovered_edges = set(
        non_sig_edges
    )

    uncovered_vertices = set(
        ranked
    )

    chosen: List[Set[str]] = []

    available = [
        set(c)
        for c in cliques
    ]

    while (
        uncovered_edges
        or uncovered_vertices
    ):

        best = None
        best_score = None

        for clique in available:

            covered_edges = sum(
                1
                for edge in uncovered_edges
                if set(edge).issubset(clique)
            )

            covered_vertices = len(
                clique & uncovered_vertices
            )

            contains_top = int(
                ranked[0] in clique
            )

            score = (
                covered_edges,
                covered_vertices,
                contains_top,
                len(clique),
                -min(
                    rank_index[p]
                    for p in clique
                ),
            )

            if (
                best_score is None
                or score > best_score
            ):
                best_score = score
                best = clique

        if (
            best is None
            or best_score is None
            or (
                best_score[0] == 0
                and best_score[1] == 0
            )
        ):
            product = min(
                uncovered_vertices,
                key=lambda p: rank_index[p],
            )
            best = {product}

        chosen.append(set(best))

        uncovered_vertices -= best

        uncovered_edges = {
            edge
            for edge in uncovered_edges
            if not set(edge).issubset(best)
        }

        available = [
            c
            for c in available
            if c != best
        ]

    chosen.sort(
        key=lambda clique: (
            min(
                rank_index[p]
                for p in clique
            ),
            -len(clique),
        )
    )

    labels: Dict[str, List[str]] = {
        product: []
        for product in ranked
    }

    for index, clique in enumerate(chosen):

        letter = letter_name(index)

        for product in sorted(
            clique,
            key=lambda p: rank_index[p],
        ):
            labels[product].append(
                letter
            )

    return {
        product: "".join(labels[product])
        for product in ranked
    }

def build_answer(
    overall_test: Tuple[float, int, float],
    means: pd.DataFrame,
    pairwise: pd.DataFrame,
    sensitivity_required: bool,
    model_sensitive: bool,
) -> Tuple[str, str]:

    chi2, df_test, p_overall = overall_test

    ranked = (
        means
        .sort_values(
            "estimated_mean",
            ascending=False,
        )
        .reset_index(drop=True)
    )

    high = ranked.iloc[0]
    low = ranked.iloc[-1]

    n_sig = int(
        pairwise["significant_holm"].sum()
    )

    spread = float(
        high["estimated_mean"]
        - low["estimated_mean"]
    )

    if model_sensitive:

        headline = (
            "Evidence for product differences in "
            "flavor impact is model-sensitive."
        )

        detail = (
            "The mixed model and the required panelist "
            "fixed-effect sensitivity analysis did not "
            "fully agree. No single significance grouping "
            "is presented. "
            f"The primary model ranked Product {high[PRODUCT]} "
            f"highest ({high['estimated_mean']:.2f}/9) and "
            f"Product {low[PRODUCT]} lowest "
            f"({low['estimated_mean']:.2f}/9), a "
            f"{spread:.2f}-point estimated span."
        )

        return headline, detail

    if p_overall < ALPHA:

        headline = (
            "Flavor impact differs across products."
        )

        sensitivity_note = (
            " The required sensitivity analysis was "
            "consistent with this conclusion."
            if sensitivity_required
            else ""
        )

        detail = (
            "The approved repeated-measures model detected "
            "an overall product effect "
            f"(Wald χ²({df_test}) = {chi2:.2f}, "
            f"p {fmt_p(p_overall)}). "
            f"Product {high[PRODUCT]} had the highest "
            "estimated flavor impact "
            f"({high['estimated_mean']:.2f}/9), while "
            f"Product {low[PRODUCT]} had the lowest "
            f"({low['estimated_mean']:.2f}/9). "
            f"{n_sig} of {len(pairwise)} pairwise contrasts "
            "remained significant after Holm correction."
            f"{sensitivity_note}"
        )

    else:

        headline = (
            "No overall product difference in flavor "
            "impact was detected."
        )

        sensitivity_note = (
            " The required sensitivity analysis was "
            "consistent with this conclusion."
            if sensitivity_required
            else ""
        )

        detail = (
            "The approved repeated-measures model did not "
            "detect an overall product effect "
            f"(Wald χ²({df_test}) = {chi2:.2f}, "
            f"p {fmt_p(p_overall)}). "
            f"Product {high[PRODUCT]} still had the highest "
            "estimated mean "
            f"({high['estimated_mean']:.2f}/9), but rank order "
            "alone is not evidence of a statistically reliable "
            "product difference."
            f"{sensitivity_note}"
        )

    return headline, detail

def is_export_index(series: pd.Series) -> bool:
    numeric = pd.to_numeric(series, errors="coerce")
    if numeric.isna().any():
        return False
    return np.array_equal(
        numeric.to_numpy(dtype=float),
        np.arange(len(series), dtype=float),
    )

def remove_export_index(df: pd.DataFrame) -> Tuple[pd.DataFrame, List[str]]:
    out = df.copy()
    removed: List[str] = []
    for col in list(out.columns):
        normalized = str(col).strip().lower()
        if normalized in {"index", "unnamed: 0"} and is_export_index(out[col]):
            out = out.drop(columns=[col])
            removed.append(str(col))
    return out, removed

def normalize_identifier_columns(df: pd.DataFrame) -> pd.DataFrame:
    out = df.copy()
    for col in [PRODUCT, PANELIST]:
        if col not in out.columns:
            continue
        out[col] = out[col].astype("string").str.strip()
        blank = out[col].fillna("").eq("")
        if blank.any():
            out.loc[blank, col] = pd.NA
    return out

def selected_attributes() -> List[str]:
    mode = ANALYSIS_MODE.strip().lower()

    if mode not in {"broad_screen", "targeted"}:
        raise ValueError(
            "ANALYSIS_MODE must be 'broad_screen' or 'targeted'."
        )

    if mode == "broad_screen":
        return list(APPROVED_ATTRIBUTES.keys())

    if not TARGET_ATTRIBUTES:
        raise ValueError(
            "ANALYSIS_MODE='targeted' requires at least one "
            "TARGET_ATTRIBUTES entry."
        )

    unknown = [
        a
        for a in TARGET_ATTRIBUTES
        if a not in APPROVED_ATTRIBUTES
    ]

    if unknown:
        raise ValueError(
            "Targeted SOP-04 attributes must be approved measured "
            "descriptive variables. "
            f"Unsupported request(s): {unknown}"
        )

    comparison_mode = TARGET_COMPARISON_MODE.strip().lower()

    if comparison_mode not in {"planned", "all_pairwise"}:
        raise ValueError(
            "TARGET_COMPARISON_MODE must be 'planned' or 'all_pairwise'."
        )

    if comparison_mode == "planned" and not TARGET_COMPARISONS:
        raise ValueError(
            "TARGET_COMPARISON_MODE='planned' requires at least one "
            "TARGET_COMPARISONS pair. "
            "For a question like 'Is F more bitter than G?', "
            "use [('F', 'G')]."
        )

    return list(dict.fromkeys(TARGET_ATTRIBUTES))

def all_product_pairs(
    products: Sequence[str],
) -> List[Tuple[str, str]]:

    p = [str(x) for x in products]

    return [
        (p[i], p[j])
        for i in range(len(p))
        for j in range(i + 1, len(p))
    ]

def pair_key(
    a: str,
    b: str,
) -> Tuple[str, str]:

    return tuple(
        sorted(
            [
                str(a),
                str(b),
            ]
        )
    )

def validate_target_pairs(
    products: Sequence[str],
) -> List[Tuple[str, str]]:

    available = {
        str(x)
        for x in products
    }

    cleaned: List[Tuple[str, str]] = []
    seen: Set[Tuple[str, str]] = set()

    for raw in TARGET_COMPARISONS:

        if (
            not isinstance(
                raw,
                (tuple, list),
            )
            or len(raw) != 2
        ):

            raise DesignStop(
                "Every TARGET_COMPARISONS entry must "
                "contain exactly two product codes."
            )

        a = str(raw[0]).strip()
        b = str(raw[1]).strip()

        if not a or not b or a == b:

            raise DesignStop(
                f"Invalid planned contrast: {raw}"
            )

        missing = [
            x
            for x in [a, b]
            if x not in available
        ]

        if missing:

            raise DesignStop(
                f"Planned contrast {a} vs {b} cannot be run "
                "because product(s) "
                f"{missing} are absent from this attribute's "
                "analyzed data."
            )

        key = pair_key(
            a,
            b,
        )

        if key not in seen:

            seen.add(
                key
            )

            cleaned.append(
                (a, b)
            )

    return cleaned

def comparison_plan(
    products: Sequence[str],
) -> Tuple[
    str,
    List[Tuple[str, str]],
    bool,
]:

    complete = all_product_pairs(
        products
    )

    if ANALYSIS_MODE.strip().lower() == "broad_screen":

        return (
            "all_pairwise",
            complete,
            True,
        )

    if TARGET_COMPARISON_MODE.strip().lower() == "all_pairwise":

        return (
            "all_pairwise",
            complete,
            True,
        )

    planned = validate_target_pairs(
        products
    )

    return (
        "planned",
        planned,
        {
            pair_key(*p)
            for p in planned
        }
        == {
            pair_key(*p)
            for p in complete
        },
    )

def capability_gate(
    raw_df: pd.DataFrame,
    requested_attributes: Sequence[str],
) -> Dict[str, object]:

    base_missing = [
        c
        for c in [PRODUCT, PANELIST]
        if c not in raw_df.columns
    ]

    present_attributes = [
        a
        for a in requested_attributes
        if a in raw_df.columns
    ]

    missing_attributes = [
        a
        for a in requested_attributes
        if a not in raw_df.columns
    ]

    if base_missing or not present_attributes:

        classification = "UNSUPPORTED"

    elif missing_attributes:

        classification = "PARTIALLY SUPPORTED"

    else:

        classification = "SUPPORTED"

    return {
        "classification": classification,
        "base_missing": base_missing,
        "requested_attributes": list(requested_attributes),
        "present_attributes": present_attributes,
        "missing_attributes": missing_attributes,
    }

def validate_global_design(
    df: pd.DataFrame,
) -> Tuple[
    List[str],
    Dict[str, object],
]:

    warnings_out: List[str] = []

    missing_ids = int(
        df[
            [PRODUCT, PANELIST]
        ]
        .isna()
        .any(axis=1)
        .sum()
    )

    if missing_ids > 0:

        raise DesignStop(
            f"{missing_ids} row(s) have missing/blank "
            f"{PRODUCT} or {PANELIST}. "
            "These identifiers are required for "
            "repeated-measures inference."
        )

    duplicate_rows = df.duplicated(
        subset=[
            PANELIST,
            PRODUCT,
        ],
        keep=False,
    )

    if duplicate_rows.any():

        n_rows = int(
            duplicate_rows.sum()
        )

        n_cells = int(
            df.loc[
                duplicate_rows,
                [
                    PANELIST,
                    PRODUCT,
                ],
            ]
            .drop_duplicates()
            .shape[0]
        )

        raise DesignStop(
            "Duplicate panelist × product evaluation cells detected "
            f"({n_rows} rows across {n_cells} duplicated cells). "
            "Resolve duplicates; the script will not silently "
            "average them."
        )

    products = sorted(
        df[PRODUCT]
        .astype(str)
        .unique()
        .tolist()
    )

    panelists = sorted(
        df[PANELIST]
        .astype(str)
        .unique()
        .tolist()
    )

    if len(products) < 2:

        raise DesignStop(
            "Fewer than two products are present."
        )

    if len(panelists) < 2:

        raise DesignStop(
            "Fewer than two panelists are present."
        )

    observed_cells = int(
        df[
            [PANELIST, PRODUCT]
        ]
        .drop_duplicates()
        .shape[0]
    )

    expected_cells = (
        len(panelists)
        * len(products)
    )

    missing_cells = (
        expected_cells
        - observed_cells
    )

    if missing_cells > 0:

        warnings_out.append(
            f"Observed {observed_cells}/{expected_cells} "
            "panelist × product cells; "
            f"{missing_cells} design cell(s) are missing. "
            "MixedLM can accommodate incomplete cells, "
            "but the imbalance is disclosed."
        )

    evals_per_panelist = (
        df.groupby(
            PANELIST,
            observed=True,
        )[PRODUCT]
        .nunique()
    )

    if evals_per_panelist.min() < 2:

        warnings_out.append(
            "At least one panelist evaluated fewer than "
            "two products; those rows do not independently "
            "establish a within-panelist product contrast."
        )

    if products != EXPECTED_PRODUCTS:

        warnings_out.append(
            f"Observed products: {products}; "
            f"workshop expectation: {EXPECTED_PRODUCTS}."
        )

    if len(panelists) != EXPECTED_PANELISTS:

        warnings_out.append(
            f"Observed {len(panelists)} panelists; "
            f"workshop expectation: "
            f"{EXPECTED_PANELISTS}."
        )

    if len(df) != EXPECTED_ROWS:

        warnings_out.append(
            f"Observed {len(df)} rows; "
            f"workshop expectation: "
            f"{EXPECTED_ROWS}."
        )

    summary = {
        "products": products,
        "n_products": len(products),
        "n_panelists": len(panelists),
        "observed_cells": observed_cells,
        "expected_cells": expected_cells,
        "missing_cells": missing_cells,
    }

    return (
        warnings_out,
        summary,
    )

def prepare_attribute_data(
    base_df: pd.DataFrame,
    attribute: str,
) -> Tuple[
    pd.DataFrame,
    List[str],
    Dict[str, int],
]:

    work = base_df[
        [
            PANELIST,
            PRODUCT,
            attribute,
        ]
    ].copy()

    warnings_out: List[str] = []

    originally_nonmissing = (
        work[attribute].notna()
    )

    work[attribute] = pd.to_numeric(
        work[attribute],
        errors="coerce",
    )

    newly_invalid = int(
        (
            originally_nonmissing
            & work[attribute].isna()
        ).sum()
    )

    if newly_invalid > 0:

        raise CapabilityStop(
            f"{newly_invalid} non-numeric value(s) "
            f"found in {attribute}; the approved "
            "descriptive outcome is numeric on a "
            "0–10 intensity scale."
        )

    values = work[
        attribute
    ].dropna()

    outside = values[
        (values < SCALE_MIN)
        | (values > SCALE_MAX)
    ]

    if not outside.empty:

        raise CapabilityStop(
            f"{attribute} contains {len(outside)} "
            "value(s) outside the approved "
            f"{int(SCALE_MIN)}–{int(SCALE_MAX)} "
            "descriptive-intensity scale."
        )

    before = len(work)

    work = work.dropna(
        subset=[
            PANELIST,
            PRODUCT,
            attribute,
        ]
    ).copy()

    missing_rows = (
        before
        - len(work)
    )

    if missing_rows > 0:

        warnings_out.append(
            f"Excluded {missing_rows} row(s) "
            f"with missing {attribute}; "
            f"{len(work)} complete rows remained "
            "for this attribute model."
        )

    if work.empty:

        raise DesignStop(
            "No complete rows remain "
            "for this attribute."
        )

    if work[attribute].nunique(
        dropna=True
    ) < 2:

        raise DesignStop(
            "The attribute has no usable "
            "outcome variation."
        )

    if work[PRODUCT].nunique() < 2:

        raise DesignStop(
            "Fewer than two products remain "
            "for this attribute."
        )

    if work[PANELIST].nunique() < 2:

        raise DesignStop(
            "Fewer than two panelists remain "
            "for this attribute."
        )

    product_counts_by_panelist = (
        work.groupby(
            PANELIST,
            observed=True,
        )[PRODUCT]
        .nunique()
    )

    repeated_panelists = int(
        (
            product_counts_by_panelist
            >= 2
        ).sum()
    )

    if repeated_panelists < 2:

        raise DesignStop(
            "Insufficient repeated product "
            "evaluations within panelist for "
            "the approved model."
        )

    cells_per_product = (
        work.groupby(
            PRODUCT,
            observed=True,
        )
        .size()
    )

    low_products = (
        cells_per_product[
            cells_per_product < 2
        ]
    )

    if not low_products.empty:

        raise DesignStop(
            "At least one product has fewer "
            "than two analyzed observations: "
            f"{low_products.to_dict()}. "
            "Required residual-variance "
            "diagnostics cannot run."
        )

    products = sorted(
        work[PRODUCT]
        .astype(str)
        .unique()
        .tolist()
    )

    panelists = sorted(
        work[PANELIST]
        .astype(str)
        .unique()
        .tolist()
    )

    observed_cells = int(
        work[
            [PANELIST, PRODUCT]
        ]
        .drop_duplicates()
        .shape[0]
    )

    expected_cells = (
        len(products)
        * len(panelists)
    )

    missing_cells = (
        expected_cells
        - observed_cells
    )

    if missing_cells > 0:

        warnings_out.append(
            f"Attribute model uses "
            f"{observed_cells}/{expected_cells} "
            "available panelist × product cells; "
            f"{missing_cells} cell(s) are incomplete "
            "for this outcome."
        )

    work[PRODUCT] = pd.Categorical(
        work[PRODUCT],
        categories=products,
        ordered=True,
    )

    work[PANELIST] = (
        work[PANELIST]
        .astype("string")
    )

    info = {
        "missing_rows": missing_rows,
        "observed_cells": observed_cells,
        "expected_cells": expected_cells,
        "missing_cells": missing_cells,
    }

    return (
        work,
        warnings_out,
        info,
    )

def fit_primary_mixed_model(
    df: pd.DataFrame,
    outcome: str,
):

    model = smf.mixedlm(
        f"{outcome} ~ C({PRODUCT})",
        data=df,
        groups=df[PANELIST],
        re_formula="1",
    )

    attempts: List[
        Dict[str, object]
    ] = []

    last_result = None
    last_warning_text: List[str] = []
    last_exception: Optional[Exception] = None

    for method in OPTIMIZER_SEQUENCE:

        try:

            with warnings.catch_warnings(
                record=True
            ) as caught:

                warnings.simplefilter(
                    "always"
                )

                result = model.fit(
                    reml=True,
                    method=method,
                    maxiter=2000,
                    disp=False,
                )

                captured = [
                    str(w.message)
                    for w in caught
                ]

            converged = bool(
                getattr(
                    result,
                    "converged",
                    False,
                )
            )

            attempts.append(
                {
                    "optimizer": method,
                    "completed": True,
                    "converged": converged,
                    "warnings": (
                        " | ".join(captured)
                        if captured
                        else "None"
                    ),
                }
            )

            last_result = result
            last_warning_text = captured

            if converged:

                return (
                    result,
                    captured,
                    attempts,
                )

        except Exception as exc:

            last_exception = exc

            attempts.append(
                {
                    "optimizer": method,
                    "completed": False,
                    "converged": False,
                    "warnings": (
                        f"Exception: {exc}"
                    ),
                }
            )

    if last_result is not None:

        return (
            last_result,
            last_warning_text,
            attempts,
        )

    raise InferenceStop(
        "All approved MixedLM optimizer "
        "attempts failed. "
        f"Last error: {last_exception}"
    )

def fixed_effect_covariance(
    result,
) -> pd.DataFrame:

    fe_names = list(
        result.fe_params.index
    )

    covariance = (
        result.cov_params()
    )

    if isinstance(
        covariance,
        pd.DataFrame,
    ):

        try:

            block = covariance.loc[
                fe_names,
                fe_names,
            ]

        except Exception:

            block = covariance.iloc[
                : len(fe_names),
                : len(fe_names),
            ]

        return pd.DataFrame(
            block,
            index=fe_names,
            columns=fe_names,
        )

    array = np.asarray(
        covariance,
        dtype=float,
    )[
        : len(fe_names),
        : len(fe_names),
    ]

    return pd.DataFrame(
        array,
        index=fe_names,
        columns=fe_names,
    )

def mixedlm_design_vector(
    result,
    product: str,
) -> np.ndarray:

    design_info = (
        result.model.data.design_info
    )

    frame = (
        result.model.data.frame
    )

    categories = list(
        frame[
            PRODUCT
        ].cat.categories
    )

    new_data = pd.DataFrame(
        {
            PRODUCT: pd.Categorical(
                [product],
                categories=categories,
                ordered=True,
            )
        }
    )

    matrix = (
        patsy.build_design_matrices(
            [design_info],
            new_data,
            return_type="dataframe",
        )[0]
    )

    return matrix.to_numpy(
        dtype=float
    ).ravel()

def estimate_linear_function(
    beta: np.ndarray,
    covariance: np.ndarray,
    lvector: np.ndarray,
) -> Tuple[
    float,
    float,
    float,
    float,
    float,
    float,
]:

    estimate = float(
        lvector @ beta
    )

    variance = float(
        lvector
        @ covariance
        @ lvector.T
    )

    if (
        not np.isfinite(
            variance
        )
        or variance
        < -NUMERIC_TOL
    ):

        raise InferenceStop(
            "Invalid variance for a "
            "required model contrast."
        )

    variance = max(
        variance,
        0.0,
    )

    se = math.sqrt(
        variance
    )

    if (
        not np.isfinite(se)
        or se <= NUMERIC_TOL
    ):

        raise InferenceStop(
            "A required model contrast "
            "has an invalid or zero SE."
        )

    z_value = estimate / se

    p_value = float(
        2
        * stats.norm.sf(
            abs(z_value)
        )
    )

    ci_low = (
        estimate
        - Z_CRITICAL * se
    )

    ci_high = (
        estimate
        + Z_CRITICAL * se
    )

    return (
        estimate,
        se,
        z_value,
        p_value,
        ci_low,
        ci_high,
    )

def mixedlm_estimated_means(
    result,
    products: Sequence[str],
    outcome: str,
) -> pd.DataFrame:

    beta = (
        result.fe_params
        .to_numpy(
            dtype=float
        )
    )

    covariance = (
        fixed_effect_covariance(
            result
        )
        .to_numpy(
            dtype=float
        )
    )

    rows: List[
        Dict[str, object]
    ] = []

    for product in products:

        lvec = (
            mixedlm_design_vector(
                result,
                product,
            )
        )

        (
            estimate,
            se,
            _,
            _,
            ci_low,
            ci_high,
        ) = estimate_linear_function(
            beta,
            covariance,
            lvec,
        )

        rows.append(
            {
                "attribute": outcome,
                PRODUCT: product,
                "estimated_mean": estimate,
                "se": se,
                "ci95_low": ci_low,
                "ci95_high": ci_high,
            }
        )

    return (
        pd.DataFrame(rows)
        .sort_values(
            "estimated_mean",
            ascending=False,
        )
        .reset_index(
            drop=True
        )
    )

def apply_holm(
    pairwise: pd.DataFrame,
) -> pd.DataFrame:

    if pairwise.empty:

        raise InferenceStop(
            "No requested product contrasts "
            "were available."
        )

    p_values = (
        pairwise[
            "p_raw"
        ]
        .to_numpy(
            dtype=float
        )
    )

    if (
        not np.isfinite(
            p_values
        ).all()
    ):

        raise InferenceStop(
            "Non-finite pairwise p-values "
            "prevent valid multiplicity correction."
        )

    (
        reject,
        p_holm,
        _,
        _,
    ) = multipletests(
        p_values,
        alpha=ALPHA,
        method=PAIRWISE_MULTIPLICITY,
    )

    out = pairwise.copy()

    out[
        "p_holm"
    ] = p_holm

    out[
        "significant_holm"
    ] = reject.astype(
        bool
    )

    return out

def mixedlm_product_contrasts(
    result,
    pairs: Sequence[Tuple[str, str]],
    outcome: str,
) -> pd.DataFrame:

    beta = (
        result.fe_params
        .to_numpy(
            dtype=float
        )
    )

    covariance = (
        fixed_effect_covariance(
            result
        )
        .to_numpy(
            dtype=float
        )
    )

    rows: List[
        Dict[str, object]
    ] = []

    for product_1, product_2 in pairs:

        lvec = (
            mixedlm_design_vector(
                result,
                product_1,
            )
            - mixedlm_design_vector(
                result,
                product_2,
            )
        )

        (
            estimate,
            se,
            z_value,
            p_raw,
            ci_low,
            ci_high,
        ) = estimate_linear_function(
            beta,
            covariance,
            lvec,
        )

        rows.append(
            {
                "attribute": outcome,
                "product_1": product_1,
                "product_2": product_2,
                "contrast": (
                    f"{product_1} - "
                    f"{product_2}"
                ),
                "difference": estimate,
                "se": se,
                "z": z_value,
                "ci95_low": ci_low,
                "ci95_high": ci_high,
                "p_raw": p_raw,
                "magnitude_points": abs(
                    estimate
                ),
            }
        )

    out = apply_holm(
        pd.DataFrame(rows)
    )

    return (
        out.sort_values(
            [
                "significant_holm",
                "magnitude_points",
            ],
            ascending=[
                False,
                False,
            ],
        )
        .reset_index(
            drop=True
        )
    )

def mixedlm_omnibus_product_test(
    result,
) -> Tuple[
    float,
    int,
    float,
]:

    names = list(
        result.fe_params.index
    )

    product_indices = [
        i
        for i, name in enumerate(
            names
        )
        if f"C({PRODUCT})"
        in str(name)
    ]

    if not product_indices:

        raise InferenceStop(
            "Could not identify product "
            "fixed-effect coefficients."
        )

    beta = (
        result.fe_params
        .to_numpy(
            dtype=float
        )
    )

    covariance = (
        fixed_effect_covariance(
            result
        )
        .to_numpy(
            dtype=float
        )
    )

    b = beta[
        product_indices
    ]

    v = covariance[
        np.ix_(
            product_indices,
            product_indices,
        )
    ]

    if not np.isfinite(
        v
    ).all():

        raise InferenceStop(
            "Invalid covariance matrix "
            "for the omnibus product test."
        )

    try:

        solved = np.linalg.solve(
            v,
            b,
        )

    except np.linalg.LinAlgError as exc:

        raise InferenceStop(
            "Covariance inversion failed "
            "for omnibus product test."
        ) from exc

    chi2 = float(
        b.T @ solved
    )

    df_test = len(
        product_indices
    )

    p_value = float(
        stats.chi2.sf(
            chi2,
            df_test,
        )
    )

    return (
        chi2,
        df_test,
        p_value,
    )

def run_g9a_diagnostics(
    df: pd.DataFrame,
    result,
    warning_text: Sequence[str],
    outcome: str,
) -> Tuple[
    pd.DataFrame,
    bool,
    bool,
    List[str],
]:

    diagnostics: List[
        Dict[str, object]
    ] = []

    hard_stop = False
    sensitivity_required = False
    triggers: List[str] = []

    warning_blob = (
        " | ".join(
            warning_text
        )
        .lower()
    )

    # -------------------------------------------------------------------------
    # A1 — optimizer convergence
    # -------------------------------------------------------------------------

    converged = bool(
        getattr(
            result,
            "converged",
            False,
        )
    )

    final_convergence_warning = any(
        phrase in warning_blob
        for phrase in [
            "failed to converge",
            "gradient optimization failed",
            "optimization failed",
            "maximum likelihood optimization failed",
        ]
    )

    a1_pass = (
        converged
        and not final_convergence_warning
    )

    if not a1_pass:

        hard_stop = True

    diagnostics.append(
        diag_row(
            f"{outcome} — A1 Optimizer convergence",
            (
                f"converged={converged}; "
                "final convergence warning="
                f"{final_convergence_warning}"
            ),
            (
                "result.converged=True and no final "
                "warning that the optimum was not reached"
            ),
            (
                "PASS"
                if a1_pass
                else "HARD STOP"
            ),
            (
                "Proceed."
                if a1_pass
                else (
                    "Suppress substantive inference "
                    "for this attribute."
                )
            ),
        )
    )

    # -------------------------------------------------------------------------
    # A2 — singular covariance / Hessian / invalid SE covariance
    # -------------------------------------------------------------------------

    singular_warning = any(
        phrase in warning_blob
        for phrase in [
            "random effects covariance is singular",
            "covariance matrix is singular",
            "singular matrix",
            "pseudo-inverse",
            "pseudoinverse",
        ]
    )

    hessian_warning = (
        "hessian" in warning_blob
        and "not positive definite"
        in warning_blob
    )

    cov_re_finite = False
    random_variance = float(
        "nan"
    )

    try:

        cov_re = np.asarray(
            result.cov_re,
            dtype=float,
        )

        cov_re_finite = bool(
            np.isfinite(
                cov_re
            ).all()
        )

        if (
            cov_re_finite
            and cov_re.size
        ):

            random_variance = float(
                cov_re.ravel()[0]
            )

    except Exception:

        cov_re_finite = False

    covariance_valid = True
    covariance_error = ""

    try:

        cov_fe = (
            fixed_effect_covariance(
                result
            )
            .to_numpy(
                dtype=float
            )
        )

        covariance_valid = bool(
            np.isfinite(
                cov_fe
            ).all()
            and np.all(
                np.diag(
                    cov_fe
                )
                >= -NUMERIC_TOL
            )
        )

    except Exception as exc:

        covariance_valid = False
        covariance_error = str(
            exc
        )

    hessian_valid = True
    hessian_singular = False
    min_neg_hessian_eig = float(
        "nan"
    )

    hessian_error = ""

    try:

        (
            hess,
            hessian_singular,
        ) = result.model.hessian(
            result.params_object
        )

        hess = np.asarray(
            hess,
            dtype=float,
        )

        neg_hess = -(
            hess
            + hess.T
        ) / 2

        if not np.isfinite(
            neg_hess
        ).all():

            hessian_valid = False

        else:

            eigvals = np.linalg.eigvalsh(
                neg_hess
            )

            min_neg_hessian_eig = float(
                np.min(
                    eigvals
                )
            )

            hessian_valid = bool(
                min_neg_hessian_eig
                > NUMERIC_TOL
            )

    except Exception as exc:

        hessian_valid = False
        hessian_error = str(
            exc
        )

    # IMPORTANT: a small positive random variance is handled by A3,
    # not promoted to A2 solely because it is near the boundary.

    invalid_random_covariance = (
        not cov_re_finite
        or not np.isfinite(
            random_variance
        )
        or random_variance
        < -NUMERIC_TOL
    )

    a2_fail = (
        singular_warning
        or hessian_warning
        or bool(
            hessian_singular
        )
        or invalid_random_covariance
        or not hessian_valid
        or not covariance_valid
    )

    if a2_fail:

        hard_stop = True

    observed_a2 = (
        "random variance="
        f"{random_variance:.6g}; "
        "min(-Hessian) eigen="
        f"{min_neg_hessian_eig:.6g}; "
        "singular warning="
        f"{singular_warning}; "
        "hessian singular="
        f"{bool(hessian_singular)}; "
        "SE covariance valid="
        f"{covariance_valid}"
    )

    if hessian_error:

        observed_a2 += (
            "; Hessian error="
            f"{hessian_error}"
        )

    if covariance_error:

        observed_a2 += (
            "; covariance error="
            f"{covariance_error}"
        )

    diagnostics.append(
        diag_row(
            (
                f"{outcome} — "
                "A2 Covariance / Hessian / "
                "SE covariance"
            ),
            observed_a2,
            (
                "No singular covariance, pseudo-inverse "
                "handling, non-PD Hessian, or invalid "
                "SE covariance"
            ),
            (
                "HARD STOP"
                if a2_fail
                else "PASS"
            ),
            (
                "Suppress inference for this attribute."
                if a2_fail
                else "Proceed."
            ),
        )
    )

    residual_variance = safe_float(
        getattr(
            result,
            "scale",
            float("nan"),
        )
    )

    if (
        np.isfinite(
            random_variance
        )
        and np.isfinite(
            residual_variance
        )
        and random_variance >= 0
        and residual_variance >= 0
        and (
            random_variance
            + residual_variance
        )
        > 0
    ):

        icc = (
            random_variance
            / (
                random_variance
                + residual_variance
            )
        )

    else:

        icc = float(
            "nan"
        )

    # -------------------------------------------------------------------------
    # A3 — boundary random-effect variance
    # -------------------------------------------------------------------------

    a3_trigger = (
        np.isfinite(
            random_variance
        )
        and random_variance
        < BOUNDARY_RANDOM_VARIANCE
    )

    if (
        a3_trigger
        and not hard_stop
    ):

        sensitivity_required = True

        triggers.append(
            "A3 random-intercept variance "
            f"{random_variance:.6f} < "
            f"{BOUNDARY_RANDOM_VARIANCE:.2f}"
        )

    diagnostics.append(
        diag_row(
            (
                f"{outcome} — "
                "A3 Random-intercept variance"
            ),
            random_variance,
            (
                "WARNING / SENSITIVITY if < "
                f"{BOUNDARY_RANDOM_VARIANCE}"
            ),
            (
                "SENSITIVITY REQUIRED"
                if a3_trigger
                else "PASS"
            ),
            (
                "Run approved repeated-measures "
                "sensitivity analysis."
                if a3_trigger
                else "Proceed."
            ),
        )
    )

    # -------------------------------------------------------------------------
    # A4 — near-zero ICC
    # -------------------------------------------------------------------------

    a4_trigger = (
        np.isfinite(
            icc
        )
        and icc
        < NEAR_ZERO_ICC
    )

    if (
        a4_trigger
        and not hard_stop
    ):

        sensitivity_required = True

        triggers.append(
            f"A4 ICC {icc:.6f} < "
            f"{NEAR_ZERO_ICC:.2f}"
        )

    if not np.isfinite(
        icc
    ):

        hard_stop = True

    diagnostics.append(
        diag_row(
            (
                f"{outcome} — "
                "A4 Intraclass correlation (ICC)"
            ),
            icc,
            (
                "WARNING / SENSITIVITY if < "
                f"{NEAR_ZERO_ICC}"
            ),
            (
                "SENSITIVITY REQUIRED"
                if a4_trigger
                else (
                    "PASS"
                    if np.isfinite(
                        icc
                    )
                    else "HARD STOP"
                )
            ),
            (
                "Run approved repeated-measures "
                "sensitivity analysis."
                if a4_trigger
                else (
                    "Proceed."
                    if np.isfinite(
                        icc
                    )
                    else "Suppress inference."
                )
            ),
        )
    )

    # -------------------------------------------------------------------------
    # Required residuals for A5–A7
    # -------------------------------------------------------------------------

    try:

        residuals = pd.Series(
            np.asarray(
                result.resid,
                dtype=float,
            ),
            index=df.index,
            name="residual",
        )

    except Exception as exc:

        hard_stop = True

        diagnostics.append(
            diag_row(
                (
                    f"{outcome} — "
                    "A5–A7 Residual diagnostics availability"
                ),
                (
                    "Residual extraction failed: "
                    f"{exc}"
                ),
                (
                    "Residual diagnostics are mandatory"
                ),
                "HARD STOP",
                (
                    "Suppress inference because required "
                    "diagnostics cannot be completed."
                ),
            )
        )

        return (
            pd.DataFrame(
                diagnostics
            ),
            hard_stop,
            sensitivity_required,
            triggers,
        )

    if (
        residuals.isna().any()
        or not np.isfinite(
            residuals.to_numpy()
        ).all()
    ):

        hard_stop = True

        diagnostics.append(
            diag_row(
                (
                    f"{outcome} — "
                    "A5–A7 Residual diagnostics availability"
                ),
                (
                    "Residual vector contains "
                    "non-finite values"
                ),
                (
                    "Residual diagnostics require "
                    "finite residuals"
                ),
                "HARD STOP",
                "Suppress inference.",
            )
        )

        return (
            pd.DataFrame(
                diagnostics
            ),
            hard_stop,
            sensitivity_required,
            triggers,
        )

    # -------------------------------------------------------------------------
    # A5 — residual distribution
    # -------------------------------------------------------------------------

    normality_p = float(
        "nan"
    )

    normality_test_name = (
        "formal normality test unavailable"
    )

    if len(
        residuals
    ) >= 3:

        try:

            if len(
                residuals
            ) <= 5000:

                (
                    _,
                    normality_p,
                ) = stats.shapiro(
                    residuals.to_numpy()
                )

                normality_test_name = (
                    "Shapiro-Wilk"
                )

            else:

                (
                    _,
                    normality_p,
                ) = stats.normaltest(
                    residuals.to_numpy()
                )

                normality_test_name = (
                    "D'Agostino K²"
                )

            normality_p = float(
                normality_p
            )

        except Exception:

            pass

    skewness = (
        float(
            stats.skew(
                residuals.to_numpy(),
                bias=False,
            )
        )
        if len(
            residuals
        ) >= 3
        else float(
            "nan"
        )
    )

    pearson_kurtosis = (
        float(
            stats.kurtosis(
                residuals.to_numpy(),
                fisher=False,
                bias=False,
            )
        )
        if len(
            residuals
        ) >= 4
        else float(
            "nan"
        )
    )

    normality_warning = (
        np.isfinite(
            normality_p
        )
        and normality_p
        < NORMALITY_WARNING_P
    )

    skew_trigger = (
        np.isfinite(
            skewness
        )
        and abs(
            skewness
        )
        > RESIDUAL_SKEW_SENSITIVITY
    )

    kurtosis_trigger = (
        np.isfinite(
            pearson_kurtosis
        )
        and pearson_kurtosis
        > PEARSON_KURTOSIS_SENSITIVITY
    )

    if (
        skew_trigger
        or kurtosis_trigger
    ) and not hard_stop:

        sensitivity_required = True

        triggers.append(
            "A5 residual shape: "
            f"skew={skewness:.3f}, "
            "Pearson kurtosis="
            f"{pearson_kurtosis:.3f}"
        )

    diagnostics.append(
        diag_row(
            (
                f"{outcome} — "
                "A5 Formal residual normality "
                f"({normality_test_name})"
            ),
            (
                f"p={fmt_p(normality_p)}"
            ),
            (
                "WARNING if p < "
                f"{NORMALITY_WARNING_P}"
            ),
            (
                "WARNING"
                if normality_warning
                else "PASS"
            ),
            (
                "Disclose; p-value alone is not "
                "a hard stop or automatic sensitivity trigger."
                if normality_warning
                else "Proceed."
            ),
        )
    )

    diagnostics.append(
        diag_row(
            (
                f"{outcome} — "
                "A5 Residual skewness"
            ),
            skewness,
            (
                "SENSITIVITY if |skew| > "
                f"{RESIDUAL_SKEW_SENSITIVITY}"
            ),
            (
                "SENSITIVITY REQUIRED"
                if skew_trigger
                else "PASS"
            ),
            (
                "Run approved repeated-measures "
                "sensitivity analysis."
                if skew_trigger
                else "Proceed."
            ),
        )
    )

    diagnostics.append(
        diag_row(
            (
                f"{outcome} — "
                "A5 Pearson residual kurtosis"
            ),
            pearson_kurtosis,
            (
                "SENSITIVITY if > "
                f"{PEARSON_KURTOSIS_SENSITIVITY}"
            ),
            (
                "SENSITIVITY REQUIRED"
                if kurtosis_trigger
                else "PASS"
            ),
            (
                "Run approved repeated-measures "
                "sensitivity analysis."
                if kurtosis_trigger
                else "Proceed."
            ),
        )
    )

    # -------------------------------------------------------------------------
    # A6 — heteroscedasticity
    # -------------------------------------------------------------------------

    residual_frame = pd.DataFrame(
        {
            PRODUCT: (
                df[
                    PRODUCT
                ]
                .astype(str)
                .to_numpy()
            ),
            "residual": (
                residuals.to_numpy()
            ),
        },
        index=df.index,
    )

    variance_by_product = (
        residual_frame
        .groupby(
            PRODUCT,
            observed=True,
        )[
            "residual"
        ]
        .var(
            ddof=1
        )
    )

    if (
        variance_by_product.isna().any()
        or (
            variance_by_product < 0
        ).any()
    ):

        hard_stop = True

        diagnostics.append(
            diag_row(
                (
                    f"{outcome} — "
                    "A6 Residual variance diagnostics"
                ),
                (
                    "Could not calculate valid residual "
                    "variance for every product"
                ),
                (
                    "Required product-level variance + "
                    "Brown-Forsythe diagnostics"
                ),
                "HARD STOP",
                (
                    "Suppress inference because required "
                    "diagnostics are unavailable."
                ),
            )
        )

    else:

        min_variance = float(
            variance_by_product.min()
        )

        max_variance = float(
            variance_by_product.max()
        )

        variance_ratio = (
            float(
                "inf"
            )
            if min_variance
            <= NUMERIC_TOL
            else (
                max_variance
                / min_variance
            )
        )

        groups = [
            group[
                "residual"
            ].to_numpy(
                dtype=float
            )
            for _, group
            in residual_frame.groupby(
                PRODUCT,
                observed=True,
            )
        ]

        try:

            (
                _,
                levene_p,
            ) = stats.levene(
                *groups,
                center="median",
            )

            levene_p = float(
                levene_p
            )

        except Exception:

            levene_p = float(
                "nan"
            )

        if not np.isfinite(
            levene_p
        ):

            hard_stop = True

            diagnostics.append(
                diag_row(
                    (
                        f"{outcome} — "
                        "A6 Brown-Forsythe "
                        "(median-centered Levene)"
                    ),
                    "p=NA",
                    (
                        "Required diagnostic "
                        "must be calculable"
                    ),
                    "HARD STOP",
                    (
                        "Suppress inference because "
                        "the required heteroscedasticity "
                        "test failed."
                    ),
                )
            )

        else:

            hetero_warning = (
                levene_p
                < HETERO_WARNING_P
                or variance_ratio
                > HETERO_WARNING_RATIO
            )

            hetero_sensitivity = (
                (
                    levene_p
                    < HETERO_WARNING_P
                    and variance_ratio
                    > HETERO_WARNING_RATIO
                )
                or variance_ratio
                > HETERO_SENSITIVITY_RATIO
            )

            if (
                hetero_sensitivity
                and not hard_stop
            ):

                sensitivity_required = True

                triggers.append(
                    "A6 heteroscedasticity: "
                    "Brown-Forsythe p="
                    f"{levene_p:.4g}, "
                    "variance ratio="
                    f"{variance_ratio:.3f}"
                )

            hetero_status = (
                "SENSITIVITY REQUIRED"
                if hetero_sensitivity
                else (
                    "WARNING"
                    if hetero_warning
                    else "PASS"
                )
            )

            hetero_action = (
                "Run approved repeated-measures "
                "sensitivity analysis."
                if hetero_sensitivity
                else (
                    "Disclose."
                    if hetero_warning
                    else "Proceed."
                )
            )

            diagnostics.append(
                diag_row(
                    (
                        f"{outcome} — "
                        "A6 Brown-Forsythe "
                        "(median-centered Levene)"
                    ),
                    (
                        f"p={fmt_p(levene_p)}"
                    ),
                    (
                        "WARNING if p < "
                        f"{HETERO_WARNING_P}; "
                        "sensitivity also considers "
                        "variance ratio"
                    ),
                    hetero_status,
                    hetero_action,
                )
            )

            diagnostics.append(
                diag_row(
                    (
                        f"{outcome} — "
                        "A6 Residual variance ratio "
                        "(max/min)"
                    ),
                    variance_ratio,
                    (
                        "WARNING if > "
                        f"{HETERO_WARNING_RATIO}; "
                        "SENSITIVITY if > "
                        f"{HETERO_SENSITIVITY_RATIO}"
                    ),
                    hetero_status,
                    hetero_action,
                )
            )

    # -------------------------------------------------------------------------
    # A7 — extreme residuals
    # -------------------------------------------------------------------------

    abs_resid = (
        residuals.abs()
    )

    n_gt3 = int(
        (
            abs_resid
            > EXTREME_RESIDUAL_FLAG
        ).sum()
    )

    pct_gt3 = (
        100.0
        * n_gt3
        / len(
            abs_resid
        )
    )

    max_abs = float(
        abs_resid.max()
    )

    extreme_sensitivity = (
        pct_gt3
        > EXTREME_RESIDUAL_PCT_SENSITIVITY
        or max_abs
        > EXTREME_RESIDUAL_HARDER
    )

    if (
        extreme_sensitivity
        and not hard_stop
    ):

        sensitivity_required = True

        triggers.append(
            "A7 extreme residuals: "
            f"{pct_gt3:.2f}% exceed |3|; "
            "max |residual|="
            f"{max_abs:.3f}"
        )

    diagnostics.append(
        diag_row(
            (
                f"{outcome} — "
                "A7 Rows with |residual| > 3.0"
            ),
            (
                f"{n_gt3}/{len(abs_resid)} "
                f"({pct_gt3:.2f}%)"
            ),
            (
                "Flag each; SENSITIVITY if > "
                f"{EXTREME_RESIDUAL_PCT_SENSITIVITY}% "
                "of rows"
            ),
            (
                "SENSITIVITY REQUIRED"
                if extreme_sensitivity
                else (
                    "WARNING"
                    if n_gt3 > 0
                    else "PASS"
                )
            ),
            (
                "Do not delete automatically; "
                "run sensitivity analysis."
                if extreme_sensitivity
                else (
                    "Flag only; do not delete."
                    if n_gt3 > 0
                    else "Proceed."
                )
            ),
        )
    )

    diagnostics.append(
        diag_row(
            (
                f"{outcome} — "
                "A7 Maximum absolute residual"
            ),
            max_abs,
            (
                "SENSITIVITY if > "
                f"{EXTREME_RESIDUAL_HARDER}"
            ),
            (
                "SENSITIVITY REQUIRED"
                if max_abs
                > EXTREME_RESIDUAL_HARDER
                else "PASS"
            ),
            (
                "Do not delete automatically; "
                "run sensitivity analysis."
                if max_abs
                > EXTREME_RESIDUAL_HARDER
                else "Proceed."
            ),
        )
    )

    return (
        pd.DataFrame(
            diagnostics
        ),
        hard_stop,
        sensitivity_required,
        triggers,
    )

def fit_sensitivity_model(
    df: pd.DataFrame,
    outcome: str,
):

    ordinary = smf.ols(
        (
            f"{outcome} ~ "
            f"C({PRODUCT}) + "
            f"C({PANELIST})"
        ),
        data=df,
    ).fit()

    return (
        ordinary
        .get_robustcov_results(
            cov_type="HC3"
        )
    )

def ols_design_vector(
    result,
    product: str,
    reference_panelist: str,
) -> np.ndarray:

    design_info = (
        result.model.data.design_info
    )

    frame = (
        result.model.data.frame
    )

    categories = list(
        frame[
            PRODUCT
        ].cat.categories
    )

    new_data = pd.DataFrame(
        {
            PRODUCT: pd.Categorical(
                [product],
                categories=categories,
                ordered=True,
            ),
            PANELIST: [
                reference_panelist
            ],
        }
    )

    matrix = (
        patsy.build_design_matrices(
            [design_info],
            new_data,
            return_type="dataframe",
        )[0]
    )

    return matrix.to_numpy(
        dtype=float
    ).ravel()

def ols_product_omnibus(
    result,
) -> Tuple[
    float,
    int,
    float,
]:

    names = list(
        result.model.exog_names
    )

    product_indices = [
        i
        for i, name in enumerate(
            names
        )
        if f"C({PRODUCT})"
        in str(name)
    ]

    if not product_indices:

        raise InferenceStop(
            "Could not identify product terms "
            "in the sensitivity model."
        )

    beta = np.asarray(
        result.params,
        dtype=float,
    )

    covariance = np.asarray(
        result.cov_params(),
        dtype=float,
    )

    b = beta[
        product_indices
    ]

    v = covariance[
        np.ix_(
            product_indices,
            product_indices,
        )
    ]

    if not np.isfinite(
        v
    ).all():

        raise InferenceStop(
            "Sensitivity-model covariance "
            "is invalid."
        )

    try:

        solved = np.linalg.solve(
            v,
            b,
        )

    except np.linalg.LinAlgError as exc:

        raise InferenceStop(
            "Sensitivity-model omnibus "
            "covariance is singular."
        ) from exc

    chi2 = float(
        b.T @ solved
    )

    df_test = len(
        product_indices
    )

    p_value = float(
        stats.chi2.sf(
            chi2,
            df_test,
        )
    )

    return (
        chi2,
        df_test,
        p_value,
    )

def sensitivity_product_contrasts(
    result,
    pairs: Sequence[Tuple[str, str]],
    reference_panelist: str,
    outcome: str,
) -> pd.DataFrame:

    beta = np.asarray(
        result.params,
        dtype=float,
    )

    covariance = np.asarray(
        result.cov_params(),
        dtype=float,
    )

    rows: List[
        Dict[str, object]
    ] = []

    for product_1, product_2 in pairs:

        lvec = (
            ols_design_vector(
                result,
                product_1,
                reference_panelist,
            )
            - ols_design_vector(
                result,
                product_2,
                reference_panelist,
            )
        )

        (
            estimate,
            se,
            z_value,
            p_raw,
            ci_low,
            ci_high,
        ) = estimate_linear_function(
            beta,
            covariance,
            lvec,
        )

        rows.append(
            {
                "attribute": outcome,
                "product_1": product_1,
                "product_2": product_2,
                "contrast": (
                    f"{product_1} - "
                    f"{product_2}"
                ),
                "difference": estimate,
                "se": se,
                "z": z_value,
                "ci95_low": ci_low,
                "ci95_high": ci_high,
                "p_raw": p_raw,
                "magnitude_points": abs(
                    estimate
                ),
            }
        )

    return apply_holm(
        pd.DataFrame(
            rows
        )
    )

def compare_pairwise_models(
    primary_pairs: pd.DataFrame,
    sensitivity_pairs: pd.DataFrame,
    compare_all_decisions: bool,
) -> Tuple[
    List[str],
    pd.DataFrame,
]:

    p_map = {
        pair_key(
            row.product_1,
            row.product_2,
        ): row
        for row
        in primary_pairs.itertuples(
            index=False
        )
    }

    s_map = {
        pair_key(
            row.product_1,
            row.product_2,
        ): row
        for row
        in sensitivity_pairs.itertuples(
            index=False
        )
    }

    if set(
        p_map
    ) != set(
        s_map
    ):

        raise InferenceStop(
            "Primary and sensitivity contrast "
            "families do not match."
        )

    reasons: List[str] = []
    rows: List[
        Dict[str, object]
    ] = []

    support_changes = 0
    direction_changes = 0

    for key in sorted(
        p_map
    ):

        p_row = p_map[
            key
        ]

        s_row = s_map[
            key
        ]

        p_diff = float(
            p_row.difference
        )

        s_diff = float(
            s_row.difference
        )

        p_sig = bool(
            p_row.significant_holm
        )

        s_sig = bool(
            s_row.significant_holm
        )

        support_changed = (
            p_sig != s_sig
        )

        direction_changed = (
            abs(
                p_diff
            )
            > NUMERIC_TOL
            and abs(
                s_diff
            )
            > NUMERIC_TOL
            and np.sign(
                p_diff
            )
            != np.sign(
                s_diff
            )
        )

        # For a complete pairwise family used to derive letters,
        # every Holm decision is substantively part of the grouping matrix.
        # For a planned contrast family, every requested contrast is
        # directly interpreted.

        if (
            compare_all_decisions
            and support_changed
        ):

            support_changes += 1

        if (
            direction_changed
            and (
                compare_all_decisions
                or p_sig
                or s_sig
            )
        ):

            direction_changes += 1

        rows.append(
            {
                "pair": (
                    f"{key[0]} vs "
                    f"{key[1]}"
                ),
                "primary_difference": p_diff,
                "sensitivity_difference": s_diff,
                "primary_p_holm": float(
                    p_row.p_holm
                ),
                "sensitivity_p_holm": float(
                    s_row.p_holm
                ),
                "primary_significant": p_sig,
                "sensitivity_significant": s_sig,
                "support_changed": support_changed,
                "direction_changed": direction_changed,
            }
        )

    if support_changes:

        reasons.append(
            "Holm-adjusted support changed for "
            f"{support_changes} interpreted comparison(s)."
        )

    if direction_changes:

        reasons.append(
            "Estimated contrast direction changed for "
            f"{direction_changes} interpreted comparison(s)."
        )

    return (
        reasons,
        pd.DataFrame(
            rows
        ),
    )

def maximal_nonsignificant_cliques(
    products: Sequence[str],
    pairwise: pd.DataFrame,
) -> List[
    Set[str]
]:

    products = [
        str(x)
        for x in products
    ]

    adjacency: Dict[
        str,
        Set[str],
    ] = {
        p: set()
        for p in products
    }

    sig_map = {
        pair_key(
            row.product_1,
            row.product_2,
        ): bool(
            row.significant_holm
        )
        for row
        in pairwise.itertuples(
            index=False
        )
    }

    for i, a in enumerate(
        products
    ):

        for b in products[
            i + 1 :
        ]:

            if not sig_map.get(
                pair_key(
                    a,
                    b,
                ),
                False,
            ):

                adjacency[
                    a
                ].add(
                    b
                )

                adjacency[
                    b
                ].add(
                    a
                )

    cliques: List[
        Set[str]
    ] = []

    def bron_kerbosch(
        r: Set[str],
        p: Set[str],
        x: Set[str],
    ) -> None:

        if (
            not p
            and not x
        ):

            cliques.append(
                set(
                    r
                )
            )

            return

        for vertex in sorted(
            list(
                p
            )
        ):

            bron_kerbosch(
                r
                | {
                    vertex
                },
                p
                & adjacency[
                    vertex
                ],
                x
                & adjacency[
                    vertex
                ],
            )

            p.remove(
                vertex
            )

            x.add(
                vertex
            )

    bron_kerbosch(
        set(),
        set(
            products
        ),
        set(),
    )

    return cliques

def compact_letter_display(
    means: pd.DataFrame,
    pairwise: pd.DataFrame,
) -> Dict[
    str,
    str,
]:

    ranked = (
        means
        .sort_values(
            "estimated_mean",
            ascending=False,
        )[
            PRODUCT
        ]
        .astype(str)
        .tolist()
    )

    rank_index = {
        product: i
        for i, product
        in enumerate(
            ranked
        )
    }

    cliques = (
        maximal_nonsignificant_cliques(
            ranked,
            pairwise,
        )
    )

    non_sig_edges: Set[
        Tuple[
            str,
            str,
        ]
    ] = set()

    for row in (
        pairwise.itertuples(
            index=False
        )
    ):

        if not bool(
            row.significant_holm
        ):

            non_sig_edges.add(
                pair_key(
                    row.product_1,
                    row.product_2,
                )
            )

    uncovered_edges = set(
        non_sig_edges
    )

    uncovered_vertices = set(
        ranked
    )

    chosen: List[
        Set[str]
    ] = []

    available = [
        set(
            c
        )
        for c in cliques
    ]

    while (
        uncovered_edges
        or uncovered_vertices
    ):

        best: Optional[
            Set[str]
        ] = None

        best_score = None

        for clique in available:

            covered_edges = sum(
                1
                for edge
                in uncovered_edges
                if set(
                    edge
                ).issubset(
                    clique
                )
            )

            covered_vertices = len(
                clique
                & uncovered_vertices
            )

            contains_top = int(
                ranked[
                    0
                ]
                in clique
            )

            score = (
                covered_edges,
                covered_vertices,
                contains_top,
                len(
                    clique
                ),
                -min(
                    rank_index[
                        p
                    ]
                    for p
                    in clique
                ),
            )

            if (
                best_score is None
                or score
                > best_score
            ):

                best_score = score
                best = clique

        if (
            best is None
            or best_score is None
            or (
                best_score[
                    0
                ]
                == 0
                and best_score[
                    1
                ]
                == 0
            )
        ):

            product = min(
                uncovered_vertices,
                key=lambda p: (
                    rank_index[
                        p
                    ]
                ),
            )

            best = {
                product
            }

        chosen.append(
            set(
                best
            )
        )

        uncovered_vertices -= (
            best
        )

        uncovered_edges = {
            edge
            for edge
            in uncovered_edges
            if not set(
                edge
            ).issubset(
                best
            )
        }

        available = [
            clique
            for clique
            in available
            if clique
            != best
        ]

    chosen.sort(
        key=lambda clique: (
            min(
                rank_index[
                    p
                ]
                for p
                in clique
            ),
            -len(
                clique
            ),
        )
    )

    labels: Dict[
        str,
        List[str],
    ] = {
        product: []
        for product
        in ranked
    }

    for index, clique in enumerate(
        chosen
    ):

        letter = letter_name(
            index
        )

        for product in sorted(
            clique,
            key=lambda p: (
                rank_index[
                    p
                ]
            ),
        ):

            labels[
                product
            ].append(
                letter
            )

    return {
        product: "".join(
            labels[
                product
            ]
        )
        for product
        in ranked
    }

def analyze_attribute(
    base_df: pd.DataFrame,
    attribute: str,
) -> AttributeResult:

    out = AttributeResult(
        attribute=attribute,
        label=APPROVED_ATTRIBUTES[
            attribute
        ],
    )

    if attribute not in (
        base_df.columns
    ):

        out.capability_status = (
            "UNSUPPORTED"
        )

        out.eligibility_status = (
            "CAPABILITY STOP"
        )

        out.hard_stop = True

        out.hard_stop_reason = (
            "Required measured attribute "
            f"'{attribute}' is absent. "
            "No neighboring attribute or "
            "proxy is authorized."
        )

        out.sensitivity_status = (
            "INFERENCE WITHHELD"
        )

        return out

    try:

        (
            df,
            attr_warnings,
            design_info,
        ) = prepare_attribute_data(
            base_df,
            attribute,
        )

    except (
        CapabilityStop,
        DesignStop,
    ) as exc:

        out.capability_status = (
            "UNSUPPORTED"
            if isinstance(
                exc,
                CapabilityStop,
            )
            else "SUPPORTED"
        )

        out.eligibility_status = (
            "CAPABILITY STOP"
            if isinstance(
                exc,
                CapabilityStop,
            )
            else "HARD STOP"
        )

        out.hard_stop = True
        out.hard_stop_reason = str(
            exc
        )

        out.sensitivity_status = (
            "INFERENCE WITHHELD"
        )

        return out

    out.analysis_n = len(
        df
    )

    out.panelists_n = int(
        df[
            PANELIST
        ].nunique()
    )

    out.products_n = int(
        df[
            PRODUCT
        ].nunique()
    )

    out.missing_rows = (
        design_info[
            "missing_rows"
        ]
    )

    out.observed_cells = (
        design_info[
            "observed_cells"
        ]
    )

    out.expected_cells = (
        design_info[
            "expected_cells"
        ]
    )

    out.missing_cells = (
        design_info[
            "missing_cells"
        ]
    )

    out.design_warnings = (
        attr_warnings
    )

    out.raw_product_means = (
        df.groupby(
            PRODUCT,
            observed=True,
        )[
            attribute
        ]
        .mean()
        .astype(float)
        .to_dict()
    )

    products = [
        str(
            x
        )
        for x
        in df[
            PRODUCT
        ].cat.categories
    ]

    try:

        (
            comparison_mode,
            pairs,
            complete_family,
        ) = comparison_plan(
            products
        )

    except DesignStop as exc:

        out.eligibility_status = (
            "HARD STOP"
        )

        out.hard_stop = True
        out.hard_stop_reason = str(
            exc
        )

        out.sensitivity_status = (
            "INFERENCE WITHHELD"
        )

        return out

    out.comparison_mode = (
        comparison_mode
    )

    out.comparison_pairs = (
        pairs
    )

    out.complete_pairwise_family = (
        complete_family
    )

    try:

        (
            model_result,
            warning_text,
            attempts,
        ) = fit_primary_mixed_model(
            df,
            attribute,
        )

    except InferenceStop as exc:

        out.eligibility_status = (
            "HARD STOP"
        )

        out.hard_stop = True
        out.hard_stop_reason = str(
            exc
        )

        out.sensitivity_status = (
            "INFERENCE WITHHELD"
        )

        return out

    out.optimizer_attempts = (
        attempts
    )

    out.model_warning_text = (
        warning_text
    )

    (
        diagnostics,
        hard_stop,
        sensitivity_required,
        triggers,
    ) = run_g9a_diagnostics(
        df,
        model_result,
        warning_text,
        attribute,
    )

    out.diagnostics = (
        diagnostics
    )

    out.hard_stop = (
        hard_stop
    )

    out.sensitivity_required = (
        sensitivity_required
    )

    out.sensitivity_triggers = (
        triggers
    )

    if hard_stop:

        out.eligibility_status = (
            "HARD STOP"
        )

        out.hard_stop_reason = (
            "A required G9-A A1/A2 or "
            "mandatory diagnostic condition failed. "
            "Substantive inference is suppressed "
            "for this attribute."
        )

        out.sensitivity_status = (
            "INFERENCE WITHHELD"
        )

        return out

    try:

        (
            chi2,
            df_test,
            p_value,
        ) = mixedlm_omnibus_product_test(
            model_result
        )

        means = (
            mixedlm_estimated_means(
                model_result,
                products,
                attribute,
            )
        )

        pairwise = (
            mixedlm_product_contrasts(
                model_result,
                pairs,
                attribute,
            )
        )

    except InferenceStop as exc:

        out.eligibility_status = (
            "HARD STOP"
        )

        out.hard_stop = True

        out.hard_stop_reason = (
            "Required omnibus/contrast "
            "calculations failed: "
            f"{exc}"
        )

        out.sensitivity_status = (
            "INFERENCE WITHHELD"
        )

        return out

    out.overall_chi2 = (
        chi2
    )

    out.overall_df = (
        df_test
    )

    out.overall_p = (
        p_value
    )

    out.means = means
    out.pairwise = pairwise

    ranked = (
        means
        .sort_values(
            "estimated_mean",
            ascending=False,
        )
        .reset_index(
            drop=True
        )
    )

    out.high_product = str(
        ranked.iloc[
            0
        ][
            PRODUCT
        ]
    )

    out.low_product = str(
        ranked.iloc[
            -1
        ][
            PRODUCT
        ]
    )

    out.high_mean = float(
        ranked.iloc[
            0
        ][
            "estimated_mean"
        ]
    )

    out.low_mean = float(
        ranked.iloc[
            -1
        ][
            "estimated_mean"
        ]
    )

    out.span = (
        out.high_mean
        - out.low_mean
    )

    if sensitivity_required:

        out.sensitivity_status = (
            "PENDING"
        )

        try:

            sensitivity_model = (
                fit_sensitivity_model(
                    df,
                    attribute,
                )
            )

            (
                s_chi2,
                s_df,
                s_p,
            ) = ols_product_omnibus(
                sensitivity_model
            )

            reference_panelist = str(
                df[
                    PANELIST
                ].iloc[
                    0
                ]
            )

            s_pairs = (
                sensitivity_product_contrasts(
                    sensitivity_model,
                    pairs,
                    reference_panelist,
                    attribute,
                )
            )

            out.sensitivity_overall_chi2 = (
                s_chi2
            )

            out.sensitivity_overall_df = (
                s_df
            )

            out.sensitivity_overall_p = (
                s_p
            )

            out.sensitivity_pairwise = (
                s_pairs
            )

        except Exception as exc:

            out.eligibility_status = (
                "HARD STOP"
            )

            out.hard_stop = True

            out.hard_stop_reason = (
                "G9-A sensitivity analysis was "
                "required but could not be completed: "
                f"{exc}"
            )

            out.sensitivity_status = (
                "INFERENCE WITHHELD"
            )

            return out

    else:

        out.sensitivity_status = (
            "NOT REQUIRED"
        )

    out.eligibility_status = (
        "SENSITIVITY COMPLETED"
        if sensitivity_required
        else "PRIMARY MODEL PASSED"
    )

    return out

def needs_attribute_family_adjustment(
    results: Dict[
        str,
        AttributeResult,
    ],
) -> bool:

    if ANALYSIS_MODE.strip().lower() == "broad_screen":

        return True

    # A single prespecified attribute does not get an artificial
    # across-attribute correction. Multiple prespecified target
    # attributes form a prespecified family.

    testable = [
        r
        for r in results.values()
        if (
            not r.hard_stop
            and np.isfinite(
                r.overall_p
            )
        )
    ]

    return len(
        testable
    ) > 1

def apply_attribute_family_multiplicity(
    results: Dict[
        str,
        AttributeResult,
    ],
) -> None:

    if not results:

        return

    adjust_family = (
        needs_attribute_family_adjustment(
            results
        )
    )

    attrs = list(
        results.keys()
    )

    if adjust_family:

        # Conservative family bookkeeping:
        # a requested attribute without a valid inferential
        # p-value contributes p=1.0; inference remains
        # suppressed for it.

        primary_input = [
            float(
                results[
                    a
                ].overall_p
            )
            if (
                not results[
                    a
                ].hard_stop
                and np.isfinite(
                    results[
                        a
                    ].overall_p
                )
            )
            else 1.0
            for a in attrs
        ]

        (
            _,
            p_adj,
            _,
            _,
        ) = multipletests(
            primary_input,
            alpha=ALPHA,
            method=ATTRIBUTE_FAMILY_MULTIPLICITY,
        )

        for attr, adjusted in zip(
            attrs,
            p_adj,
        ):

            result = results[
                attr
            ]

            result.overall_adjustment_label = (
                "Attribute-family Holm p"
            )

            if (
                not result.hard_stop
                and np.isfinite(
                    result.overall_p
                )
            ):

                result.overall_p_adjusted = float(
                    adjusted
                )

                result.screen_significant = bool(
                    adjusted
                    < ALPHA
                )

        sensitivity_input: List[
            float
        ] = []

        for attr in attrs:

            result = results[
                attr
            ]

            if result.hard_stop:

                sensitivity_input.append(
                    1.0
                )

            elif (
                result.sensitivity_required
                and np.isfinite(
                    result.sensitivity_overall_p
                )
            ):

                sensitivity_input.append(
                    float(
                        result.sensitivity_overall_p
                    )
                )

            elif np.isfinite(
                result.overall_p
            ):

                sensitivity_input.append(
                    float(
                        result.overall_p
                    )
                )

            else:

                sensitivity_input.append(
                    1.0
                )

        (
            _,
            s_adj,
            _,
            _,
        ) = multipletests(
            sensitivity_input,
            alpha=ALPHA,
            method=ATTRIBUTE_FAMILY_MULTIPLICITY,
        )

        for attr, adjusted in zip(
            attrs,
            s_adj,
        ):

            result = results[
                attr
            ]

            if not result.hard_stop:

                result.sensitivity_overall_p_adjusted = float(
                    adjusted
                )

                result.sensitivity_screen_significant = bool(
                    adjusted
                    < ALPHA
                )

    else:

        for result in results.values():

            result.overall_adjustment_label = (
                "Prespecified omnibus p"
            )

            if (
                not result.hard_stop
                and np.isfinite(
                    result.overall_p
                )
            ):

                result.overall_p_adjusted = float(
                    result.overall_p
                )

                result.screen_significant = bool(
                    result.overall_p
                    < ALPHA
                )

            if (
                not result.hard_stop
                and result.sensitivity_required
                and np.isfinite(
                    result.sensitivity_overall_p
                )
            ):

                result.sensitivity_overall_p_adjusted = float(
                    result.sensitivity_overall_p
                )

                result.sensitivity_screen_significant = bool(
                    result.sensitivity_overall_p
                    < ALPHA
                )

def finalize_sensitivity_and_letters(
    results: Dict[
        str,
        AttributeResult,
    ],
) -> None:

    """
    Classify stability only after the relevant
    omnibus family is known.
    """

    for result in results.values():

        result.significance_letters = {}

        if result.hard_stop:

            result.sensitivity_status = (
                "INFERENCE WITHHELD"
            )

            continue

        if not result.sensitivity_required:

            result.sensitivity_status = (
                "NOT REQUIRED"
            )

        else:

            reasons: List[str] = []

            primary_support = bool(
                result.screen_significant
            )

            sensitivity_support = bool(
                result.sensitivity_screen_significant
            )

            if (
                primary_support
                != sensitivity_support
            ):

                reasons.append(
                    "Overall product-effect support changed "
                    "under the approved sensitivity model."
                )

            # Pairwise conclusions are compared only when
            # they are actually part of the interpretation:
            # always for planned contrasts; for a complete
            # pairwise family only when the attribute is
            # supported in either model.

            compare_pairs = (
                result.comparison_mode
                == "planned"
                or primary_support
                or sensitivity_support
            )

            if (
                compare_pairs
                and result.pairwise
                is not None
                and result.sensitivity_pairwise
                is not None
            ):

                compare_all_decisions = (
                    result.comparison_mode
                    == "planned"
                    or result.complete_pairwise_family
                )

                (
                    pair_reasons,
                    comparison,
                ) = compare_pairwise_models(
                    result.pairwise,
                    result.sensitivity_pairwise,
                    compare_all_decisions=(
                        compare_all_decisions
                    ),
                )

                result.sensitivity_comparison = (
                    comparison
                )

                reasons.extend(
                    pair_reasons
                )

            result.model_sensitive_reasons = (
                reasons
            )

            result.model_sensitive = bool(
                reasons
            )

            result.sensitivity_status = (
                "MODEL-SENSITIVE"
                if result.model_sensitive
                else "PASSED / STABLE"
            )

        # Letters require complete pairwise coverage,
        # supported product effect, and diagnostic stability.
        # Planned single contrasts do not receive letters.

        if (
            result.screen_significant
            and not result.model_sensitive
            and result.complete_pairwise_family
            and result.comparison_mode
            == "all_pairwise"
            and result.means
            is not None
            and result.pairwise
            is not None
        ):

            result.significance_letters = (
                compact_letter_display(
                    result.means,
                    result.pairwise,
                )
            )

            result.means = (
                result.means.copy()
            )

            result.means[
                "significance_letters"
            ] = (
                result.means[
                    PRODUCT
                ]
                .astype(str)
                .map(
                    result.significance_letters
                )
            )

def finalize_inference(
    results: Dict[
        str,
        AttributeResult,
    ],
) -> None:

    apply_attribute_family_multiplicity(
        results
    )

    finalize_sensitivity_and_letters(
        results
    )

def product_attribute_mean_matrix(
    df: pd.DataFrame,
    attributes: Sequence[str],
    results: Dict[
        str,
        AttributeResult,
    ],
) -> Tuple[
    pd.DataFrame,
    Dict[str, str],
]:

    """
    Prefer model-estimated product means when available.

    If an attribute cannot support inference, retain raw
    descriptive means only for descriptive context.
    """

    products = sorted(
        df[
            PRODUCT
        ]
        .dropna()
        .astype(str)
        .unique()
        .tolist()
    )

    matrix = pd.DataFrame(
        index=products
    )

    source: Dict[
        str,
        str,
    ] = {}

    for attribute in attributes:

        result = results.get(
            attribute
        )

        if (
            result is not None
            and result.means is not None
            and not result.means.empty
        ):

            s = (
                result.means
                .set_index(
                    PRODUCT
                )[
                    "estimated_mean"
                ]
                .copy()
            )

            s.index = (
                s.index
                .astype(str)
            )

            matrix[
                attribute
            ] = s.reindex(
                products
            )

            source[
                attribute
            ] = (
                "model-estimated mean"
            )

        elif attribute in df.columns:

            tmp = df[
                [
                    PRODUCT,
                    attribute,
                ]
            ].copy()

            tmp[
                attribute
            ] = pd.to_numeric(
                tmp[
                    attribute
                ],
                errors="coerce",
            )

            s = (
                tmp.groupby(
                    PRODUCT,
                    observed=True,
                )[
                    attribute
                ]
                .mean()
            )

            s.index = (
                s.index
                .astype(str)
            )

            matrix[
                attribute
            ] = s.reindex(
                products
            )

            source[
                attribute
            ] = (
                "raw descriptive mean; "
                "inference unavailable"
            )

    return (
        matrix,
        source,
    )

def stable_supported_results(
    results: Dict[
        str,
        AttributeResult,
    ],
) -> List[
    AttributeResult
]:

    return [
        r
        for r in results.values()
        if (
            not r.hard_stop
            and not r.model_sensitive
            and r.screen_significant
        )
    ]

def diagnostic_plain_english(
    name: str,
) -> Tuple[
    str,
    str,
]:

    u = name.lower()

    if "a1 optimizer convergence" in u:

        return (
            "Optimizer convergence",
            "Checks whether the mixed model successfully "
            "found a stable solution. We need this because "
            "estimates and p-values are not trustworthy when "
            "the optimizer has not converged.",
        )

    if "a2 covariance / hessian" in u:

        return (
            "Covariance / Hessian validity",
            "Checks whether the fitted model has a "
            "mathematically usable uncertainty structure. "
            "Singular covariance, pseudo-inverse dependence, "
            "or a non-positive-definite Hessian can invalidate "
            "standard errors and inference, so A2 is a hard stop.",
        )

    if "random-intercept variance" in u:

        return (
            "Random-intercept variance",
            "Estimates how much consumers differ in their "
            "baseline use of the sensory scale. A value below "
            "0.01 is close to the model boundary, so the "
            "repeated-measures conclusion is rechecked with "
            "the approved sensitivity model.",
        )

    if "intraclass correlation" in u:

        return (
            "ICC — Intraclass Correlation",
            "Estimates how much rating variation is associated "
            "with differences between consumers rather than "
            "residual within-consumer variation. ICC below 0.01 "
            "means the panelist random effect contributes very "
            "little variance and requires sensitivity analysis.",
        )

    if "formal residual normality" in u:

        return (
            "Residual normality",
            "Checks whether model errors depart from a normal "
            "distribution. A p-value below 0.01 is reported as "
            "a warning, but under this SOP that p-value alone "
            "does not prove the model is unusable or automatically "
            "trigger sensitivity.",
        )

    if "residual skewness" in u:

        return (
            "Residual skewness",
            "Measures whether model errors are strongly asymmetric. "
            "Values near zero are more symmetric. The SOP sensitivity "
            "threshold is |skewness| > 1.0.",
        )

    if "pearson residual kurtosis" in u:

        return (
            "Residual kurtosis",
            "Measures whether model errors have unusually heavy "
            "tails or extreme observations compared with a normal "
            "distribution. Pearson kurtosis above 5.0 triggers "
            "sensitivity analysis.",
        )

    if "brown-forsythe" in u:

        return (
            "Brown-Forsythe test",
            "Checks whether residual variability differs across "
            "products. A small p-value indicates evidence of unequal "
            "residual variances and is interpreted together with "
            "the residual variance ratio.",
        )

    if "residual variance ratio" in u:

        return (
            "Residual variance ratio",
            "Compares the product with the largest residual variance "
            "with the product having the smallest. Ratios above 2 "
            "raise a warning; ratios above 4 require sensitivity analysis.",
        )

    if (
        "rows with |residual| > 3" in u
        or "maximum absolute residual" in u
    ):

        return (
            "Extreme residuals",
            "Flags ratings unusually far from the fitted model "
            "prediction. They are not automatically deleted. "
            "Sensitivity is required if more than 1% exceed |3| "
            "or any residual exceeds |4|.",
        )

    if "residual diagnostics availability" in u:

        return (
            "Residual diagnostics availability",
            "The SOP requires residual checks for every attribute "
            "model. If residuals cannot be calculated reliably, "
            "the inferential result is withheld rather than assumed "
            "acceptable.",
        )

    return (
        "Diagnostic check",
        "This check is part of the SOP model-health review "
        "and is evaluated separately for each descriptive attribute.",
    )

def inference_status(
    result: AttributeResult,
) -> Tuple[
    str,
    str,
]:

    if result.hard_stop:

        return (
            "HARD STOP",
            "danger",
        )

    if result.model_sensitive:

        return (
            "MODEL-SENSITIVE",
            "danger",
        )

    if result.screen_significant:

        return (
            "DIFFERENTIATES",
            "success",
        )

    if (
        result.sensitivity_required
        and result.sensitivity_status
        == "PASSED / STABLE"
    ):

        return (
            "STABLE / NOT SUPPORTED",
            "warning",
        )

    return (
        "NOT SUPPORTED",
        "neutral",
    )

                    "Dataset could not "
                    "be loaded."
                ),
                str(
                    exc
                ),
                data_path,
            )
        )

        write_dashboard(
            dashboard_path,
            dashboard,
        )

        print_console_summary(
            None,
            (
                "SOP-04 could not run "
                "because the dataset "
                "was unavailable."
            ),
            "HARD STOP",
            dashboard_path,
        )

        maybe_open_dashboard(
            dashboard_path
        )

        return

    cleaned_df, removed_columns = (
        remove_export_index(
            raw_df
        )
    )

    cleaned_df = (
        normalize_identifier_columns(
            cleaned_df
        )
    )

    # -------------------------------------------------------------------------
    # DATASET CAPABILITY GATE
    # -------------------------------------------------------------------------

    capability = (
        capability_gate(
            cleaned_df,
            requested,
        )
    )

    if (
        capability[
            "classification"
        ]
        == "UNSUPPORTED"
    ):

        missing = (
            capability[
                "base_missing"
            ]
            + capability[
                "missing_attributes"
            ]
        )

        message = (
            "Required SOP-04 inputs are "
            "unavailable: "
            + ", ".join(
                missing
            )
            + ". No neighboring attribute, "
            "JAR measure, comment theme, or "
            "proxy will be substituted."
        )

        dashboard = (
            build_stop_dashboard(
                (
                    "The active dataset does "
                    "not support SOP-04."
                ),
                message,
                data_path,
            )
        )

        write_dashboard(
            dashboard_path,
            dashboard,
        )

        print_console_summary(
            None,
            (
                "Required SOP-04 data "
                "were not available."
            ),
            "HARD STOP",
            dashboard_path,
        )

        maybe_open_dashboard(
            dashboard_path
        )

        return

    # -------------------------------------------------------------------------
    # GLOBAL STUDY-DESIGN VALIDATION
    # -------------------------------------------------------------------------

    try:

        (
            global_design_warnings,
            global_design_summary,
        ) = validate_global_design(
            cleaned_df
        )

    except DesignStop as exc:

        dashboard = (
            build_stop_dashboard(
                (
                    "The repeated-measures "
                    "design check failed."
                ),
                str(
                    exc
                ),
                data_path,
            )
        )

        write_dashboard(
            dashboard_path,
            dashboard,
        )

        print_console_summary(
            None,
            (
                "SOP-04 stopped at "
                "study-design validation."
            ),
            "HARD STOP",
            dashboard_path,
        )

        maybe_open_dashboard(
            dashboard_path
        )

        return

    # -------------------------------------------------------------------------
    # ATTRIBUTE-WISE PRIMARY ANALYSIS + G9-A
    # + REQUIRED SENSITIVITY MODELS
    # -------------------------------------------------------------------------

    results: Dict[
        str,
        AttributeResult,
    ] = {}

    for attribute in (
        capability[
            "present_attributes"
        ]
    ):

        results[
            attribute
        ] = analyze_attribute(
            cleaned_df,
            attribute,
        )

    for attribute in (
        capability[
            "missing_attributes"
        ]
    ):

        result = AttributeResult(
            attribute=attribute,
            label=APPROVED_ATTRIBUTES[
                attribute
            ],
        )

        result.capability_status = (
            "UNSUPPORTED"
        )

        result.eligibility_status = (
            "CAPABILITY STOP"
        )

        result.hard_stop = True

        result.hard_stop_reason = (
            "Required measured attribute "
            f"'{attribute}' is absent; "
            "no proxy is authorized."
        )

        result.sensitivity_status = (
            "INFERENCE WITHHELD"
        )

        results[
            attribute
        ] = result

    results = {
        attribute: results[
            attribute
        ]
        for attribute
        in requested
    }

    # -------------------------------------------------------------------------
    # RELEVANT MULTIPLICITY FAMILY
    # + FINAL SENSITIVITY CLASSIFICATION
    # + LETTERS
    # -------------------------------------------------------------------------

    finalize_inference(
        results
    )

    # -------------------------------------------------------------------------
    # PROFILE MATRIX
    # -------------------------------------------------------------------------

    profile_attributes = [
        attribute
        for attribute, result
        in results.items()
        if (
            attribute
            in cleaned_df.columns
            and result.capability_status
            == "SUPPORTED"
        )
    ]

    (
        profile_matrix,
        profile_source,
    ) = product_attribute_mean_matrix(
        cleaned_df,
        profile_attributes,
        results,
    )

    # -------------------------------------------------------------------------
    # HTML DASHBOARD
    # -------------------------------------------------------------------------

    dashboard = (
        build_dashboard(
            data_path=data_path,
            raw_df=raw_df,
            cleaned_df=cleaned_df,
            removed_columns=removed_columns,
            capability=capability,
            global_design_warnings=(
                global_design_warnings
            ),
            global_design_summary=(
                global_design_summary
            ),
            results=results,
            profile_matrix=(
                profile_matrix
            ),
            profile_source=(
                profile_source
            ),
        )
    )

    write_dashboard(
        dashboard_path,
        dashboard,
    )

    save_technical_exports(
        out_dir,
        results,
    )

    # -------------------------------------------------------------------------
    # SMALL CONSOLE SUMMARY
    # -------------------------------------------------------------------------

    (
        headline,
        _,
    ) = overall_dashboard_answer(
        results
    )

    hard_count = sum(
        1
        for r in results.values()
        if r.hard_stop
    )

    sensitive_count = sum(
        1
        for r in results.values()
        if (
            r.model_sensitive
            and not r.hard_stop
        )
    )

    sensitivity_required_count = sum(
        1
        for r in results.values()
        if (
            r.sensitivity_required
            and not r.hard_stop
        )
    )

    sensitivity_stable_count = sum(
        1
        for r in results.values()
        if r.sensitivity_status
        == "PASSED / STABLE"
    )

    if hard_count:

        model_status = (
            f"PARTIAL: {hard_count} "
            "attribute model(s) hard-stopped."
        )

    elif sensitive_count:

        model_status = (
            "MODEL-SENSITIVE: "
            f"{sensitive_count} attribute(s)."
        )

    elif sensitivity_required_count:

        model_status = (
            f"{sensitivity_required_count} "
            "sensitivity check(s) triggered; "
            f"{sensitivity_stable_count} stable."
        )

    else:

        model_status = (
            "Diagnostics acceptable across "
            "requested attribute models."
        )

    print_console_summary(
        cleaned_df,
        headline,
        model_status,
        dashboard_path,
    )

    # -------------------------------------------------------------------------
    # AUTOMATIC DASHBOARD OPEN
    # -------------------------------------------------------------------------

    maybe_open_dashboard(
        dashboard_path
    )

if __name__ == "__main__":
    main()

Standardized principal component analysis (PCA) on product-level means of

Unit of analysis
----------------
Rows = products.
Columns = measured descriptive sensory attributes.

The 400 consumer × product evaluation rows are NOT entered directly into the

This script operationalizes:
    1. Dataset Capability Gate.
    2. Product/design integrity checks.
    3. Approved descriptive-variable selection.
    4. 0–10 descriptive-scale validation.
    5. Product-level mean aggregation.

   12. One self-contained HTML dashboard with an inline-SVG PCA biplot.
   13. Collapsed methods/diagnostics with hover explanations.
   14. Optional technical CSV exports, disabled by default.
   15. Brief console summary and automatic dashboard opening.

IMPORTANT GUARDRAILS

- PCA is exploratory. It does not establish statistical significance.
- PCA does not establish causality, consumer preference, optimization, or

- PCA distance is not a p-value or proof that products are statistically alike.
- Only approved measured descriptive variables may enter the PCA.

def natural_sort_key(value: object) -> List[object]:
    """Sort labels such as P1, P2, P10 in intuitive order."""
    text = str(value)

    return [
        int(part) if part.isdigit() else part.lower()
        for part in re.split(r"(\d+)", text)
    ]

def selected_attributes() -> Tuple[
    List[str],
    bool,
]:
    """
    Determine the intended PCA attribute scope.

    Returns
    -------
    requested:
        Approved variables requested for the map.

    explicit:
        True when PCA_ATTRIBUTES was manually specified.
    """

    explicit = (
        PCA_ATTRIBUTES
        is not None
    )

    if PCA_ATTRIBUTES is None:

        requested = list(
            APPROVED_ATTRIBUTES.keys()
        )

    else:

        requested = list(
            PCA_ATTRIBUTES
        )

    if len(
        requested
    ) != len(
        set(
            requested
        )
    ):
        raise ValueError(
            "PCA_ATTRIBUTES contains duplicate variable names."
        )

    unknown = [
        attr
        for attr
        in requested
        if attr
        not in APPROVED_ATTRIBUTES
    ]

    if unknown:

        raise ValueError(
            "PCA_ATTRIBUTES contains variable(s) "
            "not approved by the current sensory schema: "
            + ", ".join(
                unknown
            )
        )

    if len(
        requested
    ) < 2:

        raise ValueError(
            "SOP-05 requires at least two approved "
            "descriptive sensory variables."
        )

    return (
        requested,
        explicit,
    )

def capability_gate(
    df: pd.DataFrame,
    requested: Sequence[str],
    explicit_selection: bool,
) -> CapabilityResult:
    """
    Determine whether the requested sensory PCA is supported by
    variables actually present in the current dataset.
    """

    notes: List[str] = []

    # -------------------------------------------------------------------------
    # Product identifier
    # -------------------------------------------------------------------------

    if PRODUCT not in df.columns:

        return CapabilityResult(
            classification="UNSUPPORTED",
            requested_attributes=list(
                requested
            ),
            available_attributes=[],
            missing_attributes=list(
                requested
            ),
            unapproved_desc_columns=[
                c
                for c
                in df.columns
                if c.startswith(
                    "desc_"
                )
            ],
            notes=[
                f"Required product identifier '{PRODUCT}' is missing."
            ],
        )

    # -------------------------------------------------------------------------
    # Descriptive sensory variables
    # -------------------------------------------------------------------------

    available = [
        attr
        for attr
        in requested
        if attr
        in df.columns
    ]

    missing = [
        attr
        for attr
        in requested
        if attr
        not in df.columns
    ]

    # Never auto-pull a desc_* variable merely because its name starts desc_.
    # It must also be in the approved schema dictionary.
    unapproved_desc = [
        c
        for c
        in df.columns
        if (
            c.startswith(
                "desc_"
            )
            and c
            not in APPROVED_ATTRIBUTES
        )
    ]

    if unapproved_desc:

        notes.append(
            "Descriptive-looking column(s) not defined "
            "in the approved schema were excluded: "
            + ", ".join(
                unapproved_desc
            )
            + "."
        )

    # -------------------------------------------------------------------------
    # Capability classification
    # -------------------------------------------------------------------------

    if (
        explicit_selection
        and missing
    ):

        classification = (
            "UNSUPPORTED"
        )

        notes.append(
            "An explicitly selected PCA attribute is missing; "
            "the script will not silently redefine the requested sensory space."
        )

    elif len(
        available
    ) < 2:

        classification = (
            "UNSUPPORTED"
        )

        notes.append(
            "Fewer than two approved measured descriptive variables "
            "are available, so SOP-05 cannot run."
        )

    elif missing:

        classification = (
            "PARTIALLY SUPPORTED"
        )

        notes.append(
            "The default category map can proceed using the approved measured "
            "descriptive variables that are present, but missing approved "
            "variables are disclosed and are not proxied."
        )

    else:

        classification = (
            "SUPPORTED"
        )

    return CapabilityResult(
        classification=classification,
        requested_attributes=list(
            requested
        ),
        available_attributes=available,
        missing_attributes=missing,
        unapproved_desc_columns=unapproved_desc,
        notes=notes,
    )

def prepare_data(
    df: pd.DataFrame,
    attributes: Sequence[str],
) -> PreparedData:
    """
    Validate raw sensory responses and create the approved
    product × descriptive-attribute mean matrix.
    """

    work = df.copy()
    notes: List[str] = []

    # -------------------------------------------------------------------------
    # Product-code validation
    # -------------------------------------------------------------------------

    product_text = (
        work[PRODUCT]
        .astype(
            "string"
        )
        .str.strip()
    )

    blank_product = (
        product_text.isna()
        | product_text.eq(
            ""
        )
    )

    if bool(
        blank_product.any()
    ):

        raise ValueError(
            f"HARD STOP: {int(blank_product.sum())} row(s) "
            f"have missing/blank {PRODUCT}. "
            "Product-level means cannot be constructed without "
            "a valid product identifier."
        )

    work[
        PRODUCT
    ] = product_text.astype(
        str
    )

    product_order = sorted(
        work[
            PRODUCT
        ]
        .unique()
        .tolist(),
        key=natural_sort_key,
    )

    if len(
        product_order
    ) < 2:

        raise ValueError(
            "HARD STOP: PCA requires at least two products."
        )

    # -------------------------------------------------------------------------
    # Duplicate panelist × product validation
    # -------------------------------------------------------------------------
    #
    # panelist_id is not itself a required SOP-05 PCA input because the PCA unit
    # is the product. However, when it exists it allows us to detect duplicate
    # evaluation rows that could otherwise overweight individual consumers when
    # product means are constructed.
    # -------------------------------------------------------------------------

    duplicate_count: Optional[
        int
    ] = None

    if PANELIST in work.columns:

        panelist_text = (
            work[
                PANELIST
            ]
            .astype(
                "string"
            )
            .str.strip()
        )

        valid_panelist = (
            panelist_text.notna()
            & panelist_text.ne(
                ""
            )
        )

        duplicate_frame = (
            work.loc[
                valid_panelist,
                [
                    PRODUCT,
                ],
            ]
            .copy()
        )

        duplicate_frame[
            PANELIST
        ] = (
            panelist_text.loc[
                valid_panelist
            ]
            .astype(
                str
            )
        )

        involved_mask = (
            duplicate_frame
            .duplicated(
                [
                    PANELIST,
                    PRODUCT,
                ],
                keep=False,
            )
        )

        duplicate_count = int(
            involved_mask.sum()
        )

        if duplicate_count > 0:

            duplicate_pairs = int(
                duplicate_frame.loc[
                    involved_mask,
                    [
                        PANELIST,
                        PRODUCT,
                    ],
                ]
                .drop_duplicates()
                .shape[
                    0
                ]
            )

            raise ValueError(
                "HARD STOP: duplicate panelist × product observations "
                "were detected "
                f"({duplicate_pairs} duplicated pair(s), "
                f"{duplicate_count} involved row(s)). "
                "The script will not silently average duplicates because "
                "they can overweight consumers in product means."
            )

        missing_panelists = int(
            (
                ~valid_panelist
            ).sum()
        )

        if missing_panelists:

            notes.append(
                f"{missing_panelists} row(s) have missing/blank "
                f"{PANELIST}. "
                "SOP-05 uses product-level means, so panelist ID "
                "is not a required PCA input; this is disclosed "
                "as a design note."
            )

    else:

        notes.append(
            f"{PANELIST} is not present. SOP-05 does not require it "
            "for the product-mean PCA, so the analysis can proceed, "
            "but duplicate consumer × product rows cannot be checked."
        )

    # -------------------------------------------------------------------------
    # Numeric and scale validation
    # -------------------------------------------------------------------------

    missing_value_count = 0

    for attr in attributes:

        original = (
            work[
                attr
            ]
        )

        numeric = pd.to_numeric(
            original,
            errors="coerce",
        )

        # Detect actual invalid strings separately from legitimate missing data.
        invalid_text = (
            original.notna()
            & numeric.isna()
        )

        if bool(
            invalid_text.any()
        ):

            examples = (
                original.loc[
                    invalid_text
                ]
                .astype(
                    str
                )
                .drop_duplicates()
                .head(
                    5
                )
                .tolist()
            )

            raise ValueError(
                f"HARD STOP: {attr} contains "
                f"{int(invalid_text.sum())} non-numeric value(s), "
                f"for example: {examples}. "
                "The script will not coerce invalid sensory responses "
                "into missing data."
            )

        out_of_range = (
            numeric.notna()
            & (
                (
                    numeric
                    < SCALE_MIN
                )
                |
                (
                    numeric
                    > SCALE_MAX
                )
            )
        )

        if bool(
            out_of_range.any()
        ):

            observed_min = float(
                numeric.min(
                    skipna=True
                )
            )

            observed_max = float(
                numeric.max(
                    skipna=True
                )
            )

            raise ValueError(
                f"HARD STOP: {attr} contains "
                f"{int(out_of_range.sum())} value(s) outside the approved "
                f"{SCALE_MIN:g}–{SCALE_MAX:g} descriptive-intensity scale "
                f"(observed range {observed_min:.2f} to "
                f"{observed_max:.2f})."
            )

        missing_value_count += int(
            numeric.isna().sum()
        )

        work[
            attr
        ] = numeric.astype(
            float
        )

    if missing_value_count:

        notes.append(
            f"{missing_value_count} missing descriptive response(s) "
            "were present across the selected PCA attributes. "
            "Product means use available valid observations; "
            "no missing value was imputed."
        )

    # -------------------------------------------------------------------------
    # Approved PCA unit:
    #
    # Rows    = products
    # Columns = descriptive sensory attributes
    # Values  = product means
    # -------------------------------------------------------------------------

    profile = (
        work.groupby(
            PRODUCT,
            sort=False,
        )[
            list(
                attributes
            )
        ]
        .mean()
        .reindex(
            product_order
        )
    )

    cell_counts = (
        work.groupby(
            PRODUCT,
            sort=False,
        )[
            list(
                attributes
            )
        ]
        .count()
        .reindex(
            product_order
        )
    )

    # -------------------------------------------------------------------------
    # Complete PCA matrix
    # -------------------------------------------------------------------------

    missing_product_means = (
        profile.isna()
    )

    if bool(
        missing_product_means
        .any()
        .any()
    ):

        cells: List[str] = []

        for product in profile.index:

            for attr in profile.columns:

                if bool(
                    missing_product_means.loc[
                        product,
                        attr,
                    ]
                ):

                    cells.append(
                        f"{product} × {attr}"
                    )

        raise ValueError(
            "HARD STOP: at least one product × attribute cell "
            "has no valid observations, so the complete PCA matrix "
            "cannot be formed. Missing cell(s): "
            + ", ".join(
                cells[
                    :20
                ]
            )
        )

    # -------------------------------------------------------------------------
    # G9-C1 — ZERO VARIANCE
    # -------------------------------------------------------------------------
    #
    # Evaluate variance on the actual product-level matrix that enters PCA.
    #
    # The approved rule is binary:
    #
    #     SD == 0 → HARD STOP until removed and disclosed
    #
    # Therefore the script does NOT silently remove an offending attribute.
    # -------------------------------------------------------------------------

    product_level_sd = (
        profile.std(
            axis=0,
            ddof=0,
        )
    )

    zero_variance = (
        product_level_sd.index[
            product_level_sd.abs()
            <= ZERO_TOL
        ]
        .tolist()
    )

    if zero_variance:

        raise ValueError(
            "HARD STOP — G9-C1 zero variance: "
            "the following selected attribute(s) have SD = 0 "
            "across product means: "
            + ", ".join(
                zero_variance
            )
            + ". SOP-05 does not allow the script to silently drop them. "
            "Remove them explicitly from the PCA scope or correct the "
            "source data, disclose the removal, and rerun."
        )

    return PreparedData(
        evaluation_df=work,
        profile_matrix=profile,
        cell_counts=cell_counts,
        product_order=product_order,
        attributes=list(
            attributes
        ),
        notes=notes,
        duplicate_count=duplicate_count,
        missing_value_count=missing_value_count,
    )

def variable_component_correlations(
    X_scaled: np.ndarray,
    scores: np.ndarray,
) -> np.ndarray:
    """
    Calculate correlation-based variable loadings.

    These are convenient for interpretation because they indicate how each
    standardized measured sensory variable aligns with each PC direction.
    """

    n_attributes = (
        X_scaled.shape[
            1
        ]
    )

    n_components = (
        scores.shape[
            1
        ]
    )

    out = np.zeros(
        (
            n_attributes,
            n_components,
        ),
        dtype=float,
    )

    for j in range(
        n_attributes
    ):

        x = (
            X_scaled[
                :,
                j,
            ]
        )

        x_sd = float(
            np.std(
                x,
                ddof=0,
            )
        )

        for k in range(
            n_components
        ):

            y = (
                scores[
                    :,
                    k,
                ]
            )

            y_sd = float(
                np.std(
                    y,
                    ddof=0,
                )
            )

            if (
                x_sd
                <= ZERO_TOL
                or y_sd
                <= ZERO_TOL
            ):

                out[
                    j,
                    k,
                ] = 0.0

            else:

                out[
                    j,
                    k,
                ] = float(
                    np.corrcoef(
                        x,
                        y,
                    )[
                        0,
                        1,
                    ]
                )

    return out

def classify_2d_adequacy(
    cumulative: float,
) -> Tuple[
    str,
    str,
    str,
]:
    """
    Apply the approved SOP-05 / G9-C2 thresholds.

    >= 60%      strong 2D workshop summary
    50 to <60%  usable with caution
    40 to <50%  exploratory / limited
    <40%         do not summarize category primarily from PC1/PC2

    PCA itself is not hard-stopped below 40%.
    Strong 2D interpretation is what is restricted.
    """

    if cumulative >= 0.60:

        return (
            "Strong 2D summary",
            "PASS",
            "PC1 + PC2 explain at least 60% of standardized "
            "sensory-profile variance; the 2D map is a strong "
            "workshop summary.",
        )

    if cumulative >= 0.50:

        return (
            "Usable with caution",
            "WARNING",
            "PC1 + PC2 explain 50% to <60%. Use the 2D map, "
            "but note that later components still carry "
            "meaningful structure.",
        )

    if cumulative >= 0.40:

        return (
            "Exploratory / limited 2D",
            "WARNING",
            "PC1 + PC2 explain 40% to <50%. Treat the biplot "
            "as a limited exploratory view rather than a complete "
            "category summary.",
        )

    return (
        "Do not summarize category primarily in 2D",
        "WARNING",
        "PC1 + PC2 explain <40%. PCA may still be reported, "
        "but the category should not be summarized primarily "
        "from the PC1/PC2 map.",
    )

def run_pca(
    prepared: PreparedData,
) -> PCAResult:
    """
    Standardize the product-level mean matrix and fit the approved PCA.
    """

    X = (
        prepared
        .profile_matrix
        .to_numpy(
            dtype=float
        )
    )

    # -------------------------------------------------------------------------
    # Standardization
    # -------------------------------------------------------------------------

    scaler = StandardScaler(
        with_mean=True,
        with_std=True,
    )

    X_scaled = (
        scaler.fit_transform(
            X
        )
    )

    if not np.isfinite(
        X_scaled
    ).all():

        raise ValueError(
            "HARD STOP: standardization produced non-finite values."
        )

    # -------------------------------------------------------------------------
    # PCA
    # -------------------------------------------------------------------------

    n_components = min(
        X_scaled.shape[
            0
        ],
        X_scaled.shape[
            1
        ],
    )

    if n_components < 2:

        raise ValueError(
            "HARD STOP: fewer than two PCA components can be "
            "estimated from the available matrix."
        )

    # Full SVD is deterministic.
    model = PCA(
        n_components=n_components,
        svd_solver="full",
    )

    score_array = (
        model.fit_transform(
            X_scaled
        )
    )

    if not np.isfinite(
        score_array
    ).all():

        raise ValueError(
            "HARD STOP: PCA produced non-finite product scores."
        )

    explained_ratio = np.asarray(
        model.explained_variance_ratio_,
        dtype=float,
    )

    if (
        explained_ratio.size
        < 2
        or not np.isfinite(
            explained_ratio[
                :2
            ]
        ).all()
    ):

        raise ValueError(
            "HARD STOP: PC1/PC2 explained variance could not "
            "be estimated reliably."
        )

    component_names = [
        f"PC{i + 1}"
        for i
        in range(
            n_components
        )
    ]

    # -------------------------------------------------------------------------
    # Product scores
    # -------------------------------------------------------------------------

    scores = pd.DataFrame(
        score_array,
        index=(
            prepared
            .profile_matrix
            .index
            .copy()
        ),
        columns=component_names,
    )

    scores.index.name = (
        PRODUCT
    )

    # -------------------------------------------------------------------------
    # Attribute loadings
    # -------------------------------------------------------------------------

    loading_array = (
        variable_component_correlations(
            X_scaled,
            score_array,
        )
    )

    loadings = pd.DataFrame(
        loading_array,
        index=(
            prepared
            .profile_matrix
            .columns
            .copy()
        ),
        columns=component_names,
    )

    loadings.index.name = (
        "attribute"
    )

    # -------------------------------------------------------------------------
    # Explained variance
    # -------------------------------------------------------------------------

    explained = pd.DataFrame(
        {
            "component": component_names,
            "explained_variance_ratio": explained_ratio,
            "cumulative_variance_ratio": np.cumsum(
                explained_ratio
            ),
        }
    )

    pc1 = float(
        explained_ratio[
            0
        ]
    )

    pc2 = float(
        explained_ratio[
            1
        ]
    )

    cumulative = (
        pc1
        + pc2
    )

    (
        adequacy_label,
        adequacy_status,
        adequacy_action,
    ) = classify_2d_adequacy(
        cumulative
    )

    return PCAResult(
        scaler=scaler,
        model=model,
        scaled_matrix=X_scaled,
        scores=scores,
        loadings=loadings,
        explained_variance=explained,
        pc1_variance=pc1,
        pc2_variance=pc2,
        cumulative_2d=cumulative,
        adequacy_label=adequacy_label,
        adequacy_status=adequacy_status,
        adequacy_action=adequacy_action,
    )

def friendly_attribute(
    attr: str,
) -> str:

    return (
        APPROVED_ATTRIBUTES.get(
            attr,
            attr
            .replace(
                "desc_",
                "",
            )
            .replace(
                "_",
                " ",
            )
            .title(),
        )
    )

def top_loading_names(
    series: pd.Series,
    positive: bool,
    n: int = 2,
) -> List[str]:

    if positive:

        subset = (
            series[
                series
                > 0
            ]
            .sort_values(
                ascending=False
            )
        )

    else:

        subset = (
            series[
                series
                < 0
            ]
            .sort_values(
                ascending=True
            )
        )

    return [
        friendly_attribute(
            attr
        )
        for attr
        in subset.head(
            n
        ).index.tolist()
    ]

def describe_dimension(
    pca_result: PCAResult,
    pc: str,
) -> Dict[
    str,
    str,
]:

    load = (
        pca_result
        .loadings[
            pc
        ]
    )

    score = (
        pca_result
        .scores[
            pc
        ]
    )

    positive_attrs = (
        top_loading_names(
            load,
            positive=True,
            n=2,
        )
    )

    negative_attrs = (
        top_loading_names(
            load,
            positive=False,
            n=2,
        )
    )

    high_product = str(
        score.idxmax()
    )

    low_product = str(
        score.idxmin()
    )

    return {
        "pc": pc,
        "positive_attrs": (
            ", ".join(
                positive_attrs
            )
            if positive_attrs
            else
            "No strong positive-side loading"
        ),
        "negative_attrs": (
            ", ".join(
                negative_attrs
            )
            if negative_attrs
            else
            "No strong negative-side loading"
        ),
        "high_product": high_product,
        "low_product": low_product,
    }

def build_answer(
    pca_result: PCAResult,
) -> Tuple[
    str,
    str,
]:

    dim1 = (
        describe_dimension(
            pca_result,
            "PC1",
        )
    )

    dim2 = (
        describe_dimension(
            pca_result,
            "PC2",
        )
    )

    # -------------------------------------------------------------------------
    # Headline based on G9-C2 adequacy
    # -------------------------------------------------------------------------

    if (
        pca_result
        .cumulative_2d
        >= 0.60
    ):

        headline = (
            "The first two sensory dimensions explain "
            f"{fmt_pct(pca_result.cumulative_2d)} "
            "of the standardized product-profile variance, "
            "giving a strong 2D category summary."
        )

    elif (
        pca_result
        .cumulative_2d
        >= 0.50
    ):

        headline = (
            "The first two sensory dimensions explain "
            f"{fmt_pct(pca_result.cumulative_2d)} "
            "of standardized profile variance; "
            "the 2D map is useful, but incomplete."
        )

    elif (
        pca_result
        .cumulative_2d
        >= 0.40
    ):

        headline = (
            "PC1 and PC2 explain "
            f"{fmt_pct(pca_result.cumulative_2d)} "
            "of standardized profile variance, so the "
            "2D map is exploratory and limited."
        )

    else:

        headline = (
            "PC1 and PC2 explain only "
            f"{fmt_pct(pca_result.cumulative_2d)} "
            "of standardized profile variance; the category "
            "should not be summarized primarily from this 2D map."
        )

    # -------------------------------------------------------------------------
    # Product / attribute story
    # -------------------------------------------------------------------------
    #
    # PCA signs are arbitrary, so we describe sides of the map relationally.
    # -------------------------------------------------------------------------

    detail = (
        f"On PC1, Product {dim1['high_product']} lies toward the side "
        f"associated with {dim1['positive_attrs']}, while Product "
        f"{dim1['low_product']} lies toward the opposite side associated "
        f"with {dim1['negative_attrs']}. PC2 separates Product "
        f"{dim2['high_product']} from Product {dim2['low_product']} "
        f"along a second measured sensory direction. These are exploratory "
        f"positioning patterns, not significance tests or preference drivers."
    )

    return (
        headline,
        detail,
    )

def fmt_int(value: object, na: str = "—") -> str:
    try:
        x = int(value)
    except Exception:
        return na
    return f"{x:,}"

def is_export_index(series: pd.Series) -> bool:
    numeric = pd.to_numeric(series, errors="coerce")
    if numeric.isna().any():
        return False
    expected = np.arange(len(series), dtype=float)
    return np.array_equal(numeric.to_numpy(dtype=float), expected)

def remove_export_index(df: pd.DataFrame) -> Tuple[pd.DataFrame, List[str]]:
    """Remove only obvious CSV export-index artifacts."""
    out = df.copy()
    removed: List[str] = []

    for col in list(out.columns):
        normalized = str(col).strip().lower()

        if (
            normalized in {"index", "unnamed: 0"}
            and is_export_index(out[col])
        ):
            out = out.drop(columns=[col])
            removed.append(str(col))

    return out, removed

def selected_attributes() -> List[str]:
    mode = str(ATTRIBUTE_MODE).strip().lower()

    if mode == "all_approved":
        attrs = list(APPROVED_ATTRIBUTES.keys())

    elif mode == "selected":
        attrs = list(
            dict.fromkeys(
                SELECTED_ATTRIBUTES
            )
        )

        if len(attrs) < 2:
            raise ValueError(
                "ATTRIBUTE_MODE='selected' requires at least two attributes in "
                "SELECTED_ATTRIBUTES for a multivariate similarity analysis."
            )

    else:
        raise ValueError(
            "ATTRIBUTE_MODE must be either 'all_approved' or 'selected'."
        )

    unknown = [
        attr
        for attr in attrs
        if attr not in APPROVED_ATTRIBUTES
    ]

    if unknown:
        raise ValueError(
            "The following configured clustering attributes are not approved "
            "schema-defined descriptive sensory variables: "
            + ", ".join(unknown)
        )

    return attrs

def validate_clustering_configuration() -> None:
    method = str(
        LINKAGE_METHOD
    ).strip().lower()

    metric = str(
        DISTANCE_METRIC
    ).strip().lower()

    allowed = {
        "single",
        "complete",
        "average",
        "weighted",
        "centroid",
        "median",
        "ward",
    }

    if method not in allowed:
        raise ValueError(
            f"Unsupported LINKAGE_METHOD='{LINKAGE_METHOD}'. "
            f"Choose one of: {', '.join(sorted(allowed))}."
        )

    if method == "ward" and metric != "euclidean":
        raise ValueError(
            "Ward linkage requires DISTANCE_METRIC='euclidean'."
        )

def capability_gate(
    df: pd.DataFrame,
    attrs: Sequence[str],

    str,
    List[Dict[str, str]],
    List[str],

    """
    Verify that the full joint sensory representation
    required by SOP-06 exists.

    Because clustering is multivariate, silently omitting
    a requested attribute changes the definition of sensory
    similarity. Missing configured attributes therefore
    hard-stop the current clustering plan rather than being
    replaced by neighboring constructs.
    """

    diagnostics: List[
        Dict[str, str]
    ] = []

    hard_stops: List[str] = []

    # -------------------------------------------------------------------------
    # Product identifier
    # -------------------------------------------------------------------------

    if PRODUCT in df.columns:

        diagnostics.append(
            diagnostic_row(
                "Product identifier available",
                PRODUCT,
                f"Required column '{PRODUCT}' present",
                "PASS",
                "Proceed.",
                (
                    "SOP-06 requires a product identifier "
                    "so evaluation rows can be aggregated "
                    "to product-level sensory profiles."
                ),
            )
        )

    else:

        hard_stops.append(
            f"Required product identifier '{PRODUCT}' is missing."
        )

        diagnostics.append(
            diagnostic_row(
                "Product identifier available",
                "Missing",
                f"Required column '{PRODUCT}' present",
                "HARD STOP",
                "Do not run SOP-06.",
                (
                    "Without product identity, product-level "
                    "sensory means cannot be built."
                ),
            )
        )

    # -------------------------------------------------------------------------
    # Selected descriptive attributes
    # -------------------------------------------------------------------------

    missing_attrs = [
        attr
        for attr in attrs
        if attr not in df.columns
    ]

    if not missing_attrs:

        diagnostics.append(
            diagnostic_row(
                "Selected sensory variables available",
                f"{len(attrs)} of {len(attrs)} present",
                (
                    "Every configured clustering "
                    "attribute must exist"
                ),
                "PASS",
                "Proceed.",
                (
                    "Similarity is defined jointly by "
                    "the configured measured sensory "
                    "variables."
                ),
            )
        )

    else:

        hard_stops.append(
            "Missing required configured sensory variables: "
            + ", ".join(
                missing_attrs
            )
        )

        diagnostics.append(
            diagnostic_row(
                "Selected sensory variables available",
                (
                    "Missing: "
                    + ", ".join(
                        missing_attrs
                    )
                ),
                (
                    "Every configured clustering "
                    "attribute must exist"
                ),
                "HARD STOP",
                (
                    "Restore the measured fields or "
                    "intentionally revise the clustering "
                    "input plan."
                ),
                (
                    "Silently dropping a requested attribute "
                    "would redefine sensory similarity."
                ),
            )
        )

    capability_status = (
        "UNSUPPORTED"
        if hard_stops
        else "SUPPORTED"
    )

    return (
        capability_status,
        diagnostics,
        hard_stops,
    )

def validate_and_prepare_data(
    raw_df: pd.DataFrame,
    attrs: Sequence[str],
) -> Tuple[
    pd.DataFrame,
    List[Dict[str, str]],
    List[str],
    List[str],
    Dict[str, object],
]:
    """
    Validate the evaluation structure before aggregation.

    Returns
    -------
    cleaned_df
    diagnostics
    hard_stops
    warnings
    study_summary
    """

    diagnostics: List[
        Dict[str, str]
    ] = []

    hard_stops: List[str] = []

    warning_messages: List[str] = []

    df, removed_index_cols = (
        remove_export_index(
            raw_df
        )
    )

    if removed_index_cols:

        warning_messages.append(
            "Removed obvious CSV export-index column(s): "
            + ", ".join(
                removed_index_cols
            )
        )

    n_rows = len(
        df
    )

    n_products = (
        int(
            df[
                PRODUCT
            ].nunique(
                dropna=True
            )
        )
        if PRODUCT in df.columns
        else 0
    )

    n_panelists = (
        int(
            df[
                PANELIST
            ].nunique(
                dropna=True
            )
        )
        if PANELIST in df.columns
        else None
    )

    # -------------------------------------------------------------------------
    # Row-count snapshot
    # -------------------------------------------------------------------------

    if n_rows == EXPECTED_ROWS:

        row_status = "PASS"
        row_action = "Proceed."

    else:

        row_status = "WARNING"

        row_action = (
            "Proceed only after noting that the active dataset "
            "differs from the workshop snapshot."
        )

        warning_messages.append(
            f"Dataset row count is {n_rows}, "
            f"not the workshop snapshot of {EXPECTED_ROWS}."
        )

    diagnostics.append(
        diagnostic_row(
            "Evaluation-row snapshot",
            fmt_int(
                n_rows
            ),
            (
                f"Workshop snapshot = {EXPECTED_ROWS}; "
                "difference is a QC warning, not an automatic stop"
            ),
            row_status,
            row_action,
            (
                "The script inspects the active dataset "
                "rather than assuming the snapshot counts."
            ),
        )
    )

    # -------------------------------------------------------------------------
    # Product count
    # -------------------------------------------------------------------------

    if n_products >= 2:

        product_status = (
            "PASS"
            if n_products
            == EXPECTED_PRODUCTS
            else "WARNING"
        )

        product_action = (
            "Proceed."
            if product_status
            == "PASS"
            else (
                "Proceed with the observed product set "
                "and disclose the snapshot difference."
            )
        )

        if product_status == "WARNING":

            warning_messages.append(
                f"Observed {n_products} products rather than "
                f"the workshop snapshot of {EXPECTED_PRODUCTS}."
            )

    else:

        product_status = (
            "HARD STOP"
        )

        product_action = (
            "At least two products are mathematically "
            "required for product similarity clustering."
        )

        hard_stops.append(
            "Fewer than two products have usable product identifiers."
        )

    diagnostics.append(
        diagnostic_row(
            "Usable products",
            fmt_int(
                n_products
            ),
            (
                ">= 2 products; "
                "workshop snapshot = 8"
            ),
            product_status,
            product_action,
            (
                "Hierarchical product similarity is "
                "undefined with fewer than two products."
            ),
        )
    )

    # -------------------------------------------------------------------------
    # Missing product identifiers
    # -------------------------------------------------------------------------

    if PRODUCT in df.columns:

        missing_product = int(
            df[
                PRODUCT
            ].isna().sum()
        )

        if missing_product > 0:

            hard_stops.append(
                f"{missing_product} row(s) have "
                f"missing '{PRODUCT}' values."
            )

            status = "HARD STOP"

            action = (
                "Resolve missing product identity "
                "before aggregation."
            )

        else:

            status = "PASS"
            action = "Proceed."

        diagnostics.append(
            diagnostic_row(
                "Missing product identifiers",
                fmt_int(
                    missing_product
                ),
                "0 missing product identifiers",
                status,
                action,
                (
                    "Rows without product identity "
                    "cannot be assigned to a product "
                    "sensory profile."
                ),
            )
        )

    # -------------------------------------------------------------------------
    # Panelist-level structural QC
    # -------------------------------------------------------------------------
    # panelist_id is not a formal SOP-06 clustering input.
    # When available, however, it is useful for verifying
    # that repeated observations were not duplicated or
    # structurally distorted before product aggregation.
    # -------------------------------------------------------------------------

    if PANELIST in df.columns:

        missing_panelist = int(
            df[
                PANELIST
            ].isna().sum()
        )

        if missing_panelist > 0:

            warning_messages.append(
                f"{missing_panelist} row(s) have missing panelist IDs; "
                "structural repeated-measures QC is incomplete "
                "for those rows."
            )

            diagnostics.append(
                diagnostic_row(
                    "Panelist identifiers for design QC",
                    f"{missing_panelist} missing",
                    (
                        "0 missing preferred for complete "
                        "repeated-measures QC"
                    ),
                    "WARNING",
                    (
                        "Proceed with clustering, but disclose "
                        "incomplete design QC."
                    ),
                    (
                        "Panelist ID is not a clustering feature; "
                        "it is used only to validate row structure "
                        "before aggregation."
                    ),
                )
            )

        else:

            diagnostics.append(
                diagnostic_row(
                    "Panelist identifiers for design QC",
                    (
                        f"{n_panelists} "
                        "unique panelists"
                    ),
                    (
                        f"Workshop snapshot = {EXPECTED_PANELISTS}; "
                        "panelist_id is QC-only for SOP-06"
                    ),
                    (
                        "PASS"
                        if n_panelists
                        == EXPECTED_PANELISTS
                        else "WARNING"
                    ),
                    (
                        "Proceed."
                        if n_panelists
                        == EXPECTED_PANELISTS
                        else (
                            "Proceed and disclose "
                            "the snapshot difference."
                        )
                    ),
                    (
                        "SOP-06 clusters product-level means, "
                        "not individual panelists."
                    ),
                )
            )

            if n_panelists != EXPECTED_PANELISTS:

                warning_messages.append(
                    f"Observed {n_panelists} panelists rather than "
                    f"the workshop snapshot of {EXPECTED_PANELISTS}."
                )

        # ---------------------------------------------------------------------
        # Duplicate consumer × product rows
        # ---------------------------------------------------------------------

        duplicate_rows = int(
            df.duplicated(
                subset=[
                    PANELIST,
                    PRODUCT,
                ],
                keep=False,
            ).sum()
        )

        duplicate_pairs = int(
            df.duplicated(
                subset=[
                    PANELIST,
                    PRODUCT,
                ],
                keep="first",
            ).sum()
        )

        if duplicate_pairs > 0:

            hard_stops.append(
                f"Found {duplicate_pairs} duplicate "
                "panelist × product observation(s)."
            )

            dup_status = (
                "HARD STOP"
            )

            dup_action = (
                "Resolve duplicate observations; "
                "do not silently average or overweight them."
            )

        else:

            dup_status = (
                "PASS"
            )

            dup_action = (
                "Proceed."
            )

        diagnostics.append(
            diagnostic_row(
                (
                    "Duplicate panelist × "
                    "product observations"
                ),
                (
                    f"{duplicate_pairs} duplicate pair(s); "
                    f"{duplicate_rows} row(s) involved"
                ),
                (
                    "0 duplicate panelist × "
                    "product observations"
                ),
                dup_status,
                dup_action,
                (
                    "Duplicate evaluation rows would change "
                    "product means by unintentionally giving "
                    "some evaluations extra weight."
                ),
            )
        )

        # ---------------------------------------------------------------------
        # Repeated-measures grid completeness
        # ---------------------------------------------------------------------

        if (
            n_panelists
            and n_products
        ):

            unique_grid = (
                df[
                    [
                        PANELIST,
                        PRODUCT,
                    ]
                ]
                .dropna()
                .drop_duplicates()
            )

            expected_grid = (
                n_panelists
                * n_products
            )

            actual_grid = len(
                unique_grid
            )

            completeness = (
                actual_grid
                / expected_grid
                if expected_grid > 0
                else float("nan")
            )

            if (
                np.isfinite(
                    completeness
                )
                and abs(
                    completeness
                    - 1.0
                )
                <= ZERO_TOL
            ):

                grid_status = (
                    "PASS"
                )

                grid_action = (
                    "Proceed."
                )

            else:

                grid_status = (
                    "WARNING"
                )

                grid_action = (
                    "Proceed only with explicit disclosure "
                    "that product means may be based on an "
                    "incomplete repeated-measures grid."
                )

                warning_messages.append(
                    f"Observed {actual_grid} of {expected_grid} "
                    "possible panelist × product cells "
                    f"({completeness:.1%} complete)."
                )

            diagnostics.append(
                diagnostic_row(
                    "Repeated-measures grid completeness",
                    (
                        f"{actual_grid} / {expected_grid} cells "
                        f"({completeness:.1%})"
                        if np.isfinite(
                            completeness
                        )
                        else "Not estimable"
                    ),
                    (
                        "Complete grid preferred "
                        "for this workshop design"
                    ),
                    grid_status,
                    grid_action,
                    (
                        "SOP-06 ultimately clusters product means, "
                        "but unequal evaluation coverage can affect "
                        "those means."
                    ),
                )
            )

    else:

        warning_messages.append(
            f"'{PANELIST}' is absent. SOP-06 can still use "
            "product-level sensory means, but duplicate/"
            "repeated-measures structural QC cannot be verified."
        )

        diagnostics.append(
            diagnostic_row(
                "Panelist-level structural QC",
                (
                    f"'{PANELIST}' "
                    "not available"
                ),
                (
                    "Panelist ID is not a formal SOP-06 "
                    "clustering input, but is preferred "
                    "for design QC"
                ),
                "WARNING",
                (
                    "Proceed with product-level clustering only; "
                    "disclose that repeated-measures row QC "
                    "could not be verified."
                ),
                (
                    "The clustering unit is product, "
                    "so panelist ID is not used as a feature."
                ),
            )
        )

    # -------------------------------------------------------------------------
    # Numeric conversion + scale validation
    # -------------------------------------------------------------------------
    # Do not silently replace, clip, or impute invalid sensory values.
    # -------------------------------------------------------------------------

    prepared = (
        df.copy()
    )

    parse_failures: Dict[
        str,
        int,
    ] = {}

    missing_counts: Dict[
        str,
        int,
    ] = {}

    range_violations: Dict[
        str,
        int,
    ] = {}

    for attr in attrs:

        original = (
            prepared[
                attr
            ]
        )

        numeric = pd.to_numeric(
            original,
            errors="coerce",
        )

        parse_failures[
            attr
        ] = int(
            (
                original.notna()
                & numeric.isna()
            ).sum()
        )

        missing_counts[
            attr
        ] = int(
            numeric.isna().sum()
        )

        range_violations[
            attr
        ] = int(
            (
                (
                    numeric
                    < SCALE_MIN
                )
                |
                (
                    numeric
                    > SCALE_MAX
                )
                |
                ~np.isfinite(
                    numeric
                )
            )
            .fillna(
                False
            )
            .sum()
        )

        prepared[
            attr
        ] = numeric

    total_parse_failures = sum(
        parse_failures.values()
    )

    total_missing = sum(
        missing_counts.values()
    )

    total_range_violations = sum(
        range_violations.values()
    )

    # -------------------------------------------------------------------------
    # Parse failures
    # -------------------------------------------------------------------------

    if total_parse_failures > 0:

        offenders = [
            f"{attr}: {count}"
            for attr, count
            in parse_failures.items()
            if count > 0
        ]

        hard_stops.append(
            "Non-numeric values found in required "
            "descriptive fields ("
            + "; ".join(
                offenders
            )
            + ")."
        )

        status = (
            "HARD STOP"
        )

        action = (
            "Correct the descriptive data; "
            "do not coerce invalid text to "
            "usable sensory scores."
        )

    else:

        status = (
            "PASS"
        )

        action = (
            "Proceed."
        )

    diagnostics.append(
        diagnostic_row(
            "Numeric descriptive values",
            (
                f"{total_parse_failures} "
                "parse failure(s)"
            ),
            (
                "0 non-numeric values in "
                "selected descriptive fields"
            ),
            status,
            action,
            (
                "The clustering representation requires "
                "numeric measured intensities."
            ),
        )
    )

    # -------------------------------------------------------------------------
    # Missing descriptive data
    # -------------------------------------------------------------------------

    if total_missing > 0:

        offenders = [
            f"{attr}: {count}"
            for attr, count
            in missing_counts.items()
            if count > 0
        ]

        hard_stops.append(
            "Missing values found in selected "
            "descriptive fields ("
            + "; ".join(
                offenders
            )
            + ")."
        )

        status = (
            "HARD STOP"
        )

        action = (
            "Resolve or predefine an approved "
            "missing-data plan before clustering."
        )

    else:

        status = (
            "PASS"
        )

        action = (
            "Proceed."
        )

    diagnostics.append(
        diagnostic_row(
            "Missing selected sensory values",
            fmt_int(
                total_missing
            ),
            (
                "0 missing values under this "
                "standalone SOP-06 implementation"
            ),
            status,
            action,
            (
                "SOP-06 does not define an imputation rule. "
                "Silent imputation or pairwise deletion would "
                "alter product profiles."
            ),
        )
    )

    # -------------------------------------------------------------------------
    # Scale validity
    # -------------------------------------------------------------------------

    if total_range_violations > 0:

        offenders = [
            f"{attr}: {count}"
            for attr, count
            in range_violations.items()
            if count > 0
        ]

        hard_stops.append(
            f"Values outside the approved {SCALE_LABEL} "
            "were found ("
            + "; ".join(
                offenders
            )
            + ")."
        )

        status = (
            "HARD STOP"
        )

        action = (
            "Correct the invalid scale "
            "values before clustering."
        )

    else:

        status = (
            "PASS"
        )

        action = (
            "Proceed."
        )

    diagnostics.append(
        diagnostic_row(
            "Descriptive scale validity",
            (
                f"{total_range_violations} "
                "out-of-range / non-finite value(s)"
            ),
            (
                "Every selected value must be finite "
                f"and within {SCALE_MIN:g}–{SCALE_MAX:g}"
            ),
            status,
            action,
            (
                "All approved descriptive sensory "
                "attributes use the same 0–10 "
                "intensity scale."
            ),
        )
    )

    study_summary: Dict[
        str,
        object,
    ] = {
        "rows": n_rows,
        "products": n_products,
        "panelists": n_panelists,
        "removed_index_columns": (
            removed_index_cols
        ),
    }

    return (
        prepared,
        diagnostics,
        hard_stops,
        warning_messages,
        study_summary,
    )

def build_product_sensory_representation(
    df: pd.DataFrame,
    attrs: Sequence[str],
) -> Tuple[
    pd.DataFrame,
    pd.DataFrame,
    List[Dict[str, str]],
    List[str],
]:

    diagnostics: List[
        Dict[str, str]
    ] = []

    hard_stops: List[str] = []

    working = (
        df.copy()
    )

    working[
        PRODUCT
    ] = (
        working[
            PRODUCT
        ]
        .astype(
            str
        )
        .str.strip()
    )

    # -------------------------------------------------------------------------
    # Product means
    # -------------------------------------------------------------------------

    product_means = (
        working
        .groupby(
            PRODUCT,
            observed=True,
        )[
            list(
                attrs
            )
        ]
        .mean()
        .sort_index()
    )

    if len(
        product_means
    ) < 2:

        hard_stops.append(
            "Fewer than two product-level "
            "sensory profiles could be built."
        )

        diagnostics.append(
            diagnostic_row(
                "Product-level profile matrix",
                (
                    f"{len(product_means)} "
                    "product profile(s)"
                ),
                (
                    ">= 2 complete "
                    "product profiles"
                ),
                "HARD STOP",
                "Do not cluster.",
                (
                    "Hierarchical product similarity "
                    "requires at least two products."
                ),
            )
        )

        return (
            product_means,
            pd.DataFrame(),
            diagnostics,
            hard_stops,
        )

    finite_matrix = (
        np.isfinite(
            product_means.to_numpy(
                dtype=float
            )
        )
        .all()
    )

    if finite_matrix:

        diagnostics.append(
            diagnostic_row(
                "Product-level profile matrix",
                (
                    f"{product_means.shape[0]} products × "
                    f"{product_means.shape[1]} attributes"
                ),
                (
                    "Every product × selected-attribute "
                    "mean must be finite"
                ),
                "PASS",
                "Proceed to standardization.",
                (
                    "Evaluation rows are aggregated to "
                    "product means because SOP-06's "
                    "analytical unit is the product."
                ),
            )
        )

    else:

        hard_stops.append(
            "At least one product × attribute "
            "mean is non-finite."
        )

        diagnostics.append(
            diagnostic_row(
                "Product-level profile matrix",
                "Contains non-finite product means",
                (
                    "Every product × selected-attribute "
                    "mean must be finite"
                ),
                "HARD STOP",
                (
                    "Resolve missing/invalid product "
                    "profiles before clustering."
                ),
                (
                    "A non-finite product profile cannot "
                    "be standardized or assigned a "
                    "valid distance."
                ),
            )
        )

        return (
            product_means,
            pd.DataFrame(),
            diagnostics,
            hard_stops,
        )

    # -------------------------------------------------------------------------
    # Standardization
    # -------------------------------------------------------------------------
    # Standardize across product means using population SD,
    # matching common StandardScaler-style preprocessing.
    # -------------------------------------------------------------------------

    centers = (
        product_means
        .mean(
            axis=0
        )
    )

    scales = (
        product_means
        .std(
            axis=0,
            ddof=0,
        )
    )

    zero_attrs = [
        attr
        for attr
        in attrs
        if (
            not np.isfinite(
                scales[
                    attr
                ]
            )
            or scales[
                attr
            ] <= ZERO_TOL
        )
    ]

    if zero_attrs:

        hard_stops.append(
            "The following selected sensory attributes "
            "have zero variance across product means "
            "and cannot be standardized: "
            + ", ".join(
                zero_attrs
            )
        )

        diagnostics.append(
            diagnostic_row(
                "Across-product sensory variation",
                (
                    "Zero-variance: "
                    + ", ".join(
                        zero_attrs
                    )
                ),
                (
                    "Every selected clustering attribute "
                    "must have SD > 0 across product means"
                ),
                "HARD STOP",
                (
                    "Intentionally remove/review the "
                    "zero-variance attribute(s), disclose "
                    "the change, then rerun."
                ),
                (
                    "A constant attribute contains no "
                    "product-separation information and "
                    "its z-score is undefined. The script "
                    "does not silently drop configured inputs."
                ),
            )
        )

        return (
            product_means,
            pd.DataFrame(),
            diagnostics,
            hard_stops,
        )

    diagnostics.append(
        diagnostic_row(
            "Across-product sensory variation",
            (
                f"All {len(attrs)} selected "
                "attributes have SD > 0"
            ),
            (
                "Every selected clustering attribute "
                "must have SD > 0 across product means"
            ),
            "PASS",
            "Proceed.",
            (
                "Standardization would be undefined "
                "for a constant sensory attribute."
            ),
        )
    )

    standardized = (
        product_means
        - centers
    ) / scales

    finite_standardized = (
        np.isfinite(
            standardized.to_numpy(
                dtype=float
            )
        )
        .all()
    )

    if not finite_standardized:

        hard_stops.append(
            "Standardized sensory matrix contains "
            "non-finite values."
        )

        status = (
            "HARD STOP"
        )

        action = (
            "Do not cluster until the standardization "
            "problem is resolved."
        )

    else:

        status = (
            "PASS"
        )

        action = (
            "Proceed to distance calculation."
        )

    diagnostics.append(
        diagnostic_row(
            "Standardized sensory representation",
            (
                f"{standardized.shape[0]} × "
                f"{standardized.shape[1]} z-score matrix"
            ),
            "All standardized values finite",
            status,
            action,
            (
                "Standardization prevents an attribute "
                "with larger across-product spread from "
                "dominating Euclidean distance solely "
                "because of scale dispersion."
            ),
        )
    )

    return (
        product_means,
        standardized,
        diagnostics,
        hard_stops,
    )

def run_hierarchical_clustering(
    product_means: pd.DataFrame,
    standardized: pd.DataFrame,
) -> Dict[str, object]:

    products = [
        str(
            value
        )
        for value
        in standardized.index.tolist()
    ]

    matrix = (
        standardized
        .to_numpy(
            dtype=float
        )
    )

    # -------------------------------------------------------------------------
    # Pairwise distance
    # -------------------------------------------------------------------------

    condensed = pdist(
        matrix,
        metric=DISTANCE_METRIC,
    )

    if (
        condensed.size == 0
        or not np.isfinite(
            condensed
        ).all()
    ):

        raise ValueError(
            "Pairwise standardized sensory distances "
            "are not finite/computable."
        )

    # -------------------------------------------------------------------------
    # Hierarchical linkage
    # -------------------------------------------------------------------------

    linkage_matrix = linkage(
        matrix,
        method=LINKAGE_METHOD,
        metric=DISTANCE_METRIC,
        optimal_ordering=(
            OPTIMAL_LEAF_ORDERING
        ),
    )

    if not np.isfinite(
        linkage_matrix
    ).all():

        raise ValueError(
            "Hierarchical linkage matrix "
            "contains non-finite values."
        )

    # -------------------------------------------------------------------------
    # Pairwise distance matrix
    # -------------------------------------------------------------------------

    distance_array = squareform(
        condensed
    )

    distance_matrix = pd.DataFrame(
        distance_array,
        index=products,
        columns=products,
    )

    pair_rows: List[
        Dict[str, object]
    ] = []

    for i in range(
        len(
            products
        )
    ):

        for j in range(
            i + 1,
            len(
                products
            ),
        ):

            pair_rows.append(
                {
                    "product_a": (
                        products[
                            i
                        ]
                    ),
                    "product_b": (
                        products[
                            j
                        ]
                    ),
                    "distance": float(
                        distance_array[
                            i,
                            j,
                        ]
                    ),
                }
            )

    pairwise = (
        pd.DataFrame(
            pair_rows
        )
        .sort_values(
            [
                "distance",
                "product_a",
                "product_b",
            ],
            ascending=[
                True,
                True,
                True,
            ],
        )
        .reset_index(
            drop=True
        )
    )

    # -------------------------------------------------------------------------
    # Closest pair(s)
    # -------------------------------------------------------------------------

    min_distance = float(
        pairwise.iloc[
            0
        ][
            "distance"
        ]
    )

    tie_tolerance = max(
        ZERO_TOL,
        abs(
            min_distance
        )
        * 1e-9,
    )

    closest_pairs = (
        pairwise[
            np.isclose(
                pairwise[
                    "distance"
                ],
                min_distance,
                atol=(
                    tie_tolerance
                ),
                rtol=1e-9,
            )
        ]
        .copy()
    )

    # -------------------------------------------------------------------------
    # Nearest neighbor by product
    # -------------------------------------------------------------------------

    nearest_rows: List[
        Dict[str, object]
    ] = []

    for product in products:

        row = (
            distance_matrix
            .loc[
                product
            ]
            .drop(
                index=product
            )
        )

        nearest_distance = float(
            row.min()
        )

        tied = [
            str(
                idx
            )
            for idx, value
            in row.items()
            if math.isclose(
                float(
                    value
                ),
                nearest_distance,
                rel_tol=1e-9,
                abs_tol=ZERO_TOL,
            )
        ]

        nearest_rows.append(
            {
                "product": (
                    product
                ),
                "nearest_product": (
                    ", ".join(
                        tied
                    )
                ),
                "distance": (
                    nearest_distance
                ),
            }
        )

    nearest_neighbors = (
        pd.DataFrame(
            nearest_rows
        )
        .sort_values(
            "product"
        )
        .reset_index(
            drop=True
        )
    )

    # -------------------------------------------------------------------------
    # Cophenetic correlation
    # -------------------------------------------------------------------------
    # This is an informational description of how closely
    # dendrogram merge heights preserve original pairwise
    # distances. SOP-06 defines no approved pass/fail cutoff.
    # -------------------------------------------------------------------------

    cophenetic_corr = (
        float("nan")
    )

    if (
        len(
            products
        ) >= 3
        and np.ptp(
            condensed
        ) > ZERO_TOL
    ):

        try:

            (
                cophenetic_corr,
                _,
            ) = cophenet(
                linkage_matrix,
                condensed,
            )

            cophenetic_corr = float(
                cophenetic_corr
            )

        except Exception:

            cophenetic_corr = (
                float("nan")
            )

    # -------------------------------------------------------------------------
    # Dendrogram geometry
    # -------------------------------------------------------------------------

    dendro_data = dendrogram(
        linkage_matrix,
        labels=products,
        no_plot=True,
        color_threshold=0,
    )

    # -------------------------------------------------------------------------
    # Exact sensory-profile ties
    # -------------------------------------------------------------------------

    exact_ties = (
        pairwise[
            pairwise[
                "distance"
            ] <= ZERO_TOL
        ]
        .copy()
    )

    diagnostics = [
        diagnostic_row(
            "Pairwise standardized distances",
            (
                f"{len(pairwise)} pairwise "
                "distances; all finite"
            ),
            "All distances finite",
            "PASS",
            "Proceed.",
            (
                "Euclidean distance is computed "
                "after standardizing the product-level "
                "measured sensory means."
            ),
        ),
        diagnostic_row(
            "Hierarchical linkage construction",
            (
                f"{LINKAGE_METHOD} linkage + "
                f"{DISTANCE_METRIC} distance"
            ),
            (
                "Distance and linkage must be "
                "declared transparently"
            ),
            "PASS",
            (
                "Interpret as exploratory "
                "relative similarity."
            ),
            (
                "SOP-06 does not prescribe a unique "
                "linkage; results can change under another "
                "defensible linkage choice."
            ),
        ),
        diagnostic_row(
            "Cophenetic correlation",
            (
                fmt_number(
                    cophenetic_corr,
                    3,
                )
                if np.isfinite(
                    cophenetic_corr
                )
                else "Not estimable"
            ),
            (
                "No approved SOP-06 numeric cutoff; "
                "informational only"
            ),
            (
                "PASS"
                if (
                    np.isfinite(
                        cophenetic_corr
                    )
                    or len(
                        products
                    ) < 3
                    or np.ptp(
                        condensed
                    ) <= ZERO_TOL
                )
                else "WARNING"
            ),
            (
                "Report descriptively; do not use it "
                "to claim statistically validated clusters."
            ),
            (
                "This summarizes how closely dendrogram "
                "merge distances preserve the original "
                "pairwise-distance geometry. The project "
                "has not defined a pass/fail threshold "
                "for SOP-06."
            ),
        ),
    ]

    warning_messages: List[
        str
    ] = []

    if not exact_ties.empty:

        tie_labels = [
            (
                f"{row.product_a}"
                f"–"
                f"{row.product_b}"
            )
            for row
            in exact_ties.itertuples(
                index=False
            )
        ]

        warning_messages.append(
            "Exact measured-profile tie(s) detected: "
            + ", ".join(
                tie_labels

            + "."
        )

        diagnostics.append(
            diagnostic_row(
                "Exact product-profile ties",
                (
                    ", ".join(
                        tie_labels
                    )
                ),
                (
                    f"Distance <= {ZERO_TOL:g} means "
                    "identical selected standardized "
                    "product-mean profiles"

                (
                    "Interpret as identical on the "
                    "selected measured descriptive profile "

                "None",
                (
                    f"Distance <= "
                    f"{ZERO_TOL:g}"
                ),
                "PASS",
                "Proceed.",
                (
                    "No product pair had exactly identical "
                    "selected standardized product-mean "
                    "profiles within numerical tolerance."
                ),
            )
        )

    return {
        "products": (
            products
        ),
        "product_means": (
            product_means

        "standardized": (

def closest_pair_sentence(
    result: Dict[str, object],
) -> str:

    closest = (
        result[
            "closest_pairs"
        ]
    )

    min_distance = float(
        result[
            "min_distance"
        ]
    )

    pair_labels = [
        (
            f"Products "
            f"{row.product_a} "
            f"and "
            f"{row.product_b}"
        )
        for row
        in closest.itertuples(
            index=False
        )
    ]

    if len(
        pair_labels
    ) == 1:

        return (
            f"{pair_labels[0]} are the most similar "
            "on the selected measured sensory profile "
            f"(standardized Euclidean distance = "
            f"{min_distance:.2f})."
        )

    return (
        "The closest measured sensory-profile pairs are "
        f"{join_words(pair_labels)} "
        f"(tied standardized Euclidean distance = "
        f"{min_distance:.2f})."
    )

def closest_pair_detail(
    result: Dict[str, object],
    attrs: Sequence[str],
) -> Tuple[
    str,
    pd.DataFrame,
]:

    closest = (
        result[
            "closest_pairs"
        ]
    )

    product_means: pd.DataFrame = (
        result[
            "product_means"
        ]
    )

    first = (
        closest.iloc[
            0
        ]
    )

    product_a = str(
        first[
            "product_a"
        ]
    )

    product_b = str(
        first[
            "product_b"
        ]
    )

    rows: List[
        Dict[str, object]
    ] = []

    for attr in attrs:

        mean_a = float(
            product_means.loc[
                product_a,
                attr,
            ]
        )

        mean_b = float(
            product_means.loc[
                product_b,
                attr,
            ]
        )

        rows.append(
            {
                "attribute": (
                    attr
                ),
                "label": (
                    APPROVED_ATTRIBUTES[
                        attr
                    ]
                ),
                "mean_a": (
                    mean_a
                ),
                "mean_b": (
                    mean_b
                ),
                "absolute_difference": abs(
                    mean_a
                    - mean_b
                ),
            }
        )

    detail = (
        pd.DataFrame(
            rows
        )
        .sort_values(
            [
                "absolute_difference",
                "label",
            ],
            ascending=[
                False,
                True,
            ],
        )
        .reset_index(
            drop=True
        )
    )

    max_diff = float(
        detail[
            "absolute_difference"
        ].max()
    )

    if max_diff <= ZERO_TOL:

        sentence = (
            f"Products {product_a} and {product_b} have "
            "the same product-level means across every "
            "selected measured descriptive attribute. "
            "That does not imply equality on unmeasured "
            "sensory properties, comments, liking, or "
            "other outcomes."
        )

    else:

        top = (
            detail.head(
                3
            )
        )

        pieces = [
            (
                f"{row.label} "
                f"({row.absolute_difference:.2f} "
                "intensity points apart)"
            )
            for row
            in top.itertuples(
                index=False
            )
        ]

        sentence = (
            f"Products {product_a} and {product_b} are "
            "closest overall; their largest remaining "
            "mean-profile differences are "
            + join_words(
                pieces
            )
            + "."
        )

    return (
        sentence,
        detail,
    )

def color_for_distance(
    value: float,
    min_nonzero: float,
    max_distance: float,
) -> str:

    if not np.isfinite(
        value
    ):

        return (
            "rgba(27,75,90,0.04)"
        )

    if value <= ZERO_TOL:

        return (
            "rgba(27,75,90,0.60)"
        )

    if (
        max_distance
        <= min_nonzero
        + ZERO_TOL
    ):

        closeness = 0.5

    else:

        closeness = (
            1.0
            - (
                value
                - min_nonzero
            )
            /
            (
                max_distance
                - min_nonzero
            )
        )

    closeness = min(
        max(
            float(
                closeness
            ),
            0.0,
        ),
        1.0,
    )

    alpha = (
        0.10
        + 0.42
        * closeness
    )

    return (
        f"rgba(27,75,90,{alpha:.3f})"
    )

def product_sort(values: Sequence[object]) -> List[str]:
    """Sort product labels cleanly."""
    return sorted(
        (str(v) for v in values),
        key=lambda x: (len(x), x),
    )

def jar_group(v: int) -> str:
    """Apply the approved SOP-07 grouped JAR interpretation."""
    if v in (1, 2):
        return "Too Little"

    if v == 3:
        return "JAR"

    if v in (4, 5):
        return "Too Much"

    raise ValueError(f"Invalid JAR value: {v}")

def load_data(path: Path) -> Tuple[pd.DataFrame, List[str]]:
    """
    Load the active dataset.

    Only an obvious sequential CSV-export index is removed.
    That action is disclosed in the dashboard.
    """
    df = pd.read_csv(path)

    notes: List[str] = []

    index_candidates = [
        x
        for x in df.columns
        if str(x).lower().startswith("unnamed:")
    ]

    for c in index_candidates:
        s = pd.to_numeric(
            df[c],
            errors="coerce",
        )

        zero_based = pd.Series(
            np.arange(len(df)),
            index=df.index,
            dtype=float,
        )

        one_based = pd.Series(
            np.arange(1, len(df) + 1),
            index=df.index,
            dtype=float,
        )

        if (
            s.notna().all()
            and (
                s.astype(float).equals(zero_based)
                or s.astype(float).equals(one_based)
            )
        ):
            df = df.drop(columns=c)

            notes.append(
                f"Removed obvious exported row-index column '{c}'."
            )

    return df, notes

def analysis_scope() -> Tuple[List[str], List[str]]:
    """
    Resolve the requested analytical scope.

    broad_screen:
        all six approved JAR variables

    targeted:
        only TARGET_ATTRIBUTES and optionally TARGET_PRODUCTS
    """
    mode = ANALYSIS_MODE.strip().lower()

    if mode not in {
        "broad_screen",
        "targeted",
    }:
        raise ValueError(
            "ANALYSIS_MODE must be 'broad_screen' or 'targeted'."
        )

    if mode == "broad_screen":
        return list(APPROVED_JAR), []

    if not TARGET_ATTRIBUTES:
        raise ValueError(
            "Targeted mode requires TARGET_ATTRIBUTES."
        )

    bad = [
        x
        for x in TARGET_ATTRIBUTES
        if x not in APPROVED_JAR
    ]

    if bad:
        raise ValueError(
            "Non-approved TARGET_ATTRIBUTES: "
            + ", ".join(bad)
        )

    return (
        list(TARGET_ATTRIBUTES),
        [str(x) for x in TARGET_PRODUCTS],
    )

def capability_gate(
    df: pd.DataFrame,
    requested: Sequence[str],
) -> dict:
    """
    Apply the Dataset Capability Gate before analysis.

    The method library does not establish that the active
    dataset actually contains the required JAR variables.
    """
    fatal: List[str] = []

    for col in (
        PRODUCT,
        PANELIST,
    ):
        if col not in df.columns:
            fatal.append(
                f"Required identifier '{col}' is absent."
            )

    available = [
        x
        for x in requested
        if x in df.columns
    ]

    missing = [
        x
        for x in requested
        if x not in df.columns
    ]

    if not available:
        fatal.append(
            "None of the requested approved JAR variables are present."
        )

    if fatal:
        status = "UNSUPPORTED"

    elif missing:
        status = "PARTIALLY SUPPORTED"

    else:
        status = "SUPPORTED"

    return {
        "status": status,
        "fatal": fatal,
        "available": available,
        "missing": missing,
        "notes": [],
    }

def validate_design(
    df: pd.DataFrame,
) -> dict:
    """
    Validate the panelist × product evaluation structure.

    SOP-07 itself is descriptive, but duplicate evaluations
    would distort the response distributions. Therefore they
    are treated as a structural hard stop.
    """
    out = {
        "fatal": [],
        "warnings": [],
        "rows": len(df),
        "panelists": 0,
        "products": 0,
        "duplicates": 0,
        "pp_min": 0,
        "pp_max": 0,
        "pr_min": 0,
        "pr_max": 0,
    }

    if (
        PRODUCT not in df.columns
        or PANELIST not in df.columns
    ):
        out["fatal"].append(
            "Required design identifiers are unavailable."
        )

        return out

    if df[PRODUCT].isna().any():
        out["fatal"].append(
            f"{int(df[PRODUCT].isna().sum())} row(s) "
            f"have missing {PRODUCT}."
        )

    if df[PANELIST].isna().any():
        out["fatal"].append(
            f"{int(df[PANELIST].isna().sum())} row(s) "
            f"have missing {PANELIST}."
        )

    duplicate_mask = df.duplicated(
        [PANELIST, PRODUCT],
        keep=False,
    )

    out["duplicates"] = int(
        duplicate_mask.sum()
    )

    if out["duplicates"]:
        pairs = int(
            df.loc[
                duplicate_mask,
                [PANELIST, PRODUCT],
            ]
            .drop_duplicates()
            .shape[0]
        )

        out["fatal"].append(
            f"Found {out['duplicates']} duplicate-row instances "
            f"across {pairs} panelist × product pair(s). "
            "SOP-07 will not silently average or double-count them."
        )

    d = df.dropna(
        subset=[
            PANELIST,
            PRODUCT,
        ]
    )

    out["panelists"] = int(
        d[PANELIST].nunique()
    )

    out["products"] = int(
        d[PRODUCT].nunique()
    )

    panelist_counts = (
        d.groupby(PANELIST)[PRODUCT]
        .nunique()
    )

    product_counts = (
        d.groupby(PRODUCT)[PANELIST]
        .nunique()
    )

    if not panelist_counts.empty:
        out["pp_min"] = int(
            panelist_counts.min()
        )

        out["pp_max"] = int(
            panelist_counts.max()
        )

    if not product_counts.empty:
        out["pr_min"] = int(
            product_counts.min()
        )

        out["pr_max"] = int(
            product_counts.max()
        )

    if out["rows"] != EXPECTED_ROWS:
        out["warnings"].append(
            f"Observed {out['rows']} rows; "
            f"reference snapshot has {EXPECTED_ROWS}."
        )

    if out["panelists"] != EXPECTED_PANELISTS:
        out["warnings"].append(
            f"Observed {out['panelists']} panelists; "
            f"reference snapshot has {EXPECTED_PANELISTS}."
        )

    if out["products"] != EXPECTED_PRODUCTS:
        out["warnings"].append(
            f"Observed {out['products']} products; "
            f"reference snapshot has {EXPECTED_PRODUCTS}."
        )

    if out["pp_min"] != out["pp_max"]:
        out["warnings"].append(
            "Incomplete panelist product coverage: "
            f"{out['pp_min']} to {out['pp_max']} "
            "products per panelist."
        )

    if out["pr_min"] != out["pr_max"]:
        out["warnings"].append(
            "Unequal product N: "
            f"{out['pr_min']} to {out['pr_max']} "
            "unique panelists per product."
        )

    return out

def validate_jar(
    df: pd.DataFrame,
    variable: str,
) -> dict:
    """
    Validate one approved JAR variable.

    Missing values are allowed and disclosed.

    Non-missing values that cannot be interpreted as
    integer 1–5 JAR values suppress that attribute.
    """
    raw = df[variable]

    nonmissing = raw[
        raw.notna()
    ]

    missing = int(
        raw.isna().sum()
    )

    if nonmissing.empty:
        return {
            "variable": variable,
            "label": APPROVED_JAR[variable],
            "status": "HARD STOP",
            "nonmissing": 0,
            "missing": missing,
            "invalid": 0,
            "range": "—",
            "reason": (
                "No non-missing JAR responses."
            ),
        }

    num = pd.to_numeric(
        nonmissing,
        errors="coerce",
    )

    numeric_ok = num.notna()

    integer_ok = pd.Series(
        False,
        index=num.index,
    )

    integer_ok.loc[numeric_ok] = np.isclose(
        num.loc[numeric_ok],
        np.round(
            num.loc[numeric_ok]
        ),
    )

    range_ok = (
        num.between(
            1,
            5,
            inclusive="both",
        )
        .fillna(False)
    )

    valid = (
        numeric_ok
        & integer_ok
        & range_ok
    )

    invalid = int(
        (~valid).sum()
    )

    finite = num[
        num.notna()
    ]

    observed_range = (
        f"{finite.min():g} to {finite.max():g}"
        if not finite.empty
        else "—"
    )

    if invalid:
        return {
            "variable": variable,
            "label": APPROVED_JAR[variable],
            "status": "HARD STOP",
            "nonmissing": int(
                len(nonmissing)
            ),
            "missing": missing,
            "invalid": invalid,
            "range": observed_range,
            "reason": (
                f"{invalid} non-missing response(s) are not "
                "valid integer values 1–5; attribute suppressed."
            ),
        }

    return {
        "variable": variable,
        "label": APPROVED_JAR[variable],
        "status": (
            "PASS"
            if missing == 0
            else "PASS WITH MISSINGNESS"
        ),
        "nonmissing": int(
            len(nonmissing)
        ),
        "missing": missing,
        "invalid": 0,
        "range": observed_range,
        "reason": (
            "Valid 1–5 JAR scale; "
            "valid N is used as denominator."
        ),
    }

def resolve_products(
    df: pd.DataFrame,
    requested: Sequence[str],
) -> Tuple[List[str], List[str]]:
    """Resolve observed products for broad or targeted analysis."""
    observed = product_sort(
        df[PRODUCT]
        .dropna()
        .unique()
    )

    if (
        ANALYSIS_MODE.strip().lower()
        == "broad_screen"
        or not requested
    ):
        return observed, []

    observed_set = set(
        observed
    )

    found = [
        x
        for x in requested
        if x in observed_set
    ]

    missing = [
        x
        for x in requested
        if x not in observed_set
    ]

    return found, missing

def build_long(
    df: pd.DataFrame,
    products: Sequence[str],
    attributes: Sequence[str],
) -> pd.DataFrame:
    """
    Create a separate long-form JAR analysis table.

    The original 1–5 JAR columns in df are left untouched.
    """
    d = df[
        df[PRODUCT]
        .astype(str)
        .isin(products)
    ]

    rows: List[dict] = []

    for variable in attributes:

        for idx, raw in d[variable].items():

            if pd.isna(raw):
                continue

            value = int(
                float(raw)
            )

            rows.append(
                {
                    "source_row": idx,
                    PANELIST: d.at[
                        idx,
                        PANELIST,
                    ],
                    PRODUCT: str(
                        d.at[
                            idx,
                            PRODUCT,
                        ]
                    ),
                    "attribute": variable,
                    "attribute_label": (
                        APPROVED_JAR[
                            variable
                        ]
                    ),
                    "raw_jar": value,
                    "jar_group": (
                        jar_group(value)
                    ),
                }
            )

    return pd.DataFrame(
        rows
    )

def analyze(
    df: pd.DataFrame,
    long_df: pd.DataFrame,
    products: Sequence[str],
    attributes: Sequence[str],
) -> pd.DataFrame:
    """
    Calculate Too Little / JAR / Too Much distributions
    for every requested product × attribute cell.
    """
    rows: List[dict] = []

    for variable in attributes:

        for product in products:

            product_rows = df[
                df[PRODUCT]
                .astype(str)
                == str(product)
            ]

            n_product = int(
                len(product_rows)
            )

            cell = long_df[
                (
                    long_df[PRODUCT]
                    == str(product)
                )
                & (
                    long_df["attribute"]
                    == variable
                )
            ]

            n_valid = int(
                len(cell)
            )

            grouped_counts = (
                cell["jar_group"]
                .value_counts()
                .to_dict()
                if n_valid
                else {}
            )

            raw_counts = (
                cell["raw_jar"]
                .value_counts()
                .to_dict()
                if n_valid
                else {}
            )

            row = {
                PRODUCT: str(product),
                "attribute": variable,
                "attribute_label": (
                    APPROVED_JAR[
                        variable
                    ]
                ),
                "n_product": n_product,
                "n_valid": n_valid,
                "n_missing": (
                    n_product
                    - n_valid
                ),
                "n_too_little": int(
                    grouped_counts.get(
                        "Too Little",
                        0,
                    )
                ),
                "n_jar": int(
                    grouped_counts.get(
                        "JAR",
                        0,
                    )
                ),
                "n_too_much": int(
                    grouped_counts.get(
                        "Too Much",
                        0,
                    )
                ),
            }

            for i in range(
                1,
                6,
            ):
                row[f"n_raw_{i}"] = int(
                    raw_counts.get(
                        i,
                        0,
                    )
                )

            for key in (
                "too_little",
                "jar",
                "too_much",
            ):
                row[f"pct_{key}"] = (
                    100
                    * row[f"n_{key}"]
                    / n_valid
                    if n_valid
                    else np.nan
                )

            rows.append(
                row
            )

    return pd.DataFrame(
        rows
    )

def interpret(
    summary: pd.DataFrame,
) -> Tuple[str, dict]:
    """Create the concise analytical answer shown at the top."""
    usable = summary[
        summary["n_valid"] > 0
    ]

    if usable.empty:
        return (
            "No valid JAR cells were available for interpretation.",
            {},
        )

    best = usable.loc[
        usable["pct_jar"]
        .idxmax()
    ]

    too_little = usable.loc[
        usable["pct_too_little"]
        .idxmax()
    ]

    too_much = usable.loc[
        usable["pct_too_much"]
        .idxmax()
    ]

    if (
        float(
            too_little[
                "pct_too_little"
            ]
        )
        >= float(
            too_much[
                "pct_too_much"
            ]
        )
    ):
        deviation = too_little

        direction = (
            "Too Little"
        )

        value = float(
            too_little[
                "pct_too_little"
            ]
        )

    else:
        deviation = too_much

        direction = (
            "Too Much"
        )

        value = float(
            too_much[
                "pct_too_much"
            ]
        )

    answer = (
        f"The clearest measured JAR imbalance is "
        f"Product {deviation[PRODUCT]} on "
        f"{deviation['attribute_label']}: "
        f"{value:.1f}% of valid responses are "
        f"{direction}. "
        f"The highest observed JAR incidence is "
        f"Product {best[PRODUCT]} on "
        f"{best['attribute_label']} at "
        f"{float(best['pct_jar']):.1f}%. "
        "This describes direction relative to consumers' ideal level; "
        "it does not establish a liking penalty."
    )

    return (
        answer,
        {
            "best": best,
            "tl": too_little,
            "tm": too_much,
        },
    )

def segment(
    row: pd.Series,
    key: str,
    label: str,
    css_class: str,
) -> str:
    """Create one section of the 100% stacked JAR bar."""
    value = float(
        row[f"pct_{key}"]
    )

    if not np.isfinite(
        value
    ):
        return ""

    text = (
        f"{value:.0f}%"
        if value >= 11
        else ""
    )

    tooltip = (
        f"Product {row[PRODUCT]} — "
        f"{row['attribute_label']}: "
        f"{int(row[f'n_{key}'])} of "
        f"{int(row['n_valid'])} "
        f"({value:.1f}%) = {label}. "
        "Direction only; not liking impact."
    )

    return (
        f'<div class="seg {css_class}" '
        f'style="width:{value:.4f}%" '
        f'title="{esc(tooltip)}">'
        f"<span>{text}</span>"
        "</div>"
    )

def bar(
    row: pd.Series,
) -> str:
    """Create a complete 100% stacked JAR bar."""
    if int(
        row["n_valid"]
    ) == 0:
        return (
            '<div class="empty">'
            "No valid responses"
            "</div>"
        )

    return (
        '<div class="bar">'
        + segment(
            row,
            "too_little",
            "Too Little",
            "tlc",
        )
        + segment(
            row,
            "jar",
            "JAR",
            "jc",
        )
        + segment(
            row,
            "too_much",
            "Too Much",
            "tmc",
        )
        + "</div>"
    )

def distribution_grid(
    summary: pd.DataFrame,
    attributes: Sequence[str],
    products: Sequence[str],
) -> str:
    """Build the primary JAR visual."""
    cards: List[str] = []

    for variable in attributes:

        rows: List[str] = []

        d = summary[
            summary["attribute"]
            == variable
        ]

        for product in products:

            match = d[
                d[PRODUCT]
                == str(product)
            ]

            if match.empty:
                continue

            row = match.iloc[0]

            n_text = (
                f"N={int(row['n_valid'])}"
            )

            if int(
                row["n_missing"]
            ):
                n_text += (
                    f" · "
                    f"{int(row['n_missing'])} "
                    "missing"
                )

            rows.append(
                '<div class="jr">'
                f'<div class="pl">'
                f"Product {esc(product)}"
                "</div>"
                f"<div>{bar(row)}</div>"
                f'<div class="nl">'
                f"{esc(n_text)}"
                "</div>"
                "</div>"
            )

        cards.append(
            '<section class="card">'
            f"<h3>{esc(APPROVED_JAR[variable])}</h3>"
            '<div class="micro">'
            "Percent of valid responses within product."
            "</div>"
            + "".join(rows)
            + "</section>"
        )

    return (
        '<div class="grid">'
        + "".join(cards)
        + "</div>"
    )

def deviation_table(
    summary: pd.DataFrame,
    top_n: int = 10,
) -> str:
    """
    Rank observed Too Little / Too Much incidence descriptively.

    This is not a liking penalty or optimization-priority ranking.
    """
    items = []

    for _, row in summary[
        summary["n_valid"] > 0
    ].iterrows():

        for direction, key in (
            (
                "Too Little",
                "too_little",
            ),
            (
                "Too Much",
                "too_much",
            ),
        ):
            items.append(
                (
                    float(
                        row[
                            f"pct_{key}"
                        ]
                    ),
                    row,
                    direction,
                    key,
                )
            )

    items.sort(
        key=lambda x: x[0],
        reverse=True,
    )

    rows: List[str] = []

    for (
        value,
        row,
        direction,
        key,
    ) in items[:top_n]:

        cls = (
            "btl"
            if direction
            == "Too Little"
            else "btm"
        )

        rows.append(
            "<tr>"
            f"<td>Product "
            f"{esc(row[PRODUCT])}"
            "</td>"
            f"<td>"
            f"{esc(row['attribute_label'])}"
            "</td>"
            f'<td><span class="badge {cls}">'
            f"{direction}"
            "</span></td>"
            f'<td class="num">'
            f"{value:.1f}%"
            "</td>"
            f'<td class="num">'
            f"{int(row[f'n_{key}'])}"
            " / "
            f"{int(row['n_valid'])}"
            "</td>"
            "</tr>"
        )

    return (
        '<div class="scroll">'
        "<table>"
        "<thead>"
        "<tr>"
        "<th>Product</th>"
        "<th>Attribute</th>"
        "<th>Direction</th>"
        "<th>Incidence</th>"
        "<th>N / valid N</th>"
        "</tr>"
        "</thead>"
        "<tbody>"
        + "".join(rows)
        + "</tbody>"
        "</table>"
        "</div>"
    )

def save_exports(
    output_dir: Path,
    summary: pd.DataFrame,
    validations: Sequence[dict],
) -> None:
    """
    Technical exports are disabled by default
    to preserve the smallest-complete-output rule.
    """
    if not SAVE_TECHNICAL_EXPORTS:
        return

    summary.to_csv(
        output_dir
        / "SOP07_jar_distribution_summary.csv",
        index=False,
    )

    pd.DataFrame(
        validations
    ).to_csv(
        output_dir
        / "SOP07_jar_variable_validation.csv",
        index=False,
    )

def console(
    design: dict,
    answer: str,
    path: Path,
    status: str,
    n_attrs: int,
) -> None:
    """Print only the small workshop-ready console summary."""
    print(
        f"{SOP_ID} — "
        f"{SOP_NAME}\n"
    )

    print(
        f"Analyzing "
        f"{design['rows']:,} evaluations "
        f"from {design['panelists']:,} consumers "
        f"across {design['products']:,} products..."
    )

    print(
        f"{n_attrs} measured JAR "
        "attribute(s) eligible.\n"
    )

    print(
        "=" * 72
    )

    print(
        f"ANSWER: {answer}"
    )

    print(
        "MODEL STATUS: "
        "Not applicable — SOP-07 is diagnostic/descriptive."
    )

    print(
        f"CAPABILITY: {status}"
    )

    print(
        "Dashboard:"
    )

    print(
        path.resolve()
    )

    print(
        "=" * 72
    )

def main() -> None:
    """
    Run the complete SOP-07 workflow.
    """
    output_dir = (
        Path.cwd()
        / OUTPUT_DIR
    )

    output_dir.mkdir(
        parents=True,
        exist_ok=True,
    )

    dashboard_path = (
        output_dir
        / DASHBOARD_FILENAME
    )

    try:
        # ---------------------------------------------------------------------
        # Load active data
        # ---------------------------------------------------------------------

        data_path = (
            resolve_data_path()
        )

        df, load_notes = (
            load_data(
                data_path
            )
        )

        # ---------------------------------------------------------------------
        # Determine requested analytical scope
        # ---------------------------------------------------------------------

        requested_attributes, requested_products = (
            analysis_scope()
        )

        # ---------------------------------------------------------------------
        # Dataset Capability Gate
        # ---------------------------------------------------------------------

        capability = (
            capability_gate(
                df,
                requested_attributes,
            )
        )

        # ---------------------------------------------------------------------
        # Study-design validation
        # ---------------------------------------------------------------------

        design = (
            validate_design(
                df
            )
        )

        # ---------------------------------------------------------------------
        # Capability / structural hard stop
        # ---------------------------------------------------------------------

        if (
            capability["fatal"]
            or design["fatal"]
        ):
            reasons = (
                capability["fatal"]
                + design["fatal"]
            )

            page = (
                stop_dashboard(
                    reasons=reasons,
                    capability=capability,
                    design=design,
                    data_path=data_path,
                    notes=(
                        load_notes
                        + design["warnings"]
                    ),
                )
            )

            dashboard_path.write_text(
                page,
                encoding="utf-8",
            )

            answer = (
                "Analysis stopped because required "
                "data or design conditions failed."
            )

            console(
                design=design,
                answer=answer,
                path=dashboard_path,
                status=capability["status"],
                n_attrs=0,
            )

            open_dashboard(
                dashboard_path
            )

            return

        # ---------------------------------------------------------------------
        # Product scope
        # ---------------------------------------------------------------------

        products, missing_products = (
            resolve_products(
                df,
                requested_products,
            )
        )

        if not products:
            reason = (
                "None of the requested products "
                "are present in the dataset."
            )

            page = (
                stop_dashboard(
                    reasons=[reason],
                    capability=capability,
                    design=design,
                    data_path=data_path,
                    notes=load_notes,
                )
            )

            dashboard_path.write_text(
                page,
                encoding="utf-8",
            )

            console(
                design=design,
                answer=reason,
                path=dashboard_path,
                status=capability["status"],
                n_attrs=0,
            )

            open_dashboard(
                dashboard_path
            )

            return

        # ---------------------------------------------------------------------
        # JAR scale validation
        # ---------------------------------------------------------------------

        validations = [
            validate_jar(
                df,
                variable,
            )
            for variable
            in capability["available"]
        ]

        valid_attributes = [
            v["variable"]
            for v in validations
            if str(
                v["status"]
            ).startswith("PASS")
        ]

        if not valid_attributes:
            reason = (
                "No requested JAR attribute passed "
                "the required 1–5 scale validation."
            )

            page = (
                stop_dashboard(
                    reasons=[reason],
                    capability=capability,
                    design=design,
                    data_path=data_path,
                    notes=(
                        load_notes
                        + [
                            v["reason"]
                            for v in validations
                        ]
                    ),
                )
            )

            dashboard_path.write_text(
                page,
                encoding="utf-8",
            )

            console(
                design=design,
                answer=reason,
                path=dashboard_path,
                status=capability["status"],
                n_attrs=0,
            )

            open_dashboard(
                dashboard_path
            )

            return

        # ---------------------------------------------------------------------
        # Partial support if one separable JAR item is unusable
        # ---------------------------------------------------------------------

        if (
            len(valid_attributes)
            < len(
                capability["available"]
            )
        ):
            capability[
                "status"
            ] = "PARTIALLY SUPPORTED"

            capability[
                "notes"
            ].append(
                "One or more requested JAR attributes "
                "failed scale validation and were suppressed."
            )

        # ---------------------------------------------------------------------
        # Required preprocessing
        # ---------------------------------------------------------------------

        long_df = (
            build_long(
                df=df,
                products=products,
                attributes=valid_attributes,
            )
        )

        # ---------------------------------------------------------------------
        # Primary JAR distribution analysis
        # ---------------------------------------------------------------------

        summary = (
            analyze(
                df=df,
                long_df=long_df,
                products=products,
                attributes=valid_attributes,
            )
        )

        # ---------------------------------------------------------------------
        # Build dashboard
        # ---------------------------------------------------------------------

        page, answer = (
            dashboard(
                summary=summary,
                validations=validations,
                capability=capability,
                design=design,
                products=products,
                attributes=valid_attributes,
                missing_products=missing_products,
                notes=(
                    load_notes
                    + capability["notes"]
                ),
                data_path=data_path,
            )
        )

        dashboard_path.write_text(
            page,
            encoding="utf-8",
        )

        # ---------------------------------------------------------------------
        # Optional technical exports
        # ---------------------------------------------------------------------

        save_exports(
            output_dir=output_dir,
            summary=summary,
            validations=validations,
        )

        # ---------------------------------------------------------------------
        # Small console summary
        # ---------------------------------------------------------------------

        console(
            design=design,
            answer=answer,
            path=dashboard_path,
            status=capability["status"],
            n_attrs=len(
                valid_attributes
            ),
        )

        # ---------------------------------------------------------------------
        # Open dashboard
        # ---------------------------------------------------------------------

        open_dashboard(
            dashboard_path
        )

    except Exception as exc:
        print(
            f"{SOP_ID} — "
            f"{SOP_NAME}"
        )

        print(
            f"ERROR: "
            f"{type(exc).__name__}: "
            f"{exc}"
        )

        raise

if __name__ == "__main__":
    main()
