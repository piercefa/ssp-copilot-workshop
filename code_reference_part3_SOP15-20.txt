def normalize_comment(value: object) -> Optional[str]:
    """
    Clean usable text while preserving missingness.

    Missing comments remain missing.
    They are NOT interpreted as "no complaint".
    """
    if pd.isna(value):
        return None

    text = re.sub(
        r"\s+",
        " ",
        str(value),
    ).strip()

    if not text:
        return None

    if text.lower() in {
        "nan",
        "na",
        "n/a",
        "null",
        "missing",
    }:
        return None

    return text

def remove_harmless_index_column(
    df: pd.DataFrame,
) -> pd.DataFrame:
    """
    Remove an exported 0..N-1 CSV index only when it is
    obviously a saved index.
    """
    out = df.copy()

    for col in list(out.columns):

        name = str(col).strip().lower()

        if name not in {
            "unnamed: 0",
            "index",
        }:
            continue

        numeric = pd.to_numeric(
            out[col],
            errors="coerce",
        )

        expected = np.arange(
            len(out)
        )

        if (
            numeric.notna().all()
            and np.array_equal(
                numeric.to_numpy(),
                expected,
            )
        ):
            out = out.drop(
                columns=[col]
            )

    return out

def rule_matches(
    text: Optional[str],
    rule: ThemeRule,
) -> bool:
    """
    Apply one frozen coding rule.

    This is transparent exact/phrase-rule coding.
    """
    if text is None:
        return False

    lowered = text.lower()

    return any(
        re.search(
            pattern,
            lowered,
            flags=re.IGNORECASE,
        )
        for pattern in rule.patterns
    )

class CapabilityResult:
    status: str
    available_comment_fields: List[str]
    missing_fields: List[str]
    notes: List[str]

def dataset_capability_gate(
    df: pd.DataFrame,
) -> CapabilityResult:
    """
    Determine whether SOP-15 can run.

    Required:
        product_code
        at least one of:
            likes
            dislikes

    Preferred:
        both likes and dislikes

    Optional:
        panelist_id
        overall_liking
    """

    available = [
        col
        for col in (
            LIKES_COL,
            DISLIKES_COL,
        )
        if col in df.columns
    ]

    missing: List[str] = []
    notes: List[str] = []

    if PRODUCT_COL not in df.columns:
        missing.append(
            PRODUCT_COL
        )

    if not available:
        missing.extend(
            [
                LIKES_COL,
                DISLIKES_COL,
            ]
        )

    if (
        PRODUCT_COL not in df.columns
        or not available
    ):

        status = "UNSUPPORTED"

    elif len(available) == 1:

        status = "PARTIALLY SUPPORTED"

        if available[0] == LIKES_COL:
            missing.append(
                DISLIKES_COL
            )
        else:
            missing.append(
                LIKES_COL
            )

        notes.append(
            "Only the available open-ended field "
            "can be coded and interpreted."
        )

    else:

        status = "SUPPORTED"

    if LIKING_COL not in df.columns:

        notes.append(
            "overall_liking is absent; optional "
            "descriptive triangulation will be skipped."
        )

    return CapabilityResult(
        status=status,
        available_comment_fields=available,
        missing_fields=sorted(
            set(missing)
        ),
        notes=notes,
    )

def validate_study_design(
    df: pd.DataFrame,
    capability: CapabilityResult,
) -> List[str]:
    """
    Validate the study structure relevant to SOP-15.

    SOP-15 theme frequencies are descriptive at the
    comment level. We therefore recognize the
    repeated-measures design without pretending rows
    are independent consumers for inference.
    """

    notes: List[str] = [
        f"Rows loaded: {len(df)}."
    ]

    if PRODUCT_COL in df.columns:

        n_products = int(
            df[PRODUCT_COL]
            .nunique(
                dropna=True
            )
        )

        notes.append(
            f"Products observed: {n_products}."
        )

    if PANELIST_COL in df.columns:

        n_panelists = int(
            df[PANELIST_COL]
            .nunique(
                dropna=True
            )
        )

        notes.append(
            f"Consumers observed: {n_panelists}."
        )

    if (
        PANELIST_COL in df.columns
        and PRODUCT_COL in df.columns
    ):

        valid_keys = df.dropna(
            subset=[
                PANELIST_COL,
                PRODUCT_COL,
            ]
        )

        duplicate_rows = int(
            valid_keys.duplicated(
                [
                    PANELIST_COL,
                    PRODUCT_COL,
                ],
                keep=False,
            ).sum()
        )

        if duplicate_rows:

            notes.append(
                "WARNING: "
                f"{duplicate_rows} row(s) participate "
                "in duplicated consumer × product cells."
            )

        else:

            notes.append(
                "No duplicated consumer × product "
                "evaluation rows detected."
            )

        if not valid_keys.empty:

            products_per_consumer = (
                valid_keys
                .groupby(
                    PANELIST_COL
                )[PRODUCT_COL]
                .nunique()
            )

            if (
                products_per_consumer.max()
                > 1
            ):

                notes.append(
                    "Repeated-measures structure confirmed. "
                    "SOP-15 theme prevalence remains descriptive "
                    "at the comment level; no independence-based "
                    "significance test is added."
                )

    for field_name in (
        capability.available_comment_fields
    ):

        cleaned = df[field_name].map(
            normalize_comment
        )

        valid = int(
            cleaned.notna().sum()
        )

        missing = len(df) - valid

        notes.append(
            f"{field_name}: "
            f"{valid} valid comments and "
            f"{missing} missing responses."
        )

    return notes

def preprocess_comments(
    df: pd.DataFrame,
    comment_fields: Sequence[str],
) -> pd.DataFrame:
    """
    Add cleaned comment columns without overwriting
    the original verbatim text.
    """

    out = df.copy()

    for field_name in comment_fields:

        clean_col = (
            f"__{field_name}_clean"
        )

        out[clean_col] = (
            out[field_name]
            .map(
                normalize_comment
            )
        )

    return out

def corpus_supported_codebook(
    df: pd.DataFrame,
    comment_fields: Sequence[str],
) -> Tuple[
    List[ThemeRule],
    List[ThemeRule],
]:
    """
    Inspect the full in-scope corpus BEFORE producing
    product-specific results.

    Candidate themes that occur at least once are
    activated.

    Themes not observed in the current corpus are
    retained as inactive rather than displayed as
    data-supported findings.
    """

    corpus: List[str] = []

    for field_name in comment_fields:

        clean_col = (
            f"__{field_name}_clean"
        )

        corpus.extend(
            [
                text
                for text
                in df[clean_col].tolist()
                if text is not None
            ]
        )

    active: List[ThemeRule] = []
    inactive: List[ThemeRule] = []

    for rule in CODEBOOK:

        supported = any(
            rule_matches(
                text,
                rule,
            )
            for text in corpus
        )

        if supported:
            active.append(rule)
        else:
            inactive.append(rule)

    return active, inactive

def apply_codebook(
    df: pd.DataFrame,
    comment_fields: Sequence[str],
    codebook: Sequence[ThemeRule],
) -> pd.DataFrame:
    """
    Apply identical frozen coding rules to every
    product and every valid comment.

    Multiple themes per comment are allowed.
    """

    out = df.copy()

    for field_name in comment_fields:

        clean_col = (
            f"__{field_name}_clean"
        )

        theme_cols: List[str] = []

        for rule in codebook:

            theme_col = (
                f"theme__{field_name}"
                f"__{rule.theme}"
            )

            out[theme_col] = (
                out[clean_col]
                .map(
                    lambda value,
                    current_rule=rule:
                    rule_matches(
                        value,
                        current_rule,
                    )
                )
            )

            theme_cols.append(
                theme_col
            )

        coded_col = (
            f"__{field_name}_coded"
        )

        if theme_cols:

            out[coded_col] = (
                out[clean_col].notna()
                & out[theme_cols].any(
                    axis=1
                )
            )

        else:

            out[coded_col] = False

    return out

class CoverageResult:
    field_name: str
    expected: int
    valid: int
    missing: int
    coded: int
    uncoded: int
    coverage_rate: float
    uncoded_rate: float
    status: str

def calculate_coverage(
    df: pd.DataFrame,
    field_name: str,
) -> CoverageResult:
    """
    Report coding coverage separately for each
    comment field.

    SOP-15 warning:
        uncoded rate > 10%
    """

    clean_col = (
        f"__{field_name}_clean"
    )

    coded_col = (
        f"__{field_name}_coded"
    )

    valid_mask = (
        df[clean_col].notna()
    )

    valid = int(
        valid_mask.sum()
    )

    expected = len(df)

    missing = (
        expected - valid
    )

    coded = int(
        (
            valid_mask
            & df[coded_col]
        ).sum()
    )

    uncoded = (
        valid - coded
    )

    if valid > 0:

        coverage_rate = (
            coded / valid
        )

        uncoded_rate = (
            uncoded / valid
        )

    else:

        coverage_rate = float(
            "nan"
        )

        uncoded_rate = float(
            "nan"
        )

    if valid == 0:

        status = "HARD STOP"

    elif (
        uncoded_rate
        > UNCODED_WARNING_THRESHOLD
    ):

        status = "WARNING"

    else:

        status = "PASS"

    return CoverageResult(
        field_name=field_name,
        expected=expected,
        valid=valid,
        missing=missing,
        coded=coded,
        uncoded=uncoded,
        coverage_rate=coverage_rate,
        uncoded_rate=uncoded_rate,
        status=status,
    )

def summarize_themes(
    df: pd.DataFrame,
    comment_fields: Sequence[str],
    codebook: Sequence[ThemeRule],
) -> Tuple[
    pd.DataFrame,
    pd.DataFrame,
]:
    """
    Produce:

    1. overall theme prevalence
    2. product × theme prevalence

    Denominator:
        valid nonmissing comments within the
        corresponding comment field.
    """

    overall_rows: List[
        Dict[str, object]
    ] = []

    product_rows: List[
        Dict[str, object]
    ] = []

    products = sorted(
        df[PRODUCT_COL]
        .dropna()
        .astype(str)
        .unique()
        .tolist()
    )

    for field_name in comment_fields:

        clean_col = (
            f"__{field_name}_clean"
        )

        valid_mask = (
            df[clean_col].notna()
        )

        valid_total = int(
            valid_mask.sum()
        )

        for rule in codebook:

            theme_col = (
                f"theme__{field_name}"
                f"__{rule.theme}"
            )

            mentions = int(
                (
                    valid_mask
                    & df[theme_col]
                ).sum()
            )

            if valid_total > 0:

                prevalence = (
                    mentions
                    / valid_total
                )

            else:

                prevalence = float(
                    "nan"
                )

            overall_rows.append(
                {
                    "field":
                        field_name,

                    "theme":
                        rule.theme,

                    "definition":
                        rule.definition,

                    "mentions":
                        mentions,

                    "valid_comments":
                        valid_total,

                    "prevalence":
                        prevalence,
                }
            )

            for product in products:

                product_mask = (
                    df[PRODUCT_COL]
                    .astype(str)
                    .eq(product)
                )

                product_valid = (
                    product_mask
                    & valid_mask
                )

                product_n = int(
                    product_valid.sum()
                )

                product_mentions = int(
                    (
                        product_valid
                        & df[theme_col]
                    ).sum()
                )

                if product_n > 0:

                    product_prevalence = (
                        product_mentions
                        / product_n
                    )

                else:

                    product_prevalence = (
                        float("nan")
                    )

                product_rows.append(
                    {
                        "field":
                            field_name,

                        "product":
                            product,

                        "theme":
                            rule.theme,

                        "mentions":
                            product_mentions,

                        "valid_comments":
                            product_n,

                        "prevalence":
                            product_prevalence,
                    }
                )

    overall = pd.DataFrame(
        overall_rows
    )

    product = pd.DataFrame(
        product_rows
    )

    if not overall.empty:

        overall = (
            overall
            .sort_values(
                [
                    "field",
                    "prevalence",
                    "mentions",
                    "theme",
                ],
                ascending=[
                    True,
                    False,
                    False,
                    True,
                ],
            )
            .reset_index(
                drop=True
            )
        )

    return overall, product

def recurring_uncoded_terms(
    comments: Sequence[str],
) -> List[
    Tuple[str, int]
]:
    """
    Surface recurring words among uncoded comments.

    This is a review aid ONLY.

    Terms found here are NOT automatically added
    to the codebook.
    """

    counter: Counter = Counter()

    for comment in comments:

        tokens = re.findall(
            r"[a-z][a-z'-]{2,}",
            comment.lower(),
        )

        tokens = [
            token
            for token in tokens
            if token not in STOPWORDS
        ]

        # Count each word once per comment,
        # approximating comment prevalence.
        counter.update(
            set(tokens)
        )

    return [
        (
            term,
            count,
        )
        for term, count
        in counter.most_common(
            MAX_RECURRING_UNCODED_TERMS
        )
        if count >= 2
    ]

def review_uncoded(
    df: pd.DataFrame,
    field_name: str,
) -> Dict[str, object]:
    """
    Identify valid comments that received no code.
    """

    clean_col = (
        f"__{field_name}_clean"
    )

    coded_col = (
        f"__{field_name}_coded"
    )

    mask = (
        df[clean_col].notna()
        & ~df[coded_col]
    )

    cols = [
        PRODUCT_COL,
        clean_col,
    ]

    if PANELIST_COL in df.columns:

        cols.insert(
            0,
            PANELIST_COL,
        )

    uncoded = (
        df.loc[
            mask,
            cols,
        ]
        .copy()
    )

    comments = (
        uncoded[clean_col]
        .astype(str)
        .tolist()
    )

    examples: List[
        Dict[str, str]
    ] = []

    for _, row in (
        uncoded
        .head(
            MAX_UNCODED_EXAMPLES
        )
        .iterrows()
    ):

        examples.append(
            {
                "product":
                    str(
                        row[
                            PRODUCT_COL
                        ]
                    ),

                "comment":
                    str(
                        row[
                            clean_col
                        ]
                    ),
            }
        )

    return {
        "count":
            len(uncoded),

        "terms":
            recurring_uncoded_terms(
                comments
            ),

        "examples":
            examples,
    }

def liking_by_theme_presence(
    df: pd.DataFrame,
    comment_fields: Sequence[str],
    codebook: Sequence[ThemeRule],
) -> pd.DataFrame:
    """
    Optional descriptive triangulation.

    This DOES NOT estimate a causal effect.

    No significance test is added because SOP-15 is
    structured qualitative / diagnostic-exploratory
    and the repeated-measures observations should not
    be treated as independent observations.
    """

    if LIKING_COL not in df.columns:

        return pd.DataFrame()

    liking = pd.to_numeric(
        df[LIKING_COL],
        errors="coerce",
    )

    rows: List[
        Dict[str, object]
    ] = []

    for field_name in comment_fields:

        valid_comment = (
            df[
                f"__{field_name}_clean"
            ]
            .notna()
        )

        for rule in codebook:

            theme_col = (
                f"theme__{field_name}"
                f"__{rule.theme}"
            )

            present = (
                valid_comment
                & df[theme_col]
                & liking.notna()
            )

            absent = (
                valid_comment
                & ~df[theme_col]
                & liking.notna()
            )

            n_present = int(
                present.sum()
            )

            n_absent = int(
                absent.sum()
            )

            if n_present:

                mean_present = float(
                    liking[
                        present
                    ].mean()
                )

            else:

                mean_present = float(
                    "nan"
                )

            if n_absent:

                mean_absent = float(
                    liking[
                        absent
                    ].mean()
                )

            else:

                mean_absent = float(
                    "nan"
                )

            if (
                np.isfinite(
                    mean_present
                )
                and np.isfinite(
                    mean_absent
                )
            ):

                difference = (
                    mean_present
                    - mean_absent
                )

            else:

                difference = float(
                    "nan"
                )

            rows.append(
                {
                    "field":
                        field_name,

                    "theme":
                        rule.theme,

                    "n_present":
                        n_present,

                    "n_absent":
                        n_absent,

                    "mean_liking_present":
                        mean_present,

                    "mean_liking_absent":
                        mean_absent,

                    "raw_difference":
                        difference,
                }
            )

    return pd.DataFrame(
        rows
    )

class AnalysisResult:
    capability: CapabilityResult
    structural_notes: List[str]
    active_codebook: List[ThemeRule]
    inactive_codebook: List[ThemeRule]
    coded_df: pd.DataFrame
    coverage: Dict[
        str,
        CoverageResult,
    ]
    overall_summary: pd.DataFrame
    product_summary: pd.DataFrame
    uncoded_review: Dict[
        str,
        Dict[str, object],
    ]
    liking_summary: pd.DataFrame
    warnings: List[str]

def run_analysis(
    df: pd.DataFrame,
) -> AnalysisResult:
    """
    Run SOP-15 from capability gate through
    robustness review.
    """

    capability = (
        dataset_capability_gate(
            df
        )
    )

    structural_notes = (
        validate_study_design(
            df,
            capability,
        )
    )

    # -------------------------------------------------------------------------
    # HARD STOP
    # -------------------------------------------------------------------------

    if (
        capability.status
        == "UNSUPPORTED"
    ):

        return AnalysisResult(

            capability=capability,

            structural_notes=
                structural_notes,

            active_codebook=[],

            inactive_codebook=[],

            coded_df=df.copy(),

            coverage={},

            overall_summary=
                pd.DataFrame(),

            product_summary=
                pd.DataFrame(),

            uncoded_review={},

            liking_summary=
                pd.DataFrame(),

            warnings=[
                "Required SOP-15 input fields "
                "are missing; analysis stopped."
            ],
        )

    # -------------------------------------------------------------------------
    # PREPROCESS
    # -------------------------------------------------------------------------

    processed = (
        preprocess_comments(
            df,
            capability
            .available_comment_fields,
        )
    )

    # -------------------------------------------------------------------------
    # INSPECT CORPUS + FREEZE CODEBOOK
    # -------------------------------------------------------------------------

    (
        active_codebook,
        inactive_codebook,
    ) = corpus_supported_codebook(
        processed,
        capability
        .available_comment_fields,
    )

    # -------------------------------------------------------------------------
    # APPLY FROZEN CODEBOOK
    # -------------------------------------------------------------------------

    coded_df = (
        apply_codebook(
            processed,
            capability
            .available_comment_fields,
            active_codebook,
        )
    )

    # -------------------------------------------------------------------------
    # COVERAGE
    # -------------------------------------------------------------------------

    warnings: List[str] = []

    coverage: Dict[
        str,
        CoverageResult,
    ] = {}

    uncoded_review: Dict[
        str,
        Dict[str, object],
    ] = {}

    for field_name in (
        capability
        .available_comment_fields
    ):

        cov = calculate_coverage(
            coded_df,
            field_name,
        )

        coverage[
            field_name
        ] = cov

        uncoded_review[
            field_name
        ] = review_uncoded(
            coded_df,
            field_name,
        )

        if (
            cov.status
            == "HARD STOP"
        ):

            warnings.append(
                f"{field_name}: "
                "no valid comments "
                "were available."
            )

        elif (
            cov.status
            == "WARNING"
        ):

            warnings.append(
                f"{field_name}: "
                f"{cov.uncoded}/"
                f"{cov.valid} valid comments "
                "were uncoded "
                f"({fmt_pct(cov.uncoded_rate)}), "
                "exceeding the 10% "
                "SOP-15 warning threshold."
            )

    # -------------------------------------------------------------------------
    # THEME PREVALENCE
    # -------------------------------------------------------------------------

    (
        overall_summary,
        product_summary,
    ) = summarize_themes(

        coded_df,

        capability
        .available_comment_fields,

        active_codebook,
    )

    # -------------------------------------------------------------------------
    # OPTIONAL LIKING TRIANGULATION
    # -------------------------------------------------------------------------

    liking_summary = (
        liking_by_theme_presence(

            coded_df,

            capability
            .available_comment_fields,

            active_codebook,
        )
    )

    return AnalysisResult(

        capability=
            capability,

        structural_notes=
            structural_notes,

        active_codebook=
            active_codebook,

        inactive_codebook=
            inactive_codebook,

        coded_df=
            coded_df,

        coverage=
            coverage,

        overall_summary=
            overall_summary,

        product_summary=
            product_summary,

        uncoded_review=
            uncoded_review,

        liking_summary=
            liking_summary,

        warnings=
            warnings,
    )

def field_label(
    field_name: str,
) -> str:

    if field_name == LIKES_COL:
        return "Likes"

    if field_name == DISLIKES_COL:
        return "Dislikes"

    return field_name

def top_theme(
    result: AnalysisResult,
    field_name: str,
) -> Optional[pd.Series]:
    """
    Return the highest-prevalence coded theme
    for one comment field.
    """

    if result.overall_summary.empty:
        return None

    subset = (
        result.overall_summary[
            (
                result
                .overall_summary[
                    "field"
                ]
                == field_name
            )
            & (
                result
                .overall_summary[
                    "mentions"
                ]
                > 0
            )
        ]
    )

    if subset.empty:
        return None

    return subset.iloc[0]

def build_answer(
    result: AnalysisResult,
) -> str:
    """
    Build the dashboard's plain-language answer.
    """

    if (
        result.capability.status
        == "UNSUPPORTED"
    ):

        missing = ", ".join(
            result
            .capability
            .missing_fields
        )

        return (
            "SOP-15 cannot run because "
            "required field(s) are missing: "
            f"{missing}."
        )

    pieces: List[str] = []

    if (
        LIKES_COL
        in result
        .capability
        .available_comment_fields
    ):

        row = top_theme(
            result,
            LIKES_COL,
        )

        if row is not None:

            pieces.append(
                "The most frequent coded "
                "like theme was "
                f"{row['theme']} "
                f"({int(row['mentions'])}/"
                f"{int(row['valid_comments'])}; "
                f"{fmt_pct(float(row['prevalence']))})."
            )

    if (
        DISLIKES_COL
        in result
        .capability
        .available_comment_fields
    ):

        row = top_theme(
            result,
            DISLIKES_COL,
        )

        if row is not None:

            pieces.append(
                "The most frequent coded "
                "dislike theme was "
                f"{row['theme']} "
                f"({int(row['mentions'])}/"
                f"{int(row['valid_comments'])}; "
                f"{fmt_pct(float(row['prevalence']))})."
            )

    if not pieces:

        pieces.append(
            "Valid comments were available, "
            "but no frozen codebook theme "
            "was identified."
        )

    if result.warnings:

        pieces.append(
            "At least one comment field has "
            "a coding-coverage warning, so "
            "uncoded language should be reviewed "
            "before treating the codebook "
            "as complete."
        )

    return " ".join(
        pieces
    )

def build_theme_bar_chart(
    summary: pd.DataFrame,
    field_name: str,
) -> str:
    """
    Build an HTML/CSS horizontal bar chart.
    """

    subset = summary[
        (
            summary[
                "field"
            ]
            == field_name
        )
        & (
            summary[
                "mentions"
            ]
            > 0
        )
    ].copy()

    subset = (
        subset
        .sort_values(
            [
                "prevalence",
                "mentions",
            ],
            ascending=[
                False,
                False,
            ],
        )
        .head(
            TOP_THEMES_TO_SHOW
        )
    )

    if subset.empty:

        return (
            '<div class="empty-state">'
            'No coded themes available.'
            '</div>'
        )

    max_prev = max(
        float(
            subset[
                "prevalence"
            ].max()
        ),
        0.01,
    )

    rows: List[str] = []

    for _, row in (
        subset.iterrows()
    ):

        prevalence = float(
            row[
                "prevalence"
            ]
        )

        width_pct = (
            100.0
            * prevalence
            / max_prev
        )

        rows.append(
            '<div class="bar-row">'

            '<div class="bar-label">'
            f'{esc(row["theme"])}'
            '</div>'

            '<div class="bar-track">'
            '<div '
            'class="bar-fill" '
            f'style="width:{width_pct:.1f}%">'
            '</div>'
            '</div>'

            '<div class="bar-value">'
            f'{fmt_pct(prevalence)} '
            '<span>'
            f'({int(row["mentions"])}/'
            f'{int(row["valid_comments"])})'
            '</span>'
            '</div>'

            '</div>'
        )

    return (
        '<div class="bar-chart">'
        + "".join(rows)
        + '</div>'
    )

def heat_class(
    prevalence: float,
) -> str:

    if (
        not np.isfinite(
            prevalence
        )
        or prevalence <= 0
    ):

        return "hm0"

    if prevalence < 0.10:
        return "hm1"

    if prevalence < 0.25:
        return "hm2"

    if prevalence < 0.50:
        return "hm3"

    if prevalence < 0.75:
        return "hm4"

    return "hm5"

def build_product_heatmap(
    product_summary: pd.DataFrame,
    field_name: str,
) -> str:
    """
    Build product × theme prevalence heatmap.
    """

    subset = (
        product_summary[
            product_summary[
                "field"
            ]
            == field_name
        ]
        .copy()
    )

    subset = subset[
        subset[
            "mentions"
        ]
        > 0
    ]

    if subset.empty:

        return (
            '<div class="empty-state">'
            'No product-level theme '
            'results available.'
            '</div>'
        )

    themes = (
        subset
        .groupby(
            "theme"
        )[
            "mentions"
        ]
        .sum()
        .sort_values(
            ascending=False
        )
        .index
        .tolist()
    )

    products = sorted(
        subset[
            "product"
        ]
        .astype(str)
        .unique()
        .tolist()
    )

    lookup: Dict[
        Tuple[str, str],
        pd.Series,
    ] = {}

    for _, row in (
        subset.iterrows()
    ):

        lookup[
            (
                str(
                    row["product"]
                ),
                str(
                    row["theme"]
                ),
            )
        ] = row

    header_cells = [
        '<th>Product</th>'
    ]

    for theme in themes:

        header_cells.append(
            f'<th>{esc(theme)}</th>'
        )

    header = (
        '<tr>'
        + "".join(
            header_cells
        )
        + '</tr>'
    )

    body_rows: List[str] = []

    for product in products:

        cells: List[str] = [
            f'<th>{esc(product)}</th>'
        ]

        for theme in themes:

            row = lookup.get(
                (
                    product,
                    theme,
                )
            )

            if row is None:

                prevalence = 0.0
                mentions = 0
                valid = 0

            else:

                prevalence = float(
                    row[
                        "prevalence"
                    ]
                )

                mentions = int(
                    row[
                        "mentions"
                    ]
                )

                valid = int(
                    row[
                        "valid_comments"
                    ]
                )

            if valid:

                prevalence_text = (
                    fmt_pct(
                        prevalence
                    )
                )

            else:

                prevalence_text = "—"

            title = (
                f"Product {product} · "
                f"{theme}: "
                f"{mentions}/{valid} "
                "valid comments "
                f"({prevalence_text})."
            )

            if (
                valid
                and mentions
            ):

                display = (
                    f"{100 * prevalence:.0f}%"
                )

            else:

                display = ""

            cells.append(
                '<td '
                f'class="{heat_class(prevalence)}" '
                f'title="{esc(title)}">'
                f'{esc(display)}'
                '</td>'
            )

        body_rows.append(
            '<tr>'
            + "".join(cells)
            + '</tr>'
        )

    return (
        '<div class="table-scroll">'

        '<table class="heatmap-table">'

        '<thead>'
        f'{header}'
        '</thead>'

        '<tbody>'
        f'{"".join(body_rows)}'
        '</tbody>'

        '</table>'

        '</div>'
    )

def build_uncoded_review(
    result: AnalysisResult,
    field_name: str,
) -> str:
    """
    Show recurring uncoded language and examples.

    This is review support, not automatic
    codebook modification.
    """

    review = (
        result
        .uncoded_review
        .get(
            field_name,
            {},
        )
    )

    count = int(
        review.get(
            "count",
            0,
        )
    )

    if count == 0:

        return (
            '<div class="review-box">'

            f'<strong>'
            f'{esc(field_label(field_name))}:'
            f'</strong> '

            'no valid comments remained uncoded.'

            '</div>'
        )

    terms = review.get(
        "terms",
        [],
    )

    examples = review.get(
        "examples",
        [],
    )

    chip_parts: List[str] = []

    for (
        term,
        count_value,
    ) in terms:

        chip_parts.append(
            '<span class="chip">'
            f'{esc(term)} · '
            f'{count_value}'
            '</span>'
        )

    chips = "".join(
        chip_parts
    )

    if not chips:

        chips = (
            '<span class="small-note">'
            'No recurring uncoded term '
            'appeared in at least two comments.'
            '</span>'
        )

    example_rows: List[str] = []

    for item in examples:

        example_rows.append(
            '<tr>'

            f'<td>'
            f'{esc(item["product"])}'
            f'</td>'

            f'<td>'
            f'{esc(item["comment"])}'
            f'</td>'

            '</tr>'
        )

    examples_html = ""

    if example_rows:

        examples_html = (
            '<div class="table-scroll">'

            '<table class="data-table">'

            '<thead>'
            '<tr>'
            '<th>Product</th>'
            '<th>Example uncoded comment</th>'
            '</tr>'
            '</thead>'

            '<tbody>'
            f'{"".join(example_rows)}'
            '</tbody>'

            '</table>'

            '</div>'
        )

    return (
        '<div class="review-box">'

        f'<h3>'
        f'{esc(field_label(field_name))}: '
        'uncoded-comment review'
        '</h3>'

        f'<p>'
        f'{count} valid comment(s) '
        'received no frozen code.'
        '</p>'

        '<p class="small-note">'
        'Recurring uncoded language is surfaced '
        'only for review. It is not automatically '
        'converted into a new theme.'
        '</p>'

        f'<div class="chip-row">'
        f'{chips}'
        f'</div>'

        f'{examples_html}'

        '</div>'
    )

def build_liking_table(
    liking_summary: pd.DataFrame,
) -> str:
    """
    Optional descriptive relationship between
    comment theme presence and overall liking.
    """

    if liking_summary.empty:

        return (
            '<div class="empty-state">'
            'overall_liking was unavailable '
            'for optional descriptive triangulation.'
            '</div>'
        )

    show = liking_summary.copy()

    show = show[
        (
            show[
                "n_present"
            ]
            > 0
        )
        & (
            show[
                "n_absent"
            ]
            > 0
        )
        & (
            show[
                "raw_difference"
            ]
            .notna()
        )
    ]

    if show.empty:

        return (
            '<div class="empty-state">'
            'No theme had both present and '
            'absent liking observations.'
            '</div>'
        )

    show[
        "abs_difference"
    ] = (
        show[
            "raw_difference"
        ]
        .abs()
    )

    show = (
        show
        .sort_values(
            "abs_difference",
            ascending=False,
        )
        .head(12)
    )

    rows: List[str] = []

    for _, row in (
        show.iterrows()
    ):

        rows.append(
            '<tr>'

            f'<td>'
            f'{esc(field_label(str(row["field"])))}'
            f'</td>'

            f'<td>'
            f'{esc(row["theme"])}'
            f'</td>'

            f'<td>'
            f'{int(row["n_present"])}'
            f'</td>'

            f'<td>'
            f'{float(row["mean_liking_present"]):.2f}'
            f'</td>'

            f'<td>'
            f'{int(row["n_absent"])}'
            f'</td>'

            f'<td>'
            f'{float(row["mean_liking_absent"]):.2f}'
            f'</td>'

            f'<td>'
            f'{float(row["raw_difference"]):+.2f}'
            f'</td>'

            '</tr>'
        )

    return (
        '<div class="callout">'

        '<strong>'
        'Descriptive association only.'
        '</strong> '

        'These raw liking differences do not '
        'establish that mentioning a theme '
        'caused liking to change.'

        '</div>'

        '<div class="table-scroll">'

        '<table class="data-table">'

        '<thead>'
        '<tr>'
        '<th>Field</th>'
        '<th>Theme</th>'
        '<th>N present</th>'
        '<th>Mean liking present</th>'
        '<th>N absent</th>'
        '<th>Mean liking absent</th>'
        '<th>Raw difference</th>'
        '</tr>'
        '</thead>'

        '<tbody>'
        f'{"".join(rows)}'
        '</tbody>'

        '</table>'

        '</div>'
    )

    Run SOP-15 and create the dashboard.
    """

        resolve_data_path()
    )

    df = pd.read_csv(
        data_path
    )

    df = (
        remove_harmless_index_column(
            df
        )
    )

    # -------------------------------------------------------------------------
    # ANALYSIS
    # -------------------------------------------------------------------------

    result = (
        run_analysis(
            df
        )
    )

    # -------------------------------------------------------------------------
    # OUTPUT DIRECTORY
    # -------------------------------------------------------------------------

    try:

        base_dir = (
            Path(__file__)
            .resolve()
            .parent
        )

    except NameError:

        base_dir = (
            Path.cwd()
        )

    output_dir = (
        base_dir
        / OUTPUT_DIR
    )

    output_dir.mkdir(
        parents=True,
        exist_ok=True,
    )

        output_dir

    # CONSOLE
    # -------------------------------------------------------------------------

    print_console_summary(
        df,
        result,
        dashboard_path,

    # -------------------------------------------------------------------------
    # AUTO OPEN
    # -------------------------------------------------------------------------

    if AUTO_OPEN_DASHBOARD:

        try:

            webbrowser.open(
                dashboard_path.as_uri()
            )

        except Exception as exc:

            print(
                "Dashboard created, but "
                "automatic browser opening failed: "
                f"{exc}"
            )

    main()

    name: str
    observed: str
    threshold: str
    status: str
    action: str

class CandidateResult:
    n_topics: int

    mean_similarity: float
    min_similarity: float

    topic_prevalence: np.ndarray

    passed: bool

    W: np.ndarray
    H: np.ndarray

    terms: np.ndarray

    top_terms: List[List[str]]

    dominant_topic: np.ndarray

class CorpusResult:
    field_name: str
    display_name: str

    status: str

    stop_stage: Optional[str]
    stop_reason: Optional[str]

    n_expected: int
    n_valid: int
    n_missing: int

    retained_terms: int = 0

    informative_docs: int = 0
    informative_prop: float = 0.0

    max_repeat_count: int = 0
    max_repeat_prop: float = 0.0

    diagnostics: List[Diagnostic] = field(default_factory=list)

    candidates: List[CandidateResult] = field(default_factory=list)

    selected: Optional[CandidateResult] = None

    analysis_df: Optional[pd.DataFrame] = None

    product_topic: Optional[pd.DataFrame] = None

    product_topic_counts: Optional[pd.DataFrame] = None

def display_field(
    field_name: str,
) -> str:
    """
    Friendly field name for the dashboard.
    """
    if field_name == "likes":
        return "Liked comments"

    if field_name == "dislikes":
        return "Disliked comments"

    return field_name

def resolve_paths() -> Tuple[Path, Path]:
    """
    Resolve the input CSV and output HTML paths.

    Default project structure:

        project/
        ├── SOP16_exploratory_topic_extraction.py
        └── data/
            └── protein_shake_category_appraisal_mock_redesigned.csv

    An alternative CSV may be supplied as the first command-line argument.
    """

    parser = argparse.ArgumentParser(
        description=f"{SOP_ID} — {SOP_NAME}"
    )

    parser.add_argument(
        "data_path",
        nargs="?",
        help="Optional path to the CSV dataset",
    )

    parser.add_argument(
        "--no-open",
        action="store_true",
        help="Create the HTML dashboard without opening it",
    )

    args = parser.parse_args()

    if args.data_path:

        data_path = (
            Path(args.data_path)
            .expanduser()
            .resolve()
        )

        if not data_path.exists():
            raise FileNotFoundError(
                f"Dataset not found: {data_path}"
            )

    else:

        candidates = [
            Path.cwd() / DATA_FILENAME,
        ]

        try:
            candidates.append(
                Path(__file__).resolve().parent
                / DATA_FILENAME
            )
        except NameError:
            pass

        data_path = next(
            (
                candidate.resolve()
                for candidate in candidates
                if candidate.exists()
            ),
            None,
        )

        if data_path is None:

            searched = "\n".join(
                f"  - {candidate}"
                for candidate in candidates
            )

            raise FileNotFoundError(
                f"Could not find '{DATA_FILENAME}'.\n"
                f"Searched:\n{searched}\n\n"
                "Place the CSV in the project's data folder "
                "or pass a CSV path as the first command-line "
                "argument."
            )

    try:
        base_dir = Path(__file__).resolve().parent
    except NameError:
        base_dir = Path.cwd()

    output_dir = base_dir / OUTPUT_DIR

    output_dir.mkdir(
        parents=True,
        exist_ok=True,
    )

    dashboard_path = (
        output_dir
        / DASHBOARD_FILENAME
    )

    if args.no_open:

        global AUTO_OPEN_DASHBOARD

        AUTO_OPEN_DASHBOARD = False

    return data_path, dashboard_path

def drop_export_index(
    df: pd.DataFrame,
) -> pd.DataFrame:
    """
    Drop a CSV export-index column only when it is demonstrably
    just 0, 1, 2, ..., N-1.

    This protects against accidentally deleting a real variable
    merely because its column name resembles an index.
    """

    out = df.copy()

    for col in list(out.columns):

        if str(col).lower() not in {
            "index",
            "unnamed: 0",
            "unnamed: 0.1",
        }:
            continue

        numeric = pd.to_numeric(
            out[col],
            errors="coerce",
        )

        if numeric.isna().any():
            continue

        expected = np.arange(
            len(out),
            dtype=float,
        )

        if np.array_equal(
            numeric.to_numpy(dtype=float),
            expected,
        ):
            out = out.drop(columns=[col])

    return out

def clean_text(
    value: object,
) -> str:
    """
    Transparent SOP-16 text cleaning.

    Important principles:
    - preserve sensory-relevant vocabulary;
    - do not replace words with model-generated concepts;
    - do not use product metadata;
    - do not interpret missing comments;
    - normalize case and punctuation only.
    """

    if pd.isna(value):
        return ""

    text = unicodedata.normalize(
        "NFKC",
        str(value),
    )

    text = (
        text
        .lower()
        .strip()
    )

    # Remove apostrophe variants.
    text = re.sub(
        r"[\u2018\u2019'`]",
        "",
        text,
    )

    # Keep letters, numbers, whitespace, and hyphens.
    text = re.sub(
        r"[^a-z0-9\s-]",
        " ",
        text,
    )

    # Treat hyphenated constructions as separate words so
    # terms such as "protein off-note" remain discoverable.
    text = text.replace(
        "-",
        " ",
    )

    text = re.sub(
        r"\s+",
        " ",
        text,
    )

    return text.strip()

def normalize_exact_comment(
    value: object,
) -> str:
    """
    Normalization used only to assess exact-comment repetition.
    """
    return clean_text(value)

def capability_gate(
    df: pd.DataFrame,
) -> Tuple[str, List[str]]:
    """
    Determine whether the dataset can support SOP-16.

    SOP-16 requires:
    - a product identifier;
    - a panelist identifier for structural traceability;
    - at least one approved qualitative comment field.

    Missing constructs are never proxied with other variables.
    """

    core = [
        PANELIST,
        PRODUCT,
    ]

    missing_core = [
        col
        for col in core
        if col not in df.columns
    ]

    available_text = [
        col
        for col in TEXT_FIELDS
        if col in df.columns
    ]

    messages: List[str] = []

    if missing_core:

        messages.append(
            "Missing required structural field(s): "
            + ", ".join(missing_core)
        )

        return (
            "UNSUPPORTED",
            messages,
        )

    if not available_text:

        messages.append(
            "Neither approved text field "
            "('likes', 'dislikes') is present."
        )

        return (
            "UNSUPPORTED",
            messages,
        )

    if len(available_text) < len(TEXT_FIELDS):

        missing = [
            col
            for col in TEXT_FIELDS
            if col not in df.columns
        ]

        messages.append(
            "Missing in-scope text field(s): "
            + ", ".join(missing)
        )

        return (
            "PARTIALLY SUPPORTED",
            messages,
        )

    messages.append(
        "Required structural fields and both approved "
        "open-ended text fields are present."
    )

    return (
        "SUPPORTED",
        messages,
    )

def validate_structure(
    df: pd.DataFrame,
) -> List[Diagnostic]:
    """
    Inspect the actual study structure instead of assuming 50 × 8.

    Topic modeling treats each nonmissing comment as a document.
    However, panelist/product structure remains visible for data QC
    and product-level topic summaries.
    """

    diagnostics: List[Diagnostic] = []

    duplicate_count = int(
        df.duplicated(
            [
                PANELIST,
                PRODUCT,
            ]
        ).sum()
    )

    duplicate_status = (
        "PASS"
        if duplicate_count == 0
        else "WARNING"
    )

    diagnostics.append(
        Diagnostic(
            name=(
                "Duplicate panelist × product rows"
            ),
            observed=str(duplicate_count),
            threshold=(
                "0 expected for the workshop design"
            ),
            status=duplicate_status,
            action=(
                "Topic modeling uses comments as documents; "
                "duplicates are disclosed rather than "
                "silently removed."
            ),
        )
    )

    n_panelists = int(
        df[PANELIST].nunique(
            dropna=True
        )
    )

    n_products = int(
        df[PRODUCT].nunique(
            dropna=True
        )
    )

    panelist_counts = (
        df.groupby(
            PANELIST,
            dropna=False,
        )[PRODUCT]
        .count()
    )

    product_counts = (
        df.groupby(
            PRODUCT,
            dropna=False,
        )[PANELIST]
        .count()
    )

    panelist_range = (
        f"{int(panelist_counts.min())}"
        f"–{int(panelist_counts.max())}"
        if len(panelist_counts)
        else "—"
    )

    product_range = (
        f"{int(product_counts.min())}"
        f"–{int(product_counts.max())}"
        if len(product_counts)
        else "—"
    )

    diagnostics.append(
        Diagnostic(
            name="Observed study structure",
            observed=(
                f"{len(df)} rows; "
                f"{n_panelists} panelists; "
                f"{n_products} products"
            ),
            threshold=(
                "Inspect the actual dataset"
            ),
            status="PASS",
            action=(
                "Use observed structure; do not assume "
                "the nominal 50-consumer × 8-product design."
            ),
        )
    )

    diagnostics.append(
        Diagnostic(
            name=(
                "Evaluations per panelist"
            ),
            observed=panelist_range,
            threshold=(
                "Describe observed repeated-measures coverage"
            ),
            status="PASS",
            action=(
                "Repeated evaluations belong to the same "
                "consumer and must not be described as "
                "independent consumers."
            ),
        )
    )

    diagnostics.append(
        Diagnostic(
            name=(
                "Evaluations per product"
            ),
            observed=product_range,
            threshold=(
                "Describe observed product coverage"
            ),
            status="PASS",
            action=(
                "Use valid comment denominators within each "
                "product when reporting topic prevalence."
            ),
        )
    )

    return diagnostics

def prepare_corpus(
    df: pd.DataFrame,
    field_name: str,
) -> Tuple[
    pd.DataFrame,
    TfidfVectorizer,
    object,
    List[Diagnostic],
    Dict[str, float],
]:
    """
    Construct one qualitative corpus and apply every SOP-16
    pre-fit eligibility rule.

    IMPORTANT:
    If a pre-fit hard stop is triggered, NMF must not be fit.
    """

    valid_mask = (
        df[field_name].notna()
        & df[field_name]
        .astype(str)
        .str.strip()
        .ne("")
    )

    corpus_df = (
        df.loc[
            valid_mask,
            [
                PANELIST,
                PRODUCT,
                field_name,
            ],
        ]
        .copy()
        .reset_index(drop=False)
    )

    corpus_df["clean_text"] = (
        corpus_df[field_name]
        .map(clean_text)
    )

    # A value that became empty after basic cleaning contains
    # no usable text and is not treated as a valid document.
    corpus_df = (
        corpus_df[
            corpus_df["clean_text"]
            .str.len()
            .gt(0)
        ]
        .reset_index(drop=True)
    )

    n_expected = len(df)

    n_valid = len(
        corpus_df
    )

    n_missing = (
        n_expected
        - n_valid
    )

    diagnostics: List[Diagnostic] = []


    # -------------------------------------------------------------------------
    # Corpus-size eligibility
    # -------------------------------------------------------------------------

    diagnostics.append(
        Diagnostic(
            name="Valid comments",
            observed=str(n_valid),
            threshold=(
                f">= {MIN_VALID_COMMENTS}"
            ),
            status=(
                "PASS"
                if n_valid >= MIN_VALID_COMMENTS
                else "HARD STOP"
            ),
            action=(
                "Fit TF-IDF/NMF only if the corpus contains "
                "at least 100 valid comments."
            ),
        )
    )


    # -------------------------------------------------------------------------
    # Exact-comment repetition
    # -------------------------------------------------------------------------

    normalized = (
        corpus_df[
            "clean_text"
        ]
    )

    counts = (
        normalized
        .value_counts(
            dropna=False
        )
    )

    max_repeat_count = (
        int(counts.iloc[0])
        if len(counts)
        else 0
    )

    max_repeat_prop = (
        max_repeat_count
        / n_valid
        if n_valid
        else math.nan
    )

    repeat_observed = (
        f"{max_repeat_count}/{n_valid} "
        f"({pct(max_repeat_prop, 1)})"
        if n_valid
        else "0/0"
    )

    diagnostics.append(
        Diagnostic(
            name=(
                "Largest exact normalized-comment repetition"
            ),
            observed=repeat_observed,
            threshold=(
                f"<= {pct(MAX_EXACT_REPEAT_PROP)}"
            ),
            status=(
                "PASS"
                if (
                    n_valid
                    and max_repeat_prop
                    <= MAX_EXACT_REPEAT_PROP
                )
                else "HARD STOP"
            ),
            action=(
                "Halt before NMF if one exact normalized "
                "response represents more than 40% of "
                "valid comments."
            ),
        )
    )


    # -------------------------------------------------------------------------
    # TF-IDF
    # -------------------------------------------------------------------------

    vectorizer = TfidfVectorizer(
        lowercase=False,

        # Generic stop words are removed, while sensory words
        # themselves remain available to the model.
        stop_words="english",

        min_df=TFIDF_MIN_DF,
        max_df=TFIDF_MAX_DF,

        # Unigrams make the retained-token coverage criterion
        # straightforward and auditable.
        token_pattern=(
            r"(?u)\b[a-z][a-z0-9]+\b"
        ),

        norm="l2",
        use_idf=True,
        smooth_idf=True,
        sublinear_tf=False,
    )

    X = None

    retained_terms = 0

    informative_docs = 0
    informative_prop = 0.0

    if n_valid > 0:

        try:

            X = vectorizer.fit_transform(
                corpus_df[
                    "clean_text"
                ].tolist()
            )

            retained_terms = int(
                X.shape[1]
            )

            analyzer = (
                vectorizer
                .build_analyzer()
            )

            vocabulary = set(
                vectorizer
                .get_feature_names_out()
            )

            retained_token_counts = np.array(
                [
                    sum(
                        1
                        for token
                        in analyzer(text)
                        if token in vocabulary
                    )
                    for text
                    in corpus_df[
                        "clean_text"
                    ]
                ],
                dtype=int,
            )

            corpus_df[
                "retained_token_count"
            ] = retained_token_counts

            informative_docs = int(
                (
                    retained_token_counts
                    >= MIN_RETAINED_TOKENS_PER_DOC
                ).sum()
            )

            informative_prop = (
                informative_docs
                / n_valid
            )

        except ValueError:

            # A vocabulary-pruning failure is converted into
            # the formal vocabulary hard-stop state.
            X = None

            retained_terms = 0

            informative_docs = 0
            informative_prop = 0.0


    # -------------------------------------------------------------------------
    # Usable vocabulary
    # -------------------------------------------------------------------------

    diagnostics.append(
        Diagnostic(
            name=(
                "Retained TF-IDF vocabulary"
            ),
            observed=str(
                retained_terms
            ),
            threshold=(
                f">= {MIN_RETAINED_TERMS} terms"
            ),
            status=(
                "PASS"
                if retained_terms
                >= MIN_RETAINED_TERMS
                else "HARD STOP"
            ),
            action=(
                f"TF-IDF uses min_df={TFIDF_MIN_DF} "
                f"and max_df={TFIDF_MAX_DF}. "
                "At least 50 terms must remain after filtering."
            ),
        )
    )


    # -------------------------------------------------------------------------
    # Document information
    # -------------------------------------------------------------------------

    informative_observed = (
        f"{informative_docs}/{n_valid} "
        f"({pct(informative_prop, 1)})"
        if n_valid
        else "0/0"
    )

    diagnostics.append(
        Diagnostic(
            name=(
                "Comments with >=3 retained tokens"
            ),
            observed=informative_observed,
            threshold=(
                f">= {pct(MIN_INFORMATIVE_DOC_PROP)}"
            ),
            status=(
                "PASS"
                if informative_prop
                >= MIN_INFORMATIVE_DOC_PROP
                else "HARD STOP"
            ),
            action=(
                "At least 70% of valid comments must contain "
                "three or more vocabulary terms retained by "
                "the TF-IDF preprocessing."
            ),
        )
    )


    stats: Dict[str, float] = {
        "n_expected": n_expected,
        "n_valid": n_valid,
        "n_missing": n_missing,
        "retained_terms": retained_terms,
        "informative_docs": informative_docs,
        "informative_prop": informative_prop,
        "max_repeat_count": max_repeat_count,
        "max_repeat_prop": max_repeat_prop,
    }

    return (
        corpus_df,
        vectorizer,
        X,
        diagnostics,
        stats,
    )

def fit_nmf(
    X,
    n_topics: int,
    seed: int,
) -> Tuple[
    np.ndarray,
    np.ndarray,
]:
    """
    Fit one NMF model.

    init='random' is intentional because SOP-16 requires
    evaluation of reproducibility across random starts.
    """

    model = NMF(
        n_components=n_topics,
        init="random",
        random_state=seed,
        max_iter=NMF_MAX_ITER,
        solver="cd",
        beta_loss="frobenius",
        tol=1e-4,
    )

    W = model.fit_transform(
        X
    )

    H = model.components_

    return (
        W,
        H,
    )

def normalized_topics(
    H: np.ndarray,
) -> np.ndarray:
    """
    Unit-normalize topic-term vectors for cosine matching.
    """

    norms = np.linalg.norm(
        H,
        axis=1,
        keepdims=True,
    )

    norms[
        norms == 0
    ] = 1.0

    return (
        H
        / norms
    )

def reorder_to_reference(
    reference_H: np.ndarray,
    H: np.ndarray,
) -> Tuple[
    np.ndarray,
    np.ndarray,
]:
    """
    Match topics in a candidate NMF run to topics in the
    reference run using maximum cosine similarity.

    The Hungarian algorithm ensures a one-to-one assignment.
    """

    similarities = cosine_similarity(
        normalized_topics(
            reference_H
        ),
        normalized_topics(
            H
        ),
    )

    row_ind, col_ind = (
        linear_sum_assignment(
            -similarities
        )
    )

    order = np.empty(
        reference_H.shape[0],
        dtype=int,
    )

    matched = np.empty(
        reference_H.shape[0],
        dtype=float,
    )

    for row, col in zip(
        row_ind,
        col_ind,
    ):

        order[row] = col

        matched[row] = (
            similarities[
                row,
                col,
            ]
        )

    return (
        order,
        matched,
    )

def top_terms_from_H(
    H: np.ndarray,
    terms: np.ndarray,
    n: int = N_TOP_TERMS,
) -> List[List[str]]:
    """
    Return the highest-weight terms for each NMF topic.

    These are shown for human review.

    The script intentionally does NOT invent semantic topic
    names automatically because SOP-16 requires human review
    before descriptive labeling.
    """

    results: List[
        List[str]
    ] = []

    for topic_weights in H:

        indices = (
            np.argsort(
                topic_weights
            )[::-1][:n]
        )

        topic_terms = [
            str(
                terms[index]
            )
            for index
            in indices
        ]

        results.append(
            topic_terms
        )

    return results

def evaluate_candidate(
    X,
    vectorizer: TfidfVectorizer,
    n_topics: int,
) -> CandidateResult:
    """
    Evaluate one candidate topic count across 20 random starts.

    Required pass conditions:
        mean matched-topic cosine >= 0.80
        minimum matched-topic cosine >= 0.65
        every retained topic >= 5% dominant-topic prevalence
    """

    seeds = [
        RANDOM_STATE + offset
        for offset
        in range(
            N_STABILITY_SEEDS
        )
    ]

    runs: List[
        Tuple[
            np.ndarray,
            np.ndarray,
        ]
    ] = []

    for seed in seeds:

        W, H = fit_nmf(
            X=X,
            n_topics=n_topics,
            seed=seed,
        )

        runs.append(
            (
                W,
                H,
            )
        )


    # The first reproducible random-start fit is used as the
    # reference orientation for topic matching.
    reference_W, reference_H = (
        runs[0]
    )


    # -------------------------------------------------------------------------
    # Stability across starts
    # -------------------------------------------------------------------------

    all_matched: List[
        float
    ] = []

    for _, H in runs[1:]:

        _, matched = (
            reorder_to_reference(
                reference_H,
                H,
            )
        )

        all_matched.extend(
            matched.tolist()
        )

    if all_matched:

        mean_similarity = float(
            np.mean(
                all_matched
            )
        )

        min_similarity = float(
            np.min(
                all_matched
            )
        )

    else:

        mean_similarity = 1.0
        min_similarity = 1.0


    # -------------------------------------------------------------------------
    # Topic prevalence
    # -------------------------------------------------------------------------

    dominant_topic = np.argmax(
        reference_W,
        axis=1,
    )

    topic_prevalence = np.array(
        [
            (
                dominant_topic
                == topic_index
            ).mean()
            for topic_index
            in range(
                n_topics
            )
        ],
        dtype=float,
    )


    # -------------------------------------------------------------------------
    # Candidate eligibility
    # -------------------------------------------------------------------------

    passes = (
        mean_similarity
        >= MIN_MEAN_TOPIC_COSINE

        and min_similarity
        >= MIN_TOPIC_COSINE

        and bool(
            np.all(
                topic_prevalence
                >= MIN_TOPIC_PREVALENCE
            )
        )
    )

    terms = (
        vectorizer
        .get_feature_names_out()
    )

    top_terms = (
        top_terms_from_H(
            reference_H,
            terms,
        )
    )

    return CandidateResult(
        n_topics=n_topics,

        mean_similarity=(
            mean_similarity
        ),

        min_similarity=(
            min_similarity
        ),

        topic_prevalence=(
            topic_prevalence
        ),

        passed=passes,

        W=reference_W,
        H=reference_H,

        terms=terms,

        top_terms=top_terms,

        dominant_topic=(
            dominant_topic
        ),
    )

def select_candidate(
    candidates: Sequence[
        CandidateResult
    ],
) -> Optional[
    CandidateResult
]:
    """
    Select among topic counts that ALREADY pass every approved
    SOP-16 validity rule.

    The SOP defines candidate eligibility but not an explicit
    tie-breaking objective among multiple passing solutions.

    This script therefore uses a transparent, deterministic rule:
    1. strongest mean matched-topic reproducibility;
    2. exact ties favor the smaller topic count.

    No candidate that fails an SOP threshold can be selected.
    """

    passing = [
        candidate
        for candidate
        in candidates
        if candidate.passed
    ]

    if not passing:
        return None

    return sorted(
        passing,
        key=lambda candidate: (
            -candidate.mean_similarity,
            candidate.n_topics,
        ),
    )[0]

def analyze_corpus(
    df: pd.DataFrame,
    field_name: str,
) -> CorpusResult:
    """
    Execute the complete SOP-16 workflow for one approved
    comment field.

    likes and dislikes are kept separate because they answer
    different qualitative prompts and have different missingness.
    """

    (
        corpus_df,
        vectorizer,
        X,
        diagnostics,
        stats,
    ) = prepare_corpus(
        df=df,
        field_name=field_name,
    )

    result = CorpusResult(
        field_name=field_name,

        display_name=(
            display_field(
                field_name
            )
        ),

        status="SUPPORTED",

        stop_stage=None,
        stop_reason=None,

        n_expected=int(
            stats[
                "n_expected"
            ]
        ),

        n_valid=int(
            stats[
                "n_valid"
            ]
        ),

        n_missing=int(
            stats[
                "n_missing"
            ]
        ),

        retained_terms=int(
            stats[
                "retained_terms"
            ]
        ),

        informative_docs=int(
            stats[
                "informative_docs"
            ]
        ),

        informative_prop=float(
            stats[
                "informative_prop"
            ]
        ),

        max_repeat_count=int(
            stats[
                "max_repeat_count"
            ]
        ),

        max_repeat_prop=float(
            stats[
                "max_repeat_prop"
            ]
        ),

        diagnostics=diagnostics,

        analysis_df=corpus_df,
    )


    # =========================================================================
    # PRE-FIT HARD STOP
    # =========================================================================

    failed_pre_fit = [
        diagnostic
        for diagnostic
        in diagnostics
        if diagnostic.status
        == "HARD STOP"
    ]

    if failed_pre_fit:

        result.status = (
            "HARD STOP"
        )

        result.stop_stage = (
            "Pre-fit eligibility"
        )

        result.stop_reason = (
            "; ".join(
                (
                    f"{diagnostic.name}: "
                    f"{diagnostic.observed} "
                    f"vs {diagnostic.threshold}"
                )
                for diagnostic
                in failed_pre_fit
            )
        )

        # Critical SOP-16 behavior:
        # do NOT fit NMF after a pre-fit eligibility failure.
        return result


    if X is None:

        result.status = (
            "HARD STOP"
        )

        result.stop_stage = (
            "Pre-fit eligibility"
        )

        result.stop_reason = (
            "TF-IDF could not construct an eligible "
            "document-term matrix."
        )

        return result


    # =========================================================================
    # POST-FIT STABILITY GATE
    # =========================================================================

    candidates: List[
        CandidateResult
    ] = []

    for n_topics in CANDIDATE_TOPICS:

        # Defensive protection for a future very small matrix.
        if n_topics > min(
            X.shape
        ):
            continue

        candidate = (
            evaluate_candidate(
                X=X,
                vectorizer=vectorizer,
                n_topics=n_topics,
            )
        )

        candidates.append(
            candidate
        )

    result.candidates = (
        candidates
    )


    # =========================================================================
    # CANDIDATE DIAGNOSTIC DISCLOSURE
    # =========================================================================

    for candidate in candidates:

        min_prevalence = (
            float(
                np.min(
                    candidate
                    .topic_prevalence
                )
            )
            if len(
                candidate
                .topic_prevalence
            )
            else math.nan
        )

        observed = (
            f"mean cosine="
            f"{candidate.mean_similarity:.3f}; "
            f"min cosine="
            f"{candidate.min_similarity:.3f}; "
            f"min prevalence="
            f"{pct(min_prevalence, 1)}"
        )

        threshold = (
            f"mean >= "
            f"{MIN_MEAN_TOPIC_COSINE:.2f}; "
            f"min >= "
            f"{MIN_TOPIC_COSINE:.2f}; "
            f"every topic >= "
            f"{pct(MIN_TOPIC_PREVALENCE)}"
        )

        diagnostics.append(
            Diagnostic(
                name=(
                    f"{candidate.n_topics}-topic "
                    "stability"
                ),
                observed=observed,
                threshold=threshold,
                status=(
                    "PASS"
                    if candidate.passed
                    else "HARD STOP (candidate)"
                ),
                action=(
                    "Retain this topic count only "
                    "if all stability and prevalence "
                    "requirements pass."
                ),
            )
        )


    # =========================================================================
    # FINAL ELIGIBILITY
    # =========================================================================

    selected = (
        select_candidate(
            candidates
        )
    )

    if selected is None:

        result.status = (
            "HARD STOP"
        )

        result.stop_stage = (
            "Post-fit stability"
        )

        result.stop_reason = (
            "No candidate topic count from 2–5 passed "
            "all reproducibility and prevalence thresholds."
        )

        # Critical SOP behavior:
        # candidate models were fitted only to test eligibility.
        # Failed topic solutions are not interpreted.
        return result


    result.selected = (
        selected
    )


    # =========================================================================
    # DOCUMENT ASSIGNMENTS
    # =========================================================================

    corpus_df = (
        corpus_df.copy()
    )

    corpus_df[
        "dominant_topic"
    ] = (
        selected
        .dominant_topic
        + 1
    )

    result.analysis_df = (
        corpus_df
    )


    # =========================================================================
    # PRODUCT × TOPIC PREVALENCE
    # =========================================================================

    product_order = sorted(
        corpus_df[
            PRODUCT
        ]
        .astype(str)
        .unique()
        .tolist()
    )

    topic_columns = [
        f"Topic {index + 1}"
        for index
        in range(
            selected.n_topics
        )
    ]

    prevalence_rows: List[
        List[float]
    ] = []

    count_rows: List[
        List[int]
    ] = []

    for product in product_order:

        product_df = (
            corpus_df[
                corpus_df[
                    PRODUCT
                ]
                .astype(str)
                .eq(
                    str(product)
                )
            ]
        )

        denominator = len(
            product_df
        )

        product_prevalence: List[
            float
        ] = []

        product_counts: List[
            int
        ] = []

        for topic_index in range(
            1,
            selected.n_topics + 1,
        ):

            count = int(
                (
                    product_df[
                        "dominant_topic"
                    ]
                    == topic_index
                ).sum()
            )

            product_counts.append(
                count
            )

            product_prevalence.append(
                (
                    count
                    / denominator
                    if denominator
                    else math.nan
                )
            )

        prevalence_rows.append(
            product_prevalence
        )

        count_rows.append(
            product_counts
        )


    result.product_topic = (
        pd.DataFrame(
            prevalence_rows,
            index=product_order,
            columns=topic_columns,
        )
    )

    result.product_topic_counts = (
        pd.DataFrame(
            count_rows,
            index=product_order,
            columns=topic_columns,
        )
    )

    return result

def topic_cards(
    result: CorpusResult,
) -> str:
    """
    Show top terms without assigning unreviewed semantic labels.
    """

    if not result.selected:
        return ""

    cards: List[str] = []

    for index, terms in enumerate(
        result.selected.top_terms,
        start=1,
    ):

        prevalence = float(
            result
            .selected
            .topic_prevalence[
                index - 1
            ]
        )

        chips = "".join(
            (
                "<span class='term'>"
                f"{h(term)}"
                "</span>"
            )
            for term
            in terms
        )

        cards.append(
            "<div class='topic-card'>"

            "<div class='topic-head'>"

            f"<strong>Topic {index}</strong>"

            "<span>"
            f"{pct(prevalence, 0)} "
            "of valid comments"
            "</span>"

            "</div>"

            f"<div class='terms'>{chips}</div>"

            "<p class='tiny'>"
            "Machine-extracted word pattern. "
            "Descriptive sensory labeling requires "
            "human review."
            "</p>"

            "</div>"
        )

    return (
        "<div class='topic-grid'>"
        + "".join(cards)
        + "</div>"
    )

def heat_cell(
    value: float,
    count: int,
    denominator: int,
) -> str:
    """
    Build one self-contained HTML heatmap cell.
    """

    if not np.isfinite(
        value
    ):

        return (
            "<td class='heat na'>"
            "—"
            "</td>"
        )

    alpha = (
        0.08
        + 0.72
        * min(
            max(
                value,
                0.0,
            ),
            1.0,
        )
    )

    text_color = (
        "#ffffff"
        if alpha > 0.48
        else "#173042"
    )

    style = (
        "background:"
        f"rgba(27,75,90,{alpha:.3f});"
        f"color:{text_color};"
    )

    title = (
        f"{count} of {denominator} valid "
        f"product comments "
        f"({100 * value:.1f}%) "
        "had this as the dominant topic."
    )

    return (
        "<td "
        "class='heat' "
        f"style='{style}' "
        f"title='{h(title)}'>"
        f"{100 * value:.0f}%"
        "</td>"
    )

def topic_heatmap(
    result: CorpusResult,
) -> str:
    """
    Approved product × exploratory-topic heatmap.

    Each cell represents dominant-topic prevalence within
    valid comments for that product.
    """

    if (
        result.product_topic
        is None
        or result.product_topic_counts
        is None
    ):
        return ""

    headers = "".join(
        f"<th>{h(column)}</th>"
        for column
        in result.product_topic.columns
    )

    rows: List[str] = []

    for product in (
        result
        .product_topic
        .index
    ):

        denominator = int(
            result
            .product_topic_counts
            .loc[
                product
            ]
            .sum()
        )

        cells = "".join(
            heat_cell(
                value=float(
                    result
                    .product_topic
                    .loc[
                        product,
                        column,
                    ]
                ),
                count=int(
                    result
                    .product_topic_counts
                    .loc[
                        product,
                        column,
                    ]
                ),
                denominator=denominator,
            )
            for column
            in result
            .product_topic
            .columns
        )

        rows.append(
            "<tr>"
            "<th class='rowhead'>"
            f"Product {h(product)}"
            "</th>"
            f"{cells}"
            "</tr>"
        )

    return (
        "<div class='table-wrap heat-wrap'>"

        "<table class='heatmap'>"

        "<thead>"
        "<tr>"
        "<th>Product</th>"
        f"{headers}"
        "</tr>"
        "</thead>"

        "<tbody>"
        + "".join(rows)
        + "</tbody>"

        "</table>"

        "</div>"

        "<p class='section-note'>"
        "Cells show the percentage of valid comments "
        "for each product assigned most strongly to each "
        "topic. Hover over a cell for the numerator and "
        "denominator."
        "</p>"
    )

    """
    Execute SOP-16 from start to finish.
    """

    data_path, dashboard_path = (
        resolve_paths()
    )

    # =========================================================================

    df = pd.read_csv(
        data_path
    )

    df = drop_export_index(
        df
    )


    # =========================================================================
    # DATASET CAPABILITY GATE
    # =========================================================================

    (
        capability_status,
        capability_messages,
    ) = capability_gate(
        df
    )


    # =========================================================================
    # UNSUPPORTED DATASET
    # =========================================================================

    if (
        capability_status
        == "UNSUPPORTED"
    ):

        corpus_results: List[
            CorpusResult
        ] = []

        for field_name in TEXT_FIELDS:

            if field_name not in df.columns:

                corpus_results.append(
                    CorpusResult(

                        ),

                        display_name=(
                            display_field(
                                field_name
                            )
                        ),

                        status=(
                            "HARD STOP"
                        ),

                        stop_stage=(
                            "Dataset capability gate"
                        ),

                        stop_reason=(
                            f"Required text field "
                            f"'{field_name}' is absent."
                        ),

                        n_expected=(
                            len(df)
                        ),

                        n_valid=0,

                        n_missing=(
                            len(df)
                        ),
                    )
                )

        if all(
            column in df.columns
            for column
            in [
                PANELIST,
                PRODUCT,
            ]
        ):

            structure_diags = (
                validate_structure(
                    df
                )
            )

        else:

            structure_diags = [
                Diagnostic(
                    name=(
                        "Dataset capability"
                    ),

                    observed=(
                        "; ".join(
                            capability_messages
                        )
                    ),

                    threshold=(
                        "Required SOP-16 "
                        "fields present"
                    ),

                    status=(
                        "HARD STOP"
                    ),

                    action=(
                        "Do not run SOP-16."
                    ),
                )
            ]


    # =========================================================================
    # SUPPORTED / PARTIALLY SUPPORTED DATASET
    # =========================================================================

    else:

        structure_diags = (
            validate_structure(
                df
            )
        )

        corpus_results = [
            analyze_corpus(
                df=df,
                field_name=field_name,
            )
            for field_name
            in TEXT_FIELDS
            if field_name
            in df.columns
        ]


    # =========================================================================
    # HTML DASHBOARD
    # =========================================================================

    dashboard_html = (
        build_dashboard(
            df=df,

            data_path=(
                data_path
            ),

            capability_status=(
                capability_status
            ),

            capability_messages=(
                capability_messages
            ),

            structure_diags=(
                structure_diags
            ),

            corpus_results=(
                corpus_results
            ),
        )
    )

    dashboard_path.write_text(
        dashboard_html,
        encoding="utf-8",
    )


    # =========================================================================
    # OPTIONAL TECHNICAL OUTPUT
    # =========================================================================

    save_technical_exports(
        output_dir=(
            dashboard_path.parent
        ),
        results=(
            corpus_results
        ),
    )


    # =========================================================================
    # CONSOLE SUMMARY
    # =========================================================================

    passed = [
        result
        for result
        in corpus_results
        if result.status
        != "HARD STOP"
    ]

    stopped = [
        result
        for result
        in corpus_results
        if result.status
        == "HARD STOP"
    ]

    if passed and not stopped:

        answer = (
            "Stable exploratory text-topic structure "
            "was detected in the in-scope comment "
            "corpora."
        )

        model_status = (
            "All SOP-16 eligibility and stability "
            "checks passed."
        )

    elif passed:

        answer = (
            "Stable topic structure was detected for "
            "only part of the in-scope comment data."
        )

        model_status = (
            "PARTIAL — at least one corpus "
            "hard-stopped."
        )

    else:

        answer = (
            "The comments do not support a meaningful "
            "unsupervised topic model under SOP-16."
        )

        model_status = (
            "HARD STOP."
        )


    n_panelists = (
        df[PANELIST].nunique()
        if PANELIST in df.columns
        else 0
    )

    n_products = (
        df[PRODUCT].nunique()
        if PRODUCT in df.columns
        else 0
    )


    print(
        f"{SOP_ID} — {SOP_NAME}\n"
    )

    print(
        f"Analyzing {len(df):,} evaluations "
        f"from {n_panelists:,} consumers "
        f"across {n_products:,} products...\n"
    )

    print(
        "=" * 72
    )

    print(
        f"ANSWER: {answer}"
    )

    print(
        f"MODEL STATUS: {model_status}"
    )

    print(
        "Dashboard:"
    )

    print(
        dashboard_path.resolve()
    )

    print(
        "=" * 72
    )


    # =========================================================================
    # AUTO-OPEN DASHBOARD
    # =========================================================================

    if AUTO_OPEN_DASHBOARD:

        try:

            webbrowser.open(
                dashboard_path
                .resolve()
                .as_uri()
            )

        except Exception as exc:

            print(
                "Dashboard created, but automatic "
                "browser opening failed: "
                f"{exc}"
            )

if __name__ == "__main__":
    main()

"""
SOP-17 — Product-Level Comment Sentiment Profile

Secondary exploratory analysis of HOW STRONGLY consumers express positive
language in prompted LIKE comments and negative language in prompted DISLIKE
comments, by product.

This revised SOP intentionally does NOT emphasize the overall percentage of
positive vs. negative comments because the questionnaire explicitly asks every
consumer for both a like and a dislike. The useful signal is the PRODUCT-RELATIVE
strength of the language within each prompt.

Primary quantities
------------------
1. Like enthusiasm deviation:
       product mean VADER compound for likes
       minus category mean VADER compound for likes

   Positive values = more enthusiastic positive-language tendency than the
   category average for like comments.

2. Dislike severity deviation:
       category mean VADER compound for dislikes
       minus product mean VADER compound for dislikes

   Positive values = more negative / harsher language than the category average
   for dislike comments.

3. 95% consumer-level bootstrap confidence intervals around those deviations.
   Whole consumers are resampled so the repeated-measures product structure is
   preserved.

IMPORTANT GUARDRAILS
--------------------
- VADER is a generic-language tool, not a validated sensory model.
- Sensory words such as bitter, dark, strong, intense, thick, sweet, or mild can
  have context-specific meaning that generic sentiment may misread.
- Sentiment is secondary exploratory evidence only.
- Do not interpret these scores as overall liking, purchase intent, quality,
  repurchase likelihood, causality, or sensory intensity.
- Like and dislike prompts are NEVER pooled into a single acceptance score.
- Missing comments remain missing; they are not recoded as neutral or favorable.
- Bootstrap intervals quantify sampling uncertainty in the descriptive
  product-relative sentiment metric. They do not validate VADER as a sensory
  construct and are not used to create significance letters.

Dependencies
------------
pip install pandas numpy nltk

Default project structure
-------------------------
project/
├── SOP17_product_sentiment_profile.py
└── data/
    └── protein_shake_category_appraisal_mock_redesigned.csv
"""

import argparse

import html

import sys

import webbrowser

from dataclasses import dataclass, field

from pathlib import Path

from typing import Iterable, Optional

import numpy as np

import pandas as pd

try:
    import nltk
    from nltk.sentiment.vader import SentimentIntensityAnalyzer
except ImportError as exc:
    raise SystemExit(
        "SOP-17 requires NLTK. Install dependencies with:\n"
        "    pip install pandas numpy nltk"
    ) from exc

SOP_ID = "SOP-17"

SOP_NAME = "Product-Level Comment Sentiment Profile"

ANALYSIS_CLASSIFICATION = "Secondary exploratory"

DATA_FILENAME = "data/protein_shake_category_appraisal_mock_redesigned.csv"

OUTPUT_DIR = "sop17_output"

DASHBOARD_FILENAME = "SOP17_dashboard.html"

AUTO_OPEN_DASHBOARD = True

SAVE_TECHNICAL_EXPORTS = False

PRODUCT_COL = "product_code"

PANELIST_COL = "panelist_id"

LIKE_COL = "likes"

DISLIKE_COL = "dislikes"

REQUIRED_COLUMNS = (PANELIST_COL, PRODUCT_COL, LIKE_COL, DISLIKE_COL)

RANDOM_STATE = 42

BOOTSTRAP_REPS = 2000

CI_LEVEL = 0.95

MIN_VALID_COMMENTS_HARD_STOP = 5

MIN_VALID_COMMENTS_WARNING = 20

EXPECTED_ROWS = 400

EXPECTED_PRODUCTS = 8

EXPECTED_PANELISTS = 50

EXPECTED_MISSING_LIKES = 0

EXPECTED_MISSING_DISLIKES = 5

class GateResult:
    status: str
    messages: list[str] = field(default_factory=list)
    warnings: list[str] = field(default_factory=list)
    hard_stops: list[str] = field(default_factory=list)

class AnalysisResult:
    scored: pd.DataFrame
    product_summary: pd.DataFrame
    coverage_summary: pd.DataFrame
    answer: str
    model_status: str
    warnings: list[str]
    hard_stops: list[str]
    diagnostics: list[str]

def is_missing_comment(value: object) -> bool:
    if pd.isna(value):
        return True
    return str(value).strip() == ""

def clean_comment(value: object) -> Optional[str]:
    if is_missing_comment(value):
        return None
    return " ".join(str(value).split())

def safe_product_sort(values: Iterable[object]) -> list[str]:
    cleaned = [str(v) for v in values if not pd.isna(v)]
    return sorted(set(cleaned), key=lambda x: (len(x), x.casefold()))

def source_label(source: str) -> str:
    return "Like comments" if source == LIKE_COL else "Dislike comments"

def load_data(path: Path) -> tuple[pd.DataFrame, list[str]]:
    df = pd.read_csv(path)
    removed: list[str] = []

    for col in list(df.columns):
        name = str(col).strip().lower()
        if name.startswith("unnamed:") or name == "index":
            numeric = pd.to_numeric(df[col], errors="coerce")
            if numeric.notna().all():
                vals = numeric.astype(int).to_numpy()
                n = len(vals)
                if np.array_equal(vals, np.arange(n)) or np.array_equal(vals, np.arange(1, n + 1)):
                    df = df.drop(columns=[col])
                    removed.append(str(col))

    return df, removed

def dataset_capability_gate(df: pd.DataFrame) -> GateResult:
    missing = [c for c in REQUIRED_COLUMNS if c not in df.columns]
    messages: list[str] = []
    warnings: list[str] = []
    hard_stops: list[str] = []

    if missing:
        hard_stops.append(
            "The revised product-level SOP-17 requires all of: "
            f"{', '.join(REQUIRED_COLUMNS)}. Missing: {', '.join(missing)}. "
            "No other variable may be substituted for the missing construct."
        )
        return GateResult("UNSUPPORTED", messages, warnings, hard_stops)

    if df[PANELIST_COL].isna().any():
        hard_stops.append(
            f"`{PANELIST_COL}` contains missing values. Consumer-level resampling cannot "
            "preserve repeated-measures structure without a consumer identifier."
        )

    if df[PRODUCT_COL].isna().any():
        hard_stops.append(
            f"`{PRODUCT_COL}` contains missing values. Product-relative sentiment cannot "
            "be assigned for those evaluations."
        )

    for source in (LIKE_COL, DISLIKE_COL):
        valid = int(df[source].map(lambda x: not is_missing_comment(x)).sum())
        messages.append(f"{source}: {valid:,} nonmissing/nonblank comments.")
        if valid == 0:
            hard_stops.append(f"`{source}` contains no usable observed comments.")

    if hard_stops:
        return GateResult("UNSUPPORTED", messages, warnings, hard_stops)

    messages.append(
        "Product and consumer identifiers are present, so category-relative product sentiment "
        "and consumer-level bootstrap resampling are structurally available."
    )
    return GateResult("SUPPORTED", messages, warnings, hard_stops)

def validate_study_design(
    df: pd.DataFrame,
    removed_index_columns: list[str],
) -> tuple[list[str], list[str], list[str]]:
    diagnostics: list[str] = []
    warnings: list[str] = []
    hard_stops: list[str] = []

    diagnostics.append(f"Rows loaded: {len(df):,}.")
    diagnostics.append(f"Columns analyzed: {df.shape[1]:,}.")

    if removed_index_columns:
        diagnostics.append(
            "Removed CSV export-index column(s): " + ", ".join(removed_index_columns) + "."
        )

    n_panelists = int(df[PANELIST_COL].nunique(dropna=True))
    n_products = int(df[PRODUCT_COL].nunique(dropna=True))
    diagnostics.append(f"Unique consumers: {n_panelists:,}.")
    diagnostics.append(f"Unique products: {n_products:,}.")

    if len(df) != EXPECTED_ROWS:
        warnings.append(
            f"Dataset snapshot differs from the current workshop dataset: found {len(df):,} rows "
            f"rather than {EXPECTED_ROWS:,}. Analysis uses the data actually loaded."
        )

    if n_panelists != EXPECTED_PANELISTS:
        warnings.append(
            f"Found {n_panelists} consumers rather than the current workshop snapshot of "
            f"{EXPECTED_PANELISTS}. Analysis uses the observed consumer count."
        )

    if n_products != EXPECTED_PRODUCTS:
        warnings.append(
            f"Found {n_products} products rather than the current workshop snapshot of "
            f"{EXPECTED_PRODUCTS}. Analysis uses the observed products."
        )

    duplicated_mask = df.duplicated([PANELIST_COL, PRODUCT_COL], keep=False)
    duplicated_rows = int(duplicated_mask.sum())
    diagnostics.append(
        f"Rows involved in duplicate consumer × product cells: {duplicated_rows:,}."
    )
    if duplicated_rows > 0:
        hard_stops.append(
            "Duplicate consumer × product rows were detected. The revised SOP-17 assumes one "
            "evaluation per consumer per product so that whole-consumer bootstrap resampling does "
            "not silently over-weight duplicate evaluations. Resolve duplicates before analysis."
        )

    products_per_consumer = df.groupby(PANELIST_COL)[PRODUCT_COL].nunique()
    if not products_per_consumer.empty:
        diagnostics.append(
            "Distinct products evaluated per consumer: "
            f"{int(products_per_consumer.min())}–{int(products_per_consumer.max())}."
        )
        if (products_per_consumer < n_products).any():
            warnings.append(
                "At least one consumer has an incomplete product set. Bootstrap resampling remains "
                "consumer-level, but product-specific comment denominators may differ."
            )

    for source in (LIKE_COL, DISLIKE_COL):
        missing_n = int(df[source].map(is_missing_comment).sum())
        valid_n = len(df) - missing_n
        diagnostics.append(
            f"{source}: {valid_n:,} valid comments; {missing_n:,} missing/blank."
        )

        expected_missing = (
            EXPECTED_MISSING_LIKES if source == LIKE_COL else EXPECTED_MISSING_DISLIKES
        )
        if len(df) == EXPECTED_ROWS and missing_n != expected_missing:
            warnings.append(
                f"`{source}` missingness differs from the current workshop snapshot: found "
                f"{missing_n}, expected {expected_missing}. Missing comments remain missing."
            )

    # Product × source coverage thresholds.
    coverage_rows: list[tuple[str, str, int]] = []
    for product in safe_product_sort(df[PRODUCT_COL]):
        product_df = df.loc[df[PRODUCT_COL].astype(str) == product]
        for source in (LIKE_COL, DISLIKE_COL):
            valid_n = int(product_df[source].map(lambda x: not is_missing_comment(x)).sum())
            coverage_rows.append((product, source, valid_n))

            if valid_n < MIN_VALID_COMMENTS_HARD_STOP:
                hard_stops.append(
                    f"Product {product} has only {valid_n} valid {source} comments; the project "
                    f"hard-stop threshold is {MIN_VALID_COMMENTS_HARD_STOP}."
                )
            elif valid_n < MIN_VALID_COMMENTS_WARNING:
                warnings.append(
                    f"Product {product} has {valid_n} valid {source} comments, below the project "
                    f"warning threshold of {MIN_VALID_COMMENTS_WARNING}; bootstrap uncertainty may be wide."
                )

    diagnostics.append(
        f"Project text-coverage thresholds: HARD STOP < {MIN_VALID_COMMENTS_HARD_STOP} valid comments "
        f"per product × prompt; WARNING < {MIN_VALID_COMMENTS_WARNING}."
    )
    diagnostics.append(
        "Whole consumers, not individual comments, are the resampling unit for uncertainty."
    )

    return diagnostics, warnings, hard_stops

def build_comment_long(df: pd.DataFrame) -> pd.DataFrame:
    records: list[dict[str, object]] = []

    for row_index, row in df.iterrows():
        for source in (LIKE_COL, DISLIKE_COL):
            comment = clean_comment(row[source])
            if comment is None:
                continue

            records.append(
                {
                    "row_index": row_index,
                    PANELIST_COL: row[PANELIST_COL],
                    PRODUCT_COL: str(row[PRODUCT_COL]),
                    "source": source,
                    "comment": comment,
                }
            )

    return pd.DataFrame.from_records(records)

def initialize_vader() -> SentimentIntensityAnalyzer:
    try:
        return SentimentIntensityAnalyzer()
    except LookupError:
        print("VADER lexicon not found locally. Attempting one-time NLTK download...")
        try:
            ok = nltk.download("vader_lexicon", quiet=True)
        except Exception:
            ok = False

        if not ok:
            raise RuntimeError(
                "NLTK's VADER lexicon is not installed and automatic download failed. "
                "Run `python -m nltk.downloader vader_lexicon` and rerun SOP-17."
            )

        try:
            return SentimentIntensityAnalyzer()
        except LookupError as exc:
            raise RuntimeError(
                "The VADER lexicon still could not be loaded after the download attempt. "
                "Run `python -m nltk.downloader vader_lexicon` and rerun SOP-17."
            ) from exc

def score_comments(
    comment_long: pd.DataFrame,
    analyzer: SentimentIntensityAnalyzer,
) -> pd.DataFrame:
    scored = comment_long.copy()
    if scored.empty:
        return scored

    polarity = scored["comment"].map(analyzer.polarity_scores)
    vader_df = pd.DataFrame(polarity.tolist(), index=scored.index)
    scored = pd.concat(
        [scored, vader_df[["neg", "neu", "pos", "compound"]]],
        axis=1,
    )

    # Common-direction metric for the two prompts:
    #   likes    -> larger compound = more enthusiastic
    #   dislikes -> more negative compound = harsher, so multiply by -1
    scored["profile_score"] = np.where(
        scored["source"].eq(LIKE_COL),
        scored["compound"],
        -scored["compound"],
    )
    return scored

def summarize_coverage(df: pd.DataFrame) -> pd.DataFrame:
    rows: list[dict[str, object]] = []

    for product in safe_product_sort(df[PRODUCT_COL]):
        product_df = df.loc[df[PRODUCT_COL].astype(str) == product]
        for source in (LIKE_COL, DISLIKE_COL):
            valid_n = int(product_df[source].map(lambda x: not is_missing_comment(x)).sum())
            total_n = len(product_df)
            rows.append(
                {
                    PRODUCT_COL: product,
                    "source": source,
                    "valid_n": valid_n,
                    "missing_n": total_n - valid_n,
                    "coverage_pct": 100.0 * valid_n / total_n if total_n else np.nan,
                }
            )

    return pd.DataFrame(rows)

def calculate_product_relative_summary(scored: pd.DataFrame) -> pd.DataFrame:
    """
    Category baseline is calculated separately within likes and dislikes.

    For likes:
        relative_deviation = product mean compound - category mean compound

    For dislikes:
        relative_deviation = category mean compound - product mean compound
        (positive = harsher / more negative language than category average)
    """
    if scored.empty:
        return pd.DataFrame()

    rows: list[dict[str, object]] = []

    for source in (LIKE_COL, DISLIKE_COL):
        source_df = scored.loc[scored["source"].eq(source)].copy()
        if source_df.empty:
            continue

        category_mean_compound = float(source_df["compound"].mean())
        category_median_compound = float(source_df["compound"].median())

        for product, group in source_df.groupby(PRODUCT_COL, sort=False):
            mean_compound = float(group["compound"].mean())
            median_compound = float(group["compound"].median())

            if source == LIKE_COL:
                deviation = mean_compound - category_mean_compound
            else:
                deviation = category_mean_compound - mean_compound

            rows.append(
                {
                    PRODUCT_COL: str(product),
                    "source": source,
                    "n_valid": int(len(group)),
                    "mean_compound": mean_compound,
                    "median_compound": median_compound,
                    "category_mean_compound": category_mean_compound,
                    "category_median_compound": category_median_compound,
                    "relative_deviation": float(deviation),
                }
            )

    return pd.DataFrame(rows)

def bootstrap_relative_deviations(
    scored: pd.DataFrame,
    products: list[str],
    reps: int = BOOTSTRAP_REPS,
    random_state: int = RANDOM_STATE,
    ci_level: float = CI_LEVEL,
) -> pd.DataFrame:
    """
    Resample whole consumers with replacement.

    Each selected consumer contributes all of their observed product comments.
    This preserves the within-consumer repeated-measures structure in the
    resampling scheme.
    """
    rng = np.random.default_rng(random_state)
    panelists = pd.unique(scored[PANELIST_COL].dropna())

    if len(panelists) < 2:
        raise RuntimeError(
            "Consumer-level bootstrap requires at least two unique consumers."
        )

    # Pre-index each consumer's scored rows so resampling only concatenates
    # small arrays rather than filtering the full DataFrame repeatedly.
    panelist_frames = {
        panelist: scored.loc[scored[PANELIST_COL].eq(panelist)].copy()
        for panelist in panelists
    }

    stored: dict[tuple[str, str], list[float]] = {
        (product, source): []
        for product in products
        for source in (LIKE_COL, DISLIKE_COL)
    }

    n_panelists = len(panelists)

    for _ in range(reps):
        sampled = rng.choice(panelists, size=n_panelists, replace=True)
        sample_df = pd.concat(
            [panelist_frames[p] for p in sampled],
            ignore_index=True,
        )

        for source in (LIKE_COL, DISLIKE_COL):
            source_df = sample_df.loc[sample_df["source"].eq(source)]
            if source_df.empty:
                continue

            category_mean = float(source_df["compound"].mean())
            product_means = source_df.groupby(PRODUCT_COL)["compound"].mean()

            for product in products:
                if product not in product_means.index:
                    continue

                product_mean = float(product_means.loc[product])
                deviation = (
                    product_mean - category_mean
                    if source == LIKE_COL
                    else category_mean - product_mean
                )
                stored[(product, source)].append(float(deviation))

    alpha = 1.0 - ci_level
    lower_q = 100.0 * alpha / 2.0
    upper_q = 100.0 * (1.0 - alpha / 2.0)

    rows: list[dict[str, object]] = []
    for (product, source), values in stored.items():
        arr = np.asarray(values, dtype=float)
        arr = arr[np.isfinite(arr)]

        if arr.size == 0:
            ci_low = np.nan
            ci_high = np.nan
            bootstrap_valid = 0
        else:
            ci_low = float(np.percentile(arr, lower_q))
            ci_high = float(np.percentile(arr, upper_q))
            bootstrap_valid = int(arr.size)

        rows.append(
            {
                PRODUCT_COL: product,
                "source": source,
                "ci_low": ci_low,
                "ci_high": ci_high,
                "bootstrap_valid_reps": bootstrap_valid,
            }
        )

    return pd.DataFrame(rows)

def add_profile_labels(summary: pd.DataFrame) -> pd.DataFrame:
    """
    Create descriptive quadrant labels from the SIGN of the two deviations.

    No small/medium/large magnitude thresholds are invented.
    """
    if summary.empty:
        return summary

    wide = summary.pivot(
        index=PRODUCT_COL,
        columns="source",
        values="relative_deviation",
    ).reset_index()

    if LIKE_COL not in wide.columns or DISLIKE_COL not in wide.columns:
        return summary

    labels: dict[str, str] = {}
    for _, row in wide.iterrows():
        product = str(row[PRODUCT_COL])
        like_dev = float(row[LIKE_COL])
        dislike_dev = float(row[DISLIKE_COL])

        if like_dev >= 0 and dislike_dev < 0:
            label = "More enthusiastic likes / milder dislikes"
        elif like_dev >= 0 and dislike_dev >= 0:
            label = "More enthusiastic likes / harsher dislikes"
        elif like_dev < 0 and dislike_dev < 0:
            label = "Less enthusiastic likes / milder dislikes"
        else:
            label = "Less enthusiastic likes / harsher dislikes"

        labels[product] = label

    out = summary.copy()
    out["profile_label"] = out[PRODUCT_COL].map(labels)
    return out

def build_answer(summary: pd.DataFrame) -> str:
    if summary.empty:
        return "No supported product-level sentiment profile was produced."

    likes = summary.loc[summary["source"].eq(LIKE_COL)].copy()
    dislikes = summary.loc[summary["source"].eq(DISLIKE_COL)].copy()

    if likes.empty or dislikes.empty:
        return (
            "Product-level sentiment scoring was incomplete because one prompted comment source "
            "was unavailable."
        )

    like_top = likes.loc[likes["relative_deviation"].idxmax()]
    dislike_harsh = dislikes.loc[dislikes["relative_deviation"].idxmax()]
    dislike_mild = dislikes.loc[dislikes["relative_deviation"].idxmin()]

    like_top_supported = bool(
        np.isfinite(like_top["ci_low"])
        and np.isfinite(like_top["ci_high"])
        and like_top["ci_low"] > 0
    )
    harsh_supported = bool(
        np.isfinite(dislike_harsh["ci_low"])
        and np.isfinite(dislike_harsh["ci_high"])
        and dislike_harsh["ci_low"] > 0
    )
    mild_supported = bool(
        np.isfinite(dislike_mild["ci_low"])
        and np.isfinite(dislike_mild["ci_high"])
        and dislike_mild["ci_high"] < 0
    )

    parts: list[str] = []

    if like_top_supported:
        parts.append(
            f"Product {like_top[PRODUCT_COL]} generated the strongest evidence of more enthusiastic "
            "like-language than the category average."
        )
    else:
        parts.append(
            f"Product {like_top[PRODUCT_COL]} had the highest observed like-language enthusiasm, "
            "although its bootstrap interval should be considered before treating that difference "
            "as clearly separated from the category average."
        )

    if harsh_supported:
        parts.append(
            f"Product {dislike_harsh[PRODUCT_COL]} generated clearly harsher dislike-language than "
            "the category average."
        )
    else:
        parts.append(
            f"Product {dislike_harsh[PRODUCT_COL]} had the harshest observed dislike-language profile."
        )

    if mild_supported:
        parts.append(
            f"Product {dislike_mild[PRODUCT_COL]} showed comparatively milder dislike-language."
        )

    parts.append(
        "These are product-relative language patterns from a generic sentiment model, not measures "
        "of acceptance or causal sensory drivers."
    )
    return " ".join(parts)

def run_analysis(
    df: pd.DataFrame,
    gate: GateResult,
    diagnostics: list[str],
    design_warnings: list[str],
    design_hard_stops: list[str],
) -> AnalysisResult:
    coverage = summarize_coverage(df) if all(c in df.columns for c in REQUIRED_COLUMNS) else pd.DataFrame()
    warnings = list(dict.fromkeys(gate.warnings + design_warnings))
    hard_stops = list(dict.fromkeys(gate.hard_stops + design_hard_stops))

    if hard_stops:
        return AnalysisResult(
            scored=pd.DataFrame(),
            product_summary=pd.DataFrame(),
            coverage_summary=coverage,
            answer="SOP-17 stopped because the product-level sentiment profile is not structurally supported.",
            model_status="HARD STOP",
            warnings=warnings,
            hard_stops=hard_stops,
            diagnostics=diagnostics,
        )

    comment_long = build_comment_long(df)
    if comment_long.empty:
        hard_stops.append("No usable like/dislike comments remained after preprocessing.")
        return AnalysisResult(
            scored=pd.DataFrame(),
            product_summary=pd.DataFrame(),
            coverage_summary=coverage,
            answer="SOP-17 stopped because no usable observed comments remained after preprocessing.",
            model_status="HARD STOP",
            warnings=warnings,
            hard_stops=hard_stops,
            diagnostics=diagnostics,
        )

    try:
        analyzer = initialize_vader()
    except RuntimeError as exc:
        hard_stops.append(str(exc))
        return AnalysisResult(
            scored=pd.DataFrame(),
            product_summary=pd.DataFrame(),
            coverage_summary=coverage,
            answer="SOP-17 could not run because the approved VADER resource is unavailable.",
            model_status="HARD STOP",
            warnings=warnings,
            hard_stops=hard_stops,
            diagnostics=diagnostics,
        )

    scored = score_comments(comment_long, analyzer)
    point_summary = calculate_product_relative_summary(scored)
    products = safe_product_sort(scored[PRODUCT_COL])

    try:
        boot = bootstrap_relative_deviations(scored, products)
    except RuntimeError as exc:
        hard_stops.append(str(exc))
        return AnalysisResult(
            scored=scored,
            product_summary=pd.DataFrame(),
            coverage_summary=coverage,
            answer="SOP-17 stopped because consumer-level bootstrap uncertainty could not be estimated.",
            model_status="HARD STOP",
            warnings=warnings,
            hard_stops=hard_stops,
            diagnostics=diagnostics,
        )

    summary = point_summary.merge(
        boot,
        on=[PRODUCT_COL, "source"],
        how="left",
        validate="one_to_one",
    )
    summary = add_profile_labels(summary)

    invalid_boot = summary.loc[
        summary["bootstrap_valid_reps"].fillna(0).astype(int) < BOOTSTRAP_REPS
    ]
    if not invalid_boot.empty:
        warnings.append(
            "At least one product × prompt had fewer valid bootstrap replicates than requested; "
            "review product-level missingness and the diagnostics table."
        )

    diagnostics.extend(
        [
            f"VADER scored {len(scored):,} observed comments.",
            f"Consumer-level bootstrap replicates requested: {BOOTSTRAP_REPS:,}; random seed = {RANDOM_STATE}.",
            "Category baselines are calculated separately for likes and dislikes.",
            "Like deviation = product mean VADER compound − category like mean.",
            "Dislike severity deviation = category dislike mean − product mean VADER compound; positive = harsher language.",
            "95% percentile bootstrap intervals quantify uncertainty in the product-relative descriptive metric.",
            "No p-values, Holm correction, significance letters, or generic small/medium/large sentiment thresholds are used.",
            "Like and dislike prompts are never pooled into a synthetic acceptance score.",
        ]
    )

    return AnalysisResult(
        scored=scored,
        product_summary=summary,
        coverage_summary=coverage,
        answer=build_answer(summary),
        model_status="COMPLETED — WARNING" if warnings else "COMPLETED",
        warnings=list(dict.fromkeys(warnings)),
        hard_stops=[],
        diagnostics=diagnostics,
    )

def alert_box(title: str, items: list[str], kind: str = "warning") -> str:
    if not items:
        return ""
    lis = "".join(f"<li>{h(item)}</li>" for item in items)
    return f"""
    <div class="alert {kind}">
        <strong>{h(title)}</strong>
        <ul>{lis}</ul>
    </div>
    """

def build_kpis(result: AnalysisResult) -> str:
    if result.product_summary.empty:
        return "".join(
            [
                kpi_card("Analysis status", "Hard stop", "See diagnostic explanation"),
                kpi_card("Comments scored", "0", "Product profile not released"),
                kpi_card("Bootstrap", "Not run", "Consumer-level uncertainty unavailable"),
                kpi_card("Interpretation", "Suppressed", "Guardrail preserved"),
            ]
        )

    summary = result.product_summary
    likes = summary.loc[summary["source"].eq(LIKE_COL)]
    dislikes = summary.loc[summary["source"].eq(DISLIKE_COL)]

    like_top = likes.loc[likes["relative_deviation"].idxmax()]
    harsh = dislikes.loc[dislikes["relative_deviation"].idxmax()]
    mild = dislikes.loc[dislikes["relative_deviation"].idxmin()]

    return "".join(
        [
            kpi_card(
                "Most enthusiastic likes",
                f"Product {like_top[PRODUCT_COL]}",
                f"Δ {like_top['relative_deviation']:+.3f} vs category",
            ),
            kpi_card(
                "Harshest dislike language",
                f"Product {harsh[PRODUCT_COL]}",
                f"Δ {harsh['relative_deviation']:+.3f} severity",
            ),
            kpi_card(
                "Mildest dislike language",
                f"Product {mild[PRODUCT_COL]}",
                f"Δ {mild['relative_deviation']:+.3f} severity",
            ),
            kpi_card(
                "Bootstrap",
                f"{BOOTSTRAP_REPS:,} reps",
                "Whole-consumer resampling",
            ),
        ]
    )

def main() -> int:
    args = parse_args()

    try:
        data_path = discover_data_file(args.data_path)
    except FileNotFoundError as exc:
        print(str(exc), file=sys.stderr)
        return 2

    try:
        df, removed_index_columns = load_data(data_path)
    except Exception as exc:
        print(f"Could not read CSV:\n{exc}", file=sys.stderr)
        return 2

    gate = dataset_capability_gate(df)

    if all(c in df.columns for c in REQUIRED_COLUMNS):
        diagnostics, design_warnings, design_hard_stops = validate_study_design(
            df,
            removed_index_columns,
        )
    else:
        diagnostics = [f"Rows loaded: {len(df):,}."]
        design_warnings = []
        design_hard_stops = []

    result = run_analysis(
        df=df,
        gate=gate,
        diagnostics=diagnostics,
        design_warnings=design_warnings,
        design_hard_stops=design_hard_stops,
    )

    script_dir = Path(__file__).resolve().parent if "__file__" in globals() else Path.cwd()
    output_dir = script_dir / OUTPUT_DIR
    output_dir.mkdir(parents=True, exist_ok=True)
    dashboard_path = output_dir / DASHBOARD_FILENAME

    dashboard_path.write_text(
        build_dashboard(data_path, df, gate, result),
        encoding="utf-8",
    )

    save_optional_exports(output_dir, result)
    print_console_summary(dashboard_path, df, gate, result)

    if AUTO_OPEN_DASHBOARD and not args.no_open:
        try:
            webbrowser.open(dashboard_path.resolve().as_uri())
        except Exception:
            pass

    return 1 if result.hard_stops else 0

if __name__ == "__main__":
    raise SystemExit(main())

def panelist_text(
    value: object,
) -> str:
    """
    Make panelist identifiers readable without
    unnecessarily displaying .0 for integer IDs.
    """

    if pd.isna(
        value
    ):
        return "<missing>"

    try:

        result = float(
            value
        )

        if result.is_integer():
            return str(
                int(
                    result
                )
            )

    except Exception:
        pass

    return str(
        value
    )

def numeric_series(
    df: pd.DataFrame,
    column: str,
) -> pd.Series:
    """
    Convert a column to numeric without modifying
    the original dataframe datatype.
    """

    if column not in df.columns:

        return pd.Series(
            np.nan,
            index=df.index,
            dtype=float,
        )

    return pd.to_numeric(
        df[
            column
        ],
        errors="coerce",
    )

def load_data(
    path: Path,
) -> pd.DataFrame:

    df = pd.read_csv(
        path
    )

    # Remove only clearly accidental
    # exported dataframe index columns.
    #
    # A legitimate field literally named
    # "index" is preserved.

    accidental_columns = [
        column
        for column in df.columns
        if str(
            column
        ).startswith(
            "Unnamed:"
        )
    ]

    if accidental_columns:

        df = df.drop(
            columns=(
                accidental_columns
            )
        )

    return df

def validate_study_design(
    df: pd.DataFrame,
) -> StudyDesignResult:

    result = (
        StudyDesignResult()
    )

    result.n_rows = int(
        len(
            df
        )
    )

    result.n_panelists = int(
        df[
            PANELIST
        ].nunique(
            dropna=True
        )
    )

    result.n_products = int(
        df[
            PRODUCT
        ].nunique(
            dropna=True
        )
    )

    result.expected_products = (
        result.n_products
    )

    duplicate_mask = (
        df.duplicated(
            [
                PANELIST,
                PRODUCT,
            ],
            keep=False,
        )
    )

    result.duplicate_pair_rows = int(
        duplicate_mask.sum()
    )

    counts = (
        df
        .dropna(
            subset=[
                PANELIST,
                PRODUCT,
            ]
        )
        .groupby(
            PANELIST
        )[
            PRODUCT
        ]
        .nunique()
    )

    if not counts.empty:

        result.min_products_per_panelist = int(
            counts.min()
        )

        result.max_products_per_panelist = int(
            counts.max()
        )

        result.incomplete_panelists = [
            panelist_text(
                value
            )
            for value
            in counts[
                counts
                < result.expected_products
            ].index.tolist()
        ]

    for (
        column,
        scale,
    ) in KNOWN_SCALE_RANGES.items():

        if column not in df.columns:
            continue

        low, high = scale

        values = numeric_series(
            df,
            column,
        )

        invalid = (
            values.notna()
            & (
                (
                    values
                    < low
                )
                |
                (
                    values
                    > high
                )
            )
        )

        result.scale_violations[
            column
        ] = int(
            invalid.sum()
        )

    return result

def available_qc_numeric_columns(
    df: pd.DataFrame,
) -> List[
    str
]:

    preferred = (
        [
            LIKING,
            FLAVOR_IMPACT,
        ]
        + JAR_COLUMNS
        + DESCRIPTIVE_COLUMNS
    )

    available: List[
        str
    ] = []

    for column in preferred:

        if column not in df.columns:
            continue

        converted = (
            numeric_series(
                df,
                column,
            )
        )

        if converted.notna().any():

            available.append(
                column
            )

    return available

def add_within_product_z(
    df: pd.DataFrame,
    value_column: str,
) -> pd.Series:
    """
    Standardize each liking rating relative to
    consumers evaluating the same product.

    This avoids treating a legitimately low-rated
    product as though all low scores on that product
    are respondent anomalies.
    """

    values = numeric_series(
        df,
        value_column,
    )

    working = pd.DataFrame(
        {
            PRODUCT: (
                df[
                    PRODUCT
                ]
            ),
            "value": (
                values
            ),
        }
    )

    product_means = (
        working
        .groupby(
            PRODUCT
        )[
            "value"
        ]
        .transform(
            "mean"
        )
    )

    product_sd = (
        working
        .groupby(
            PRODUCT
        )[
            "value"
        ]
        .transform(
            "std"
        )
    )

    product_sd = (
        product_sd
        .replace(
            0,
            np.nan,
        )
    )

    standardized = (
        (
            values
            - product_means
        )
        / product_sd
    )

    return standardized

def make_liking_matrix(
    df: pd.DataFrame,
) -> Optional[
    pd.DataFrame
]:

    if LIKING not in df.columns:

        return None

    working = (
        df[
            [
                PANELIST,
                PRODUCT,
                LIKING,
            ]
        ]
        .copy()
    )

    working[
        LIKING
    ] = pd.to_numeric(
        working[
            LIKING
        ],
        errors="coerce",
    )

    duplicate_cells = (
        working
        .duplicated(
            [
                PANELIST,
                PRODUCT,
            ],
            keep=False,
        )
        .any()
    )

    if duplicate_cells:

        # Do not silently average duplicate
        # panelist × product evaluations.
        return None

    matrix = (
        working
        .pivot(
            index=PANELIST,
            columns=PRODUCT,
            values=LIKING,
        )
    )

    ordered_columns = sorted(
        matrix.columns,
        key=lambda value: str(
            value
        ),
    )

    matrix = matrix[
        ordered_columns
    ]

    return matrix

def compute_pca_qc(
    liking_matrix: Optional[
        pd.DataFrame
    ],
) -> Tuple[
    Optional[
        pd.DataFrame
    ],
    Optional[
        float
    ],
]:

    if liking_matrix is None:

        return (
            None,
            None,
        )

    if (
        liking_matrix.shape[
            1
        ]
        < 2
    ):

        return (
            None,
            None,
        )

    complete = (
        liking_matrix
        .dropna(
            axis=0,
            how="any",
        )
    )

    if len(
        complete
    ) < 5:

        return (
            None,
            None,
        )

    across_consumer_sd = (
        complete.std(
            axis=0,
            ddof=1,
        )
    )

    if (
        across_consumer_sd
        == 0
    ).any():

        # Do not silently remove a zero-variance
        # product from respondent-profile PCA.
        return (
            None,
            None,
        )

    scaler = (
        StandardScaler()
    )

    X = scaler.fit_transform(
        complete.to_numpy(
            dtype=float
        )
    )

    n_components = min(
        2,
        X.shape[
            0
        ],
        X.shape[
            1
        ],
    )

    if n_components < 2:

        return (
            None,
            None,
        )

    pca = PCA(
        n_components=2,
        random_state=(
            RANDOM_STATE
        ),
    )

    scores = (
        pca.fit_transform(
            X
        )
    )

    pca_df = pd.DataFrame(
        scores,
        index=(
            complete.index
        ),
        columns=[
            "PC1",
            "PC2",
        ],
    )

    pca_df[
        "distance"
    ] = np.sqrt(
        (
            pca_df[
                "PC1"
            ]
            ** 2
        )
        +
        (
            pca_df[
                "PC2"
            ]
            ** 2
        )
    )

    q1 = float(
        pca_df[
            "distance"
        ].quantile(
            0.25
        )
    )

    q3 = float(
        pca_df[
            "distance"
        ].quantile(
            0.75
        )
    )

    iqr = (
        q3
        - q1
    )

    threshold = (
        q3
        + (
            PCA_DISTANCE_IQR_MULTIPLIER
            * iqr
        )
    )

    pca_df[
        "pca_tail_flag"
    ] = (
        pca_df[
            "distance"
        ]
        > threshold
    )

    return (
        pca_df,
        threshold,
    )

def respondent_level_qc(
    df: pd.DataFrame,
    design: StudyDesignResult,
) -> QCResult:

    numeric_columns = (
        available_qc_numeric_columns(
            df
        )
    )

    if LIKING in df.columns:

        liking_z = (
            add_within_product_z(
                df,
                LIKING,
            )
        )

    else:

        liking_z = (
            pd.Series(
                np.nan,
                index=df.index,
            )
        )

    working = (
        df.copy()
    )

    working[
        "__liking_z__"
    ] = liking_z

    liking_matrix = (
        make_liking_matrix(
            df
        )
    )

    (
        pca_scores,
        pca_threshold,
    ) = compute_pca_qc(
        liking_matrix
    )

    records: List[
        Dict[
            str,
            object,
        ]
    ] = []

    for (
        panelist,
        group,
    ) in working.groupby(
        PANELIST,
        dropna=False,
    ):

        label = (
            panelist_text(
                panelist
            )
        )

        structural_reasons: List[
            str
        ] = []

        pattern_reasons: List[
            str
        ] = []

        evaluation_rows = int(
            len(
                group
            )
        )

        unique_products = int(
            group[
                PRODUCT
            ].nunique(
                dropna=True
            )
        )

        duplicate_cells = int(
            group
            .duplicated(
                [
                    PANELIST,
                    PRODUCT,
                ],
                keep=False,
            )
            .sum()
        )

        if (
            unique_products
            < design.expected_products
        ):

            structural_reasons.append(
                (
                    "Incomplete product set: "
                    f"{unique_products}/"
                    f"{design.expected_products} "
                    "unique products."
                )
            )

        if duplicate_cells > 0:

            structural_reasons.append(
                (
                    "Duplicate panelist × product "
                    f"structure: {duplicate_cells} "
                    "row(s) involved."
                )
            )

        # ---------------------------------------------------------------------
        # Numeric-response missingness
        # ---------------------------------------------------------------------

        if numeric_columns:

            numeric_block = (
                group[
                    numeric_columns
                ]
                .apply(
                    pd.to_numeric,
                    errors="coerce",
                )
            )

        else:

            numeric_block = (
                pd.DataFrame(
                    index=(
                        group.index
                    )
                )
            )

        numeric_total = int(
            numeric_block.size
        )

        if numeric_total:

            numeric_missing = int(
                numeric_block
                .isna()
                .sum()
                .sum()
            )

            missing_pct = (
                100.0
                * numeric_missing
                / numeric_total
            )

        else:

            numeric_missing = 0

            missing_pct = float(
                "nan"
            )

        if numeric_missing > 0:

            pattern_reasons.append(
                (
                    "Missing measured numeric "
                    f"responses: {numeric_missing}/"
                    f"{numeric_total} "
                    f"({missing_pct:.1f}%)."
                )
            )

        # ---------------------------------------------------------------------
        # Overall liking scale-use summaries
        # ---------------------------------------------------------------------

        if LIKING in group.columns:

            liking = (
                numeric_series(
                    group,
                    LIKING,
                )
                .dropna()
            )

        else:

            liking = (
                pd.Series(
                    dtype=float
                )
            )

        liking_n = int(
            liking.size
        )

        if liking_n:

            liking_mean = float(
                liking.mean()
            )

            liking_min = float(
                liking.min()
            )

            liking_max = float(
                liking.max()
            )

            liking_range = (
                liking_max
                - liking_min
            )

            liking_unique = int(
                liking.nunique()
            )

            endpoint_share = float(
                (
                    (
                        liking
                        == 1
                    )
                    |
                    (
                        liking
                        == 9
                    )
                )
                .mean()
            )

        else:

            liking_mean = float(
                "nan"
            )

            liking_min = float(
                "nan"
            )

            liking_max = float(
                "nan"
            )

            liking_range = float(
                "nan"
            )

            liking_unique = 0

            endpoint_share = float(
                "nan"
            )

        if liking_n >= 2:

            liking_sd = float(
                liking.std(
                    ddof=1
                )
            )

        else:

            liking_sd = float(
                "nan"
            )

        # Exact invariant responding is directly observable
        # and does not require an arbitrary low-SD cutoff.

        if (
            liking_n
            >= 2
            and liking_unique
            == 1
        ):

            pattern_reasons.append(
                (
                    "Invariant overall liking: "
                    "the same hedonic score was "
                    "used for every observed product."
                )
            )

        # ---------------------------------------------------------------------
        # Within-product standardized liking screen
        # ---------------------------------------------------------------------

        absolute_z = (
            group[
                "__liking_z__"
            ]
            .abs()
        )

        if absolute_z.notna().any():

            max_abs_z = float(
                absolute_z.max()
            )

        else:

            max_abs_z = float(
                "nan"
            )

        n_z_flags = int(
            (
                absolute_z
                >= WITHIN_PRODUCT_Z_REVIEW
            )
            .sum()
        )

        if n_z_flags > 0:

            pattern_reasons.append(
                (
                    "Unusual within-product "
                    "liking value(s): "
                    f"{n_z_flags} rating(s) "
                    "had |z| ≥ "
                    f"{WITHIN_PRODUCT_Z_REVIEW:.1f}."
                )
            )

        # ---------------------------------------------------------------------
        # PCA profile-tail screen
        # ---------------------------------------------------------------------

        pca_distance = float(
            "nan"
        )

        pca_tail = False

        if (
            pca_scores
            is not None
            and panelist
            in pca_scores.index
        ):

            pca_distance = float(
                pca_scores.loc[
                    panelist,
                    "distance",
                ]
            )

            pca_tail = bool(
                pca_scores.loc[
                    panelist,
                    "pca_tail_flag",
                ]
            )

            if pca_tail:

                pattern_reasons.append(
                    (
                        "Liking profile lies in "
                        "the outer tail of the "
                        "two-dimensional PCA "
                        "score-distance screen."
                    )
                )

        all_reasons = (
            structural_reasons
            + pattern_reasons
        )

        records.append(
            {
                PANELIST: (
                    panelist
                ),
                "panelist_label": (
                    label
                ),
                "evaluation_rows": (
                    evaluation_rows
                ),
                "unique_products": (
                    unique_products
                ),
                "numeric_missing_n": (
                    numeric_missing
                ),
                "numeric_missing_pct": (
                    missing_pct
                ),
                "liking_n": (
                    liking_n
                ),
                "liking_mean": (
                    liking_mean
                ),
                "liking_sd": (
                    liking_sd
                ),
                "liking_min": (
                    liking_min
                ),
                "liking_max": (
                    liking_max
                ),
                "liking_range": (
                    liking_range
                ),
                "liking_unique_values": (
                    liking_unique
                ),
                "liking_endpoint_share": (
                    endpoint_share
                ),
                "max_abs_within_product_z": (
                    max_abs_z
                ),
                "n_abs_z_ge_3": (
                    n_z_flags
                ),
                "pca_distance": (
                    pca_distance
                ),
                "pca_tail_flag": (
                    pca_tail
                ),
                "structural_flag": (
                    bool(
                        structural_reasons
                    )
                ),
                "review_flag": (
                    bool(
                        all_reasons
                    )
                ),
                "flag_count": (
                    len(
                        all_reasons
                    )
                ),
                "flag_reasons": (

def build_answer(
    capability: CapabilityResult,
    design: StudyDesignResult,
    qc: Optional[
        QCResult
    ],
) -> str:

    if (
        capability.status
        == "UNSUPPORTED"
    ):

        missing = ", ".join(
            capability.missing_core
        )

        return (
            "SOP-18 cannot run because "
            "required respondent-structure "
            "field(s) are missing: "
            f"{missing}."
        )

    if qc is None:

        return (
            "Respondent-level QC could "
            "not be completed."
        )

    n_flagged = len(
        qc.flagged_panelists
    )

    n_structural = len(
        qc.structural_panelists
    )

    if n_flagged == 0:

        return (
            "No respondents were flagged "
            "by the available structural "
            "and response-pattern screens. "
            "This does not prove every "
            "response is valid; it means "
            "no unusual pattern crossed "
            "the review screens used here."
        )

    if n_structural > 0:

        return (
            f"{n_flagged} respondent(s) "
            "were flagged for review, "
            f"including {n_structural} "
            "with structural completeness "
            "or duplication issues. "
            "These are QC flags only; "
            "SOP-18 does not authorize "
            "automatic exclusion."
        )

    return (
        f"{n_flagged} respondent(s) "
        "showed unusual response-pattern "
        "signals worth reviewing. "
        "No respondent was automatically "
        "excluded because unusual consumer "
        "preference is not itself evidence "
        "of invalid data."
    )

def sign_with_tolerance(value: float, tolerance: float = 1e-12) -> int:
    """Return -1, 0, or +1 for a finite value."""
    if not np.isfinite(value):
        return 0
    if value > tolerance:
        return 1
    if value < -tolerance:
        return -1
    return 0

def strength_label(r: float) -> str:
    """
    Apply the project G9-H interpretation convention.

    These thresholds are project conventions, not universal statistical laws.
    """
    if not np.isfinite(r):
        return "Not estimable"

    absolute = abs(r)

    if absolute < RHO_NEGLIGIBLE:
        return "Negligible / very weak"
    if absolute < RHO_WEAK:
        return "Weak"
    if absolute < RHO_MODERATE:
        return "Moderate"
    return "Strong"

def direction_label(r: float) -> str:
    """Plain-language direction label."""
    if not np.isfinite(r):
        return "not estimable"
    if r > 0:
        return "positive"
    if r < 0:
        return "negative"
    return "no directional"

def load_dataset(path: Path) -> pd.DataFrame:
    """Load the CSV and remove obvious accidental unnamed index columns."""
    df = pd.read_csv(path)

def run_capability_gate(df: pd.DataFrame) -> CapabilityResult:
    """
    SOP-19 requires directly measured variables.

    No missing construct is replaced with another variable.
    """
    required_structural = ["panelist_id", "product_code"]

    available_analysis_variables = [
        variable
        for variable in DEFAULT_CORRELATION_VARIABLES
        if variable in df.columns
    ]

    missing_analysis_variables = [
        variable
        for variable in DEFAULT_CORRELATION_VARIABLES
        if variable not in df.columns
    ]

    missing_structural = [
        variable
        for variable in required_structural
        if variable not in df.columns
    ]

    messages: list[str] = []

    if missing_structural:
        messages.append(
            "Required structural field(s) missing: "
            + ", ".join(missing_structural)
            + "."
        )

        return CapabilityResult(
            status="UNSUPPORTED",
            available_variables=available_analysis_variables,
            missing_variables=missing_structural + missing_analysis_variables,
            messages=messages,
        )

    # Need at least two approved measured variables to calculate an association.
    if len(available_analysis_variables) < 2:
        messages.append(
            "Fewer than two approved measured correlation variables are "
            "available. SOP-19 cannot calculate an association."
        )

        return CapabilityResult(
            status="UNSUPPORTED",
            available_variables=available_analysis_variables,
            missing_variables=missing_analysis_variables,
            messages=messages,
        )

    if missing_analysis_variables:
        messages.append(
            "Some variables in the configured SOP-19 sweep are absent and "
            "will not be analyzed: "
            + ", ".join(missing_analysis_variables)
            + "."
        )
        status = "PARTIALLY SUPPORTED"
    else:
        status = "SUPPORTED"

    nonnumeric = []

    for variable in available_analysis_variables:
        converted = pd.to_numeric(df[variable], errors="coerce")

        # If all nonmissing source values fail numeric conversion, the variable
        # is not usable as configured.
        source_nonmissing = int(df[variable].notna().sum())
        converted_nonmissing = int(converted.notna().sum())

        if source_nonmissing > 0 and converted_nonmissing == 0:
            nonnumeric.append(variable)

    if nonnumeric:
        messages.append(
            "Configured variables that are present but nonnumeric were "
            "excluded: "
            + ", ".join(nonnumeric)
            + "."
        )

        available_analysis_variables = [
            v for v in available_analysis_variables if v not in nonnumeric
        ]

        missing_analysis_variables.extend(nonnumeric)

        if len(available_analysis_variables) < 2:
            status = "UNSUPPORTED"
        else:
            status = "PARTIALLY SUPPORTED"

    if status == "SUPPORTED":
        messages.append(
            "All configured SOP-19 variables and structural identifiers "
            "required for this analysis are present."
        )

    return CapabilityResult(
        status=status,
        available_variables=available_analysis_variables,
        missing_variables=missing_analysis_variables,
        messages=messages,
    )

def validate_study_design(df: pd.DataFrame) -> DesignResult:
    """Validate the repeated-measures product-evaluation structure."""
    messages: list[str] = []

    n_rows = len(df)
    n_panelists = int(df["panelist_id"].nunique(dropna=True))
    n_products = int(df["product_code"].nunique(dropna=True))

    duplicate_cells = int(
        df.duplicated(subset=["panelist_id", "product_code"]).sum()
    )

    evaluations_per_panelist = (
        df.groupby("panelist_id", dropna=True)["product_code"].nunique()
    )

    repeated_measures = bool(
        len(evaluations_per_panelist) > 0
        and evaluations_per_panelist.max() > 1
    )

    complete_panelists = int(
        (evaluations_per_panelist == n_products).sum()
    )

    total_panelists = int(len(evaluations_per_panelist))

    status = "PASS"

    if duplicate_cells > 0:
        status = "HARD STOP"
        messages.append(
            f"Found {duplicate_cells} duplicate panelist × product cells. "
            "The analytical grain is ambiguous."
        )

    if not repeated_measures:
        if status != "HARD STOP":
            status = "WARNING"
        messages.append(
            "The dataset does not appear to contain repeated product "
            "evaluations within panelist."
        )
    else:
        messages.append(
            "Repeated measures confirmed: the same panelists evaluated "
            "multiple products."
        )

    if complete_panelists < total_panelists:
        if status == "PASS":
            status = "WARNING"
        messages.append(
            f"{complete_panelists} of {total_panelists} panelists have "
            f"evaluations for all {n_products} observed products."
        )
    else:
        messages.append(
            f"All {total_panelists} panelists have evaluations for all "
            f"{n_products} observed products."
        )

    if n_products < 2:
        status = "HARD STOP"
        messages.append(
            "Fewer than two products are present, so product-structure "
            "diagnostics cannot be performed."
        )

    return DesignResult(
        status=status,
        n_rows=n_rows,
        n_panelists=n_panelists,
        n_products=n_products,
        duplicate_cells=duplicate_cells,
        repeated_measures=repeated_measures,
        complete_panelists=complete_panelists,
        total_panelists=total_panelists,
        messages=messages,
    )

def prepare_analysis_data(
    df: pd.DataFrame,
    variables: list[str],
) -> pd.DataFrame:
    """
    Preserve original dataframe structure while coercing configured analysis
    variables safely to numeric.
    """
    analysis_df = df.copy()

    analysis_df["product_code"] = analysis_df["product_code"].astype(str)

    for variable in variables:
        analysis_df[variable] = pd.to_numeric(
            analysis_df[variable],
            errors="coerce",
        )

    return analysis_df

def variance_eligible_variables(
    df: pd.DataFrame,
    variables: list[str],
) -> tuple[list[str], list[str]]:
    """Remove no-variance variables with explicit disclosure."""
    eligible: list[str] = []
    excluded: list[str] = []

    for variable in variables:
        values = df[variable].dropna()

        if len(values) < MIN_PAIR_N:
            excluded.append(variable)
            continue

        if values.nunique() < 2:
            excluded.append(variable)
            continue

        eligible.append(variable)

    return eligible, excluded

def calculate_correlation(
    x: pd.Series,
    y: pd.Series,
) -> tuple[float, float, int]:
    """
    Calculate a pairwise-complete correlation.

    Returns:
        correlation coefficient
        p-value
        paired N
    """
    pair = pd.DataFrame({"x": x, "y": y}).dropna()

    n = len(pair)

    if n < MIN_PAIR_N:
        return np.nan, np.nan, n

    if pair["x"].nunique() < 2 or pair["y"].nunique() < 2:
        return np.nan, np.nan, n

    with warnings.catch_warnings():
        warnings.simplefilter("ignore")

        if CORRELATION_METHOD == "spearman":
            coefficient, p_value = stats.spearmanr(
                pair["x"],
                pair["y"],
                nan_policy="omit",
            )
        elif CORRELATION_METHOD == "pearson":
            coefficient, p_value = stats.pearsonr(
                pair["x"],
                pair["y"],
            )
        else:
            raise ValueError(
                "CORRELATION_METHOD must be 'spearman' or 'pearson'."
            )

    return float(coefficient), float(p_value), int(n)

def all_unique_pairs(
    variables: list[str],
) -> list[tuple[str, str]]:
    """Return all unique unordered pairs."""
    pairs = []

    for i, first in enumerate(variables):
        for second in variables[i + 1:]:
            pairs.append((first, second))

    return pairs

def calculate_within_product_results(
    df: pd.DataFrame,
    var_x: str,
    var_y: str,
) -> pd.DataFrame:
    """Calculate the requested association separately within each product."""
    rows = []

    for product, product_df in df.groupby("product_code", sort=True):
        r, p_value, n = calculate_correlation(
            product_df[var_x],
            product_df[var_y],
        )

        eligible = (
            n >= MIN_WITHIN_PRODUCT_N
            and np.isfinite(r)
        )

        rows.append(
            {
                "product_code": product,
                "var_x": var_x,
                "var_y": var_y,
                "n": n,
                "correlation": r,
                "p_value": p_value,
                "eligible": eligible,
                "strength": (
                    strength_label(r)
                    if eligible
                    else "Not estimable"
                ),
            }
        )

    return pd.DataFrame(rows)

def count_direction_reversals(
    pooled_r: float,
    within: pd.DataFrame,
) -> int:
    """
    Count products whose correlation direction opposes the pooled direction.

    A zero or non-estimable coefficient is not counted as a reversal.
    """
    pooled_sign = sign_with_tolerance(pooled_r)

    if pooled_sign == 0:
        return 0

    count = 0

    for value in within.loc[
        within["eligible"], "correlation"
    ]:
        product_sign = sign_with_tolerance(float(value))

        if product_sign != 0 and product_sign != pooled_sign:
            count += 1

    return count

def build_pair_results(
    df: pd.DataFrame,
    variables: list[str],
) -> tuple[pd.DataFrame, pd.DataFrame]:
    """
    Calculate pooled and within-product associations for the complete family.
    """
    pair_rows: list[dict] = []
    within_frames: list[pd.DataFrame] = []

    configured_pairs = all_unique_pairs(variables)

    for var_x, var_y in configured_pairs:
        pooled_r, pooled_p, pooled_n = calculate_correlation(
            df[var_x],
            df[var_y],
        )

        within = calculate_within_product_results(
            df,
            var_x,
            var_y,
        )

        within_frames.append(within)

        valid_within = within[
            within["eligible"] & within["correlation"].notna()
        ]

        if len(valid_within) > 0:
            median_within = float(
                valid_within["correlation"].median()
            )
            min_within = float(
                valid_within["correlation"].min()
            )
            max_within = float(
                valid_within["correlation"].max()
            )
        else:
            median_within = np.nan
            min_within = np.nan
            max_within = np.nan

        reversals = count_direction_reversals(
            pooled_r,
            valid_within,
        )

        if np.isfinite(pooled_r) and np.isfinite(median_within):
            pooled_within_difference = abs(
                pooled_r - median_within
            )
        else:
            pooled_within_difference = np.nan

        product_structure_flag = bool(
            (
                np.isfinite(pooled_within_difference)
                and pooled_within_difference
                > POOLED_WITHIN_DIFFERENCE_WARNING
            )
            or reversals >= MIN_DIRECTION_REVERSALS_WARNING
        )

        pair_rows.append(
            {
                "var_x": var_x,
                "var_y": var_y,
                "pooled_r": pooled_r,
                "pooled_p": pooled_p,
                "pooled_n": pooled_n,
                "median_within_r": median_within,
                "min_within_r": min_within,
                "max_within_r": max_within,
                "valid_product_count": int(len(valid_within)),
                "direction_reversal_count": int(reversals),
                "pooled_within_difference": pooled_within_difference,
                "product_structure_flag": product_structure_flag,
                "strength_label": strength_label(median_within),
            }
        )

    pair_df = pd.DataFrame(pair_rows)

    within_df = (
        pd.concat(within_frames, ignore_index=True)
        if within_frames
        else pd.DataFrame()
    )

    return pair_df, within_df

def apply_holm_correction(pair_df: pd.DataFrame) -> pd.DataFrame:
    """
    Apply Holm correction to the family of pooled correlation tests.

    SOP-19 is exploratory. Adjusted p-values are supportive evidence rather
    than permission to make causal or confirmatory claims.
    """
    result = pair_df.copy()

    result["adjusted_p"] = np.nan
    result["significant_adjusted"] = False

    eligible = (
        result["pooled_p"].notna()
        & np.isfinite(result["pooled_p"])
    )

    if eligible.sum() == 0:
        return result

    reject, adjusted, _, _ = multipletests(
        result.loc[eligible, "pooled_p"].to_numpy(),
        alpha=ALPHA,
        method="holm",
    )

    result.loc[eligible, "adjusted_p"] = adjusted
    result.loc[eligible, "significant_adjusted"] = reject

    return result

def build_pair_interpretation(row: pd.Series) -> str:
    """Generate a guarded plain-language interpretation."""
    x = variable_label(row["var_x"])
    y = variable_label(row["var_y"])

    pooled = row["pooled_r"]
    within = row["median_within_r"]

    if not np.isfinite(pooled):
        return (
            f"The association between {x} and {y} could not be "
            "estimated reliably."
        )

    if not np.isfinite(within):
        return (
            f"The pooled {correlation_symbol()} was "
            f"{pooled:.2f}, but there were insufficient estimable "
            "within-product associations to determine whether the pattern "
            "persists within products."
        )

    strength = strength_label(within).lower()
    direction = direction_label(within)

    if bool(row["product_structure_flag"]):
        return (
            f"The pooled relationship between {x} and {y} does not "
            "closely represent the typical within-product relationship. "
            f"Median within-product {correlation_symbol()} = "
            f"{within:.2f} ({strength}, {direction}). Product identity "
            "may be contributing materially to the pooled pattern."
        )

    return (
        f"{x} and {y} showed a {strength} {direction} exploratory "
        f"association within products "
        f"(median {correlation_symbol()} = {within:.2f}). "
        "This is associative evidence only."
    )

def add_interpretations(pair_df: pd.DataFrame) -> pd.DataFrame:
    result = pair_df.copy()
    result["interpretation"] = result.apply(
        build_pair_interpretation,
        axis=1,
    )
    return result

def select_notable_pairs(pair_df: pd.DataFrame) -> pd.DataFrame:
    """
    Rank primarily by absolute median within-product association.

    This intentionally avoids ranking sensory variables by pooled correlation
    alone.
    """
    ranked = pair_df.copy()

    ranked["abs_within"] = ranked["median_within_r"].abs()
    ranked["abs_pooled"] = ranked["pooled_r"].abs()

    ranked = ranked.sort_values(
        by=[
            "abs_within",
            "abs_pooled",
        ],
        ascending=[False, False],
        na_position="last",
    )

    return ranked.head(TOP_PAIR_COUNT)

def target_pair_mask(
    pair_df: pd.DataFrame,
    pair: tuple[str, str],
) -> pd.Series:
    a, b = pair

    return (
        ((pair_df["var_x"] == a) & (pair_df["var_y"] == b))
        | ((pair_df["var_x"] == b) & (pair_df["var_y"] == a))
    )

def choose_lead_pair(pair_df: pd.DataFrame) -> pd.Series | None:
    """
    Choose the lead pair.

    Explicit TARGET_PAIRS have priority. Otherwise choose the strongest
    estimable median within-product association.
    """
    for pair in TARGET_PAIRS:
        matches = pair_df[target_pair_mask(pair_df, pair)]

        if len(matches) > 0:
            return matches.iloc[0]

    eligible = pair_df[pair_df["median_within_r"].notna()].copy()

    if len(eligible) == 0:
        return None

    eligible["absolute_within"] = eligible["median_within_r"].abs()

    return eligible.sort_values(
        "absolute_within",
        ascending=False,
    ).iloc[0]

def build_overall_answer(
    pair_df: pd.DataFrame,
    lead_pair: pd.Series | None,
) -> str:
    if lead_pair is None:
        return (
            "No correlation pair had sufficient usable information for a "
            "defensible within-product exploratory interpretation."
        )

    x = variable_label(lead_pair["var_x"])
    y = variable_label(lead_pair["var_y"])
    r = float(lead_pair["median_within_r"])

    strength = strength_label(r).lower()
    direction = direction_label(r)

    if bool(lead_pair["product_structure_flag"]):
        return (
            f"The strongest observed within-product pattern was between "
            f"{x} and {y}, but pooled and within-product evidence differed "
            "enough that product structure must be considered."
        )

    return (
        f"The strongest typical within-product association was between "
        f"{x} and {y}: a {strength} {direction} exploratory relationship."
    )

def correlation_fill(value: float) -> str:
    """
    Return restrained diverging fills.

    No external plotting dependency is needed.
    """
    if not np.isfinite(value):
        return "#F3F5F7"

    value = max(-1.0, min(1.0, float(value)))

    # Negative: muted rust.
    if value < 0:
        intensity = abs(value)
        light = np.array([250, 245, 242])
        dark = np.array([176, 91, 58])
    else:
        # Positive: muted teal.
        intensity = value
        light = np.array([244, 248, 248])
        dark = np.array([30, 91, 99])

    rgb = (
        light * (1.0 - intensity)
        + dark * intensity
    ).astype(int)

    return "#{:02X}{:02X}{:02X}".format(*rgb)

def text_color_for_correlation(value: float) -> str:
    if np.isfinite(value) and abs(value) >= 0.55:
        return "#FFFFFF"
    return "#172533"

def build_within_product_table(
    within_df: pd.DataFrame,
    lead_pair: pd.Series | None,
) -> str:
    if lead_pair is None or within_df.empty:
        return '<div class="empty-state">No lead pair available.</div>'

    x = lead_pair["var_x"]
    y = lead_pair["var_y"]

    subset = within_df[
        (
            (within_df["var_x"] == x)
            & (within_df["var_y"] == y)
        )
        | (
            (within_df["var_x"] == y)
            & (within_df["var_y"] == x)
        )
    ].copy()

    subset = subset.sort_values("product_code")

    rows = []

    pooled_sign = sign_with_tolerance(
        float(lead_pair["pooled_r"])
    )

    for _, row in subset.head(
        MAX_WITHIN_PRODUCT_TABLE_ROWS
    ).iterrows():
        r = row["correlation"]

        reversed_direction = (
            row["eligible"]
            and sign_with_tolerance(r) != 0
            and pooled_sign != 0
            and sign_with_tolerance(r) != pooled_sign
        )

        if not row["eligible"]:
            status = '<span class="pill muted">Insufficient</span>'
        elif reversed_direction:
            status = '<span class="pill warning">Opposite direction</span>'
        else:
            status = '<span class="pill pass">Same direction</span>'

        rows.append(
            f"""
            <tr>
                <td><strong>{esc(row["product_code"])}</strong></td>
                <td>{int(row["n"])}</td>
                <td>{fmt_number(r, 2)}</td>
                <td>{esc(row["strength"])}</td>
                <td>{fmt_p(row["p_value"])}</td>
                <td>{status}</td>
            </tr>
            """
        )

    return f"""
    <div class="table-scroll">
        <table class="results-table compact">
            <thead>
                <tr>
                    <th>Product</th>
                    <th>Paired N</th>
                    <th>{correlation_symbol()}</th>
                    <th>Magnitude</th>
                    <th>Raw exploratory p</th>
                    <th>Direction vs pooled</th>
                </tr>
            </thead>
            <tbody>
                {''.join(rows)}
            </tbody>
        </table>
    </div>
    """

def main() -> None:
    # ---------------------------------------------------------------------
    # Validate script-level settings
    # ---------------------------------------------------------------------

    if CORRELATION_METHOD not in {"spearman", "pearson"}:
        raise ValueError(
            "CORRELATION_METHOD must be either 'spearman' or 'pearson'."
        )

    args = parse_args()

    # ---------------------------------------------------------------------
    # Data discovery / loading
    # ---------------------------------------------------------------------

    try:
        data_path = resolve_data_path(args.data_path)
        df = load_dataset(data_path)
    except Exception as exc:
        print(f"\nSOP-19 could not load the dataset:\n{exc}\n")
        raise

    output_dir = resolve_output_dir()

    # ---------------------------------------------------------------------
    # Dataset Capability Gate
    # ---------------------------------------------------------------------

    capability = run_capability_gate(df)

    if capability.status == "UNSUPPORTED":
        dashboard_html = build_hard_stop_dashboard(
            data_path=data_path,
            capability=capability,
            design=None,
            reason=(
                "The Dataset Capability Gate failed. SOP-19 requires at "
                "least two directly measured numeric variables plus "
                "panelist_id and product_code."
            ),
        )

        dashboard_path = write_dashboard(
            output_dir,
            dashboard_html,
        )

        print_hard_stop_summary(
            reason="Dataset Capability Gate failed.",
            dashboard_path=dashboard_path,
        )

        open_dashboard(dashboard_path)
        return

    # ---------------------------------------------------------------------
    # Study-design validation
    # ---------------------------------------------------------------------

    design = validate_study_design(df)

    if design.status == "HARD STOP":
        dashboard_html = build_hard_stop_dashboard(
            data_path=data_path,
            capability=capability,
            design=design,
            reason=(
                "Study-design validation found a structural problem that "
                "prevents defensible SOP-19 interpretation."
            ),
        )

        dashboard_path = write_dashboard(
            output_dir,
            dashboard_html,
        )

        print_hard_stop_summary(
            reason="Study-design validation failed.",
            dashboard_path=dashboard_path,
        )

        open_dashboard(dashboard_path)
        return

    # ---------------------------------------------------------------------
    # Preprocessing
    # ---------------------------------------------------------------------

    analysis_df = prepare_analysis_data(
        df,
        capability.available_variables,
    )

    variables, excluded_variables = variance_eligible_variables(
        analysis_df,
        capability.available_variables,
    )

    if len(variables) < 2:
        dashboard_html = build_hard_stop_dashboard(
            data_path=data_path,
            capability=capability,
            design=design,
            reason=(
                "Fewer than two configured variables had sufficient "
                "nonmissing variation for correlation analysis."
            ),
        )

        dashboard_path = write_dashboard(
            output_dir,
            dashboard_html,
        )

        print_hard_stop_summary(
            reason="Insufficient usable variable variation.",
            dashboard_path=dashboard_path,
        )

        open_dashboard(dashboard_path)
        return

    # ---------------------------------------------------------------------
    # Primary statistical analysis
    # ---------------------------------------------------------------------

    pair_df, within_df = build_pair_results(
        analysis_df,
        variables,
    )

    if pair_df.empty:
        dashboard_html = build_hard_stop_dashboard(
            data_path=data_path,
            capability=capability,
            design=design,
            reason=(
                "No eligible variable pairs could be constructed."
            ),
        )

        dashboard_path = write_dashboard(
            output_dir,
            dashboard_html,
        )

        print_hard_stop_summary(
            reason="No eligible correlation pairs.",
            dashboard_path=dashboard_path,
        )

        open_dashboard(dashboard_path)
        return

    # ---------------------------------------------------------------------
    # Multiplicity correction
    # ---------------------------------------------------------------------

    pair_df = apply_holm_correction(pair_df)

    # ---------------------------------------------------------------------
    # Plain-language interpretation
    # ---------------------------------------------------------------------

    pair_df = add_interpretations(pair_df)

    lead_pair = choose_lead_pair(pair_df)

    answer = build_overall_answer(
        pair_df,
        lead_pair,
    )

    # ---------------------------------------------------------------------
    # Dashboard
    # ---------------------------------------------------------------------

    dashboard_html = build_dashboard(
        data_path=data_path,
        capability=capability,
        design=design,
        variables=variables,
        excluded_variables=excluded_variables,
        pair_df=pair_df,
        within_df=within_df,
    )

    dashboard_path = write_dashboard(
        output_dir,
        dashboard_html,
    )

    # ---------------------------------------------------------------------
    # Optional technical exports
    # ---------------------------------------------------------------------

    save_technical_exports(
        output_dir,
        pair_df,
        within_df,
    )

    # ---------------------------------------------------------------------
    # Console summary
    # ---------------------------------------------------------------------

    print_console_summary(
        design=design,
        answer=answer,
        pair_df=pair_df,
        dashboard_path=dashboard_path,
    )

    # ---------------------------------------------------------------------
    # Automatic dashboard opening
    # ---------------------------------------------------------------------

    open_dashboard(dashboard_path)

if __name__ == "__main__":
    main()

"""
===============================================================================
SOP-20 — EFFECT SIZE AND ESTIMATED PRODUCT DIFFERENCE
Protein Shake Category Appraisal
===============================================================================

Purpose
-------
Answer questions such as:
    - How big is the product difference?
    - Is the difference meaningful in the original sensory/consumer scale?
    - How much higher is one product than another?

SOP-20 is a confirmatory companion / estimation analysis. For repeated ratings,
it uses the same mixed-model product contrast structure as the corresponding
approved inferential SOP and prioritizes:
    1. estimated product difference in original measurement units;
    2. 95% confidence interval;
    3. Holm-adjusted inferential support for the complete pairwise family;
    4. optional paired standardized effect (d_z), with denominator documented.

Default analysis
----------------
The workshop default is overall liking, corresponding to SOP-02:

    overall_liking ~ C(product_code) + (1 | panelist_id)

The script can also be pointed to flavor_impact or a measured desc_* attribute
using --outcome, provided the requested field actually exists in the dataset.
It does not invent or proxy missing constructs.

Dependencies
------------
pip install pandas numpy scipy statsmodels patsy

Default project structure
-------------------------
project/
├── SOP20_effect_size_product_difference.py
└── data/
    └── protein_shake_category_appraisal_mock_redesigned.csv

Examples
--------
Press Run in an IDE for the default overall_liking analysis.

Optional command line:
    python SOP20_effect_size_product_difference.py --outcome flavor_impact
    python SOP20_effect_size_product_difference.py --product-a G --product-b F
    python SOP20_effect_size_product_difference.py path/to/data.csv --outcome desc_bitterness

Important guardrails
--------------------
- Repeated product ratings are never treated as independent samples.
- Independent-sample Cohen's d is not used.
- Statistical significance is not equated with practical importance.
- No arbitrary small/medium/large effect-size labels are imposed.
- overall_liking is not purchase intent, repurchase intent, quality, or WTP.
- A missing construct is a capability stop, not an invitation to substitute.
- G9-A diagnostics are inherited from the corresponding mixed-model workflow.

class GateResult:
    status: str
    messages: List[str] = field(default_factory=list)
    warnings: List[str] = field(default_factory=list)
    hard_stops: List[str] = field(default_factory=list)

class ModelBundle:
    result: object
    warning_text: List[str]
    attempts: List[Dict[str, object]]

class DiagnosticBundle:
    table: pd.DataFrame
    hard_stop: bool
    sensitivity_required: bool
    triggers: List[str]
    warnings: List[str]
    icc: float

def finite(value: object) -> bool:
    try:
        return bool(np.isfinite(float(value)))
    except Exception:
        return False

def is_export_index(series: pd.Series) -> bool:
    numeric = pd.to_numeric(series, errors="coerce")
    if numeric.isna().any():
        return False
    expected = np.arange(len(series), dtype=float)
    return np.array_equal(
        numeric.to_numpy(dtype=float),
        expected,
    )

def remove_export_index(
    df: pd.DataFrame,
) -> Tuple[pd.DataFrame, List[str]]:
    out = df.copy()
    removed: List[str] = []

    for col in list(out.columns):
        normalized = str(col).strip().lower()

        if (
            normalized in {"index", "unnamed: 0"}
            and is_export_index(out[col])
        ):
            out = out.drop(columns=[col])
            removed.append(str(col))

    return out, removed

def outcome_label(outcome: str) -> str:
    return (
        outcome.replace("desc_", "")
        .replace("_", " ")
        .strip()
        .title()
    )

def infer_companion_sop(
    outcome: str,
) -> Tuple[Optional[str], str]:

    if outcome == "overall_liking":
        return "SOP-02", "9-point hedonic units"

    if outcome == "flavor_impact":
        return "SOP-03", "flavor-impact scale units"

    if outcome.startswith("desc_"):
        return "SOP-04", "descriptive intensity units"

    return None, "original measurement units"

def dataset_capability_gate(
    raw_df: pd.DataFrame,
    outcome: str,
) -> GateResult:

    required = [
        PANELIST_COL,
        PRODUCT_COL,
        outcome,
    ]

    missing = [
        c for c in required
        if c not in raw_df.columns
    ]

    messages = [
        f"Requested measured outcome: {outcome}",
        f"Required fields: {', '.join(required)}",
    ]

    companion_sop, _ = infer_companion_sop(
        outcome
    )

    if companion_sop is None:
        return GateResult(
            status="UNSUPPORTED",
            messages=messages,
            hard_stops=[
                f"'{outcome}' is not automatically authorized "
                "by this standalone SOP-20 script. "
                "Supported default companion structures are "
                "overall_liking (SOP-02), "
                "flavor_impact (SOP-03), "
                "or an explicitly measured desc_* attribute "
                "(SOP-04)."
            ],
        )

    if missing:
        return GateResult(
            status="UNSUPPORTED",
            messages=messages,
            hard_stops=[
                "Missing required field(s): "
                + ", ".join(missing)
                + ". SOP-20 will not invent, infer, "
                  "substitute, or proxy the missing construct."
            ],
        )

    return GateResult(
        status="SUPPORTED",
        messages=messages
        + [
            f"Companion inferential structure: "
            f"{companion_sop}"
        ],
    )

def prepare_analysis_data(
    raw_df: pd.DataFrame,
    outcome: str,
) -> Tuple[
    pd.DataFrame,
    List[str],
    List[str],
]:

    df, removed = remove_export_index(
        raw_df
    )

    required = [
        PANELIST_COL,
        PRODUCT_COL,
        outcome,
    ]

    work = df[required].copy()
    design_warnings: List[str] = []

    # Product/panelist identifiers must exist at row level.
    work[PRODUCT_COL] = (
        work[PRODUCT_COL]
        .astype("string")
        .str.strip()
    )

    work[PANELIST_COL] = (
        work[PANELIST_COL]
        .astype("string")
        .str.strip()
    )

    blank_product = (
        work[PRODUCT_COL].isna()
        | work[PRODUCT_COL].eq("")
    )

    blank_panelist = (
        work[PANELIST_COL].isna()
        | work[PANELIST_COL].eq("")
    )

    if bool(
        (blank_product | blank_panelist).any()
    ):
        n = int(
            (
                blank_product
                | blank_panelist
            ).sum()
        )

        raise ValueError(
            f"HARD STOP: {n} row(s) have "
            f"missing/blank {PRODUCT_COL} "
            f"or {PANELIST_COL}."
        )

    # Duplicate panelist × product cells violate
    # the one-evaluation-per-cell design.
    duplicate_mask = work.duplicated(
        [
            PANELIST_COL,
            PRODUCT_COL,
        ],
        keep=False,
    )

    if duplicate_mask.any():
        n = int(
            duplicate_mask.sum()
        )

        raise ValueError(
            f"HARD STOP: {n} row(s) belong to "
            "duplicate panelist × product cells. "
            "Resolve duplicates before estimating "
            "repeated-measures product differences."
        )

    original_nonmissing = (
        work[outcome].notna()
    )

    work[outcome] = pd.to_numeric(
        work[outcome],
        errors="coerce",
    )

    newly_invalid = int(
        (
            original_nonmissing
            & work[outcome].isna()
        ).sum()
    )

    if newly_invalid > 0:
        raise ValueError(
            f"HARD STOP: {newly_invalid} "
            f"nonmissing '{outcome}' value(s) "
            "are nonnumeric."
        )

    # Known scale checks only where the workshop
    # schema gives a clear scale.
    valid = work[outcome].dropna()

    if outcome == "overall_liking":
        bad = valid[
            (valid < 1)
            | (valid > 9)
        ]

        if not bad.empty:
            raise ValueError(
                "HARD STOP: overall_liking "
                "contains values outside the "
                "approved 1–9 scale."
            )

    elif outcome.startswith("desc_"):
        bad = valid[
            (valid < 0)
            | (valid > 10)
        ]

        if not bad.empty:
            raise ValueError(
                f"HARD STOP: {outcome} contains "
                "values outside the approved "
                "0–10 descriptive scale."
            )

    before = len(work)

    work = work.dropna(
        subset=[outcome]
    ).copy()

    dropped = before - len(work)

    if dropped:
        design_warnings.append(
            f"Excluded {dropped} row(s) with "
            f"missing {outcome}; no missing "
            "values were imputed."
        )

    if len(work) == 0:
        raise ValueError(
            "HARD STOP: no analyzable rows remain."
        )

    if (
        work[outcome]
        .nunique(dropna=True)
        < 2
    ):
        raise ValueError(
            f"HARD STOP: '{outcome}' has "
            "no usable variation."
        )

    if work[PRODUCT_COL].nunique() < 2:
        raise ValueError(
            "HARD STOP: fewer than two "
            "products remain for comparison."
        )

    if work[PANELIST_COL].nunique() < 2:
        raise ValueError(
            "HARD STOP: fewer than two "
            "panelists remain."
        )

    product_counts_by_panelist = (
        work.groupby(PANELIST_COL)[
            PRODUCT_COL
        ].nunique()
    )

    repeated_panelists = int(
        (
            product_counts_by_panelist
            >= 2
        ).sum()
    )

    if repeated_panelists < 2:
        raise ValueError(
            "HARD STOP: insufficient repeated "
            "product evaluations within panelist."
        )

    cells_per_product = (
        work.groupby(PRODUCT_COL).size()
    )

    low = cells_per_product[
        cells_per_product < 2
    ]

    if not low.empty:
        raise ValueError(
            "HARD STOP: at least one product "
            "has fewer than two analyzed "
            f"observations: {low.to_dict()}."
        )

    products = sorted(
        work[PRODUCT_COL]
        .astype(str)
        .unique()
        .tolist()
    )

    work[PRODUCT_COL] = pd.Categorical(
        work[PRODUCT_COL],
        categories=products,
        ordered=True,
    )

    n_panelists = int(
        work[PANELIST_COL].nunique()
    )

    if products != EXPECTED_PRODUCTS:
        design_warnings.append(
            f"Observed products are {products}; "
            f"workshop expectation is "
            f"{EXPECTED_PRODUCTS}."
        )

    if n_panelists != EXPECTED_PANELISTS:
        design_warnings.append(
            f"Observed {n_panelists} panelists; "
            f"workshop expectation is "
            f"{EXPECTED_PANELISTS}."
        )

    if len(work) != EXPECTED_ROWS:
        design_warnings.append(
            f"Analyzed {len(work)} rows; "
            "workshop complete-design "
            f"expectation is {EXPECTED_ROWS}."
        )

    evals_per_panelist = (
        work.groupby(PANELIST_COL).size()
    )

    if evals_per_panelist.nunique() != 1:
        design_warnings.append(
            "The analyzed data are unbalanced "
            "across panelists. MixedLM can "
            "accommodate imbalance, but "
            "product-specific information differs "
            "and is disclosed."
        )

    full_n_products = len(products)

    incomplete = int(
        (
            product_counts_by_panelist
            < full_n_products
        ).sum()
    )

    if incomplete > 0:
        design_warnings.append(
            f"{incomplete} panelist(s) do not "
            f"contribute all {full_n_products} "
            "products."
        )

    return (
        work,
        removed,
        design_warnings,
    )

def fit_primary_mixed_model(
    df: pd.DataFrame,
    outcome: str,
) -> ModelBundle:

    formula = (
        f"{outcome} ~ "
        f"C({PRODUCT_COL})"
    )

    model = smf.mixedlm(
        formula,
        data=df,
        groups=df[PANELIST_COL],
        re_formula="1",
    )

    attempts: List[
        Dict[str, object]
    ] = []

    last_result = None
    last_warning_text: List[str] = []
    last_exception: Optional[
        Exception
    ] = None

    for method in OPTIMIZER_SEQUENCE:

        try:
            with warnings.catch_warnings(
                record=True
            ) as caught:

                warnings.simplefilter(
                    "always"
                )

                result = model.fit(
                    reml=True,
                    method=method,
                    maxiter=2000,
                    disp=False,
                )

            captured = [
                str(w.message)
                for w in caught
            ]

            converged = bool(
                getattr(
                    result,
                    "converged",
                    False,
                )
            )

            attempts.append(
                {
                    "optimizer": method,
                    "completed": True,
                    "converged": converged,
                    "warnings": (
                        " | ".join(captured)
                        if captured
                        else "None"
                    ),
                }
            )

            last_result = result
            last_warning_text = captured

            if converged:
                return ModelBundle(
                    result,
                    captured,
                    attempts,
                )

        except Exception as exc:
            last_exception = exc

            attempts.append(
                {
                    "optimizer": method,
                    "completed": False,
                    "converged": False,
                    "warnings": (
                        f"Exception: {exc}"
                    ),
                }
            )

    if last_result is not None:
        return ModelBundle(
            last_result,
            last_warning_text,
            attempts,
        )

    raise RuntimeError(
        "A1 HARD STOP: all approved MixedLM "
        "optimizer attempts failed. "
        f"Last error: {last_exception}"
    )

def fixed_effect_covariance(
    result,
) -> pd.DataFrame:

    fe_names = list(
        result.fe_params.index
    )

    cov = result.cov_params()

    if isinstance(
        cov,
        pd.DataFrame,
    ):
        return cov.loc[
            fe_names,
            fe_names,
        ].copy()

    arr = np.asarray(
        cov,
        dtype=float,
    )[
        : len(fe_names),
        : len(fe_names),
    ]

    return pd.DataFrame(
        arr,
        index=fe_names,
        columns=fe_names,
    )

def design_vector(
    result,
    product: str,
) -> np.ndarray:

    design_info = (
        result.model.data.design_info
    )

    categories = list(
        result.model.data.frame[
            PRODUCT_COL
        ].cat.categories
    )

    new_data = pd.DataFrame(
        {
            PRODUCT_COL: pd.Categorical(
                [product],
                categories=categories,
                ordered=True,
            )
        }
    )

    matrix = (
        patsy.build_design_matrices(
            [design_info],
            new_data,
            return_type="dataframe",
        )[0]
    )

    return matrix.to_numpy(
        dtype=float
    ).ravel()

def estimate_linear_contrast(
    result,
    lvector: np.ndarray,
) -> Dict[str, float]:

    beta = (
        result.fe_params
        .to_numpy(dtype=float)
    )

    cov_beta = (
        fixed_effect_covariance(result)
        .to_numpy(dtype=float)
    )

    estimate = float(
        lvector @ beta
    )

    variance = float(
        lvector
        @ cov_beta
        @ lvector.T
    )

    if (
        not np.isfinite(variance)
        or variance < -NUMERIC_TOL
    ):
        return {
            "estimate": estimate,
            "se": np.nan,
            "z": np.nan,
            "p_raw": np.nan,
            "ci_low": np.nan,
            "ci_high": np.nan,
        }

    variance = max(
        variance,
        0.0,
    )

    se = math.sqrt(
        variance
    )

    if se <= NUMERIC_TOL:
        return {
            "estimate": estimate,
            "se": se,
            "z": np.nan,
            "p_raw": np.nan,
            "ci_low": np.nan,
            "ci_high": np.nan,
        }

    z = estimate / se

    p_raw = float(
        2
        * stats.norm.sf(
            abs(z)
        )
    )

    return {
        "estimate": estimate,
        "se": se,
        "z": z,
        "p_raw": p_raw,
        "ci_low": (
            estimate
            - Z_CRITICAL * se
        ),
        "ci_high": (
            estimate
            + Z_CRITICAL * se
        ),
    }

def estimated_product_means(
    result,
    products: Sequence[str],
) -> pd.DataFrame:

    rows = []

    for product in products:

        x = design_vector(
            result,
            product,
        )

        est = (
            estimate_linear_contrast(
                result,
                x,
            )
        )

        rows.append(
            {
                "product": product,
                "estimated_mean": (
                    est["estimate"]
                ),
                "se": est["se"],
                "ci_low": est["ci_low"],
                "ci_high": est["ci_high"],
            }
        )

    return (
        pd.DataFrame(rows)
        .sort_values(
            "estimated_mean",
            ascending=False,
        )
    )

def paired_standardized_effect(
    df: pd.DataFrame,
    outcome: str,
    product_a: str,
    product_b: str,
) -> Tuple[
    float,
    float,
    float,
    int,
]:
    """
    Repeated/paired standardized effect d_z.

    Denominator = sample SD of within-panelist
    difference scores (A - B).

    CI = t-based CI for the raw paired mean
    difference, standardized by the same
    observed SD(diff).

    This is supplementary descriptive
    standardization; the mixed-model raw-unit
    contrast remains primary.
    """

    wide = (
        df.loc[
            df[PRODUCT_COL]
            .astype(str)
            .isin(
                [
                    product_a,
                    product_b,
                ]
            )
        ]
        .assign(
            _product=lambda x: (
                x[PRODUCT_COL]
                .astype(str)
            )
        )
        .pivot(
            index=PANELIST_COL,
            columns="_product",
            values=outcome,
        )
    )

    if (
        product_a
        not in wide.columns
        or product_b
        not in wide.columns
    ):
        return (
            np.nan,
            np.nan,
            np.nan,
            0,
        )

    paired = (
        wide[
            [
                product_a,
                product_b,
            ]
        ]
        .dropna()
    )

    n = len(paired)

    if n < 2:
        return (
            np.nan,
            np.nan,
            np.nan,
            n,
        )

    diff = (
        paired[product_a]
        - paired[product_b]
    )

    sd_diff = float(
        diff.std(ddof=1)
    )

    if (
        not np.isfinite(sd_diff)
        or sd_diff <= NUMERIC_TOL
    ):
        return (
            np.nan,
            np.nan,
            np.nan,
            n,
        )

    mean_diff = float(
        diff.mean()
    )

    dz = (
        mean_diff
        / sd_diff
    )

    tcrit = stats.t.ppf(
        1 - ALPHA / 2,
        df=n - 1,
    )

    raw_half = (
        tcrit
        * sd_diff
        / math.sqrt(n)
    )

    return (
        dz,
        (
            mean_diff
            - raw_half
        )
        / sd_diff,
        (
            mean_diff
            + raw_half
        )
        / sd_diff,
        n,
    )

def pairwise_contrasts(
    df: pd.DataFrame,
    result,
    outcome: str,
    products: Sequence[str],
) -> pd.DataFrame:

    rows: List[
        Dict[str, object]
    ] = []

    for i, product_a in enumerate(
        products
    ):

        xa = design_vector(
            result,
            product_a,
        )

        for product_b in products[
            i + 1:
        ]:

            xb = design_vector(
                result,
                product_b,
            )

            est = (
                estimate_linear_contrast(
                    result,
                    xa - xb,
                )
            )

            dz = np.nan
            dz_low = np.nan
            dz_high = np.nan
            paired_n = 0

            if (
                CALCULATE_PAIRED_STANDARDIZED_EFFECT
            ):
                (
                    dz,
                    dz_low,
                    dz_high,
                    paired_n,
                ) = paired_standardized_effect(
                    df,
                    outcome,
                    product_a,
                    product_b,
                )

            rows.append(
                {
                    "product_a": product_a,
                    "product_b": product_b,
                    "contrast": (
                        f"{product_a} − "
                        f"{product_b}"
                    ),
                    "difference": (
                        est["estimate"]
                    ),
                    "se": est["se"],
                    "ci_low": est["ci_low"],
                    "ci_high": est["ci_high"],
                    "p_raw": est["p_raw"],
                    "paired_dz": dz,
                    "dz_ci_low": dz_low,
                    "dz_ci_high": dz_high,
                    "paired_n": paired_n,
                }
            )

    out = pd.DataFrame(
        rows
    )

    valid_p = (
        out["p_raw"].notna()
        & np.isfinite(
            out["p_raw"]
        )
    )

    out["p_holm"] = np.nan
    out["significant_holm"] = False

    if valid_p.any():

        (
            reject,
            p_adj,
            _,
            _,
        ) = multipletests(
            out.loc[
                valid_p,
                "p_raw",
            ].to_numpy(
                dtype=float
            ),
            alpha=ALPHA,
            method="holm",
        )

        out.loc[
            valid_p,
            "p_holm",
        ] = p_adj

        out.loc[
            valid_p,
            "significant_holm",
        ] = reject

    out["abs_difference"] = (
        out["difference"].abs()
    )

    return (
        out.sort_values(
            "abs_difference",
            ascending=False,
        )
        .reset_index(
            drop=True
        )
    )

    return {

    }

def run_g9a_diagnostics(
    df: pd.DataFrame,
    result,
    warning_text: Sequence[str],
) -> DiagnosticBundle:

    diagnostics: List[
        Dict[str, object]
    ] = []

    hard_stop = False
    sensitivity_required = False
    triggers: List[str] = []
    warnings_out: List[str] = []

    warning_blob = (
        " | ".join(
            warning_text
        ).lower()
    )

    # -------------------------------------------------------------------------
    # A1 — convergence
    # -------------------------------------------------------------------------

    converged = bool(
        getattr(
            result,
            "converged",
            False,
        )
    )

    if converged:
        diagnostics.append(
            diag_row(
                "A1 Optimizer convergence",
                "True",
                "True",
                "PASS",
                "Proceed.",
            )
        )

    else:
        hard_stop = True

        diagnostics.append(
            diag_row(
                "A1 Optimizer convergence",
                "False",
                "True",
                "HARD STOP",
                "Suppress inference.",
            )
        )

    # -------------------------------------------------------------------------
    # A2 — covariance / Hessian validity
    # -------------------------------------------------------------------------

    a2_reasons: List[str] = []

    try:
        cov_re = np.asarray(
            result.cov_re,
            dtype=float,
        )

        if not np.isfinite(
            cov_re
        ).all():

            a2_reasons.append(
                "random-effects covariance "
                "contains non-finite values"
            )

        else:
            eig_re = (
                np.linalg.eigvalsh(
                    (
                        cov_re
                        + cov_re.T
                    )
                    / 2
                )
            )

            if (
                float(
                    np.min(eig_re)
                )
                <= NUMERIC_TOL
            ):
                a2_reasons.append(
                    "random-effects covariance "
                    "is singular/non-positive"
                )

    except Exception as exc:
        a2_reasons.append(
            "random-effects covariance "
            "could not be validated: "
            f"{exc}"
        )

    singular_phrases = [
        "random effects covariance is singular",
        "covariance matrix is singular",
        "singular matrix",
    ]

    if any(
        phrase in warning_blob
        for phrase in singular_phrases
    ):
        a2_reasons.append(
            "singularity warning emitted"
        )

    if (
        "hessian" in warning_blob
        and "not positive definite"
        in warning_blob
    ):
        a2_reasons.append(
            "Hessian not positive definite "
            "warning emitted"
        )

    try:
        hess, hess_singular = (
            result.model.hessian(
                result.params_object
            )
        )

        hess = np.asarray(
            hess,
            dtype=float,
        )

        if bool(
            hess_singular
        ):
            a2_reasons.append(
                "MixedLM Hessian singularity flag"
            )

        if not np.isfinite(
            hess
        ).all():

            a2_reasons.append(
                "Hessian contains "
                "non-finite values"
            )

        else:
            neg_hess = -(
                (
                    hess
                    + hess.T
                )
                / 2
            )

            min_eig = float(
                np.min(
                    np.linalg.eigvalsh(
                        neg_hess
                    )
                )
            )

            if (
                min_eig
                <= NUMERIC_TOL
            ):
                a2_reasons.append(
                    "-Hessian minimum eigenvalue "
                    f"{min_eig:.3g} <= "
                    f"{NUMERIC_TOL:g}"
                )

    except Exception as exc:
        a2_reasons.append(
            "Hessian validation failed: "
            f"{exc}"
        )

    try:
        cov_fe = (
            fixed_effect_covariance(
                result
            )
            .to_numpy(
                dtype=float
            )
        )

        if (
            not np.isfinite(
                cov_fe
            ).all()
            or np.any(
                np.diag(cov_fe)
                < 0
            )
        ):
            a2_reasons.append(
                "fixed-effect covariance invalid "
                "for standard errors"
            )

    except Exception as exc:
        a2_reasons.append(
            "fixed-effect covariance "
            "unavailable: "
            f"{exc}"
        )

    if a2_reasons:

        hard_stop = True

        diagnostics.append(
            diag_row(
                "A2 Covariance / Hessian validity",
                "; ".join(
                    a2_reasons
                ),
                (
                    "No singular covariance; "
                    "valid positive-definite "
                    "information matrix"
                ),
                "HARD STOP",
                "Suppress mixed-model inference.",
            )
        )

    else:
        diagnostics.append(
            diag_row(
                "A2 Covariance / Hessian validity",
                "Valid",
                "Valid",
                "PASS",
                "Proceed.",
            )
        )

    # Remaining diagnostics require fitted quantities.
    resid = np.asarray(
        result.resid,
        dtype=float,
    )

    fitted = np.asarray(
        result.fittedvalues,
        dtype=float,
    )

    _ = fitted

    if len(resid) > 1:
        residual_var = float(
            np.var(
                resid,
                ddof=max(
                    1,
                    len(result.fe_params),
                ),
            )
        )
    else:
        residual_var = np.nan

    try:
        random_var = float(
            np.asarray(
                result.cov_re,
                dtype=float,
            ).ravel()[0]
        )
    except Exception:
        random_var = np.nan

    if (
        finite(random_var)
        and finite(residual_var)
        and (
            random_var
            + residual_var
        )
        > 0
    ):
        icc = (
            random_var
            / (
                random_var
                + residual_var
            )
        )
    else:
        icc = np.nan

    # -------------------------------------------------------------------------
    # A3 — boundary random-effect variance
    # -------------------------------------------------------------------------

    if (
        finite(random_var)
        and random_var
        < BOUNDARY_RANDOM_VARIANCE
    ):

        sensitivity_required = True

        triggers.append(
            "A3 boundary random-effect variance"
        )

        diagnostics.append(
            diag_row(
                "A3 Random-intercept variance",
                fmt(
                    random_var,
                    4,
                ),
                (
                    f">= "
                    f"{BOUNDARY_RANDOM_VARIANCE}"
                ),
                (
                    "WARNING / "
                    "SENSITIVITY REQUIRED"
                ),
                (
                    "Run panelist fixed-effect "
                    "sensitivity."
                ),
            )
        )

    else:
        diagnostics.append(
            diag_row(
                "A3 Random-intercept variance",
                fmt(
                    random_var,
                    4,
                ),
                (
                    f">= "
                    f"{BOUNDARY_RANDOM_VARIANCE}"
                ),
                "PASS",
                "Proceed.",
            )
        )

    # -------------------------------------------------------------------------
    # A4 — ICC
    # -------------------------------------------------------------------------

    if (
        finite(icc)
        and icc
        < NEAR_ZERO_ICC
    ):

        sensitivity_required = True

        triggers.append(
            "A4 near-zero ICC"
        )

        diagnostics.append(
            diag_row(
                "A4 ICC",
                fmt(
                    icc,
                    4,
                ),
                f">= {NEAR_ZERO_ICC}",
                (
                    "WARNING / "
                    "SENSITIVITY REQUIRED"
                ),
                (
                    "Run panelist fixed-effect "
                    "sensitivity."
                ),
            )
        )

    else:
        diagnostics.append(
            diag_row(
                "A4 ICC",
                fmt(
                    icc,
                    4,
                ),
                f">= {NEAR_ZERO_ICC}",
                "PASS",
                "Proceed.",
            )
        )

    # -------------------------------------------------------------------------
    # A5 — residual distribution
    # -------------------------------------------------------------------------

    finite_resid = resid[
        np.isfinite(resid)
    ]

    normality_p = np.nan

    if (
        3
        <= len(finite_resid)
        <= 5000
    ):
        try:
            normality_p = float(
                stats.shapiro(
                    finite_resid
                ).pvalue
            )
        except Exception:
            normality_p = np.nan

    elif len(finite_resid) > 5000:
        try:
            normality_p = float(
                stats.normaltest(
                    finite_resid
                ).pvalue
            )
        except Exception:
            normality_p = np.nan

    if len(finite_resid) >= 3:
        skew = float(
            stats.skew(
                finite_resid,
                bias=False,
            )
        )
    else:
        skew = np.nan

    if len(finite_resid) >= 4:
        kurt = float(
            stats.kurtosis(
                finite_resid,
                fisher=False,
                bias=False,
            )
        )
    else:
        kurt = np.nan

    a5_sens = (
        (
            finite(skew)
            and abs(skew)
            > RESIDUAL_SKEW_SENSITIVITY
        )
        or (
            finite(kurt)
            and kurt
            > PEARSON_KURTOSIS_SENSITIVITY
        )
    )

    a5_warn = (
        finite(normality_p)
        and normality_p
        < NORMALITY_WARNING_P
    )

    if a5_sens:

        sensitivity_required = True

        triggers.append(
            "A5 residual shape"
        )

        status = (
            "WARNING / "
            "SENSITIVITY REQUIRED"
        )

        action = (
            "Run panelist fixed-effect "
            "sensitivity."
        )

    elif a5_warn:

        status = "WARNING"

        action = (
            "Disclose; formal normality p-value "
            "alone does not invalidate the model."
        )

        warnings_out.append(
            "Residual normality test p < 0.01."
        )

    else:
        status = "PASS"
        action = "Proceed."

    diagnostics.append(
        diag_row(
            "A5 Residual distribution",
            (
                f"normality p="
                f"{fmt_p(normality_p)}, "
                f"skew={fmt(skew, 3)}, "
                f"Pearson kurtosis="
                f"{fmt(kurt, 3)}"
            ),
            (
                "Warning p<0.01; "
                "sensitivity if |skew|>1 "
                "or kurtosis>5"
            ),
            status,
            action,
        )
    )

    # -------------------------------------------------------------------------
    # A6 — heteroscedasticity by product
    # -------------------------------------------------------------------------

    tmp = df[
        [PRODUCT_COL]
    ].copy()

    tmp["_resid"] = resid

    var_by_product = (
        tmp.groupby(
            PRODUCT_COL,
            observed=True,
        )["_resid"]
        .var(ddof=1)
    )

    positive_vars = (
        var_by_product[
            np.isfinite(
                var_by_product
            )
            & (
                var_by_product
                > NUMERIC_TOL
            )
        ]
    )

    if len(positive_vars) >= 2:
        variance_ratio = float(
            positive_vars.max()
            / positive_vars.min()
        )
    else:
        variance_ratio = np.nan

    levene_p = np.nan

    try:
        groups = [
            group["_resid"]
            .to_numpy(
                dtype=float
            )
            for _, group in tmp.groupby(
                PRODUCT_COL,
                observed=True,
            )
            if len(group) >= 2
        ]

        if len(groups) >= 2:
            levene_p = float(
                stats.levene(
                    *groups,
                    center="median",
                ).pvalue
            )

    except Exception:
        pass

    a6_sens = (
        (
            finite(levene_p)
            and levene_p
            < HETERO_WARNING_P
            and finite(
                variance_ratio
            )
            and variance_ratio
            > HETERO_WARNING_RATIO
        )
        or (
            finite(
                variance_ratio
            )
            and variance_ratio
            > HETERO_SENSITIVITY_RATIO
        )
    )

    a6_warn = (
        (
            finite(levene_p)
            and levene_p
            < HETERO_WARNING_P
        )
        or (
            finite(
                variance_ratio
            )
            and variance_ratio
            > HETERO_WARNING_RATIO
        )
    )

    if a6_sens:

        sensitivity_required = True

        triggers.append(
            "A6 residual heteroscedasticity"
        )

        status = (
            "WARNING / "
            "SENSITIVITY REQUIRED"
        )

        action = (
            "Run panelist fixed-effect "
            "sensitivity."
        )

    elif a6_warn:

        status = "WARNING"

        action = (
            "Disclose heteroscedasticity signal."
        )

        warnings_out.append(
            "Residual heteroscedasticity "
            "warning threshold crossed."
        )

    else:
        status = "PASS"
        action = "Proceed."

    diagnostics.append(
        diag_row(
            "A6 Residual heteroscedasticity",
            (
                f"Brown-Forsythe p="
                f"{fmt_p(levene_p)}, "
                f"variance ratio="
                f"{fmt(variance_ratio, 2)}"
            ),
            (
                "Warning p<0.01 or ratio>2; "
                "sensitivity if both "
                "or ratio>4"
            ),
            status,
            action,
        )
    )

    # -------------------------------------------------------------------------
    # A7 — extreme standardized residuals
    # -------------------------------------------------------------------------

    if len(finite_resid) >= 2:
        resid_sd = float(
            np.std(
                finite_resid,
                ddof=1,
            )
        )
    else:
        resid_sd = np.nan

    if (
        finite(resid_sd)
        and resid_sd
        > NUMERIC_TOL
    ):
        zresid = (
            finite_resid
            / resid_sd
        )
    else:
        zresid = np.full_like(
            finite_resid,
            np.nan,
        )

    if len(zresid):
        n_gt3 = int(
            np.sum(
                np.abs(zresid)
                > EXTREME_RESIDUAL_FLAG
            )
        )
    else:
        n_gt3 = 0

    if len(zresid):
        pct_gt3 = (
            100
            * n_gt3
            / len(zresid)
        )
    else:
        pct_gt3 = np.nan

    if (
        len(zresid)
        and np.isfinite(
            zresid
        ).any()
    ):
        max_abs = float(
            np.nanmax(
                np.abs(zresid)
            )
        )
    else:
        max_abs = np.nan

    a7_sens = (
        (
            finite(pct_gt3)
            and pct_gt3
            > EXTREME_RESIDUAL_PCT_SENSITIVITY
        )
        or (
            finite(max_abs)
            and max_abs
            > EXTREME_RESIDUAL_HARDER
        )
    )

    if a7_sens:

        sensitivity_required = True

        triggers.append(
            "A7 extreme residuals"
        )

        status = (
            "WARNING / "
            "SENSITIVITY REQUIRED"
        )

        action = (
            "Run panelist fixed-effect sensitivity; "
            "do not delete rows automatically."
        )

    elif n_gt3 > 0:

        status = "WARNING"

        action = (
            "Flag observations; "
            "do not delete automatically."
        )

        warnings_out.append(
            f"{n_gt3} observation(s) have "
            "|standardized residual| > 3."
        )

    else:
        status = "PASS"
        action = "Proceed."

    diagnostics.append(
        diag_row(
            "A7 Extreme residuals",
            (
                f">3: {n_gt3} "
                f"({fmt(pct_gt3, 2)}%); "
                f"max |z|="
                f"{fmt(max_abs, 2)}"
            ),
            (
                ">1% above 3 or any above 4 "
                "=> sensitivity"
            ),
            status,
            action,
        )
    )

    return DiagnosticBundle(
        table=pd.DataFrame(
            diagnostics
        ),
        hard_stop=hard_stop,
        sensitivity_required=(
            sensitivity_required
        ),
        triggers=triggers,
        warnings=warnings_out,
        icc=icc,
    )

def fit_panelist_fixed_effect_sensitivity(
    df: pd.DataFrame,
    outcome: str,
    products: Sequence[str],
) -> pd.DataFrame:
    """
    Panelist fixed-effect OLS sensitivity:

        outcome ~ product + panelist
    """

    model = smf.ols(
        (
            f"{outcome} ~ "
            f"C({PRODUCT_COL}) + "
            f"C({PANELIST_COL})"
        ),
        data=df,
    ).fit()

    design_info = (
        model.model.data.design_info
    )

    cov = np.asarray(
        model.cov_params(),
        dtype=float,
    )

    beta = np.asarray(
        model.params,
        dtype=float,
    )

    # Product contrast evaluated at a common panelist;
    # panelist terms therefore cancel.
    baseline_panelist = str(
        df[PANELIST_COL]
        .astype(str)
        .iloc[0]
    )

    def x_for(
        product: str,
    ) -> np.ndarray:

        new_data = pd.DataFrame(
            {
                PRODUCT_COL: pd.Categorical(
                    [product],
                    categories=list(
                        df[PRODUCT_COL]
                        .cat.categories
                    ),
                    ordered=True,
                ),
                PANELIST_COL: [
                    baseline_panelist
                ],
            }
        )

        return (
            patsy.build_design_matrices(
                [design_info],
                new_data,
                return_type="dataframe",
            )[0]
            .to_numpy(
                dtype=float
            )
            .ravel()
        )

    rows = []

    for i, a in enumerate(
        products
    ):

        xa = x_for(
            a
        )

        for b in products[
            i + 1:
        ]:

            lvec = (
                xa
                - x_for(b)
            )

            diff = float(
                lvec
                @ beta
            )

            var = float(
                lvec
                @ cov
                @ lvec.T
            )

            if np.isfinite(var):
                se = math.sqrt(
                    max(
                        var,
                        0.0,
                    )
                )
            else:
                se = np.nan

            if (
                finite(se)
                and se
                > NUMERIC_TOL
            ):
                p = float(
                    2
                    * stats.t.sf(
                        abs(
                            diff
                            / se
                        ),
                        df=model.df_resid,
                    )
                )
            else:
                p = np.nan

            rows.append(
                {
                    "product_a": a,
                    "product_b": b,
                    "sensitivity_difference": diff,
                    "sensitivity_p_raw": p,
                }
            )

    out = pd.DataFrame(
        rows
    )

    out[
        "sensitivity_p_holm"
    ] = np.nan

    out[
        "sensitivity_significant_holm"
    ] = False

    valid = (
        out[
            "sensitivity_p_raw"
        ].notna()
        & np.isfinite(
            out[
                "sensitivity_p_raw"
            ]
        )
    )

    if valid.any():

        (
            reject,
            p_adj,
            _,
            _,
        ) = multipletests(
            out.loc[
                valid,
                "sensitivity_p_raw",
            ],
            alpha=ALPHA,
            method="holm",
        )

        out.loc[
            valid,
            "sensitivity_p_holm",
        ] = p_adj

        out.loc[
            valid,
            "sensitivity_significant_holm",
        ] = reject

    return out

def compare_primary_sensitivity(
    primary: pd.DataFrame,
    sensitivity: pd.DataFrame,
) -> Tuple[
    pd.DataFrame,
    bool,
]:

    merged = primary.merge(
        sensitivity,
        on=[
            "product_a",
            "product_b",
        ],
        how="left",
    )

    merged[
        "direction_consistent"
    ] = (
        np.sign(
            merged["difference"]
        )
        == np.sign(
            merged[
                "sensitivity_difference"
            ]
        )
    )

    merged[
        "support_consistent"
    ] = (
        merged[
            "significant_holm"
        ].astype(bool)
        == merged[
            "sensitivity_significant_holm"
        ].astype(bool)
    )

    finite_rows = (
        merged[
            "sensitivity_difference"
        ].notna()
    )

    model_sensitive = bool(
        finite_rows.any()
        and (
            (
                ~merged.loc[
                    finite_rows,
                    "direction_consistent",
                ]
            ).any()
            or (
                ~merged.loc[
                    finite_rows,
                    "support_consistent",
                ]
            ).any()
        )
    )

    return (
        merged,
        model_sensitive,
    )

def select_focal_contrast(
    contrasts: pd.DataFrame,
    product_a: Optional[str],
    product_b: Optional[str],
) -> Optional[pd.Series]:

    if (
        product_a is None
        and product_b is None
    ):
        return (
            contrasts.iloc[0]
            if not contrasts.empty
            else None
        )

    if (
        not product_a
        or not product_b
    ):
        raise ValueError(
            "Provide both --product-a "
            "and --product-b, or neither."
        )

    a = str(
        product_a
    ).strip()

    b = str(
        product_b
    ).strip()

    direct = contrasts[
        (
            contrasts[
                "product_a"
            ]
            == a
        )
        & (
            contrasts[
                "product_b"
            ]
            == b
        )
    ]

    if not direct.empty:
        return direct.iloc[0]

    reverse = contrasts[
        (
            contrasts[
                "product_a"
            ]
            == b
        )
        & (
            contrasts[
                "product_b"
            ]
            == a
        )
    ]

    if not reverse.empty:

        row = (
            reverse
            .iloc[0]
            .copy()
        )

        row["product_a"] = a
        row["product_b"] = b
        row["contrast"] = (
            f"{a} − {b}"
        )

        # Reverse the signed interval correctly.
        old_diff = float(
            reverse.iloc[0][
                "difference"
            ]
        )

        old_low = float(
            reverse.iloc[0][
                "ci_low"
            ]
        )

        old_high = float(
            reverse.iloc[0][
                "ci_high"
            ]
        )

        row["difference"] = (
            -old_diff
        )

        row["ci_low"] = (
            -old_high
        )

        row["ci_high"] = (
            -old_low
        )

        if finite(
            reverse.iloc[0][
                "paired_dz"
            ]
        ):

            row[
                "paired_dz"
            ] = -float(
                reverse.iloc[0][
                    "paired_dz"
                ]
            )

            row[
                "dz_ci_low"
            ] = -float(
                reverse.iloc[0][
                    "dz_ci_high"
                ]
            )

            row[
                "dz_ci_high"
            ] = -float(
                reverse.iloc[0][
                    "dz_ci_low"
                ]
            )

        return row

    raise ValueError(
        f"Requested contrast {a} vs {b} "
        "is not available in analyzed products."
    )

def make_answer(
    focal: Optional[pd.Series],
    outcome: str,
    hard_stop: bool,
    model_sensitive: bool,
) -> str:

    if hard_stop:
        return (
            "Inference stopped because the "
            "repeated-measures model did not pass "
            "required diagnostics."
        )

    if focal is None:
        return (
            "No estimable product contrast "
            "was available."
        )

    a = str(
        focal["product_a"]
    )

    b = str(
        focal["product_b"]
    )

    diff = float(
        focal["difference"]
    )

    low = float(
        focal["ci_low"]
    )

    high = float(
        focal["ci_high"]
    )

    label = (
        outcome_label(outcome)
        .lower()
    )

    if diff >= 0:

        direction = (
            f"Product {a} is estimated "
            f"{abs(diff):.2f} {label} units "
            f"higher than Product {b}"
        )

    else:

        direction = (
            f"Product {a} is estimated "
            f"{abs(diff):.2f} {label} units "
            f"lower than Product {b}"
        )

    if model_sensitive:

        return (
            direction
            + f" (95% CI {low:.2f} to {high:.2f}), "
              "but the inferential conclusion "
              "is model-sensitive."
        )

    return (
        direction
        + f" (95% CI {low:.2f} to {high:.2f})."
    )

                )

                contrasts = (
                    pairwise_contrasts(
                        work,
                        model_bundle.result,
                        outcome,
                        products,
                    )
                )

                if (
                    diagnostics
                    .sensitivity_required
                ):

                    sensitivity_ran = True

                    sensitivity = (
                        fit_panelist_fixed_effect_sensitivity(
                            work,
                            outcome,
                            products,
                        )
                    )

                    (
                        contrasts,
                        model_sensitive,
                    ) = compare_primary_sensitivity(
                        contrasts,
                        sensitivity,
                    )

                focal = (
                    select_focal_contrast(
                        contrasts,
                        args.product_a,
                        args.product_b,
                    )
                )

        except Exception as exc:

            fatal_messages.append(
                str(exc)
            )

    hard_stop = (
        bool(
            fatal_messages
        )
        or (
            diagnostics.hard_stop
            if diagnostics
            else False
        )
    )

    answer = make_answer(
        focal,
        outcome,
        hard_stop,
        model_sensitive,
    )

    dashboard_html = (
        build_dashboard(
            data_path=data_path,
            outcome=outcome,
            gate=gate,
            work=work,
            design_warnings=design_warnings,
            removed_columns=removed_columns,
            attempts=attempts,
            diagnostics=diagnostics,
            means=means,
            contrasts=contrasts,
            focal=focal,
            answer=answer,
            model_sensitive=model_sensitive,
            sensitivity_ran=sensitivity_ran,
            fatal_messages=fatal_messages,
        )
    )

    dashboard_path = (
        write_dashboard(
            dashboard_html
        )
    )

    save_technical_exports(
        dashboard_path.parent,
        means,
        contrasts,
        diagnostics,
    )

    if AUTO_OPEN_DASHBOARD:

        try:
            webbrowser.open(
                dashboard_path.as_uri()
            )

        except Exception:
            pass

    if hard_stop:

        status = (
            "HARD STOP / "
            "INFERENCE SUPPRESSED"
        )

    elif model_sensitive:

        status = (
            "MODEL-SENSITIVE"
        )

    elif sensitivity_ran:

        status = (
            "SENSITIVITY CHECK "
            "PASSED / STABLE"
        )

    else:

        status = (
            "Diagnostics acceptable"
        )

    print(
        "\n"
        + "=" * 72
    )

    print(
        f"ANSWER: {answer}"
    )

    print(
        f"MODEL STATUS: {status}"
    )

    print(
        f"Dashboard: {dashboard_path}"
    )

    print(
        "=" * 72
    )

if __name__ == "__main__":
    main()
